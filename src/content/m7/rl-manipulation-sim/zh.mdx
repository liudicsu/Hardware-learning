## 从接到达到抓取

- **状态。** 关节角、末端位姿、物体位姿（或其紧凑编码）。下面的玩具把平面位置分箱，再加一个二值 grip 位。
- **动作。** 关节增量，或末端 Δxyz 加抓取命令。玩具用五个离散动作：N、S、E、W、CLOSE。
- **奖励塑形。** 到达 / 接近 / 抓住 / 抬起可以各自加一项。**稀疏**奖励只在成功时 +1；**稠密**奖励还会为靠近目标、在目标格合爪付分。
- **成功判据。** 这里：智能体站在目标格且 grip = 1。ManiSkill3 的 `PickCubeSO100-v1` 任务卡写的是「用 SO100 机器人抓起红色方块并移到目标位置」（桌面夹爪任务表；最长 50 步）。

On-policy 方法（如 PPO）用**当前**策略采的数据更新。Off-policy 方法（如 <Term id="sac">SAC</Term>，Haarnoja 等，arXiv 1801.01290）重用回放缓冲——Stable-Baselines3 把 Soft Actor-Critic 标成 "Off-Policy Maximum Entropy Deep Reinforcement Learning with a Stochastic Actor"。本页一句对照即可：样本昂贵时 off-policy 重用有帮助；已经有数千并行环境时，on-policy PPO 常常够用。

## 网格上的稀疏 vs 稠密

```python
#!/usr/bin/env python3
"""Sparse vs dense reward shaping on a discrete reach–grasp MDP (stdlib).

Lesson 7.2 used a clipped-ratio PPO toy for a 1-D reach. Here the task adds a
gripper channel: reach a cube, then close. We compare **sparse** (terminal +1)
vs **dense** (distance + grip shaping) under identical tabular Q-learning so the
success gap is seed-stable on CPU.

ManiSkill3 ``PickCubeSO100-v1`` / SB3 SAC are the GPU / off-policy counterparts
described in the MDX — this file does not import them (「本页未强制」).

State bins: x∈[0,1], y∈[0,1], grip∈{0,1}  →  N x N x 2
Actions: 5 moves {N,S,E,W,CLOSE}
Success: cell contains the goal and grip==1
"""
from __future__ import annotations

import random
from typing import Literal

SEED = 7
N = 6
GOAL = (4, 3)  # grid cell
EPISODES = 400
MAX_STEPS = 30
ALPHA = 0.35
GAMMA = 0.95
EPS_START = 0.5
EPS_END = 0.05

Action = Literal[0, 1, 2, 3, 4]  # N S E W CLOSE
N_ACTIONS = 5


def idx(x: int, y: int, g: int) -> int:
    return (g * N + y) * N + x


def clip_cell(v: int) -> int:
    return max(0, min(N - 1, v))


def step(x: int, y: int, g: int, a: int, dense: bool):
    nx, ny, ng = x, y, g
    if a == 0:
        ny = clip_cell(y - 1)
    elif a == 1:
        ny = clip_cell(y + 1)
    elif a == 2:
        nx = clip_cell(x + 1)
    elif a == 3:
        nx = clip_cell(x - 1)
    elif a == 4:
        ng = 1
    success = (nx, ny) == GOAL and ng == 1
    if dense:
        # Manhattan distance penalty + grip bonus on goal cell + terminal
        d = abs(nx - GOAL[0]) + abs(ny - GOAL[1])
        r = -0.1 * d
        if (nx, ny) == GOAL:
            r += 0.5 * ng
        if success:
            r += 5.0
    else:
        r = 1.0 if success else 0.0
    return nx, ny, ng, r, success


def train(dense: bool, seed: int = SEED):
    rng = random.Random(seed)
    q = [[0.0] * N_ACTIONS for _ in range(N * N * 2)]
    successes: list[int] = []
    returns: list[float] = []
    for ep in range(EPISODES):
        x = rng.randint(0, 1)
        y = rng.randint(0, 1)
        g = 0
        ep_ret = 0.0
        ok = False
        eps = EPS_START + (EPS_END - EPS_START) * (ep / max(1, EPISODES - 1))
        for _ in range(MAX_STEPS):
            s = idx(x, y, g)
            if rng.random() < eps:
                a = rng.randrange(N_ACTIONS)
            else:
                row = q[s]
                m = max(row)
                # tie-break deterministically by lowest index among maxima
                cands = [i for i, v in enumerate(row) if v == m]
                a = cands[0]
            nx, ny, ng, r, success = step(x, y, g, a, dense=dense)
            ns = idx(nx, ny, ng)
            best_next = max(q[ns])
            q[s][a] = (1 - ALPHA) * q[s][a] + ALPHA * (r + GAMMA * best_next)
            ep_ret += r
            x, y, g = nx, ny, ng
            if success:
                ok = True
                break
        successes.append(1 if ok else 0)
        returns.append(ep_ret)
    window = 50
    last_s = successes[-window:]
    last_r = returns[-window:]
    return {
        "mean_return": sum(last_r) / window,
        "success_rate": sum(last_s) / window,
        "last_ok": sum(last_s),
        "q": q,
    }


def greedy_rollout(q, seed: int = 99) -> bool:
    rng = random.Random(seed)
    x = rng.randint(0, 1)
    y = rng.randint(0, 1)
    g = 0
    for _ in range(MAX_STEPS):
        s = idx(x, y, g)
        row = q[s]
        m = max(row)
        a = min(i for i, v in enumerate(row) if v == m)
        x, y, g, _, success = step(x, y, g, a, dense=True)
        if success:
            return True
    return False


def main() -> None:
    sparse = train(dense=False)
    dense = train(dense=True)
    print("reward_shaping — tabular Q, sparse vs dense (seed=%d, episodes=%d, grid=%dx%d)"
          % (SEED, EPISODES, N, N))
    print("goal_cell=%s  actions=N,S,E,W,CLOSE" % (GOAL,))
    print("mode    mean_return  success_rate  last%d_ok" % 50)
    print("sparse  %11.4f  %12.3f  %9d" % (sparse["mean_return"], sparse["success_rate"], sparse["last_ok"]))
    print("dense   %11.4f  %12.3f  %9d" % (dense["mean_return"], dense["success_rate"], dense["last_ok"]))
    # evaluate greedy policy from dense Q on a few starts
    greeds = sum(1 for s in range(20) if greedy_rollout(dense["q"], seed=100 + s))
    print("dense_greedy_ok %d/20" % greeds)
    assert dense["success_rate"] > sparse["success_rate"], (
        "dense should beat sparse on success (%.3f vs %.3f)"
        % (dense["success_rate"], sparse["success_rate"])
    )
    assert dense["success_rate"] >= 0.5, "dense success_rate too low: %.3f" % dense["success_rate"]
    assert greeds >= 10, "dense greedy rollouts too weak: %d/20" % greeds
    print("ok: dense shaped reward outperforms sparse under identical Q-learning")


if __name__ == "__main__":
    main()
```

## 常见错误

- **奖励很稀疏却几乎不探索。** 表说明 400 回合不够让稀疏碰巧发现「在目标格合爪」。
- **状态里丢掉 grip 通道。** 只有到达的状态无法表示「我在方块上但还张着」。
- **只汇报稠密曲线。** 同一种子、同一超参下，永远把稀疏和稠密并排打印。
- **以为 ManiSkill GPU 仿真在每台笔记本都能跑。** 安装矩阵里 GPU 仿真是 Linux + NVIDIA；更宽的路径是 CPU 仿真。

## 事后经验回放：从实际到达的目标里学习

上面的网格 toy 显示，稀疏奖励下的表格 Q-learning 学不会，稠密塑形却能成功。塑形是一种办法，但它要求写出一个已经知道怎么走的奖励。**事后经验回放**（Hindsight Experience Replay，HER）保留稀疏的成功奖励，改的是回放。一个没到达目标的回合，毕竟到达了某个地方；HER 把这个地方当作目标重新回放，所以学习者在真正到达目标之前很久就能看到成功。本节在同一个网格 MDP 上用**合成**的表格模型加入 HER。它不是 SB3、ManiSkill 或 DDPG。

<Resource id="her-andrychowicz-2017" />

**实际打开的来源（已读 PDF 正文，arXiv 1707.01495 v3）。**

- **Andrychowicz 等人，《Hindsight Experience Replay》**（OpenAI），要点如下。
  - 方法能从稀疏、二值的奖励中学习，可以和任何 off-policy 算法结合，并且“可以看作一种隐式课程”。
  - 目标“影响智能体的动作，但不影响环境动力学”，所以只要算法是 off-policy 的，每条轨迹都可以换一个任意目标回放。最简单的版本用最终状态达到的目标回放。
  - 策略有 `final`、`future`（同一回合中之后观察到的 k 个状态）、`episode` 和 `random`；`future` 在 `k` 取 4 或 8 时最好，`k` 超过 8 会变差，因为普通回放数据剩得太少。
  - 翻转比特任务中，没有 HER 的 DQN 只能解 `n ≤ 13`，有 HER 的 DQN 能解到 `n = 50`。
  - 第 4.4 节中，用他们试过的塑形奖励，DDPG 和 DDPG+HER 都没有解出任何任务。论文补充说，每个问题都存在一个让它变容易的奖励，但设计它需要大量领域知识。
  - 论文的奖励是 `−[f_g(s) = 0]`，即成功之前都是 −1。以论文为准。

### 用达到的目标重新标注

<Formula label="稀疏目标奖励（本 toy）">r(s′, g) = 1 if cell(s′) = g and grip(s′) = 1, else 0</Formula>

<Formula label="future 重新标注">g′ = cell(s_j), with j drawn from the same episode at or after the replayed step, used with probability k / (k + 1)</Formula>

<Formula label="Q-learning 更新">Q(s, g, a) ← Q(s, g, a) + α · (r + γ · max Q(s′, g, ·) − Q(s, g, a))</Formula>

只有夹爪闭合时才算达到了某个目标，所以 `grip = 0` 的第 `j` 步会保留原目标。`k = 4` 时重新标注的比例是 4 / 5 = 0.80。论文是给每条转移加 `k` 个额外目标；本 toy 每次回放采样只抽一个目标，比例同样是 4 : 1。

**算例。** 一个回合从 (0, 0) 出发，目标 (4, 3)。智能体在第 3 步合上夹爪，然后乱走到第 30 步，停在 (3, 1)。稀疏奖励下每条转移都得 0。用 `g′ = (3, 1)` 重新标注后，最后一条从 (2, 1) 向东的转移得 1 并结束回合。在本页使用的相对目标表格中，这次更新落在“偏移 (−1, 0)、夹爪闭合、动作 E”这一项上。这正是智能体位于 (4, 3) 西边一格、夹爪闭合时需要的那一项。

**这里没有积分步长。** 环境是离散网格，没有任何东西在连续时间里步进，所以不涉及步长稳定性判据（独立测试会检查脚本里没有时间步）。学习率 `α = 0.35` 和 `γ = 0.95` 让每个 Q 值都不超过最大回报，稀疏奖励下最多是 1。

### 公平设置

`her_sparse_grasp_toy.py` 是**合成**的，只用标准库。**模型假设**如下。
- MDP 就是 `reward_shaping.py` 的那个，改成以目标为条件。状态是 N × N 网格上的 (x, y, grip)；动作是 N、S、E、W 和 CLOSE；CLOSE 让 `grip = 1` 并一直保持。成功指 30 步内夹爪在目标格上闭合。
- Q 是以相对目标的偏移 `(x − gx, y − gy, grip)` 为索引的表格。它代替论文中以目标为条件的网络：对一个目标学到的东西，会用到偏移相同的所有目标上。
- 每组**只从回放中学习**。每个回合之后，从所有存储的回合中均匀抽 64 条转移做更新。设置为 `α = 0.35`、`γ = 0.95`，ε 从 0.5 线性降到 0.05，贪心选择遇到相等时取编号最小的动作。

| 组 | 奖励 | 回放 |
|---|---|---|
| A_sparse | 成功得 1，否则 0 | 普通 |
| B_her | 同样的稀疏奖励 | `future`，k = 4 |
| C_dense | `reward_shaping.py` 的稠密奖励：−0.1 × 曼哈顿距离，在目标格上闭合夹爪 +0.5，成功 +5 | 普通 |

**共用随机数**，对每个（情形、种子），每个回合的起点和目标、每个（回合、步）的两个探索随机数只抽一次，三组共用。回放随机流在每组都从同一个种子开始，但因为缓冲区不同，消耗方式也不同。共 5 个种子。

各指标含义如下。
- `final` 是最后一个回合之后，在该情形所有（起点、目标）对上的贪心成功率。
- `auc` 是每 50 个回合一次检查点上贪心成功率的平均（样本效率）。
- `train` 是最后 50 个训练回合的成功率。

| 情形 | 网格 | 回合数 | 目标 |
|---|---|---|---|
| near_goal | 6 × 6 | 400 | x, y ≤ 2 的 9 个格子 |
| fixed_goal | 6 × 6 | 400 | 只有 `reward_shaping.py` 的目标 (4, 3) |
| far_goals | 6 × 6 | 400 | x + y ≥ 6 的 15 个格子 |
| big_grid | 10 × 10 | 400 | x + y ≥ 12 的 28 个格子 |

所有情形的起点都是 (0, 0)、(0, 1)、(1, 0) 和 (1, 1) 这四个格子。

**看过结果后做的选择。** 下面这些不在最初的设计里；toy 打印了每个备选，方便你看改了什么。
- **相对目标表格。** 第一次运行用 `(goal, cell, grip)` 做表格索引。这样的表格对从未到达过的目标什么也学不到，所以 HER 几乎没帮上忙：`near_goal` 中 final 0.9333 对稀疏的 0.7556，`fixed_goal` 中两个稀疏组都是 0.0000。论文的网络能在目标之间泛化，相对目标表格是最接近的表格替代。两种表格都会打印。
- **去掉的 `wall` 情形。** 第一次运行还有一个带五个墙格的情形，每组都接近零（final A 0.0000，B 0.0000，C 0.0250）。它被去掉了，因为相对目标表格看不到墙在哪里。
- **`big_grid`。** 它是在第一次运行之后加的，用来找 HER 不再有帮助的地方。先试的是 8 × 8 版本（目标 x + y ≥ 9），也会打印（final A 0.0000，B 0.4643，C 0.9571）。
- **试过但没有采用的设置。** 用第一种表格时，还试过每回合 256 次更新和 1 000 个回合；`far_goals` 中 HER 仍远低于稠密（final 0.333 对 1.000）。

#!/usr/bin/env python3
"""Synthetic Hindsight Experience Replay toy on the reach-grasp grid. Stdlib only.

The MDP is the one in reward_shaping.py, made goal-conditioned: state (x, y, grip) on an
N x N grid, actions N, S, E, W, CLOSE (CLOSE sets grip = 1 and it stays closed); the goal g
is a cell, and the episode succeeds when the gripper is closed on g. The predicate
f_g(s) = [cell(s) == g and grip(s) == 1]; the achieved goal of a state with grip = 1 is its
cell (states with grip = 0 achieve no goal).

Q is a table indexed by the goal-relative offset (x - gx, y - gy) and grip, a stand-in for the
goal-conditioned network of the paper: what is learned about one goal carries over to every
goal at the same offset. This representation was chosen after the first run, which indexed
the table by (goal, cell, grip); that run is printed as an alternative.

Arms (identical representation, hyperparameters and replay budget):
  A_sparse  reward 1 on success, 0 otherwise; replay without relabelling
  B_her     the same sparse reward; replay with the "future" strategy (Andrychowicz et al.):
            with probability K / (K + 1) a sampled transition is replayed with a goal achieved
            later in the same episode, its reward and termination recomputed for that goal
  C_dense   reward_shaping.py's dense reward (-0.1 Manhattan distance, +0.5 grip on the goal
            cell, +5 on success); replay without relabelling

Every arm learns only from replay: after each episode, UPDATES transitions are sampled
uniformly from all stored episodes and backed up with Q-learning (ALPHA, GAMMA).
Common random numbers: per (regime, seed) the start cell and goal of every episode and the
two exploration draws of every (episode, step) are drawn once and shared by all arms; the
replay-sampling stream is seeded identically per arm but is consumed differently because the
buffers differ. No integration step: the environment is a discrete grid.
Regimes: near_goal (goals within 2 cells), fixed_goal (reward_shaping.py's single goal (4, 3)),
far_goals (x + y >= 6 on 6 x 6), big_grid (x + y >= 12 on 10 x 10; added after the first run,
the 8 x 8 version tried before it is printed as an alternative).
Metrics per seed: final = greedy success over every (start, goal) pair of the regime after
the last episode; auc = mean greedy success over the checkpoints every EVAL_EVERY episodes
(sample efficiency); train = success share over the last 50 training episodes.
Labels: tie = identical on every seed; numerically zero = |paired mean| and sd both < 1e-9
(effectively a tie, no sign count); |m| <= sd = near-tie; |m| > sd = consistent gap (small if
|m| < 20 % of the smaller arm mean, else clear), with a sign count.
ManiSkill3 / SB3 HER / DDPG / a GPU: 本页未运行. Paper results: 以原文为准.
"""
import math
import random

SEEDS = (0, 1, 2, 3, 4)
N_ACTIONS = 5  # N S E W CLOSE
ALPHA = 0.35
GAMMA = 0.95
EPS_START, EPS_END = 0.5, 0.05
MAX_STEPS = 30
UPDATES = 64
K_FUTURE = 4
EVAL_EVERY = 50
NUM_ZERO = 1e-9
ARMS = ("A_sparse", "B_her", "C_dense")
# name: (grid N, episodes, start cells, goal cells)
REGIMES = {
    "near_goal": (6, 400, [(0, 0), (0, 1), (1, 0), (1, 1)], [(x, y) for x in range(3) for y in range(3)]),
    "fixed_goal": (6, 400, [(0, 0), (0, 1), (1, 0), (1, 1)], [(4, 3)]),
    "far_goals": (6, 400, [(0, 0), (0, 1), (1, 0), (1, 1)], [(x, y) for x in range(6) for y in range(6) if x + y >= 6]),
    "big_grid": (10, 400, [(0, 0), (0, 1), (1, 0), (1, 1)], [(x, y) for x in range(10) for y in range(10) if x + y >= 12]),
}
# big_grid was added after the first run; the 8 x 8 version tried before it is printed as an alternative
ALT_BIG = (8, 400, [(0, 0), (0, 1), (1, 0), (1, 1)], [(x, y) for x in range(8) for y in range(8) if x + y >= 9])
REPR = "relative"  # Q indexed by the goal-relative offset (chosen after the first run; ALT_REPR printed)
ALT_REPR = "absolute"  # Q indexed by (goal, cell, grip): the first run's table


def key(goal, cell, grip, repr_=REPR):
    if repr_ == "relative":
        return (cell[0] - goal[0], cell[1] - goal[1], grip)
    return (goal, cell, grip)


def step(n, cell, grip, a, goal, dense):
    x, y = cell
    nx, ny, ng = x, y, grip
    if a == 0:
        ny = max(0, y - 1)
    elif a == 1:
        ny = min(n - 1, y + 1)
    elif a == 2:
        nx = min(n - 1, x + 1)
    elif a == 3:
        nx = max(0, x - 1)
    else:
        ng = 1
    return (nx, ny), ng, reward((nx, ny), ng, goal, dense)


def reward(cell, grip, goal, dense):
    """Reward and termination of arriving in (cell, grip) when pursuing goal."""
    success = cell == goal and grip == 1
    if not dense:
        return (1.0 if success else 0.0), success
    r = -0.1 * (abs(cell[0] - goal[0]) + abs(cell[1] - goal[1]))
    if cell == goal:
        r += 0.5 * grip
    if success:
        r += 5.0
    return r, success


def shared_draws(regime, seed):
    """Common random numbers: episode (start, goal) and exploration draws, shared by every arm."""
    n, episodes, starts, goals = REGIMES[regime] if regime in REGIMES else ALT_BIG
    rng = random.Random(f"{regime}-{seed}")
    eps = [(rng.choice(starts), rng.choice(goals)) for _ in range(episodes)]
    explore = [[(rng.random(), rng.randrange(N_ACTIONS)) for _ in range(MAX_STEPS)] for _ in range(episodes)]
    return eps, explore


def greedy(row):
    m = max(row)
    return row.index(m)


def evaluate(q, regime, repr_=REPR):
    n, _, starts, goals = REGIMES[regime] if regime in REGIMES else ALT_BIG
    ok = 0
    for s0 in starts:
        for g in goals:
            cell, grip = s0, 0
            for _ in range(MAX_STEPS):
                a = greedy(q[key(g, cell, grip, repr_)])
                cell, grip, (_, done) = step(n, cell, grip, a, g, False)
                if done:
                    ok += 1
                    break
    return ok / (len(starts) * len(goals))


def train(arm, regime, seed, k_future=K_FUTURE, repr_=REPR):
    n, episodes, starts, goals = REGIMES[regime] if regime in REGIMES else ALT_BIG
    cells = [(x, y) for x in range(n) for y in range(n)]
    q = {key(g, c, gr, repr_): [0.0] * N_ACTIONS for g in cells for c in cells for gr in (0, 1)}
    eps_draws, explore = shared_draws(regime, seed)
    replay_rng = random.Random(f"replay-{regime}-{seed}")
    dense = arm == "C_dense"
    her = arm == "B_her"
    buffer, curve, train_ok = [], [], []
    for ep in range(episodes):
        start, goal = eps_draws[ep]
        eps = EPS_START + (EPS_END - EPS_START) * ep / max(1, episodes - 1)
        cell, grip, traj, ok = start, 0, [], False
        for t in range(MAX_STEPS):
            u, ra = explore[ep][t]
            a = ra if u < eps else greedy(q[key(goal, cell, grip, repr_)])
            ncell, ngrip, (_, done) = step(n, cell, grip, a, goal, dense)
            traj.append((cell, grip, a, ncell, ngrip))
            cell, grip = ncell, ngrip
            if done:
                ok = True
                break
        train_ok.append(ok)
        buffer.append((goal, traj))
        for _ in range(UPDATES):
            g, tr = buffer[replay_rng.randrange(len(buffer))]
            i = replay_rng.randrange(len(tr))
            c, gr, a, nc, ngr = tr[i]
            if her and replay_rng.random() < k_future / (k_future + 1.0):
                j = replay_rng.randrange(i, len(tr))
                if tr[j][4] == 1:  # a goal is achieved only with the gripper closed
                    g = tr[j][3]
            r, done = reward(nc, ngr, g, dense)
            target = r if done else r + GAMMA * max(q[key(g, nc, ngr, repr_)])
            row = q[key(g, c, gr, repr_)]
            row[a] += ALPHA * (target - row[a])
        if (ep + 1) % EVAL_EVERY == 0:
            curve.append(evaluate(q, regime, repr_))
    return {"final": curve[-1], "auc": sum(curve) / len(curve), "train": sum(train_ok[-50:]) / 50.0}


def mean_sd(xs):
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    if x != 0.0 and abs(x) < 5e-5:
        return f"{x:.3e}"
    return f"{x:.4f}"


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{num(m)}+-{num(s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{num(m)}+-{num(s)}: {label}; {sign_note(diffs, m)}"

## 基于势的奖励塑形

上面的稀疏对稠密网格，以及事后经验回放一节，已经在比较手写的稠密奖励和“保留稀疏奖励、改标注目标”。这一节**不是**第二条稀疏–稠密曲线，**不是**第二个 HER 玩具，也**不是** DrQ。它问的是更尖锐的问题：什么时候可以加上稠密奖励，却仍然保持原来的最优策略？

Ng、Daishi Harada、Stuart Russell；Computer Science Division, University of California, Berkeley。这份副本的标题页**本身没有**印出会议或期刊栏，本页也不编造一栏。论文中的等式与表述以论文为准。

定理 1（同一份副本）：若存在定义在状态上的实值 Phi，使得对每个可允许的转移都有

<Formula label="基于势的塑形（定理 1）">F(s, a, s') = γΦ(s') - Φ(s)</Formula>

则称 F 是**基于势的**塑形函数。当 γ < 1 时，该等式对每个状态都要成立（论文里那个可区分的吸收状态不必再挖出去）。于是，当从 M' = (S, A, T, γ, R+F) 而不是从 M = (S, A, T, γ, R) 学习时，F 是最优策略保持一致的充分必要条件：

- 充分性：M' 里的每个最优策略也是 M 的最优策略，反过来也成立。
- 必要性：若不存在这样的 Phi，则存在转移与奖励，使得 M' 的任何最优策略都不是 M 的最优策略。

引言里的“毛病”说明了为什么朴素的稠密奖励可能改变最优解：Randløv 与 Alstrøm 的自行车在靠近起点处学会绕小圈，因为朝目标前进会得到奖励；一个因碰到球而得分的足球智能体，学会了在球上“振动”的策略。这些例子以论文为准，它们不是玩具输出。

推论 2：在同一基于势的 F 下，若 γ = 1 时吸收状态上的 Phi 为 0，则 Q_M'(s, a) = Q_M(s, a) - Phi(s)，并且 V_M'(s) = V_M(s) - Phi(s)。下面的三状态手算检查无折扣环路求和以及这一平移。ManiSkill、SB3 与真实机械臂：本页未运行。

### 一条走廊、一个干扰环，以及常数势

**计算例子（手算）**。 带折扣的玩具使用 `GAMMA = 0.9000`，以及只在进入 G 时支付的稀疏 `GOAL_REWARD = 1.0000`。因此唯一最短路 S->J->P->G 的无塑形回报是 `hand_calc unshaped return from S along S->J->P->G = GAMMA^2 * GOAL_REWARD = 0.9000^2 * 1.0000 = 0.8100`。臂 B 使用 `Phi_B(s) = -distance_to_goal`，打印的距离为 S=3、J=2、P=1、G=0，在环路布局里还有 B=3。臂 C 在每次访问干扰格 B 时支付 `ILLEGAL_CYCLE_BONUS = 0.2500`，或在无环走廊的前进一步上支付 `ILLEGAL_FORWARD_BONUS = 0.0500`。常数势那一臂使用 `CONST_PHI = 3.0000`。

无折扣手算使用 `gamma_undiscounted = 1.0000`，三状态链 0、1、G，且 `Phi(0)=-2.0000`、`Phi(1)=-1.0000`、`Phi(G)=0.0000`。这时基于势的 F 化为 Phi(s') - Phi(s)。沿 0->1->G，各项是 `1.0000, 1.0000`，路径和是 `2.0000`。沿 2-环 1->0->1，各项是 `-1.0000, 1.0000`，**环路和是 0.0000**。在进入 1 时加非法奖励 `1.0000`，环路项是 `0.0000, 1.0000`，环路和是 `1.0000`（不是 0）。同一条链若每步代价是 `-1.0000`，则 `V_M(0)=-2.0000`、`V_M(1)=-1.0000`、`V_M(G)=0.0000`，于是推论 2 给出每个状态上 `V_M'(s) = 0.0000`。

**计数的正确性标准**。 从 S 出发的贪心路径到达 G 则 success 为 1。`original_return` 是该贪心策略在未塑形的 R 上的回报，不是在 R+F 上。当每个非目标状态上的贪心动作都与 A 一致时，`policy_agree_A` 为 1。贪心路径经过干扰格 B 时，`uses_cycle` 为 1。价值迭代一直做到 Bellman 残差低于 `VI_TOL = 1e-14`；贪心打平时取 `(fwd, back, ball, stay)` 里的第一个动作。

**这里没有积分步**。 这是离散的表格 MDP，每次比较都是精确的，没有步长判据。

### 公平设定

`potential_based_shaping_toy.py` 是**合成的**、只用标准库的表格价值迭代，**不是** ManiSkill、SB3 或真实机械臂。**模型假定**如下。

- 共享的确定性转移：四状态走廊 S-J-P-G，以及在同一走廊上把干扰格 B 接在 J 上（J 可以去 B；从 B 出发，`back` 回到 J，其余动作留在 B）。
- 臂 A 只用稀疏的未塑形 R。臂 B 加上 F = γΦ(s') - Φ(s)，势函数是事先声明的距离。臂 `B_const` 使用常数 Phi。臂 C 加上没有 -Phi(s) 对应项的非势奖励。

**共用随机数**。 MDP 是确定性的，并在各臂之间共享。五个种子 `0,1,2,3,4` 重复同一张表，以便把精确打平打成每个种子都相同。种子不添加抖动。

| 情形 | 布局 | 事先声明的期望（第一次运行之前写好） |
|---|---|---|
| acyclic_corridor | 仅 S-J-P-G | B 与 C 在 success 上可以与 A 接近打平（保留） |
| distractor_cycle | 走廊加上格 B | B 保持未塑形的最优（与 A 策略一致，并能到达目标）；C 稳定地更偏好环路（保留） |
| constant_phi | 同一环路布局，常数 Phi | B_const 与 A 精确打平（保留）；当非法环路有吸引力时，C 在原任务回报上落败（保留） |

标签：每个种子都相同是**打平**；若配对均值与标准差的绝对值都低于 `1e-9`，则是数值上为零；`|m| <= sd` 是**接近打平**；`|m| > sd` 是稳定差距，若 `|m|` 低于较小臂均值的 20 %（玩具打印 `0.2000`）则为**小**，否则为**明显**，并附带符号计数。

**看到结果之后做的选择**。 无。三臂、三种情形、γ、奖励和势函数都是第一次运行之前写好的，没有改过。未采用的替代：`Phi_alt = -2 * distance`，它仍然基于势。该替代在同一组 MDP 上打印，**没有采用**。

#!/usr/bin/env python3
"""Potential-based reward shaping stand-in (Ng, Harada, Russell Theorem 1).

A shared tabular MDP compares an unshaped sparse reward, a potential-based
shaping term F(s,a,s') = gamma * Phi(s') - Phi(s), and a non-potential
bonus that can lock onto a distractor cycle.  Optimal policies come from
value iteration, not a noisy learner.  Synthetic; not ManiSkill, SB3, or
a real arm.

Common random numbers: transitions are deterministic and shared across
arms.  Five seeds repeat the same MDP so exact-tie cells can be printed
as identical on every seed.  Seeds do not add jitter.

Metrics per seed: success (greedy path from the start reaches the goal),
original_return (the arm's greedy policy evaluated on the unshaped R),
policy_agree_A (greedy action agrees with A in every non-goal state),
uses_cycle (the greedy path from the start visits the distractor cell).

Regimes and expectations, written before the first run:
  acyclic_corridor   no return edge, so a mild illegal forward bonus
                     cannot form a cycle: B and C may near-tie A on
                     success (keep)
  distractor_cycle   B preserves the unshaped optimum (policy agreement
                     with A; reaches the goal); C consistently prefers
                     the cycle (keep)
  constant_phi       B with a constant Phi exact-ties A (keep); C loses
                     on original-task return when the illegal cycle is
                     attractive (keep)

Named alternative, same MDPs, not adopted: Phi_alt = -2 * distance, still
potential-based.

Paper identities, theorem wording, and the introduction's shaping bugs
are 以论文为准.  ManiSkill / SB3 / a real arm: 本页未运行.
"""
from __future__ import annotations

import math
from decimal import ROUND_HALF_UP, Decimal, getcontext

getcontext().rounding = ROUND_HALF_UP

SEEDS = (0, 1, 2, 3, 4)
NUM_ZERO = 1e-9
SMALL_FRAC = 0.2000  # 20 % of the smaller-arm mean; printed as 0.2000
VI_TOL = 1e-14
VI_MAX_ITERS = 10000

# Discounted main MDPs (gamma < 1, so S \\ {s0} = S in Theorem 1).
GAMMA = 0.9000
GOAL_REWARD = 1.0000
ILLEGAL_CYCLE_BONUS = 0.2500  # non-potential bonus on every visit to the ball
ILLEGAL_FORWARD_BONUS = 0.0500  # mild illegal bonus on a forward step
CONST_PHI = 3.0000
VI_START = "S"

# Undiscounted 3-state hand_calc (gamma = 1, absorbing G, Phi(G) = 0).
HC_GAMMA = 1.0000
HC_STEP_COST = -1.0000
HC_ILLEGAL_ENTER = 1.0000

STATES_CYCLE = ("S", "J", "P", "G", "B")
STATES_CORRIDOR = ("S", "J", "P", "G")
ACTIONS = ("fwd", "back", "ball", "stay")

# Distance-to-goal potential on each layout (declared).
DIST_CYCLE = {"S": 3, "J": 2, "P": 1, "G": 0, "B": 3}
DIST_CORRIDOR = {"S": 3, "J": 2, "P": 1, "G": 0}


def mean_sd(xs):
    if all(x == xs[0] for x in xs):
        return xs[0], 0.0
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    if x != 0.0 and abs(x) < 5e-4:
        return format(Decimal(repr(x)), ".3e")
    s = format(Decimal(repr(x)), ".4f")
    return "0.0000" if s == "-0.0000" else s


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def fmt_pair(m, s):
    return f"{num(m)}+-{num(s)}"


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{fmt_pair(m, s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < SMALL_FRAC * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{fmt_pair(m, s)}: {label}; {sign_note(diffs, m)}"


def step(state: str, action: str, states: tuple[str, ...]) -> str:
    """Deterministic next state; unavailable moves stay put."""
    if state == "G":
        return "G"
    if state == "S":
        if action == "fwd":
            return "J"
        return "S"
    if state == "J":
        if action == "fwd":
            return "P"
        if action == "back":
            return "S"
        if action == "ball" and "B" in states:
            return "B"
        return "J"
    if state == "P":
        if action == "fwd":
            return "G"
        if action == "back":
            return "J"
        return "P"
    if state == "B":
        if action == "back":
            return "J"
        return "B"
    return state


def unshaped_r(state: str, nxt: str) -> float:
    if state != "G" and nxt == "G":
        return GOAL_REWARD
    return 0.0


def phi_dist(state: str, dist: dict[str, int], scale: float = 1.0) -> float:
    return -scale * float(dist[state])


def shaping_f(state: str, nxt: str, phi_fn, gamma: float) -> float:
    return gamma * phi_fn(nxt) - phi_fn(state)


def reward(arm: str, state: str, nxt: str, dist: dict[str, int], phi_scale: float) -> float:
    r = unshaped_r(state, nxt)
    if arm == "A":
        return r
    if arm in ("B", "B_const", "B_alt"):
        if arm == "B":
            phi = lambda s: phi_dist(s, dist, 1.0)
        elif arm == "B_const":
            phi = lambda s: CONST_PHI
        else:
            phi = lambda s: phi_dist(s, dist, phi_scale)
        return r + shaping_f(state, nxt, phi, GAMMA)
    if arm == "C":
        if "B" in dist and nxt == "B":
            return r + ILLEGAL_CYCLE_BONUS
        if "B" not in dist and nxt != state and dist[nxt] < dist[state]:
            return r + ILLEGAL_FORWARD_BONUS
        return r
    raise ValueError(arm)


def value_iteration(states, dist, arm: str, phi_scale: float = 2.0):
    v = {s: 0.0 for s in states}
    for _ in range(VI_MAX_ITERS):
        nxt_v = {}
        delta = 0.0
        for s in states:
            if s == "G":
                # Absorbing goal: stay, reward is whatever R+F pays on G->G.
                r = reward(arm, "G", "G", dist, phi_scale)
                nxt_v[s] = r + GAMMA * v["G"]
            else:
                q_best = None
                for a in ACTIONS:
                    sp = step(s, a, states)
                    r = reward(arm, s, sp, dist, phi_scale)
                    q = r + GAMMA * v[sp]
                    if q_best is None or q > q_best:
                        q_best = q
                nxt_v[s] = q_best
            delta = max(delta, abs(nxt_v[s] - v[s]))
        v = nxt_v
        if delta < VI_TOL:
            break
    else:
        raise RuntimeError("value iteration did not converge")
    # Q and greedy policy (tie-break: first action in ACTIONS).
    q = {}
    pi = {}
    for s in states:
        best_a, best_q = None, None
        for a in ACTIONS:
            sp = step(s, a, states)
            r = reward(arm, s, sp, dist, phi_scale)
            qa = r + GAMMA * v[sp]
            q[(s, a)] = qa
            if best_q is None or qa > best_q + 1e-12:
                best_q, best_a = qa, a
        pi[s] = best_a
    return v, q, pi


def evaluate_policy_unshaped(states, pi, start: str) -> float:
    """Discounted return of pi on the unshaped R, from start."""
    v = {s: 0.0 for s in states}
    for _ in range(VI_MAX_ITERS):
        nxt_v = {}
        delta = 0.0
        for s in states:
            a = "stay" if s == "G" else pi[s]
            sp = step(s, a, states)
            r = unshaped_r(s, sp)
            nxt_v[s] = r + GAMMA * v[sp]
            delta = max(delta, abs(nxt_v[s] - v[s]))
        v = nxt_v
        if delta < VI_TOL:
            break
    else:
        raise RuntimeError("policy evaluation did not converge")
    return v[start]


def greedy_path(states, pi, start: str, horizon: int = 16) -> list[str]:
    path = [start]
    s = start
    for _ in range(horizon):
        if s == "G":
            break
        s = step(s, pi[s], states)
        path.append(s)
        if len(path) >= 2 and path[-1] == path[-2]:
            break
    return path


def run_arm(states, dist, arm: str, phi_scale: float = 2.0) -> dict:
    v, q, pi = value_iteration(states, dist, arm, phi_scale)
    path = greedy_path(states, pi, VI_START)
    success = 1.0 if "G" in path else 0.0
    uses_cycle = 1.0 if "B" in path else 0.0
    original_return = evaluate_policy_unshaped(states, pi, VI_START)
    return {
        "success": success,
        "original_return": original_return,
        "uses_cycle": uses_cycle,
        "path": path,
        "pi": pi,
        "v": v,
        "q": q,
    }


def policy_agree(pi_a, pi_b, states) -> float:
    for s in states:
        if s == "G":
            continue
        if pi_a[s] != pi_b[s]:
            return 0.0
    return 1.0


def fmt_policy(pi, states) -> str:
    return " ".join(f"{s}:{pi[s]}" for s in states if s != "G")


def hand_calc_block() -> None:
    """Undiscounted 3-state chain and a 2-cycle; potential cycle sum is 0."""
    # States 0, 1, G with Phi(G)=0, Phi(1)=-1, Phi(0)=-2 (minus distance).
    phi = {"0": -2.0, "1": -1.0, "G": 0.0}
    def f(s, sp):
        return HC_GAMMA * phi[sp] - phi[s]

    path = [("0", "1"), ("1", "G")]
    path_terms = [f(s, sp) for s, sp in path]
    cycle = [("1", "0"), ("0", "1")]
    cycle_terms = [f(s, sp) for s, sp in cycle]
    illegal_cycle = [HC_ILLEGAL_ENTER if sp == "1" and s != "1" else 0.0 for s, sp in cycle]

    print(
        "hand_calc 3-state chain states=0,1,G gamma_undiscounted="
        + num(HC_GAMMA)
        + " Phi(0)="
        + num(phi["0"])
        + " Phi(1)="
        + num(phi["1"])
        + " Phi(G)="
        + num(phi["G"])
        + " (Phi(G)=0)"
    )
    print(
        "hand_calc potential F(s,s') = gamma*Phi(s')-Phi(s) with gamma="
        + num(HC_GAMMA)
        + " reduces to Phi(s')-Phi(s)"
    )
    print(
        "hand_calc path 0->1->G F terms = "
        + ", ".join(num(t) for t in path_terms)
        + " path sum = "
        + num(sum(path_terms))
    )
    print(
        "hand_calc potential 2-cycle 1->0->1 F terms = "
        + ", ".join(num(t) for t in cycle_terms)
        + " cycle sum = "
        + num(sum(cycle_terms))
    )
    print(
        "hand_calc illegal bonus "
        + num(HC_ILLEGAL_ENTER)
        + " on entering 1; same 2-cycle terms = "
        + ", ".join(num(t) for t in illegal_cycle)
        + " cycle sum = "
        + num(sum(illegal_cycle))
        + " (not 0)"
    )

    # Corollary 2 style check: step cost -1 until G, one forward action,
    # V_M(0)=-2, V_M(1)=-1, V_M(G)=0; V_M'(s)=V_M(s)-Phi(s).
    v_m = {"0": -2.0, "1": -1.0, "G": 0.0}
    print(
        "hand_calc corollary-2 setup step_cost="
        + num(HC_STEP_COST)
        + " V_M(0)="
        + num(v_m["0"])
        + " V_M(1)="
        + num(v_m["1"])
        + " V_M(G)="
        + num(v_m["G"])
    )
    for s in ("0", "1", "G"):
        v_mp = v_m[s] - phi[s]
        print(
            "hand_calc corollary-2 V_M'("
            + s
            + ")=V_M("
            + s
            + ")-Phi("
            + s
            + ") = "
            + num(v_m[s])
            + "-"
            + "("
            + num(phi[s])
            + ") = "
            + num(v_mp)
        )


def q_shift_err(q_b, q_a, phi_fn, states) -> float:
    err = 0.0
    for s in states:
        for a in ACTIONS:
            err = max(err, abs(q_b[(s, a)] - (q_a[(s, a)] - phi_fn(s))))
    return err


def col(rows, metric):
    return [row[metric] for row in rows]


def print_regime(name: str, expl: str, states, dist, arms: tuple[str, ...], const_phi: bool) -> dict:
    print(f"declared expectation {name}: {expl}")
    print(f"regime {name}  (mean+-sd over seeds; deterministic MDP)")
    rows = {arm: [] for arm in arms}
    recs0 = {}
    for seed in SEEDS:
        _ = seed
        for arm in arms:
            rec = run_arm(states, dist, arm)
            rec["policy_agree_A"] = 0.0  # filled after A is known
            rows[arm].append(rec)
        # fill agreement vs this seed's A
        pi_a = rows["A"][-1]["pi"]
        for arm in arms:
            rows[arm][-1]["policy_agree_A"] = policy_agree(pi_a, rows[arm][-1]["pi"], states)
        if seed == 0:
            recs0 = {arm: rows[arm][0] for arm in arms}

    print(f"  seed 0 (all {len(SEEDS)} seeds identical):")
    for arm in arms:
        rec = recs0[arm]
        print(
            f"    {arm}: success={num(rec['success'])} original_return={num(rec['original_return'])} "
            f"policy_agree_A={num(rec['policy_agree_A'])} uses_cycle={num(rec['uses_cycle'])} "
            f"path={'->'.join(rec['path'])} policy={fmt_policy(rec['pi'], states)}"
        )

    metrics = ("success", "original_return", "policy_agree_A", "uses_cycle")
    for arm in arms:
        for metric in metrics:
            xs = col(rows[arm], metric)
            print(f"  {arm} {metric} {fmt_pair(*mean_sd(xs))}")
    for metric in metrics:
        if "B" in arms:
            print(f"  paired B-A {metric}: {verdict(col(rows['B'], metric), col(rows['A'], metric))}")
        if "B_const" in arms:
            print(
                f"  paired B_const-A {metric}: "
                + verdict(col(rows["B_const"], metric), col(rows["A"], metric))
            )
        if "C" in arms:
            print(f"  paired C-A {metric}: {verdict(col(rows['C'], metric), col(rows['A'], metric))}")
            if "B" in arms:
                print(f"  paired C-B {metric}: {verdict(col(rows['C'], metric), col(rows['B'], metric))}")

    # Potential-based Q-shift check on seed 0 (toy identity, printed).
    if "B" in arms:
        err = q_shift_err(
            recs0["B"]["q"], recs0["A"]["q"], lambda s: phi_dist(s, dist, 1.0), states
        )
        print(f"  toy Q-shift check max|Q_B(s,a)-(Q_A(s,a)-Phi(s))| = {num(err)}")
    if "B_const" in arms:
        err = q_shift_err(
            recs0["B_const"]["q"], recs0["A"]["q"], lambda s: CONST_PHI, states
        )
        print(f"  toy Q-shift check max|Q_Bconst(s,a)-(Q_A(s,a)-CONST_PHI)| = {num(err)}")
    return rows
