## MJCF next to URDF

A URDF describes links and joints for ROS. An MJCF file describes the same rigid tree, but contact and actuation are first-class:

| MJCF | Rough URDF cousin | What it adds |
|------|-------------------|--------------|
| `body` | `link` | Inertia lives on the body; children nest as a tree |
| `joint` | `joint` | Hinge, slide, free, ball, … with ranges |
| `geom` | `collision` / `visual` | Shape **and** contact parameters (`friction`, `solref`, `solimp`) |
| `actuator` | (usually separate) | Direct drives: `position`, `motor`, `velocity`, … |

Menagerie's `robotstudio_so101` (Apache-2.0, needs MuJoCo ≥ 3.1.3) is a six-hinge SO-101 adapted from TheRobotStudio's `Simulation/SO101/so101_new_calib.xml`: `implicitfast` integrator, simplified collision geoms, gripper contact parameters suited to manipulation, and a camera mount. The older `trs_so_arm100` model is five DoF and already set `impratio="10"` with `cone="elliptic"` to improve anti-slip. This lesson's runnable scene is a **minimal** two-jaw soft-contact gripper (`grip_scene.xml`) so the 3×3 table stays CPU-cheap and mesh-free; after you can read that table, open the Menagerie model on your machine.

## Soft contact is a tunable spring

MuJoCo does not treat contacts as infinitely hard. Each constraint has a `solref` pair `(timeconst, dampratio)`: a smaller time constant is a stiffer contact; `dampratio = 1` is critical damping. `solimp` shapes how the impedance ramps with penetration. At the option level, `impratio` raises the friction-to-normal impedance ratio (important when a grasp must not slip), and `cone="elliptic"` uses an elliptic friction cone instead of a pyramidal approximation. These are the knobs the Menagerie SO-101 README highlights; they are also the knobs your grasp will be sensitive to.

## A grip-and-lift sweep

`grip_scene.xml` has a vertical lift slider, left/right jaw sliders with position actuators, and a free cube. The script turns gravity off while the jaws close, then turns gravity on and lifts. For each of three timesteps and three `solref` time constants it reports the cube's final height, the worst penetration seen, and HOLD vs SLIP.

```python
#!/usr/bin/env python3
"""grip_sweep.py: 3x3 sweep of MuJoCo timestep x solref timeconst on a soft-contact grip-and-lift.

A two-jaw slider gripper closes on a free cube (gravity off), then lifts with gravity on.
For each (timestep, solref timeconst) pair we report whether the cube was held and the
maximum contact penetration seen during the run. Fixed scene: grip_scene.xml next to this file.

Requires: mujoco (private venv + pip install mujoco; Menagerie models need >= 3.1.3). Standard library + mujoco.
usage: python3 grip_sweep.py
"""
from __future__ import annotations

import os
import sys

try:
    import mujoco
    import numpy as np
except ImportError as e:
    sys.stderr.write(
        "grip_sweep.py needs mujoco (and numpy). "
        "On Ubuntu 24.04 (PEP 668) use a private venv:\n"
        "  python3 -m venv ~/Hardware-learning/.venv && "
        "source ~/Hardware-learning/.venv/bin/activate && "
        "python3 -m pip install mujoco\n"
        f"Import error: {e}\n"
    )
    sys.exit(1)

HERE = os.path.dirname(os.path.abspath(__file__))
SCENE = os.path.join(HERE, "grip_scene.xml")

# Locked so expected/ stays stable. Soft + large dt → slip; stiff → hold. Penetration grows with timeconst.
TIMESTEPS = (0.002, 0.005, 0.010)
TIMECONSTS = (0.004, 0.020, 0.080)  # solref[0]; dampratio fixed at 1.0
CLOSE = 0.0245
FRICTION = 1.2
CUBE_MASS = 0.15  # overrides grip_scene.xml cube geom mass=0.08 at runtime
HOLD_Z = 0.08
LIFT_CTRL = 0.10


def run(timestep: float, timeconst: float) -> tuple[float, float, bool]:
    m = mujoco.MjModel.from_xml_path(SCENE)
    m.opt.timestep = timestep
    m.opt.impratio = 10.0
    m.opt.cone = 1  # elliptic
    for bi in range(m.nbody):
        name = mujoco.mj_id2name(m, mujoco.mjtObj.mjOBJ_BODY, bi) or ""
        if name == "cube":
            m.body_mass[bi] = CUBE_MASS
            m.body_inertia[bi] = np.array([CUBE_MASS * 8e-5] * 3)
    for gi in range(m.ngeom):
        name = mujoco.mj_id2name(m, mujoco.mjtObj.mjOBJ_GEOM, gi) or ""
        if name in ("left_g", "right_g", "cube_geom"):
            m.geom_solref[gi] = (timeconst, 1.0)
            m.geom_friction[gi] = [FRICTION, 0.005, 0.0001]
    d = mujoco.MjData(m)

    # Close without gravity so the cube stays between the jaws.
    m.opt.gravity[:] = (0.0, 0.0, 0.0)
    d.ctrl[:] = (0.0, CLOSE, -CLOSE)
    for _ in range(max(1, int(0.8 / timestep))):
        mujoco.mj_step(m, d)
    max_pen = 0.0
    for i in range(d.ncon):
        if d.contact[i].dist < 0:
            max_pen = max(max_pen, -d.contact[i].dist)

    # Gravity on, lift the palm.
    m.opt.gravity[:] = (0.0, 0.0, -9.81)
    d.ctrl[:] = (LIFT_CTRL, CLOSE, -CLOSE)
    for _ in range(max(1, int(1.5 / timestep))):
        mujoco.mj_step(m, d)
        for i in range(d.ncon):
            if d.contact[i].dist < 0:
                max_pen = max(max_pen, -d.contact[i].dist)

    cube_z = float(d.qpos[5])  # freejoint xyz after lift/left/right
    held = cube_z > HOLD_Z
    return cube_z, max_pen, held


def main() -> None:
    print(f"scene={os.path.basename(SCENE)}")
    print(f"friction={FRICTION}  cube_mass={CUBE_MASS} kg  close={CLOSE} m")
    print("dt_s   timeconst  cube_z_m  pen_mm  result")
    holds = 0
    for dt in TIMESTEPS:
        for tc in TIMECONSTS:
            z, pen, held = run(dt, tc)
            holds += int(held)
            print(
                f"{dt:5.3f}  {tc:8.3f}  {z:8.4f}  {pen * 1e3:6.2f}  "
                f"{'HOLD' if held else 'SLIP'}"
            )
    print(f"held {holds}/9  (soft + large dt slips; penetration grows with timeconst)")


if __name__ == "__main__":
    main()
```

## Position actuators and a step

A `<position>` actuator is a PD drive toward a commanded joint coordinate (`kp`, `kv`, `ctrlrange`). In the sweep, `d.ctrl = (lift, close, −close)` is exactly that command. On the real SO-101 each STS3215 already runs a position loop; in MuJoCo you are choosing how stiff the *simulated* loop and the *simulated* contacts are. A step that looks fine at `dt = 0.002` can penetrate and slip at `dt = 0.010` with a soft `solref` — the same ctrl value, a different solver.

## Common mistakes

- **Copying Menagerie meshes into the lesson expected output.** The quoted table comes from `grip_scene.xml`. Menagerie is what you load next on your laptop.
- **Treating a HOLD at one `(dt, solref)` as proof the grasp is robust.** Change one knob and re-run the sweep before you trust a number.
- **Forgetting `cone` and `impratio` when a grasp slips with "enough" friction.** Elliptic cones and a higher friction/normal impedance ratio are exactly what the older SO-ARM100 Menagerie notes call out for anti-slip.

## Semi-implicit and implicit-in-velocity integrators versus explicit Euler

MuJoCo integrates continuous-time forward dynamics with several schemes. **Explicit Euler** is described but **not implemented**, because **semi-implicit Euler** (velocity first, then position with the *new* velocity) is strictly better and standard in physics simulation. **Implicit-in-velocity** / `implicitfast` approximates a at t+h with a first-order expansion in velocity only, and is aimed at velocity-dependent stiffness. Integrating some gyroscopic terms *explicitly* "can lead to **energy gain** and divergence of fast-spinning free bodies" (MuJoCo Computation documentation, Integrators; body read; 以原文为准). MuJoCo's `Euler` is therefore the semi-implicit scheme (with implicit joint damping when present); `implicit` and `implicitfast` are implicit only in the velocity-dependent forces. This section is **not** a redo of `grip_sweep` (timestep × `solref` on a soft-contact lift) and **not** a rewrite of the Menagerie `implicitfast` one-liner.

### Fair setup

`mujoco_integrator_energy_toy.py` is **synthetic** and stdlib only. **Model assumptions:** two shared plants, no MuJoCo binary and no quaternion update. A simple pendulum with light linear damping (gravity depends on angle; damping depends on velocity) and a torque-free rigid body with inertia diag(1, 2, 3), tumbling off the intermediate axis. Energy of the spinner is 1/2 w·I w and depends only on angular velocity, so A and B use the same velocity update there (the documented difference is the position step). Arm A is explicit Euler, B is semi-implicit, C is implicit-in-velocity with `Mhat = M + h D`, and C_bad flips the sign of `D` (an inconsistent linearisation, kept). **Common random numbers:** five seeds; per-seed initial-condition jitter is drawn once and reused by every arm. Metrics are energy drift dE from the start, the absolute drift, a blow-up flag, trajectory error versus a tiny-step reference, and |E − Eref|. After the first run showed that |dE| from E0 can be misleading under damping, |E − Eref| was added as a printed companion metric (disclosed). Real MuJoCo / Menagerie: 本页未运行.

**Stable-step criterion.** For the undamped linear oscillator x'' + ω0² x = 0, semi-implicit Euler has amplification trace 2 − h² ω0² and stays on the unit circle if and only if h < h* = 2/ω0. At exactly h* the trace is −2, so −1 is a repeated eigenvalue in a Jordan block and the solution grows linearly (hand calculation / analytic; the toy does not print a run at h = h*). Here ω0 = 3.1320919527 rad/s and h* = 0.6385508568 s. The default moderate step 0.1500 s is below h* (h/h* = 0.2349, `assert_below_hstar=1`). The toy prints a spectral radius and a 40 s energy-bounded run at 0.50×, 0.98× and 1.02× h*, and at a half default step: 0.98× is bounded with spectral radius 1.000000, 1.02× grows (spectral radius 1.490835, energy_bounded=0), and the half step is bounded. A stable integrator is not automatically a stable *contact* model — this does not contradict `grip_sweep`.

**Regimes, written before the first run:** on a lightly damped pendulum at moderate h, B and C should **consistently** show lower |dE| than A; at a very small h the three may **near-tie**; at a large h / fast spin, A should **consistently lose** (energy gain or blow-up); a flipped implicit `D` may **lose** — that cell is kept.

#!/usr/bin/env python3
"""Synthetic energy-drift comparison of explicit, semi-implicit and
implicit-in-velocity Euler, as a stdlib stand-in of the updates in the
MuJoCo Computation documentation (Integrators).

Stdlib only. Not mujoco.step, not a redo of grip_sweep (timestep x solref
on a soft-contact lift), and not a rewrite of the Menagerie implicitfast
one-liner. Real MuJoCo / Menagerie: 本页未运行.

Two shared plants, same h grid idea, initial energy printed:
  pendulum  simple pendulum with light linear damping (position-dependent
            gravity + velocity-dependent damping)
  spinner   torque-free rigid body with asymmetric inertia (gyroscopic
            bias; energy is 1/2 w·I w; no quaternion is used)

Arms:
  A_explicit     v_{t+h}=v_t+h a_t,      q_{t+h}=q_t+h v_t
  B_semimpl      v_{t+h}=v_t+h a_t,      q_{t+h}=q_t+h v_{t+h}
  C_implvel      implicit-in-velocity: v_{t+h}=v_t+h Mhat^{-1} M a_t,
                 Mhat=M+h D, D = -d(forces)/dv; then q uses v_{t+h}
  C_bad          same as C but with a sign-flipped D (inconsistent
                 linearization; declared fail arm; kept)

For the spinner, energy depends only on w, so A and B use the same
velocity update (the documented difference is the position step).

Regimes (declared before the first run):
  moderate_h     lightly damped pendulum, moderate h -> B,C consistent
                 lower |dE| than A
  tiny_h         very small h -> A,B,C may near-tie on |dE|
  large_h_spin   large h and/or a fast tumble -> A consistent lose
                 (energy gain or blow-up)
  inconsistent_D flipped gyro D -> C_bad may lose (energy gain); kept

Stable-step criterion (mandatory): for the undamped linear oscillator
x''+w0^2 x=0, semi-implicit Euler has amplification trace 2-h^2 w0^2
and stays on the unit circle iff h < h* = 2/w0. At h = h* the trace
is -2, so -1 is a repeated eigenvalue in a Jordan block and the
solution grows linearly (analytic; not printed). The default moderate
h is below h*. The toy prints a time-domain energy-bounded run and the
spectral radius at 0.50x, 0.98x and 1.02x h*.

Common random numbers: per-seed initial-condition jitter is drawn once
and reused by every arm. Seeds 0-4.
MuJoCo Computation documentation: body read; 以原文为准.
"""
from __future__ import annotations

import math
import random

SEEDS = (0, 1, 2, 3, 4)
NUM_ZERO = 1e-9
# pendulum (written before the first run)
G = 9.81
L = 1.0
MASS = 1.0
INERTIA_P = MASS * L * L
C_DAMP = 0.05
OMEGA0 = math.sqrt(G / L)
H_STAR = 2.0 / OMEGA0
H_MOD = 0.15
H_TINY = 0.005
H_LARGE = 0.55
T_PEND = 20.0
TH0 = 0.40
# spinner
I_DIAG = (1.0, 2.0, 3.0)
W_SPIN = (0.40, 5.00, 0.30)  # perturbation off the intermediate axis
H_SPIN = 0.08
T_SPIN = 6.0
BLOW = 100.0
ARMS = ("A_explicit", "B_semimpl", "C_implvel", "C_bad")
# written before the first run
REGIMES = ("moderate_h", "tiny_h", "large_h_spin", "inconsistent_D")
STREAM = {"pendulum": 0, "spinner": 1}


def mean_sd(xs: list[float]) -> tuple[float, float]:
    mu = sum(xs) / len(xs)
    if len(xs) == 1:
        return mu, 0.0
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x: float) -> str:
    if not math.isfinite(x):
        return "nan"
    if x != 0.0 and abs(x) < 5e-5:
        return f"{x:.3e}"
    return f"{x:.4f}"


def sign_note(diffs: list[float], m: float) -> str:
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def verdict(xa: list[float], xb: list[float]) -> str:
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{num(m)}+-{num(s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{num(m)}+-{num(s)}: {label}; {sign_note(diffs, m)}"


def pend_energy(th: float, w: float) -> float:
    return 0.5 * INERTIA_P * w * w + MASS * G * L * (1.0 - math.cos(th))


def pend_acc(th: float, w: float, damp: float = C_DAMP) -> float:
    return -(G / L) * math.sin(th) - (damp / INERTIA_P) * w


def tiny_step_ref(th: float, w: float, t_end: float, damp: float = C_DAMP) -> tuple[float, float, float]:
    # semi-implicit at a very small step, for trajectory and energy error
    h = 1e-4
    n = int(round(t_end / h))
    for _ in range(n):
        a = pend_acc(th, w, damp)
        w = w + h * a
        th = th + h * w
    return th, w, pend_energy(th, w)


def step_pend(th: float, w: float, h: float, arm: str, damp: float = C_DAMP) -> tuple[float, float]:
    a = pend_acc(th, w, damp)
    if arm == "A_explicit":
        th_n = th + h * w
        w_n = w + h * a
        return th_n, w_n
    if arm == "B_semimpl":
        w_n = w + h * a
        th_n = th + h * w_n
        return th_n, w_n
    # implicit-in-velocity: D = damp, M = INERTIA_P
    # a = M^{-1}(tau_grav - damp w), da/dw = -M^{-1} D with D = damp
    sign = -1.0 if arm == "C_bad" else 1.0
    D = sign * damp
    mhat = INERTIA_P + h * D
    w_n = w + h * (INERTIA_P / mhat) * a
    th_n = th + h * w_n
    return th_n, w_n


def run_pend(th0: float, w0: float, h: float, t_end: float, arm: str, damp: float = C_DAMP) -> dict:
    th, w = th0, w0
    e0 = pend_energy(th, w)
    e_max, e_last = e0, e0
    n = int(round(t_end / h))
    blow = False
    for _ in range(n):
        th, w = step_pend(th, w, h, arm, damp)
        if not math.isfinite(th) or not math.isfinite(w):
            blow = True
            e_last = float("inf")
            break
        e = pend_energy(th, w)
        e_last = e
        if e > e_max:
            e_max = e
        if e > BLOW * e0 or e < -BLOW * e0:
            blow = True
            break
    dE = e_last - e0
    th_r, w_r, e_ref = tiny_step_ref(th0, w0, t_end, damp)
    err = abs((th - th_r + math.pi) % (2 * math.pi) - math.pi) if math.isfinite(th) else float("inf")
    e_err = abs(e_last - e_ref) if math.isfinite(e_last) else float("inf")
    return {"dE": dE, "absdE": abs(dE) if math.isfinite(dE) else float("inf"), "emax": e_max, "e0": e0, "blow": int(blow), "traj_err": err, "e_err": e_err, "e_ref": e_ref}


def cross(a, b):
    return (a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0])


def imul(w):
    return (I_DIAG[0] * w[0], I_DIAG[1] * w[1], I_DIAG[2] * w[2])


def spin_energy(w) -> float:
    iw = imul(w)
    return 0.5 * (w[0] * iw[0] + w[1] * iw[1] + w[2] * iw[2])


def spin_acc(w):
    # I wdot = - w x (I w)
    tau = cross(w, imul(w))  # w x I w
    return (-tau[0] / I_DIAG[0], -tau[1] / I_DIAG[1], -tau[2] / I_DIAG[2])


def add(a, b, s=1.0):
    return (a[0] + s * b[0], a[1] + s * b[1], a[2] + s * b[2])


def D_gyro(w):
    # derivative of (w x I w) applied to basis; D = d(w x I w)/dw
    # J(u) = u x (I w) + w x (I u)
    rows = []
    for k in range(3):
        u = (1.0 if k == 0 else 0.0, 1.0 if k == 1 else 0.0, 1.0 if k == 2 else 0.0)
        ju = add(cross(u, imul(w)), cross(w, imul(u)))
        rows.append(ju)
    # rows are J(e_k); we want columns = J(e_k)
    return [[rows[k][i] for k in range(3)] for i in range(3)]


def solve3(A, b):
    # Gaussian elimination with partial pivot
    M = [A[i][:] + [b[i]] for i in range(3)]
    for k in range(3):
        piv = max(range(k, 3), key=lambda r: abs(M[r][k]))
        M[k], M[piv] = M[piv], M[k]
        if abs(M[k][k]) < 1e-18:
            return None
        inv = 1.0 / M[k][k]
        for j in range(k, 4):
            M[k][j] *= inv
        for i in range(3):
            if i == k:
                continue
            f = M[i][k]
            for j in range(k, 4):
                M[i][j] -= f * M[k][j]
    return (M[0][3], M[1][3], M[2][3])


def step_spin(w, h, arm: str):
    a = spin_acc(w)
    if arm in ("A_explicit", "B_semimpl"):
        return add(w, a, h)
    # implicit-in-velocity, eq. (6): (I + h D) (w' - w) = h I a
    # a = I^{-1}(- w x I w), D = d(w x I w)/dw  so da/dw = -I^{-1} D
    Dg = D_gyro(w)
    sign = -1.0 if arm == "C_bad" else 1.0
    Mhat = [[(I_DIAG[i] if i == j else 0.0) + h * sign * Dg[i][j] for j in range(3)] for i in range(3)]
    rhs = (h * I_DIAG[0] * a[0], h * I_DIAG[1] * a[1], h * I_DIAG[2] * a[2])
    dw = solve3(Mhat, rhs)
    if dw is None:
        return (float("inf"), float("inf"), float("inf"))
    return add(w, dw)


def run_spin(w0, h, t_end, arm: str) -> dict:
    w = w0
    e0 = spin_energy(w)
    e_last, e_max = e0, e0
    blow = False
    n = int(round(t_end / h))
    for _ in range(n):
        w = step_spin(w, h, arm)
        if not all(math.isfinite(x) for x in w):
            blow = True
            e_last = float("inf")
            break
        e = spin_energy(w)
        e_last = e
        if e > e_max:
            e_max = e
        if e > BLOW * e0:
            blow = True
            break
    dE = e_last - e0
    return {"dE": dE, "absdE": abs(dE) if math.isfinite(dE) else float("inf"), "emax": e_max, "e0": e0, "blow": int(blow)}

def amp_matrix(h: float, omega: float) -> list[list[float]]:
    # semi-implicit on x'' + omega^2 x = 0
    # v' = v - h omega^2 x; x' = x + h v'
    w2 = omega * omega
    return [[1.0 - h * h * w2, h], [-h * w2, 1.0]]


def spectral_radius(M: list[list[float]]) -> float:
    # 2x2 eigenvalues
    tr = M[0][0] + M[1][1]
    det = M[0][0] * M[1][1] - M[0][1] * M[1][0]
    disc = tr * tr - 4.0 * det
    if disc >= 0:
        s = math.sqrt(disc)
        evs = [abs(0.5 * (tr + s)), abs(0.5 * (tr - s))]
    else:
        r = math.sqrt(abs(det))  # for this symplectic map det=1, |λ|=1
        evs = [r, r]
    return max(evs)


def energy_bounded_run(h: float, t_end: float) -> int:
    # undamped linear oscillator, semi-implicit, unit energy
    # x'' + OMEGA0^2 x = 0, E = 0.5 (v^2 + w0^2 x^2)
    x, v = 1.0, 0.0
    e0 = 0.5 * OMEGA0 * OMEGA0 * x * x
    n = int(round(t_end / h))
    for _ in range(n):
        v = v - h * OMEGA0 * OMEGA0 * x
        x = x + h * v
        if not math.isfinite(x) or not math.isfinite(v):
            return 0
        e = 0.5 * (v * v + OMEGA0 * OMEGA0 * x * x)
        if e > BLOW * e0:
            return 0
    return 1
