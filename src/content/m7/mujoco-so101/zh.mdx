## 把 MJCF 放在 URDF 旁边看

URDF 为 ROS 描述 link 与 joint。MJCF 描述同一棵刚体树，但接触和驱动是一等公民：

| MJCF | 大致对应的 URDF | 多出来的部分 |
|------|----------------|--------------|
| `body` | `link` | 惯性在 body 上；子体嵌套成树 |
| `joint` | `joint` | hinge、slide、free、ball… 带范围 |
| `geom` | `collision` / `visual` | 形状**以及**接触参数（`friction`、`solref`、`solimp`） |
| `actuator` | （通常另写） | 直接驱动：`position`、`motor`、`velocity`… |

Menagerie 的 `robotstudio_so101`（Apache-2.0，要求 MuJoCo ≥ 3.1.3）是六轴 SO-101，改写自 TheRobotStudio 的 `Simulation/SO101/so101_new_calib.xml`：`implicitfast` 积分器、简化碰撞几何、适合操作的夹爪接触参数，以及相机支架。更早的 `trs_so_arm100` 是五自由度，并且已经写了 `impratio="10"` 与 `cone="elliptic"` 来改善防滑。本课可运行的场景是一个**最小**双指软接触夹爪（`grip_scene.xml`），好让 3×3 表在 CPU 上又快又不依赖网格；读懂这张表之后，再在本机打开 Menagerie 模型。

## 软接触是可调的弹簧

MuJoCo 不把接触当成无限硬。每个约束有一对 `solref`：`(timeconst, dampratio)`——时间常数越小接触越硬；`dampratio = 1` 是临界阻尼。`solimp` 决定阻抗如何随穿透加深。在 `<option>` 一层，`impratio` 提高摩擦相对法向的阻抗比（抓取防滑时很关键），`cone="elliptic"` 使用椭圆摩擦锥而不是金字塔近似。这些正是 Menagerie SO-101 README 强调的旋钮，也是你的抓取会敏感的旋钮。

```python
#!/usr/bin/env python3
"""grip_sweep.py: 3x3 sweep of MuJoCo timestep x solref timeconst on a soft-contact grip-and-lift.

A two-jaw slider gripper closes on a free cube (gravity off), then lifts with gravity on.
For each (timestep, solref timeconst) pair we report whether the cube was held and the
maximum contact penetration seen during the run. Fixed scene: grip_scene.xml next to this file.

Requires: mujoco (private venv + pip install mujoco; Menagerie models need >= 3.1.3). Standard library + mujoco.
usage: python3 grip_sweep.py
"""
from __future__ import annotations

import os
import sys

try:
    import mujoco
    import numpy as np
except ImportError as e:
    sys.stderr.write(
        "grip_sweep.py needs mujoco (and numpy). "
        "On Ubuntu 24.04 (PEP 668) use a private venv:\n"
        "  python3 -m venv ~/Hardware-learning/.venv && "
        "source ~/Hardware-learning/.venv/bin/activate && "
        "python3 -m pip install mujoco\n"
        f"Import error: {e}\n"
    )
    sys.exit(1)

HERE = os.path.dirname(os.path.abspath(__file__))
SCENE = os.path.join(HERE, "grip_scene.xml")

# Locked so expected/ stays stable. Soft + large dt → slip; stiff → hold. Penetration grows with timeconst.
TIMESTEPS = (0.002, 0.005, 0.010)
TIMECONSTS = (0.004, 0.020, 0.080)  # solref[0]; dampratio fixed at 1.0
CLOSE = 0.0245
FRICTION = 1.2
CUBE_MASS = 0.15  # overrides grip_scene.xml cube geom mass=0.08 at runtime
HOLD_Z = 0.08
LIFT_CTRL = 0.10


def run(timestep: float, timeconst: float) -> tuple[float, float, bool]:
    m = mujoco.MjModel.from_xml_path(SCENE)
    m.opt.timestep = timestep
    m.opt.impratio = 10.0
    m.opt.cone = 1  # elliptic
    for bi in range(m.nbody):
        name = mujoco.mj_id2name(m, mujoco.mjtObj.mjOBJ_BODY, bi) or ""
        if name == "cube":
            m.body_mass[bi] = CUBE_MASS
            m.body_inertia[bi] = np.array([CUBE_MASS * 8e-5] * 3)
    for gi in range(m.ngeom):
        name = mujoco.mj_id2name(m, mujoco.mjtObj.mjOBJ_GEOM, gi) or ""
        if name in ("left_g", "right_g", "cube_geom"):
            m.geom_solref[gi] = (timeconst, 1.0)
            m.geom_friction[gi] = [FRICTION, 0.005, 0.0001]
    d = mujoco.MjData(m)

    # Close without gravity so the cube stays between the jaws.
    m.opt.gravity[:] = (0.0, 0.0, 0.0)
    d.ctrl[:] = (0.0, CLOSE, -CLOSE)
    for _ in range(max(1, int(0.8 / timestep))):
        mujoco.mj_step(m, d)
    max_pen = 0.0
    for i in range(d.ncon):
        if d.contact[i].dist < 0:
            max_pen = max(max_pen, -d.contact[i].dist)

    # Gravity on, lift the palm.
    m.opt.gravity[:] = (0.0, 0.0, -9.81)
    d.ctrl[:] = (LIFT_CTRL, CLOSE, -CLOSE)
    for _ in range(max(1, int(1.5 / timestep))):
        mujoco.mj_step(m, d)
        for i in range(d.ncon):
            if d.contact[i].dist < 0:
                max_pen = max(max_pen, -d.contact[i].dist)

    cube_z = float(d.qpos[5])  # freejoint xyz after lift/left/right
    held = cube_z > HOLD_Z
    return cube_z, max_pen, held


def main() -> None:
    print(f"scene={os.path.basename(SCENE)}")
    print(f"friction={FRICTION}  cube_mass={CUBE_MASS} kg  close={CLOSE} m")
    print("dt_s   timeconst  cube_z_m  pen_mm  result")
    holds = 0
    for dt in TIMESTEPS:
        for tc in TIMECONSTS:
            z, pen, held = run(dt, tc)
            holds += int(held)
            print(
                f"{dt:5.3f}  {tc:8.3f}  {z:8.4f}  {pen * 1e3:6.2f}  "
                f"{'HOLD' if held else 'SLIP'}"
            )
    print(f"held {holds}/9  (soft + large dt slips; penetration grows with timeconst)")


if __name__ == "__main__":
    main()
```

## 半隐式与对速度隐式的积分器，对照显式 Euler

MuJoCo 用几种格式积分连续时间的正向动力学。**显式 Euler** 有说明但**没有实现**，因为**半隐式 Euler**（先更新速度，再用*新*速度更新位置）严格更好，也是物理仿真里的常规做法。**对速度隐式** / `implicitfast` 用只对速度的一阶展开来逼近 t+h 处的加速度，针对的是速度相关的刚度。把部分陀螺项*显式*积分 “can lead to **energy gain** and divergence of fast-spinning free bodies”（MuJoCo Computation 文档，Integrators；正文已读；以原文为准）。因此 MuJoCo 的 `Euler` 就是半隐式方案（有关节阻尼时还带隐式阻尼）；`implicit` 与 `implicitfast` 只对速度相关力隐式。在无阻尼线性振子上，**显式 Euler 对任意 h > 0 都不稳定。** 放大特征值满足 |λ| = √(1 + (h ω0)²) > 1（手算 / 解析）。只有**全隐式后向 Euler**（速度与加速度都取在 t+h；Computation 式 4）是无条件稳定的，并且会引入数值阻尼（手算 / 解析）。那并不是 MuJoCo 的格式：`implicit` 与 `implicitfast` 仍然只对速度相关力隐式，因此当 D = 0 时 M̂ = M，该更新与半隐式 Euler 相同（Computation，Integrators；正文已读；以原文为准）。同一页还写到，没有关节阻尼时，`Euler` 用的是半隐式更新（3）而不是（6）。玩具已经在 `C_damp = 0` 时把 B 与 C 打成 **tie: identical on every seed**。本节**不是** `grip_sweep` 的重写（时间步 × `solref` 的软接触举升），也**不是**把 Menagerie 的 `implicitfast` 那一行再抄一遍。

### 公平设置

`mujoco_integrator_energy_toy.py` 是**合成**的，只用标准库。**模型假设:** 两套共用对象，没有 MuJoCo 二进制，也没有四元数更新。带轻微线性阻尼的单摆（重力依赖角度，阻尼依赖速度），以及惯性为 diag(1, 2, 3)、偏离中轴翻滚的无力矩刚体。转子的能量是 1/2 w·I w，只依赖角速度，所以在那里 A 与 B 的速度更新相同（文档里的差别在位置那一步）。臂 A 是显式 Euler，B 是半隐式，C 是对速度隐式（`Mhat = M + h D`），C_bad 把 `D` 的符号翻转（不一致的线性化，保留）。**共用随机数:** 五个种子；每个种子的初值抖动只抽一次，所有臂共用。指标是相对起点的能量漂移 dE、绝对漂移、发散标志、相对极小步参考的轨迹误差，以及 |E − Eref|。首次运行显示在有阻尼时从 E0 算的 |dE| 会误导，因此加印了 |E − Eref| 作为伴随指标（事后披露）。真实 MuJoCo / Menagerie：本页未运行。

**稳定步长判据。** 对无阻尼线性振子 x'' + ω0² x = 0，半隐式 Euler 的放大矩阵迹是 2 − h² ω0²，当且仅当 h < h* = 2/ω0 时特征值留在单位圆上。恰在 h* 时迹为 −2，特征值 −1 是重根且为 Jordan 块，解线性增长（手算 / 解析；玩具未在 h = h* 处打印运行）。这里 ω0 = 3.1320919527 rad/s，h* = 0.6385508568 s。默认中等步长 0.1500 s 低于 h*（h/h* = 0.2349，`assert_below_hstar=1`）。玩具在 0.50×、0.98×、1.02× h* 以及默认步长的一半上打印谱半径和 40 s 能量有界运行：0.98× 有界、谱半径 1.000000，1.02× 增长（谱半径 1.490835，energy_bounded=0），半步有界。

**有阻尼判据，与本玩具的更新式一致。** 臂 B 在*旧*速度处计算线性阻尼（阻尼写在 a_t 里），再用新速度更新位置：v(t+h) = (1 − h c) v(t) − h ω0² x(t)，x(t+h) = x(t) + h v(t+h)，其中 c = 2 ζ ω0。对该映射，Jury 判据给出渐近稳定当且仅当 (ω0 h)² + 4 ζ ω0 h < 4（严格不等式；手算，再由玩具核对）。等号处有一个单重特征值 −1，周期为 2 的模态既不增长也不衰减；重根的 Jordan 块和线性增长只出现在 ζ = 0 的无阻尼情形（手算 / 解析）。本页 `C_damp = 0.05` 时，玩具打印 ζ = 0.0079818857，h*_damped = 0.6334743577 s。0.98× h*_damped 的谱半径是 0.984358（小于 1），1.02× 是 1.469468（大于 1），默认中等步长是 0.996243。上面的无阻尼 0.50× / 0.98× / 1.02× 检查保持不变。稳定的积分器并不自动等于稳定的*接触*模型——这与 `grip_sweep` 并不矛盾。

**在首次运行前写好的情形:** 轻度阻尼的摆在中等 h 下，B、C 的 |dE| 应**稳定地**小于 A；在很小的 h 下三者**可能接近打平**；在大步长 / 快速旋转下，A 应**稳定地输掉**（能量增长或发散）；翻转的隐式 `D` 可能**输掉**——该格保留。

```python
#!/usr/bin/env python3
"""Synthetic energy-drift comparison of explicit, semi-implicit and
implicit-in-velocity Euler, as a stdlib stand-in of the updates in the
MuJoCo Computation documentation (Integrators).

Stdlib only. Not mujoco.step, not a redo of grip_sweep (timestep x solref
on a soft-contact lift), and not a rewrite of the Menagerie implicitfast
one-liner. Real MuJoCo / Menagerie: 本页未运行.

Two shared plants, same h grid idea, initial energy printed:
  pendulum  simple pendulum with light linear damping (position-dependent
            gravity + velocity-dependent damping)
  spinner   torque-free rigid body with asymmetric inertia (gyroscopic
            bias; energy is 1/2 w·I w; no quaternion is used)

Arms:
  A_explicit     v_{t+h}=v_t+h a_t,      q_{t+h}=q_t+h v_t
  B_semimpl      v_{t+h}=v_t+h a_t,      q_{t+h}=q_t+h v_{t+h}
  C_implvel      implicit-in-velocity: v_{t+h}=v_t+h Mhat^{-1} M a_t,
                 Mhat=M+h D, D = -d(forces)/dv; then q uses v_{t+h}
  C_bad          same as C but with a sign-flipped D (inconsistent
                 linearization; declared fail arm; kept)

For the spinner, energy depends only on w, so A and B use the same
velocity update (the documented difference is the position step).

Regimes (declared before the first run):
  moderate_h     lightly damped pendulum, moderate h -> B,C consistent
                 lower |dE| than A
  tiny_h         very small h -> A,B,C may near-tie on |dE|
  large_h_spin   large h and/or a fast tumble -> A consistent lose
                 (energy gain or blow-up)
  inconsistent_D flipped gyro D -> C_bad may lose (energy gain); kept

Stable-step criterion (mandatory): for the undamped linear oscillator
x''+w0^2 x=0, semi-implicit Euler has amplification trace 2-h^2 w0^2
and stays on the unit circle iff h < h* = 2/w0. At h = h* the trace
is -2, so -1 is a repeated eigenvalue in a Jordan block and the
solution grows linearly (analytic; not printed). The default moderate
h is below h*. The toy prints a time-domain energy-bounded run and the
spectral radius at 0.50x, 0.98x and 1.02x h*.

Damped criterion (this toy's update: damping sits in a_t, i.e. the old
velocity): v_n=(1-h c)v - h w0^2 x, x_n=x+h v_n, c=2 zeta w0. Jury:
asymptotically stable iff (w0 h)^2 + 4 zeta w0 h < 4 (strict). The toy
prints zeta, h_star_damped and spectral radii at 0.98x / 1.02x, plus a
small (zeta, h) table. At equality a simple eigenvalue is -1 when
zeta>0 (analytic; the Jordan block is only the zeta=0 case).

Common random numbers: per-seed initial-condition jitter is drawn once
and reused by every arm. Seeds 0-4.
MuJoCo Computation documentation: body read; 以原文为准.

def mean_sd(xs: list[float]) -> tuple[float, float]:
    mu = sum(xs) / len(xs)
    if len(xs) == 1:
        return mu, 0.0
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x: float) -> str:
    if not math.isfinite(x):
        return "nan"
    if x != 0.0 and abs(x) < 5e-5:
        return f"{x:.3e}"
    return f"{x:.4f}"


def sign_note(diffs: list[float], m: float) -> str:
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def verdict(xa: list[float], xb: list[float]) -> str:
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{num(m)}+-{num(s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{num(m)}+-{num(s)}: {label}; {sign_note(diffs, m)}"


def pend_energy(th: float, w: float) -> float:
    return 0.5 * INERTIA_P * w * w + MASS * G * L * (1.0 - math.cos(th))


def pend_acc(th: float, w: float, damp: float = C_DAMP) -> float:
    return -(G / L) * math.sin(th) - (damp / INERTIA_P) * w


def tiny_step_ref(th: float, w: float, t_end: float, damp: float = C_DAMP) -> tuple[float, float, float]:
    # semi-implicit at a very small step, for trajectory and energy error
    h = 1e-4
    n = int(round(t_end / h))
    for _ in range(n):
        a = pend_acc(th, w, damp)
        w = w + h * a
        th = th + h * w
    return th, w, pend_energy(th, w)


def step_pend(th: float, w: float, h: float, arm: str, damp: float = C_DAMP) -> tuple[float, float]:
    a = pend_acc(th, w, damp)
    if arm == "A_explicit":
        th_n = th + h * w
        w_n = w + h * a
        return th_n, w_n
    if arm == "B_semimpl":
        w_n = w + h * a
        th_n = th + h * w_n
        return th_n, w_n
    # implicit-in-velocity: D = damp, M = INERTIA_P
    # a = M^{-1}(tau_grav - damp w), da/dw = -M^{-1} D with D = damp
    sign = -1.0 if arm == "C_bad" else 1.0
    D = sign * damp
    mhat = INERTIA_P + h * D
    w_n = w + h * (INERTIA_P / mhat) * a
    th_n = th + h * w_n
    return th_n, w_n


def run_pend(th0: float, w0: float, h: float, t_end: float, arm: str, damp: float = C_DAMP) -> dict:

def cross(a, b):
    return (a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0])


def imul(w):
    return (I_DIAG[0] * w[0], I_DIAG[1] * w[1], I_DIAG[2] * w[2])


def spin_energy(w) -> float:
    iw = imul(w)
    return 0.5 * (w[0] * iw[0] + w[1] * iw[1] + w[2] * iw[2])


def spin_acc(w):
    # I wdot = - w x (I w)
    tau = cross(w, imul(w))  # w x I w
    return (-tau[0] / I_DIAG[0], -tau[1] / I_DIAG[1], -tau[2] / I_DIAG[2])


def add(a, b, s=1.0):
    return (a[0] + s * b[0], a[1] + s * b[1], a[2] + s * b[2])


def D_gyro(w):
    # derivative of (w x I w) applied to basis; D = d(w x I w)/dw
    # J(u) = u x (I w) + w x (I u)
    rows = []
    for k in range(3):
        u = (1.0 if k == 0 else 0.0, 1.0 if k == 1 else 0.0, 1.0 if k == 2 else 0.0)
        ju = add(cross(u, imul(w)), cross(w, imul(u)))
        rows.append(ju)
    # rows are J(e_k); we want columns = J(e_k)
    return [[rows[k][i] for k in range(3)] for i in range(3)]


def solve3(A, b):
    # Gaussian elimination with partial pivot
    M = [A[i][:] + [b[i]] for i in range(3)]
    for k in range(3):
        piv = max(range(k, 3), key=lambda r: abs(M[r][k]))
        M[k], M[piv] = M[piv], M[k]
        if abs(M[k][k]) < 1e-18:
            return None
        inv = 1.0 / M[k][k]
        for j in range(k, 4):
            M[k][j] *= inv
        for i in range(3):
            if i == k:
                continue
            f = M[i][k]
            for j in range(k, 4):
                M[i][j] -= f * M[k][j]
    return (M[0][3], M[1][3], M[2][3])


def step_spin(w, h, arm: str):
    a = spin_acc(w)
    if arm in ("A_explicit", "B_semimpl"):
        return add(w, a, h)
    # implicit-in-velocity, eq. (6): (I + h D) (w' - w) = h I a
    # a = I^{-1}(- w x I w), D = d(w x I w)/dw  so da/dw = -I^{-1} D
    Dg = D_gyro(w)
    sign = -1.0 if arm == "C_bad" else 1.0
    Mhat = [[(I_DIAG[i] if i == j else 0.0) + h * sign * Dg[i][j] for j in range(3)] for i in range(3)]
    rhs = (h * I_DIAG[0] * a[0], h * I_DIAG[1] * a[1], h * I_DIAG[2] * a[2])
    dw = solve3(Mhat, rhs)
    if dw is None:
        return (float("inf"), float("inf"), float("inf"))
    return add(w, dw)

def amp_matrix(h: float, omega: float, c: float = 0.0) -> list[list[float]]:
    # semi-implicit; damping (if any) is evaluated at the old velocity:
    # v_n = (1 - h c) v - h omega^2 x; x_n = x + h v_n
    # c = 0 recovers the undamped map used by the h* tests above
    w2 = omega * omega
    return [[1.0 - h * h * w2, h * (1.0 - h * c)], [-h * w2, 1.0 - h * c]]


def spectral_radius(M: list[list[float]]) -> float:
    # 2x2 eigenvalues
    tr = M[0][0] + M[1][1]
    det = M[0][0] * M[1][1] - M[0][1] * M[1][0]
    disc = tr * tr - 4.0 * det
    if disc >= 0:
        s = math.sqrt(disc)
        evs = [abs(0.5 * (tr + s)), abs(0.5 * (tr - s))]
    else:
        r = math.sqrt(abs(det))  # for this symplectic map det=1, |λ|=1
        evs = [r, r]
    return max(evs)


def energy_bounded_run(h: float, t_end: float) -> int:
    # undamped linear oscillator, semi-implicit, unit energy
    # x'' + OMEGA0^2 x = 0, E = 0.5 (v^2 + w0^2 x^2)
    x, v = 1.0, 0.0
    e0 = 0.5 * OMEGA0 * OMEGA0 * x * x
    n = int(round(t_end / h))
    for _ in range(n):
        v = v - h * OMEGA0 * OMEGA0 * x
        x = x + h * v
        if not math.isfinite(x) or not math.isfinite(v):
            return 0
        e = 0.5 * (v * v + OMEGA0 * OMEGA0 * x * x)
        if e > BLOW * e0:
            return 0
    return 1
```
