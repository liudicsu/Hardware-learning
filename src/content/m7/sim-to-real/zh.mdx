```python
#!/usr/bin/env python3
"""Fit a 1st-order lag model to a synthetic joint step response (stdlib).

System identification (sysid) estimates real dynamics so the sim can be
calibrated before domain randomization. We synthesize:

    y[k+1] = y[k] + dt/tau * (u[k] - y[k]) + noise

then recover ``tau_hat`` by least-squares on the discrete update. Relative
error must stay under 5 % for the CI gate (proposal §2.4).
"""
from __future__ import annotations

import math
import random

SEED = 11
DT = 0.01
TRUE_TAU = 0.08  # seconds
NOISE_STD = 0.002
N_STEPS = 400
STEP_AMPLITUDE = 1.0


def simulate(tau: float, rng: random.Random) -> list[tuple[float, float, float]]:
    """Return list of (t, u, y)."""
    y = 0.0
    rows = []
    for k in range(N_STEPS):
        t = k * DT
        u = 0.0 if t < 0.2 else STEP_AMPLITUDE
        y = y + (DT / tau) * (u - y) + rng.gauss(0.0, NOISE_STD)
        rows.append((t, u, y))
    return rows


def fit_tau(rows: list[tuple[float, float, float]]) -> float:
    """LS fit of y_{k+1}-y_k = (dt/tau)(u_k - y_k) → beta=dt/tau."""
    # beta * x = dy  with x = (u - y)
    num = 0.0
    den = 0.0
    for i in range(len(rows) - 1):
        _, u, y = rows[i]
        _, _, y2 = rows[i + 1]
        x = u - y
        dy = y2 - y
        num += x * dy
        den += x * x
    beta = num / den
    if beta <= 1e-9:
        raise RuntimeError("degenerate fit")
    return DT / beta


def main() -> None:
    rng = random.Random(SEED)
    rows = simulate(TRUE_TAU, rng)
    tau_hat = fit_tau(rows)
    rel_err = abs(tau_hat - TRUE_TAU) / TRUE_TAU
    print("sysid_fit — 1st-order lag step response (seed=%d)" % SEED)
    print("dt=%.3f  true_tau=%.4f  noise_std=%.4f  n=%d" % (DT, TRUE_TAU, NOISE_STD, N_STEPS))
    print("tau_hat=%.6f  rel_err=%.4f (%.2f%%)" % (tau_hat, rel_err, 100 * rel_err))
    assert rel_err < 0.05, "sysid relative error %.4f exceeds 5%%" % rel_err
    print("ok: |tau_hat - true_tau|/true_tau < 5%")


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""Domain-randomization scan: friction / latency / sensor bias vs grasp success.

A short open-loop close-then-lift schedule that succeeds under nominal
parameters is replayed under friction×{1.0,0.7,0.4}, command delay {0,2,5}
steps, and vertical sensor bias {0, ±1.5 cm}. Per-cell rates use a Wilson
score interval (same idea as month-6 ``eval_report.py``).

Takeaway for lesson 7.4: latency and friction misspecification open the
reality gap even when the policy schedule is unchanged — motivate sysid + DR
before any SO-101 deployment.
"""
from __future__ import annotations

import math
import random
from dataclasses import dataclass

SEED = 13
N_TRIALS = 40
DT = 0.02
HORIZON = 60
LIFT_TARGET = 0.10  # metres
PALM_SPEED = 0.35   # m/s commanded when "lifting"


def wilson(successes: int, n: int, z: float = 1.96) -> tuple[float, float, float]:
    if n <= 0:
        return 0.0, 0.0, 0.0
    p = successes / n
    den = 1.0 + z * z / n
    centre = (p + z * z / (2 * n)) / den
    half = (z / den) * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n))
    return p, max(0.0, centre - half), min(1.0, centre + half)


@dataclass
class TrialResult:
    success: bool
    final_z: float
    slipped: bool


def simulate(friction_scale: float, delay: int, z_bias: float, rng: random.Random) -> TrialResult:
    """Kinematic lift with a single slip draw when the lift starts.

    Slip probability rises when friction is low **or** the close command is
    delayed relative to the lift (gripper still open when the palm rises).
    ``z_bias`` affects only the measured height used for the success check.
    """
    true_z = 0.0
    # delayed commands
    close_q = [False] * delay
    lift_q = [0.0] * delay
    closed = False
    lift_started = False
    slipped = False
    for t in range(HORIZON):
        intend_close = t >= 5
        intend_lift = PALM_SPEED if t >= 12 else 0.0
        close_q.append(intend_close)
        lift_q.append(intend_lift)
        do_close = close_q.pop(0)
        lift_cmd = lift_q.pop(0)
        if do_close:
            closed = True
        if lift_cmd > 0.0 and not lift_started:
            lift_started = True
            # one-shot slip draw at lift onset
            # base fail ~ 0.05 at fric=1, delay=0; grows with (1-fric) and delay
            fail_p = 0.05 + 0.55 * (1.0 - friction_scale) + 0.08 * delay
            # if close has not yet arrived when lift starts, almost certain slip
            if not closed:
                fail_p = max(fail_p, 0.9)
            fail_p = max(0.0, min(0.95, fail_p))
            if rng.random() < fail_p:
                slipped = True
        if lift_cmd > 0.0 and closed and not slipped:
            true_z += lift_cmd * DT
        elif slipped:
            true_z = max(0.0, true_z - 0.4 * DT)
    measured = true_z + z_bias
    success = (not slipped) and closed and measured >= LIFT_TARGET
    return TrialResult(success=success, final_z=true_z, slipped=slipped)


def scan_cell(friction_scale: float, delay: int, z_bias: float, seed: int):
    rng = random.Random(seed)
    ok = 0
    for _ in range(N_TRIALS):
        r = random.Random(rng.randint(0, 10**9))
        if simulate(friction_scale, delay, z_bias, r).success:
            ok += 1
    p, lo, hi = wilson(ok, N_TRIALS)
    return ok, p, lo, hi


def main() -> None:
    cells = [
        # fric, delay, bias
        (1.0, 0, 0.0),
        (0.7, 0, 0.0),
        (0.4, 0, 0.0),
        (1.0, 2, 0.0),
        (1.0, 5, 0.0),
        (1.0, 0, -0.015),
        (0.4, 5, -0.015),
    ]
    print("randomize_eval — DR scan friction×delay×z_bias (seed=%d, trials/cell=%d)"
          % (SEED, N_TRIALS))
    print("lift_target=%.2f m  palm_speed=%.2f m/s" % (LIFT_TARGET, PALM_SPEED))
    print("fric  delay  z_bias  ok/%d  rate   wilson95" % N_TRIALS)
    rates = {}
    for i, (fs, dly, zb) in enumerate(cells):
        ok, p, lo, hi = scan_cell(fs, dly, zb, seed=SEED + 19 * i)
        rates[(fs, dly, zb)] = p
        print(" %.1f  %5d  %+6.3f  %3d    %.3f  [%.3f, %.3f]" % (fs, dly, zb, ok, p, lo, hi))
    nom = rates[(1.0, 0, 0.0)]
    assert nom >= 0.85, "nominal success too low: %.3f" % nom
    assert rates[(1.0, 5, 0.0)] < nom - 0.2, "delay=5 should clearly hurt vs nominal"
    assert rates[(0.4, 0, 0.0)] < nom - 0.2, "low friction should clearly hurt"
    assert rates[(0.4, 5, -0.015)] <= 0.15, "worst cell should be near floor"
    print("ok: DR scan shows friction/delay/bias sensitivity (Wilson intervals)")


if __name__ == "__main__":
    main()
```

## 接触参数 DR 与仅延迟随机化

上面的扫描只是在**测**一套冻结的动作安排会在哪里失效。接下来的问题是：**调参时**就做随机化有没有用，代价是什么。`randomize_eval.py` 把摩擦和延迟放在同一张表里；这里把两者拆开。一个臂**只随机化延迟**，接触参数保持标称值。另一个臂在同样的延迟之上，再随机化**接触参数**：摩擦系数 µ 和物块质量。

玩具任务：一个有力限的推杆以指令速度 `v_cmd` 沿直线推物块。推杆按**标称**模型规划松手点：松手后物块应滑行 `v_cmd² / (2 µ₀ g)`，再加上平均延迟期间多走的距离。停止命令晚 L 秒生效，之后物块在库仑摩擦替身（减速度 µ·g）下滑行直到停住。成功的条件是停在 0.30 m 的 ±15 mm 以内，且 1.5 s 内静止。同样 5 N 的力限下，更重的物块加速更慢。

比较为什么公平：

- **只有一个控制器参数** `v_cmd`，两个臂都在同一个 41 点网格（0.20–1.00 m/s）上调。
- **预算相同**：两个臂对每个候选值都跑 200 个训练回合。
- **噪声种子流相同**：每个臂、每个候选值的第 k 个回合使用同一个延迟抽样、同一个 ±3 % 摩擦抖动和同一组均匀随机数。DR 臂把这些均匀随机数映射到 µ ∈ [0.25, 0.60]、m ∈ [0.30, 0.90] kg。
- **评分相同**：先比成功次数，平手时比平均周期时间（更短者胜）。
- **评估回合共享**（另一个种子），分别在标称 µ = 0.40 / m = 0.50 kg 以及偏移后的 µ 和质量下打分。

这里什么都不拟合：不重拟 τ，也不估计任何参数——那是上面的 `sysid_fit.py`，持续激励属于 5.1。这里做的是在一个分布上调控制器。摩擦只是一行库仑替身，**不是** MuJoCo 的软接触求解器（7.1）；真 MuJoCo 接触 DR 常扰动 geom 上的 `solref` / `solimp` / `friction`（见 [MuJoCo XML 参考](https://mujoco.readthedocs.io/en/stable/XMLreference.html)）；SO-101 上的真实推动本页未运行。

```python
#!/usr/bin/env python3
"""contact_dr_toy.py: contact-parameter domain randomisation vs latency-only (stdlib).

A force-limited pusher drives a block along a line at a commanded speed v_cmd
(the ONE controller parameter tuned here). The release mark is planned from
the NOMINAL model (slide v^2 / (2 mu0 g) plus the mean latency travel); the
stop command lands L seconds late. The block then slides with a Coulomb
friction proxy (deceleration mu * g) until it rests. Success: final position
within TOL of X_TARGET and at rest by T_MAX. A heavier block accelerates more
slowly under the same force limit; lower friction slides further.

Two arms tune v_cmd on the SAME budget: the same candidate grid, the same
number of training episodes and the SAME noise seed stream (episode k of every
arm and every candidate uses the same latency draw, friction jitter and
uniforms). Score = successes, ties broken by the shorter mean cycle time.
  latency_only  randomises latency only; mu and mass stay nominal
  contact_dr    randomises the same latency AND mu, mass (contact parameters)
Both are evaluated on shared eval episodes: nominal mu / mass (in-distribution
for latency_only), shifted mu, shifted mass, and a mu outside the DR range.
No universal "DR wins": here it pays in nominal cycle time.
Not system identification: nothing is fitted (7.4's sysid_fit owns tau; 5.1
owns PE). The friction model is a toy proxy, not the MuJoCo contact solver.
Real pushing on an SO-101 is not run on this page (本页未运行).
usage: python3 contact_dr_toy.py
"""
from __future__ import annotations

import math
import random

SEED = 74
G = 9.81
DT = 0.001
X_TARGET = 0.30     # m
TOL = 0.015         # m
T_MAX = 1.5         # s, cycle-time limit (push + slide)
K_V, F_MAX = 40.0, 5.0          # N per m/s, N (force-limited pusher)
MU0, M0 = 0.40, 0.50            # nominal friction coefficient, kg
MU_RANGE, M_RANGE = (0.25, 0.60), (0.30, 0.90)   # contact DR ranges
LAT_RANGE = (0.0, 0.03)         # s, randomised in both arms
JITTER = 0.03                   # +-3 % per-episode friction jitter (both arms)
GRID = [round(0.20 + 0.02 * i, 2) for i in range(41)]   # v_cmd candidates 0.20..1.00
N_TRAIN, N_EVAL = 200, 200


def release_mark(v_cmd):
    """Release planned from the NOMINAL model: slide v^2/(2 mu0 g) plus mean latency travel."""
    return X_TARGET - v_cmd ** 2 / (2 * MU0 * G) - v_cmd * sum(LAT_RANGE) / 2


def episode(v_cmd, mu, m, lat):
    """(final position, time to rest) for one push-release-slide episode."""
    x_rel = release_mark(v_cmd)
    x, v, t_rel = 0.0, 0.0, None
    t = 0.0
    while True:
        pushing = t_rel is None or t < t_rel
        if pushing:
            f = max(0.0, min(F_MAX, K_V * (v_cmd - v)))
            a = (f - mu * m * G) / m
            if v <= 0.0 and a < 0.0:
                a = 0.0
        else:
            a = -mu * G
        v = max(0.0, v + a * DT)
        x += v * DT
        t += DT
        if t_rel is None and x >= x_rel:
            t_rel = t + lat
        if t_rel is not None and t >= t_rel and v <= 0.0:
            return x, t
        if t > 10.0:
            return x, t


def draws(seed, n):
    """Shared noise stream: (latency, jitter, u_mu, u_m) per episode."""
    rng = random.Random(seed)
    return [(rng.uniform(*LAT_RANGE), rng.uniform(-JITTER, JITTER), rng.random(), rng.random())
            for _ in range(n)]


def params(arm, d, mu=None, m=None):
    lat, jit, u_mu, u_m = d
    if arm == "contact_dr":
        mu = MU_RANGE[0] + u_mu * (MU_RANGE[1] - MU_RANGE[0])
        m = M_RANGE[0] + u_m * (M_RANGE[1] - M_RANGE[0])
    return mu * (1.0 + jit), m, lat


def score(v_cmd, eps):
    """(successes, mean |err| m, mean cycle time s); success = within TOL and by T_MAX."""
    runs = [episode(v_cmd, mu, m, lat) for mu, m, lat in eps]
    errs = [abs(x - X_TARGET) for x, _ in runs]
    ok = sum(e <= TOL and t <= T_MAX for e, (_, t) in zip(errs, runs))
    return ok, sum(errs) / len(errs), sum(t for _, t in runs) / len(runs)


def tune(arm, train):
    eps = [params(arm, d, MU0, M0) for d in train]
    best = None
    for v in GRID:
        ok, mae, cyc = score(v, eps)
        if best is None or (ok, -cyc) > (best[1], -best[3]):   # success first, then faster cycle
            best = (v, ok, mae, cyc)
    return best


def main() -> None:
    train = draws(SEED, N_TRAIN)
    evals = draws(SEED + 1, N_EVAL)
    print(f"seed={SEED} target={X_TARGET} m tol=+-{TOL} m t_max={T_MAX} s "
          f"latency~U{LAT_RANGE} s jitter=+-{JITTER:.0%} F_max={F_MAX} N")
    print(f"budget per arm: {len(GRID)} v_cmd candidates x {N_TRAIN} train episodes, same seed stream")
    print(f"nominal mu={MU0} m={M0} kg; contact DR mu~U{MU_RANGE} m~U{M_RANGE} kg")
    tuned = {}
    for arm in ("latency_only", "contact_dr"):
        v, ok, mae, cyc = tune(arm, train)
        tuned[arm] = v
        print(f"tuned {arm:<12} v_cmd={v:.2f} m/s release_mark={release_mark(v):.3f} m  "
              f"train success={ok}/{N_TRAIN} mean|err|={mae * 1000:.1f} mm cycle={cyc:.2f} s")
    conds = (("nominal", MU0, M0), ("mu_low_0.28", 0.28, M0), ("mu_high_0.55", 0.55, M0),
             ("mass_0.85", MU0, 0.85), ("mu_out_0.15", 0.15, M0))
    print(f"{'eval (shared episodes)':<22} {'latency_only':>23} {'contact_dr':>23}")
    res = {}
    for name, mu, m in conds:
        cells = []
        eps = [params("eval", d, mu, m) for d in evals]
        for arm in ("latency_only", "contact_dr"):
            ok, mae, cyc = score(tuned[arm], eps)
            res[(name, arm)] = (ok, cyc)
            cells.append(f"{ok:3d}/{N_EVAL} {mae * 1000:5.1f}mm {cyc:4.2f}s")
        print(f"{name:<22} {cells[0]:>23} {cells[1]:>23}")
    lo, dr = "latency_only", "contact_dr"
    assert res[("nominal", lo)][0] >= res[("nominal", dr)][0], "latency_only should not lose in-distribution"
    assert res[("nominal", lo)][1] < res[("nominal", dr)][1], "DR pays cycle time at nominal"
    assert res[("mu_high_0.55", dr)][0] > res[("mu_high_0.55", lo)][0], "DR should help at shifted mu"
    assert res[("mass_0.85", dr)][0] > res[("mass_0.85", lo)][0], "DR should help at shifted mass"
    assert res[("mu_out_0.15", dr)][0] <= N_EVAL // 20, "outside the DR range DR is not a cure"
    print("ok: same budget / seed stream; DR trades nominal cycle time for shifted mu/mass; "
          "mu outside the DR range still fails; toy friction != MuJoCo; SO-101 本页未运行")


if __name__ == "__main__":
    main()
```

## 常见错误

- **只随机化外观。** Tobin 式视觉 DR 修不好错误的 τ；Peng 式动力学 DR 修不好错误的相机外参。
- **因为「DR 会兜住」而跳过 sysid。** 标称对象软了两倍，整笔 DR 预算都花在空洞区域。
- **不做急停演练。** 策略把爪子合在手指上不是「有趣的失败模式」——是实验室事故。
- **只拿偏移条件宣布 DR 获胜。** 标称那一行和代价也要报：玩具里 DR 在标称下打平（200/200），但每次推动要 1.08 s 而不是 0.72 s。
- **在随机化范围之外测试。** µ = 0.15 低于 DR 范围 [0.25, 0.60]，DR 只有 2/200。只有当更宽的范围对你的机械臂在物理上合理时，才去放宽范围。
- **把 DR 调参叫作系统辨识。** `contact_dr_toy.py` 不估计任何量，它只是在一个分布上挑一个控制器参数。系统辨识（上文）和持续激励（5.1）是另外的工作。
- **只报一个成功率、不报区间。** Wilson 列的存在，就是为了不把 21/40 和 37/40 当成同样精确。

## Tobin 式视觉域随机化

<Resource id="tobin-dr" />

本页开头引用了 Tobin 等人，《常见错误》也提醒 Tobin 式视觉 DR 修不好错误的 τ。这一节读论文正文，并补上缺少的 toy。arXiv 摘要页给出的标题是 **Domain Randomization for Transferring Deep Neural Networks from Simulation to the Real World**，作者六人：Josh Tobin、Rachel Fong、Alex Ray、Jonas Schneider、Wojciech Zaremba、Pieter Abbeel。PDF 正文写道（以论文为准）：域随机化**通过在仿真器里随机化渲染**，让在仿真图像上训练的模型迁移到真实图像；只要仿真里的变化足够多，真实世界在模型看来可能**只是又一种变化**。每个训练样本，作者随机化干扰物的数量和形状、每个物体的位置和**纹理**、桌面、地面、天空盒和机器人的纹理、相机的位置、朝向和视场、灯的数量及其位置、朝向和高光特性，以及加到图像上的噪声的类型和大小。纹理是**非写实**的：一个随机 RGB 值、两个随机 RGB 值之间的渐变，或两个随机 RGB 值组成的棋盘格。检测器**拿不到**目标物体的**颜色**，只知道它的大小和形状；他们用的 MuJoCo 渲染器本来就不追求照片级真实。

他们报告的结果（以论文为准，本页**不**复现）：只用仿真图像训练的真实世界物体检测器平均精度约 1.5 cm，对干扰物和部分遮挡稳健；Fetch 机器人用它在 40 次试验中抓起目标 38 次；消融实验中，**纹理少于 1,000 种时性能明显下降**，而加噪声对精度影响可以忽略。下面的 toy 是 **k 近邻替身**，不是他们的 CNN，它的像素误差也不是厘米。

这一节**不是**上面的接触参数 DR toy（那里为推动控制器随机化 µ 和质量），**不是**滞后 sysid 拟合，也**不是** Peng 式动力学随机化。它**只**随机化外观，然后展示外观买不到什么。

**手算例题，已由 toy 打印。** 物体在传送带上以 30.0 px/s 运动，夹爪在拍照后 0.10 s 到达，这段时间里物体移动 3.0000 px。控制器若沿用仿真里的 τ = 0.05 s，会瞄短 30.0 × (0.10 − 0.05) = 1.5000 px，**即使检测器完美**也不低于 1.0 px 的容差。任何纹理都改变不了这个数。

**这里没有积分步长。** 这个 toy 只计算闭式位置；没有微分方程，也没有步长判据。

### 公平设置

`tobin_visual_domain_randomization_toy.py` 是**合成**的，只用标准库。**模型假设:** 一条 24 像素的灰度相机行上，背景前有一个宽 4.0 px、位置连续的物体；每个表面有平涂、渐变或棋盘格纹理，光照增益乘以整行，再加高斯像素噪声。检测器是在 600 条训练行上、对原始像素做 3 近邻回归。所有尺寸、颜色、增益、传送带速度和两个 τ 值都是**声明的玩具参数**。

| 臂 | 训练时的外观 | 抓取控制器用的 τ |
|---|---|---|
| A_canonical | 一种固定纹理：暗色平涂背景上的亮色平涂物体，增益 1 | TAU_REAL（知道动力学） |
| B_random | 从调色板取 200 种随机纹理，增益在 0.6–1.4 随机（Tobin 式） | TAU_REAL（知道动力学） |
| C_rand_tau | B 的检测器，原样不动 | TAU_SIM，错误的 τ |

**共用随机数:** 每个种子只抽一次训练位置和像素噪声，A 与 B 共享，所以两者**只**差纹理和增益；每个（情形, 种子）只抽一次测试集（位置、纹理、增益、噪声），所有臂共享。C 原样复用 B 的检测器。B **从未见过**留出的测试纹理：打印的偷看检查确认，在任何种子上，留出纹理和固定纹理都不在 B 的调色板里。五个种子，每个 40 条测试行。指标是 `loc_err`（px）、`grasp_err`（px）和成功比例；只有 `grasp_err` **严格**小于 TOL 时才算抓取成功。

| 情形 | 测试外观与运动 | 预先声明 |
|---|---|---|
| heldout | 四种留出纹理，物体静止 | B 的迁移**稳定地**好于 A；物体不动，所以 C 等于 B |
| canonical | A 自己的训练纹理，静止 | A 与 B 可能**接近打平**，或 B 可能**输**（随机化过度）；**保留** |
| moving | 留出纹理，30.0 px/s | 即使有视觉 DR，C 相对 B 也**稳定地**输；**保留** |

```python
#!/usr/bin/env python3
"""Tobin-style visual domain randomisation: canonical texture vs random textures vs random textures + wrong tau.

Synthetic 1-D "camera row" and a k-nearest-neighbour detector, stdlib only.
Not a deep network, not MuJoCo rendering, not a real camera and not the
contact-DR or lag-sysid toys on this page.  GPU training / real camera /
robot: 本页未运行.  Tobin et al. (arXiv 1703.06907) statements and numbers
(for example their 1.5 cm) 以论文为准; nothing here reproduces them.

What the paper body says (and what this toy keeps): train a detector only on
simulated images whose rendering is randomised (non-realistic random
textures: a random RGB value, a gradient between two random values or a
checker between two random values; lights; noise), so that with enough
variability the real world may look like just another variation.  The
detector never sees the colour of the object, only its size and shape.  The
paper's ablation reports that fewer unique textures hurt transfer.

Toy: a W-pixel grey-scale row shows one object of width OBJ_W at a
continuous position x (anti-aliased coverage) on a background.  Each surface
has a texture: flat, gradient or checker between grey values.  A lighting
gain multiplies the row and Gaussian pixel noise is added.  The detector is
k-NN regression (K_NN neighbours, squared pixel distance) from the row to x.
Arms:
  A_canonical : trained on ONE canonical texture (bright flat object on a
                dark flat background, gain 1)
  B_random    : trained on PALETTE random textures (Tobin-style) and random
                gains; the held-out test textures are never in the palette
  C_rand_tau  : B's detector unchanged, but the grasp controller uses a wrong
                latency tau (TAU_SIM) instead of the real TAU_REAL
Grasp proxy: the object moves at U px/s on a conveyor; the gripper arrives
TAU_REAL after the frame and aims at x_hat + U * tau_model.  grasp_err =
|aim - true position at arrival|; success iff grasp_err < TOL (strict).
A and B use tau_model = TAU_REAL (dynamics-aware); C uses TAU_SIM.

Common random numbers: for each seed one training set of positions and
pixel noise is drawn once and shared by A and B (they differ only in
textures and gains); for each (regime, seed) one test set (positions,
textures, gains, noise) is shared by every arm.
No integration step: closed-form positions, no ODE, no time-step criterion.

Regimes and parameters were written before the first run:
  heldout   : held-out "real" textures, static object (U = 0)
              - B consistent better transfer than A; keep
              - C equals B (tau does not matter when nothing moves)
  canonical : test texture identical to A's training texture, U = 0
              - A and B may near-tie, or B may lose (over-randomisation); keep
  moving    : held-out textures, U = U_MOVE px/s
              - C consistent lose vs B even with visual DR (appearance does
                not fix a wrong tau); keep
Alternatives, also written before the first run, on the same seeds:
  PALETTE = 10 random textures instead of 200 (heldout);
  TOL = 2.0 px instead of 1.0 px (moving).
All sizes, colours, gains, U and tau values are declared toy parameters.
"""
from __future__ import annotations

import math
import random

SEEDS = (0, 1, 2, 3, 4)
NUM_ZERO = 1e-9
W = 24                  # pixels in the camera row (toy)
OBJ_W = 4.0             # object width, px (toy)
X_LO, X_HI = 2.0, 18.0  # object left edge range, px
N_TRAIN = 600
N_TEST = 40
K_NN = 3
PIX_SIGMA = 0.02        # pixel noise (toy)
PALETTE = 200           # unique random texturizations for B (toy)
GAIN_LO, GAIN_HI = 0.6, 1.4
U_MOVE = 30.0           # conveyor speed, px/s (toy)
TAU_REAL = 0.10         # s, real camera-to-gripper latency (toy)
TAU_SIM = 0.05          # s, the wrong latency C's controller uses (toy)
TOL = 1.0               # px, grasp succeeds iff grasp_err < TOL
ARMS = ("A_canonical", "B_random", "C_rand_tau")
CANON = (("flat", 0.80, 0.80), ("flat", 0.20, 0.20))   # (object, background)
# Held-out "real" textures: fixed, never drawn into B's palette.
REAL = (
    (("flat", 0.35, 0.35), ("checker", 0.60, 0.75)),
    (("gradient", 0.90, 0.70), ("flat", 0.50, 0.50)),
    (("checker", 0.10, 0.30), ("gradient", 0.30, 0.60)),
    (("flat", 0.60, 0.60), ("flat", 0.85, 0.85)),
)
REGIMES = {"heldout": ("real", 0.0), "canonical": ("canon", 0.0), "moving": ("real", U_MOVE)}
KINDS = ("flat", "gradient", "checker")


def texture_value(tex, i):
    kind, a, b = tex
    if kind == "flat":
        return a
    if kind == "gradient":
        return a + (b - a) * i / (W - 1)
    return a if i % 2 == 0 else b


def render(x, obj, bg, gain, noise):
    row = []
    for i in range(W):
        cover = max(0.0, min(i + 1.0, x + OBJ_W) - max(float(i), x))
        v = cover * texture_value(obj, i) + (1.0 - cover) * texture_value(bg, i)
        row.append(gain * v + noise[i])
    return row


def random_texture(rng):
    kind = KINDS[rng.randrange(3)]
    a = rng.random()
    b = a if kind == "flat" else rng.random()
    return (kind, a, b)


def palette(seed, size):
    rng = random.Random(f"tobin-palette|{seed}|{size}")
    return [(random_texture(rng), random_texture(rng)) for _ in range(size)]


def train_draws(seed):
    """Shared by A and B: positions, pixel noise, palette picks, gains."""
    rng = random.Random(f"tobin-train|{seed}")
    xs = [rng.uniform(X_LO, X_HI) for _ in range(N_TRAIN)]
    noise = [[rng.gauss(0.0, PIX_SIGMA) for _ in range(W)] for _ in range(N_TRAIN)]
    picks = [rng.random() for _ in range(N_TRAIN)]
    gains = [rng.uniform(GAIN_LO, GAIN_HI) for _ in range(N_TRAIN)]
    return xs, noise, picks, gains


def train_set(arm, seed, size=PALETTE):
    xs, noise, picks, gains = train_draws(seed)
    if arm == "A_canonical":
        return [(render(x, CANON[0], CANON[1], 1.0, n), x) for x, n in zip(xs, noise)]
    pal = palette(seed, size)
    return [(render(x, *pal[int(p * size)], g, n), x) for x, n, p, g in zip(xs, noise, picks, gains)]


def test_draws(regime, seed):
    rng = random.Random(f"tobin-test|{regime}|{seed}")
    which = REGIMES[regime][0]
    out = []
    for _ in range(N_TEST):
        x = rng.uniform(X_LO, X_HI)
        if which == "canon":
            obj, bg, gain = CANON[0], CANON[1], 1.0
        else:
            obj, bg = REAL[rng.randrange(len(REAL))]
            gain = rng.uniform(0.8, 1.2)
        noise = [rng.gauss(0.0, PIX_SIGMA) for _ in range(W)]
        out.append((render(x, obj, bg, gain, noise), x))
    return out


def knn(train, row):
    best = sorted((sum((a - b) ** 2 for a, b in zip(img, row)), x) for img, x in train)[:K_NN]
    return sum(x for _, x in best) / K_NN


def evaluate(arm, regime, seed, train, tol=TOL):
    u = REGIMES[regime][1]
    tau_model = TAU_SIM if arm == "C_rand_tau" else TAU_REAL
    loc, grasp, ok = [], [], 0
    for row, x in test_draws(regime, seed):
        xh = knn(train, row)
        aim = xh + u * tau_model
        err = abs(aim - (x + u * TAU_REAL))
        loc.append(abs(xh - x))
        grasp.append(err)
        ok += 1 if err < tol else 0
    return {"loc_err": sum(loc) / len(loc), "grasp_err": sum(grasp) / len(grasp), "success": ok / N_TEST}


def mean_sd(xs):
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    if x != 0.0 and abs(x) < 5e-4:
        return f"{x:.3e}"
    return f"{x:.4f}"


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def fmt_pair(m, s):
    return f"{num(m)}+-{num(s)}"


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{fmt_pair(m, s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{fmt_pair(m, s)}: {label}; {sign_note(diffs, m)}"


METRICS = ("loc_err", "grasp_err", "success")
_TRAIN = {}


def trained(arm, seed, size=PALETTE):
    key = ("A_canonical" if arm == "A_canonical" else "B_random", seed, size)
    if key not in _TRAIN:
        _TRAIN[key] = train_set(key[0], seed, size)
    return _TRAIN[key]


def collect(regime, size=PALETTE, tol=TOL):
    return {arm: [evaluate(arm, regime, s, trained(arm, s, size), tol) for s in SEEDS] for arm in ARMS}


def col(rows, key):
    return [r[key] for r in rows]


def report(regime):
    res = collect(regime)
    which, u = REGIMES[regime]
    print(f"{regime} (test textures={which}, U={u:.1f} px/s):")
    for arm in ARMS:
        parts = [f"{key} {fmt_pair(*mean_sd(col(res[arm], key)))}" for key in METRICS]
        print(f"  {arm} " + " ".join(parts))
    for key in METRICS:
        print(f"  paired B-A {key}: {verdict(col(res['B_random'], key), col(res['A_canonical'], key))}")
    for key in METRICS:
        print(f"  paired C-B {key}: {verdict(col(res['C_rand_tau'], key), col(res['B_random'], key))}")
    return res


def main():
    print("Synthetic 1-D camera-row toy with a k-NN detector; canonical vs random textures vs random textures + wrong tau; not a DNN")
    print(f"seeds={len(SEEDS)} W={W} px OBJ_W={OBJ_W:.1f} px N_TRAIN={N_TRAIN} N_TEST={N_TEST} K_NN={K_NN} "
          f"PIX_SIGMA={PIX_SIGMA:.2f} PALETTE={PALETTE} GAIN=[{GAIN_LO:.1f}, {GAIN_HI:.1f}] (declared toy parameters)")
    print(f"U_MOVE={U_MOVE:.1f} px/s TAU_REAL={TAU_REAL:.2f} s TAU_SIM={TAU_SIM:.2f} s TOL={TOL:.1f} px "
```
