```python
#!/usr/bin/env python3
"""Fit a 1st-order lag model to a synthetic joint step response (stdlib).

System identification (sysid) estimates real dynamics so the sim can be
calibrated before domain randomization. We synthesize:

    y[k+1] = y[k] + dt/tau * (u[k] - y[k]) + noise

then recover ``tau_hat`` by least-squares on the discrete update. Relative
error must stay under 5 % for the CI gate (proposal §2.4).
"""
from __future__ import annotations

import math
import random

SEED = 11
DT = 0.01
TRUE_TAU = 0.08  # seconds
NOISE_STD = 0.002
N_STEPS = 400
STEP_AMPLITUDE = 1.0


def simulate(tau: float, rng: random.Random) -> list[tuple[float, float, float]]:
    """Return list of (t, u, y)."""
    y = 0.0
    rows = []
    for k in range(N_STEPS):
        t = k * DT
        u = 0.0 if t < 0.2 else STEP_AMPLITUDE
        y = y + (DT / tau) * (u - y) + rng.gauss(0.0, NOISE_STD)
        rows.append((t, u, y))
    return rows


def fit_tau(rows: list[tuple[float, float, float]]) -> float:
    """LS fit of y_{k+1}-y_k = (dt/tau)(u_k - y_k) → beta=dt/tau."""
    # beta * x = dy  with x = (u - y)
    num = 0.0
    den = 0.0
    for i in range(len(rows) - 1):
        _, u, y = rows[i]
        _, _, y2 = rows[i + 1]
        x = u - y
        dy = y2 - y
        num += x * dy
        den += x * x
    beta = num / den
    if beta <= 1e-9:
        raise RuntimeError("degenerate fit")
    return DT / beta


def main() -> None:
    rng = random.Random(SEED)
    rows = simulate(TRUE_TAU, rng)
    tau_hat = fit_tau(rows)
    rel_err = abs(tau_hat - TRUE_TAU) / TRUE_TAU
    print("sysid_fit — 1st-order lag step response (seed=%d)" % SEED)
    print("dt=%.3f  true_tau=%.4f  noise_std=%.4f  n=%d" % (DT, TRUE_TAU, NOISE_STD, N_STEPS))
    print("tau_hat=%.6f  rel_err=%.4f (%.2f%%)" % (tau_hat, rel_err, 100 * rel_err))
    assert rel_err < 0.05, "sysid relative error %.4f exceeds 5%%" % rel_err
    print("ok: |tau_hat - true_tau|/true_tau < 5%")


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""Domain-randomization scan: friction / latency / sensor bias vs grasp success.

A short open-loop close-then-lift schedule that succeeds under nominal
parameters is replayed under friction×{1.0,0.7,0.4}, command delay {0,2,5}
steps, and vertical sensor bias {0, ±1.5 cm}. Per-cell rates use a Wilson
score interval (same idea as month-6 ``eval_report.py``).

Takeaway for lesson 7.4: latency and friction misspecification open the
reality gap even when the policy schedule is unchanged — motivate sysid + DR
before any SO-101 deployment.
"""
from __future__ import annotations

import math
import random
from dataclasses import dataclass

SEED = 13
N_TRIALS = 40
DT = 0.02
HORIZON = 60
LIFT_TARGET = 0.10  # metres
PALM_SPEED = 0.35   # m/s commanded when "lifting"


def wilson(successes: int, n: int, z: float = 1.96) -> tuple[float, float, float]:
    if n <= 0:
        return 0.0, 0.0, 0.0
    p = successes / n
    den = 1.0 + z * z / n
    centre = (p + z * z / (2 * n)) / den
    half = (z / den) * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n))
    return p, max(0.0, centre - half), min(1.0, centre + half)


@dataclass
class TrialResult:
    success: bool
    final_z: float
    slipped: bool


def simulate(friction_scale: float, delay: int, z_bias: float, rng: random.Random) -> TrialResult:
    """Kinematic lift with a single slip draw when the lift starts.

    Slip probability rises when friction is low **or** the close command is
    delayed relative to the lift (gripper still open when the palm rises).
    ``z_bias`` affects only the measured height used for the success check.
    """
    true_z = 0.0
    # delayed commands
    close_q = [False] * delay
    lift_q = [0.0] * delay
    closed = False
    lift_started = False
    slipped = False
    for t in range(HORIZON):
        intend_close = t >= 5
        intend_lift = PALM_SPEED if t >= 12 else 0.0
        close_q.append(intend_close)
        lift_q.append(intend_lift)
        do_close = close_q.pop(0)
        lift_cmd = lift_q.pop(0)
        if do_close:
            closed = True
        if lift_cmd > 0.0 and not lift_started:
            lift_started = True
            # one-shot slip draw at lift onset
            # base fail ~ 0.05 at fric=1, delay=0; grows with (1-fric) and delay
            fail_p = 0.05 + 0.55 * (1.0 - friction_scale) + 0.08 * delay
            # if close has not yet arrived when lift starts, almost certain slip
            if not closed:
                fail_p = max(fail_p, 0.9)
            fail_p = max(0.0, min(0.95, fail_p))
            if rng.random() < fail_p:
                slipped = True
        if lift_cmd > 0.0 and closed and not slipped:
            true_z += lift_cmd * DT
        elif slipped:
            true_z = max(0.0, true_z - 0.4 * DT)
    measured = true_z + z_bias
    success = (not slipped) and closed and measured >= LIFT_TARGET
    return TrialResult(success=success, final_z=true_z, slipped=slipped)


def scan_cell(friction_scale: float, delay: int, z_bias: float, seed: int):
    rng = random.Random(seed)
    ok = 0
    for _ in range(N_TRIALS):
        r = random.Random(rng.randint(0, 10**9))
        if simulate(friction_scale, delay, z_bias, r).success:
            ok += 1
    p, lo, hi = wilson(ok, N_TRIALS)
    return ok, p, lo, hi


def main() -> None:
    cells = [
        # fric, delay, bias
        (1.0, 0, 0.0),
        (0.7, 0, 0.0),
        (0.4, 0, 0.0),
        (1.0, 2, 0.0),
        (1.0, 5, 0.0),
        (1.0, 0, -0.015),
        (0.4, 5, -0.015),
    ]
    print("randomize_eval — DR scan friction×delay×z_bias (seed=%d, trials/cell=%d)"
          % (SEED, N_TRIALS))
    print("lift_target=%.2f m  palm_speed=%.2f m/s" % (LIFT_TARGET, PALM_SPEED))
    print("fric  delay  z_bias  ok/%d  rate   wilson95" % N_TRIALS)
    rates = {}
    for i, (fs, dly, zb) in enumerate(cells):
        ok, p, lo, hi = scan_cell(fs, dly, zb, seed=SEED + 19 * i)
        rates[(fs, dly, zb)] = p
        print(" %.1f  %5d  %+6.3f  %3d    %.3f  [%.3f, %.3f]" % (fs, dly, zb, ok, p, lo, hi))
    nom = rates[(1.0, 0, 0.0)]
    assert nom >= 0.85, "nominal success too low: %.3f" % nom
    assert rates[(1.0, 5, 0.0)] < nom - 0.2, "delay=5 should clearly hurt vs nominal"
    assert rates[(0.4, 0, 0.0)] < nom - 0.2, "low friction should clearly hurt"
    assert rates[(0.4, 5, -0.015)] <= 0.15, "worst cell should be near floor"
    print("ok: DR scan shows friction/delay/bias sensitivity (Wilson intervals)")


if __name__ == "__main__":
    main()
```

## 接触参数 DR 与仅延迟随机化

上面的扫描只是在**测**一套冻结的动作安排会在哪里失效。接下来的问题是：**调参时**就做随机化有没有用，代价是什么。`randomize_eval.py` 把摩擦和延迟放在同一张表里；这里把两者拆开。一个臂**只随机化延迟**，接触参数保持标称值。另一个臂在同样的延迟之上，再随机化**接触参数**：摩擦系数 µ 和物块质量。

玩具任务：一个有力限的推杆以指令速度 `v_cmd` 沿直线推物块。推杆按**标称**模型规划松手点：松手后物块应滑行 `v_cmd² / (2 µ₀ g)`，再加上平均延迟期间多走的距离。停止命令晚 L 秒生效，之后物块在库仑摩擦替身（减速度 µ·g）下滑行直到停住。成功的条件是停在 0.30 m 的 ±15 mm 以内，且 1.5 s 内静止。同样 5 N 的力限下，更重的物块加速更慢。

比较为什么公平：

- **只有一个控制器参数** `v_cmd`，两个臂都在同一个 41 点网格（0.20–1.00 m/s）上调。
- **预算相同**：两个臂对每个候选值都跑 200 个训练回合。
- **噪声种子流相同**：每个臂、每个候选值的第 k 个回合使用同一个延迟抽样、同一个 ±3 % 摩擦抖动和同一组均匀随机数。DR 臂把这些均匀随机数映射到 µ ∈ [0.25, 0.60]、m ∈ [0.30, 0.90] kg。
- **评分相同**：先比成功次数，平手时比平均周期时间（更短者胜）。
- **评估回合共享**（另一个种子），分别在标称 µ = 0.40 / m = 0.50 kg 以及偏移后的 µ 和质量下打分。

这里什么都不拟合：不重拟 τ，也不估计任何参数——那是上面的 `sysid_fit.py`，持续激励属于 5.1。这里做的是在一个分布上调控制器。摩擦只是一行库仑替身，**不是** MuJoCo 的软接触求解器（7.1）；真 MuJoCo 接触 DR 常扰动 geom 上的 `solref` / `solimp` / `friction`（见 [MuJoCo XML 参考](https://mujoco.readthedocs.io/en/stable/XMLreference.html)）；SO-101 上的真实推动本页未运行。

```python
#!/usr/bin/env python3
"""contact_dr_toy.py: contact-parameter domain randomisation vs latency-only (stdlib).

A force-limited pusher drives a block along a line at a commanded speed v_cmd
(the ONE controller parameter tuned here). The release mark is planned from
the NOMINAL model (slide v^2 / (2 mu0 g) plus the mean latency travel); the
stop command lands L seconds late. The block then slides with a Coulomb
friction proxy (deceleration mu * g) until it rests. Success: final position
within TOL of X_TARGET and at rest by T_MAX. A heavier block accelerates more
slowly under the same force limit; lower friction slides further.

Two arms tune v_cmd on the SAME budget: the same candidate grid, the same
number of training episodes and the SAME noise seed stream (episode k of every
arm and every candidate uses the same latency draw, friction jitter and
uniforms). Score = successes, ties broken by the shorter mean cycle time.
  latency_only  randomises latency only; mu and mass stay nominal
  contact_dr    randomises the same latency AND mu, mass (contact parameters)
Both are evaluated on shared eval episodes: nominal mu / mass (in-distribution
for latency_only), shifted mu, shifted mass, and a mu outside the DR range.
No universal "DR wins": here it pays in nominal cycle time.
Not system identification: nothing is fitted (7.4's sysid_fit owns tau; 5.1
owns PE). The friction model is a toy proxy, not the MuJoCo contact solver.
Real pushing on an SO-101 is not run on this page (本页未运行).
usage: python3 contact_dr_toy.py
"""
from __future__ import annotations

import math
import random

SEED = 74
G = 9.81
DT = 0.001
X_TARGET = 0.30     # m
TOL = 0.015         # m
T_MAX = 1.5         # s, cycle-time limit (push + slide)
K_V, F_MAX = 40.0, 5.0          # N per m/s, N (force-limited pusher)
MU0, M0 = 0.40, 0.50            # nominal friction coefficient, kg
MU_RANGE, M_RANGE = (0.25, 0.60), (0.30, 0.90)   # contact DR ranges
LAT_RANGE = (0.0, 0.03)         # s, randomised in both arms
JITTER = 0.03                   # +-3 % per-episode friction jitter (both arms)
GRID = [round(0.20 + 0.02 * i, 2) for i in range(41)]   # v_cmd candidates 0.20..1.00
N_TRAIN, N_EVAL = 200, 200


def release_mark(v_cmd):
    """Release planned from the NOMINAL model: slide v^2/(2 mu0 g) plus mean latency travel."""
    return X_TARGET - v_cmd ** 2 / (2 * MU0 * G) - v_cmd * sum(LAT_RANGE) / 2


def episode(v_cmd, mu, m, lat):
    """(final position, time to rest) for one push-release-slide episode."""
    x_rel = release_mark(v_cmd)
    x, v, t_rel = 0.0, 0.0, None
    t = 0.0
    while True:
        pushing = t_rel is None or t < t_rel
        if pushing:
            f = max(0.0, min(F_MAX, K_V * (v_cmd - v)))
            a = (f - mu * m * G) / m
            if v <= 0.0 and a < 0.0:
                a = 0.0
        else:
            a = -mu * G
        v = max(0.0, v + a * DT)
        x += v * DT
        t += DT
        if t_rel is None and x >= x_rel:
            t_rel = t + lat
        if t_rel is not None and t >= t_rel and v <= 0.0:
            return x, t
        if t > 10.0:
            return x, t


def draws(seed, n):
    """Shared noise stream: (latency, jitter, u_mu, u_m) per episode."""
    rng = random.Random(seed)
    return [(rng.uniform(*LAT_RANGE), rng.uniform(-JITTER, JITTER), rng.random(), rng.random())
            for _ in range(n)]


def params(arm, d, mu=None, m=None):
    lat, jit, u_mu, u_m = d
    if arm == "contact_dr":
        mu = MU_RANGE[0] + u_mu * (MU_RANGE[1] - MU_RANGE[0])
        m = M_RANGE[0] + u_m * (M_RANGE[1] - M_RANGE[0])
    return mu * (1.0 + jit), m, lat


def score(v_cmd, eps):
    """(successes, mean |err| m, mean cycle time s); success = within TOL and by T_MAX."""
    runs = [episode(v_cmd, mu, m, lat) for mu, m, lat in eps]
    errs = [abs(x - X_TARGET) for x, _ in runs]
    ok = sum(e <= TOL and t <= T_MAX for e, (_, t) in zip(errs, runs))
    return ok, sum(errs) / len(errs), sum(t for _, t in runs) / len(runs)


def tune(arm, train):
    eps = [params(arm, d, MU0, M0) for d in train]
    best = None
    for v in GRID:
        ok, mae, cyc = score(v, eps)
        if best is None or (ok, -cyc) > (best[1], -best[3]):   # success first, then faster cycle
            best = (v, ok, mae, cyc)
    return best


def main() -> None:
    train = draws(SEED, N_TRAIN)
    evals = draws(SEED + 1, N_EVAL)
    print(f"seed={SEED} target={X_TARGET} m tol=+-{TOL} m t_max={T_MAX} s "
          f"latency~U{LAT_RANGE} s jitter=+-{JITTER:.0%} F_max={F_MAX} N")
    print(f"budget per arm: {len(GRID)} v_cmd candidates x {N_TRAIN} train episodes, same seed stream")
    print(f"nominal mu={MU0} m={M0} kg; contact DR mu~U{MU_RANGE} m~U{M_RANGE} kg")
    tuned = {}
    for arm in ("latency_only", "contact_dr"):
        v, ok, mae, cyc = tune(arm, train)
        tuned[arm] = v
        print(f"tuned {arm:<12} v_cmd={v:.2f} m/s release_mark={release_mark(v):.3f} m  "
              f"train success={ok}/{N_TRAIN} mean|err|={mae * 1000:.1f} mm cycle={cyc:.2f} s")
    conds = (("nominal", MU0, M0), ("mu_low_0.28", 0.28, M0), ("mu_high_0.55", 0.55, M0),
             ("mass_0.85", MU0, 0.85), ("mu_out_0.15", 0.15, M0))
    print(f"{'eval (shared episodes)':<22} {'latency_only':>23} {'contact_dr':>23}")
    res = {}
    for name, mu, m in conds:
        cells = []
        eps = [params("eval", d, mu, m) for d in evals]
        for arm in ("latency_only", "contact_dr"):
            ok, mae, cyc = score(tuned[arm], eps)
            res[(name, arm)] = (ok, cyc)
            cells.append(f"{ok:3d}/{N_EVAL} {mae * 1000:5.1f}mm {cyc:4.2f}s")
        print(f"{name:<22} {cells[0]:>23} {cells[1]:>23}")
    lo, dr = "latency_only", "contact_dr"
    assert res[("nominal", lo)][0] >= res[("nominal", dr)][0], "latency_only should not lose in-distribution"
    assert res[("nominal", lo)][1] < res[("nominal", dr)][1], "DR pays cycle time at nominal"
    assert res[("mu_high_0.55", dr)][0] > res[("mu_high_0.55", lo)][0], "DR should help at shifted mu"
    assert res[("mass_0.85", dr)][0] > res[("mass_0.85", lo)][0], "DR should help at shifted mass"
    assert res[("mu_out_0.15", dr)][0] <= N_EVAL // 20, "outside the DR range DR is not a cure"
    print("ok: same budget / seed stream; DR trades nominal cycle time for shifted mu/mass; "
          "mu outside the DR range still fails; toy friction != MuJoCo; SO-101 本页未运行")


if __name__ == "__main__":
    main()
```

## 检查清单：把 7.1 / 7.3 策略迁到真机

凡真机步骤一律标 **需硬件**。在 SO-101 / Feetech 栈上启用学习策略之前：

1. **零位 / 回零。** 在机械硬限位或标记治具处记下编码器读数；写入文档；确认 URDF / MJCF 零位一致。
2. **限位。** 软件关节限位落在厂家范围内；力矩与速度上限低于堵转；急停物理可及且切断电机供电。

4. **传感路径。** 深度 / 本体感觉偏置：每次重装相机后重做小零位检查（衔接到第 7.6 课手眼）。

## 常见错误

- **只随机化外观。** Tobin 式视觉 DR 修不好错误的 τ；Peng 式动力学 DR 修不好错误的相机外参。
- **因为「DR 会兜住」而跳过 sysid。** 标称对象软了两倍，整笔 DR 预算都花在空洞区域。
- **不做急停演练。** 策略把爪子合在手指上不是「有趣的失败模式」——是实验室事故。
- **只拿偏移条件宣布 DR 获胜。** 标称那一行和代价也要报：玩具里 DR 在标称下打平（200/200），但每次推动要 1.08 s 而不是 0.72 s。
- **在随机化范围之外测试。** µ = 0.15 低于 DR 范围 [0.25, 0.60]，DR 只有 2/200。只有当更宽的范围对你的机械臂在物理上合理时，才去放宽范围。
- **把 DR 调参叫作系统辨识。** `contact_dr_toy.py` 不估计任何量，它只是在一个分布上挑一个控制器参数。系统辨识（上文）和持续激励（5.1）是另外的工作。
- **只报一个成功率、不报区间。** Wilson 列的存在，就是为了不把 21/40 和 37/40 当成同样精确。
