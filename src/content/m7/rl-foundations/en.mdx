Month 6 taught imitation learning: a table of demonstrations, a policy that copies actions, and the failure when the robot leaves the data. <Term id="mdp">Reinforcement learning</Term> drops the demonstrations. The environment returns a scalar <Term id="reward">reward</Term> each step; the learner maximises the discounted <Term id="return-discount">return</Term>. This lesson names the MDP pieces, contrasts them with Month 6, trains a clipped-ratio (<Term id="ppo">PPO</Term>-style) <Term id="policy-rl">policy</Term> on a 1-D reach toy across three seeds, and points at Stable-Baselines3 for the real library path.

## MDP in one page

- **State** `s`: what the agent is allowed to see (here: a scalar position in [−1, 1]).
- **Action** `a`: what it may do (here: a step in [−0.2, 0.2]).
- **Reward** `r`: a scalar each step (here: `r = −|x − 0.8|`).
- **Return** `G_t = r_t + γ·r_(t+1) + γ²·r_(t+2) + …`. With `γ = 0.99` and horizon 20, later rewards still count.
- **Policy** `π(a | s)`: the rule that picks actions. In Month 6 it was fitted to demo actions; here it is updated from returns.

Imitation learning needs a demonstrator and fails where the demos never went. RL needs a reward and fails when that reward is the wrong objective — **reward hacking**. A distance-only reward on a real arm can make the gripper skim the target to shave the last millimetres of error instead of settling into a stable grasp.

## A 20-step 1-D reach

The toy: start left of the origin, goal at +0.8, twenty steps, reward `−|x − goal|` each step. The policy is a linear Gaussian: mean `= tanh(w x + b) · 0.2`. Each iteration collects rollouts, normalises advantages, and takes several epochs of **clipped probability-ratio** updates — the PPO idea from Schulman et al., 2017 (arXiv 1707.06347) — without PyTorch, so the table below is standard-library reproducible.

```python
#!/usr/bin/env python3
"""ppo_reach.py: 1-D reach MDP trained with a clipped-surrogate (PPO-style) policy, 3 seeds.

Toy environment: state is position in [-1, 1], action is a step in [-0.2, 0.2], reward is
−|x − goal| each step, episode length 20, goal at +0.8. The policy is a linear Gaussian:
mean = tanh(w · x + b) · 0.2. We take several epochs of clipped-ratio updates on each
batch of rollouts (the PPO idea from Schulman et al. 2017), without a neural net or
PyTorch, so the expected table is reproducible with the standard library only.

On a learner machine with GPU optional: Stable-Baselines3's PPO is meant to run primarily
on the CPU when not using a CNN; see the SB3 PPO docs. That path is not run here.
usage: python3 ppo_reach.py
"""
from __future__ import annotations

import math
import random

GOAL = 0.8
HORIZON = 20
ACTION_SCALE = 0.2
GAMMA = 0.99
CLIP = 0.2
LR = 0.12
EPOCHS = 6
ROLLOUTS = 48
ITERS = 80
SEEDS = (0, 1, 2)


def step_env(x: float, a: float) -> tuple[float, float, bool]:
    a = max(-ACTION_SCALE, min(ACTION_SCALE, a))
    nx = max(-1.0, min(1.0, x + a))
    r = -abs(nx - GOAL)
    done = False
    return nx, r, done


def discount(rewards: list[float], gamma: float = GAMMA) -> list[float]:
    g = 0.0
    out = [0.0] * len(rewards)
    for t in range(len(rewards) - 1, -1, -1):
        g = rewards[t] + gamma * g
        out[t] = g
    return out


class Policy:
    """Linear Gaussian: a ~ N(tanh(w x + b) * ACTION_SCALE, sigma)."""

    def __init__(self, rng: random.Random):
        self.w = rng.uniform(-0.3, 0.3)
        self.b = rng.uniform(-0.3, 0.3)
        self.log_std = -1.0  # sigma ≈ 0.37 * ACTION_SCALE after tanh scale

    def mean(self, x: float) -> float:
        return math.tanh(self.w * x + self.b) * ACTION_SCALE

    def std(self) -> float:
        return math.exp(self.log_std) * ACTION_SCALE

    def act(self, x: float, rng: random.Random) -> tuple[float, float]:
        mu = self.mean(x)
        sig = self.std()
        a = mu + sig * rng.gauss(0.0, 1.0)
        a = max(-ACTION_SCALE, min(ACTION_SCALE, a))
        # log prob under unclipped Gaussian (for ratio); clamp a for env only
        logp = -0.5 * ((a - mu) / sig) ** 2 - math.log(sig) - 0.5 * math.log(2 * math.pi)
        return a, logp

    def logp(self, x: float, a: float) -> float:
        mu = self.mean(x)
        sig = self.std()
        return -0.5 * ((a - mu) / sig) ** 2 - math.log(sig) - 0.5 * math.log(2 * math.pi)


def collect(policy: Policy, rng: random.Random, n: int = ROLLOUTS):
    batch = []
    returns = []
    for _ in range(n):
        x = rng.uniform(-1.0, -0.2)
        traj = []
        rewards = []
        for _t in range(HORIZON):
            a, lp = policy.act(x, rng)
            nx, r, _ = step_env(x, a)
            traj.append((x, a, lp))
            rewards.append(r)
            x = nx
        Gs = discount(rewards)
        returns.append(sum(rewards))  # undiscounted episode return for the table
        for (x, a, lp), G in zip(traj, Gs):
            batch.append((x, a, lp, G))
    return batch, sum(returns) / len(returns)


def ppo_update(policy: Policy, batch):
    # Normalise advantages
    Gs = [b[3] for b in batch]
    mean_g = sum(Gs) / len(Gs)
    var = sum((g - mean_g) ** 2 for g in Gs) / len(Gs)
    std = math.sqrt(var) + 1e-8
    for _ in range(EPOCHS):
        for x, a, old_lp, G in batch:
            adv = (G - mean_g) / std
            new_lp = policy.logp(x, a)
            ratio = math.exp(max(-20.0, min(20.0, new_lp - old_lp)))
            unclipped = ratio * adv
            clipped = max(1.0 - CLIP, min(1.0 + CLIP, ratio)) * adv
            # surrogate to maximise; gradient via finite difference on w, b
            # analytical d logp / d mu * d mu / d param
            mu = policy.mean(x)
            sig = policy.std()
            dlogp_dmu = (a - mu) / (sig * sig)
            # mu = tanh(z)*ACTION_SCALE, z = w x + b
            z = policy.w * x + policy.b
            dmu_dz = (1.0 - math.tanh(z) ** 2) * ACTION_SCALE
            # Use clipped objective: if ratio outside clip and adv has the wrong sign, skip
            if (adv > 0 and ratio > 1.0 + CLIP) or (adv < 0 and ratio < 1.0 - CLIP):
                continue
            # ascend: param += LR * adv * dlogp/dparam  (on-policy PG inside the clip)
            policy.w += LR * adv * dlogp_dmu * dmu_dz * x
            policy.b += LR * adv * dlogp_dmu * dmu_dz


def train(seed: int) -> tuple[float, float, float]:
    rng = random.Random(seed)
    policy = Policy(rng)
    # initial return
    _, r0 = collect(policy, random.Random(seed + 1000), n=16)
    for it in range(ITERS):
        batch, _ = collect(policy, rng)
        ppo_update(policy, batch)
    _, rT = collect(policy, random.Random(seed + 2000), n=32)
    # final mean distance to goal over 32 eval rollouts (greedy mean action)
    dists = []
    for k in range(32):
        erng = random.Random(seed + 3000 + k)
        x = erng.uniform(-1.0, -0.2)
        for _ in range(HORIZON):
            a = policy.mean(x)
            x, _, _ = step_env(x, a)
        dists.append(abs(x - GOAL))
    return r0, rT, sum(dists) / len(dists)


def main() -> None:
    print(f"1-D reach  goal={GOAL}  horizon={HORIZON}  gamma={GAMMA}  clip={CLIP}")
    print("seed  return_init  return_final  mean_|x-goal|_greedy")
    finals = []
    for s in SEEDS:
        r0, rT, d = train(s)
        finals.append(rT)
        print(f"{s:4d}  {r0:11.3f}  {rT:12.3f}  {d:20.4f}")
    print(f"mean final return over {len(SEEDS)} seeds: {sum(finals)/len(finals):.3f}")


if __name__ == "__main__":
    main()
```

## Common mistakes

- **Reporting only the best seed.** Seed 0 in the table finishes worse than seed 2; both are part of the result.
- **Confusing the BC loss with a reward.** Squared error to a demo action is supervised learning; a reward is a scalar the environment invents.
- **Assuming PPO wants a GPU.** For MLP policies SB3's own docs point at the CPU.
- **Designing a reward that only looks at distance.** That is the reward-hacking example above; add a term for grasp stability or for staying still at the goal once you move to a real arm.

## Residual-action transfer: an equal-data two-stage experiment

A policy can track in synthetic **Plant-A** yet fail to track after transfer to **Plant-B**, even when its task and observation interface stay the same. This section owns a small dynamics-mismatch experiment, not another PPO derivation, the 7.4 contact-parameter DR experiment or the 5.1 PE lesson. For floating-base contacts, LIPM and the humanoid context, follow [11.6 humanoid WBC](/en/month/month-11/humanoid-wbc-overview); the scalar plant below is **not** a walking robot.

### Check the ASAP narrative, including what is NOT deployed

[ASAP, original paper arXiv 2502.01143v3](https://arxiv.org/html/2502.01143v3), Figure 2 and sections III-B/C/D, describes: pretrain a tracking policy in simulation; collect real rollouts; fit a **delta-action model** so `f_sim(s, a + delta(s,a))` matches recorded real transitions; freeze that model and fine-tune the policy in the aligned simulator; finally deploy the fine-tuned policy **without the delta model**. The delta is not an extra “fix motor command” network shipped with the robot. Its sign is chosen to make the nominal simulator imitate the real response: if a real actuator is weaker, the simulator correction can reduce action strength. This differs from cancelling the real error with a live residual controller.

**A residual is not dynamics SysID.** An action-coordinate correction does not identify a unique mass, friction coefficient or actuator constant. Multiple physical causes can have the same action effect. Our scalar supervised ridge fit and finite policy search mirror only the workflow, not ASAP's neural delta policy/PPO training, retargeting or real-data performance. All paper numbers are **以论文为准**. IsaacGym/IsaacSim/Genesis training, real robots and G1 data collection: **本页未运行 (not run on this page)**; no weights or hardware are required.

### Shared data, distinct model assumptions

The lesson-defined plant is `qddot = g*u - d*v - c*v*abs(v) + b`, integrated at **0.04 s** with semi-implicit Euler. Units are q in m, v in m/s and u an acceleration-equivalent action (the bias is a synthetic torque-bias proxy, not a measured motor torque). Plant-A uses `(g,d,c,b)=(1,0.5,0,0)`. Pretraining searches 27 simple trackers on four synthetic reference trajectories, selecting `(kp,kd,ff,bias)=(12,6,1,0)`; this is a finite controller-policy search, not a claim of new RL training. Each seed draws hidden Plant-B gain, damping, nonlinear drag and bias. Only the collector/evaluator accesses B; fitting and tuning do not get its parameters.

The collector runs that same pretrained policy, stops after exactly **160** B transitions (first full 100-step calibration trajectory plus 60 of the next), and adds the same acceleration observation noise once. Every arm receives the **same immutable data tuple**: **160/160/160**, not different rollouts selected after seeing outcomes. Four tuning trajectories are reused in simulation, while five evaluation trajectories use the reserved `seed + 10000` namespace. No evaluation trajectory or result selects models, hyperparameters or policies.

| Arm | What is learned from those 160 rows | Fine-tuning simulator |
|---|---|---|
| SysID-retune | Explicit gain/damping/bias in a linear family; nonlinear drag is omitted | One fitted physical-parameter model |
| DR-widen | Same linear fit plus its residual RMS | Four lesson-defined wide parameter corners; bounds are not calibrated confidence intervals |
| residual-action | Ridge fit of delta action on `(u,v,v*abs(v),1)` using observed-minus-A acceleration | Plant-A with the frozen action correction added only inside simulation |

All arms search the same **81** tracker candidates and spend **4 × 100** simulator steps per candidate. DR assigns a different corner to each of the four trajectories, so the objective is an empirical four-scenario average, not worst-case robustness or full distribution integration. The model families have different capacity: equal B-data and simulator step counts do **not** make the experiments identical-capacity algorithm benchmarks. Here `g_A=1`, so the acceleration discrepancy has the same numerical scale as the delta action. That special conversion is not generally valid for arbitrary actuators.

At deployment, `deploy(policy, real_proxy, eval_refs)` has **no delta argument**: B receives only the selected tracker action, with the same saturation for every arm. This separation is the key two-stage lesson.

### Results, spread and a failure rather than a promised winner

Tracking MAE is mean absolute position error over every step and then averaged over the five evaluation trajectories. `fall_proxy_episodes/5` counts trajectories whose maximum absolute position error exceeds **0.35 m**; it is a toy error threshold, **not a humanoid fall classifier**. Seeds **720–724** vary the B plant and noise. The summary reports mean, **population** standard deviation, minimum and maximum of the five seed-level MAEs, not a confidence interval.

Measured mean ± spread: SysID-retune **0.02919 ± 0.00506 m**, DR-widen **0.02817 ± 0.00256 m**, residual-action **0.02706 ± 0.00323 m**. Some seeds select the same policy and tie; others differ. On seed **722**, SysID worsens from zero-shot **0.03413** to **0.03518 m**. All fall proxies are **0/5**, so that proxy does not distinguish these runs. The residual arm has the lowest mean on this fixed setup, but that does not establish a universal winner, a significance claim or a real-robot result. Do not delete ties or a regression just to tell a success story.

```python
#!/usr/bin/env python3
"""Synthetic two-stage tracker transfer, NOT humanoid PPO or ASAP reproduction.
Equal B-data: all arms see the SAME 160 noisy (u,v,acceleration) transitions.
Fit a delta ACTION in nominal A action coordinates; freeze it in SIM tuning;
all real-proxy B evaluations call the tuned policy alone, never the delta model.
"""
from __future__ import annotations
import itertools
import math
import random
import statistics

SEED = 720
SEEDS = tuple(range(SEED, SEED + 5))
DT = 0.04
BUDGET = 160
STEPS = 100
A = (1.0, 0.5, 0.0, 0.0)  # gain, damping, nonlinear drag, torque-bias proxy


def plant(params):
    g, d, c, b = params
    return lambda u, v: g * u - d * v - c * v * abs(v) + b


def reference(phase, omega, amp):
    return tuple((amp * math.sin(omega * i * DT + phase),
                  amp * omega * math.cos(omega * i * DT + phase),
                  -amp * omega ** 2 * math.sin(omega * i * DT + phase))
                 for i in range(STEPS))


def trajectories(seed, count):
    rng = random.Random(seed)
    return tuple(reference(rng.uniform(-1, 1), rng.uniform(0.6, 1.8),
                           rng.uniform(0.3, 0.7)) for _ in range(count))


def action(policy, q, v, ref):
    kp, kd, ff, bias = policy
    x, vel, acc = ref
    return max(-5.0, min(5.0, kp * (x - q) + kd * (vel - v) + ff * acc + bias))


def rollout(policy, model, refs):
    q, v = refs[0][0], refs[0][1]
    errors, transitions = [], []
    for ref in refs:
        u = action(policy, q, v, ref)
        acc = model(u, v)
        transitions.append((u, v, acc))
        errors.append(abs(q - ref[0]))
        v += DT * acc
        q += DT * v
    return sum(errors) / len(errors), max(errors), transitions


def objective(policy, models, refs):
    # Equal 4 simulation trajectories/candidate for all arms; no B evaluation.
    return sum(rollout(policy, model, trajectory)[0]
               for model, trajectory in zip(models, refs)) / len(refs)


def tune(initial, models, refs, pretrain=False):
    if pretrain:
        candidates = itertools.product((4.0, 8.0, 12.0), (2.0, 4.0, 6.0),
                                       (0.6, 1.0, 1.4), (0.0,))
    else:
        kp, kd, ff, _ = initial
        candidates = itertools.product((kp * 0.75, kp, kp * 1.25),
                                       (kd * 0.75, kd, kd * 1.25),
                                       (ff * 0.75, ff, ff * 1.25), (-0.3, 0.0, 0.3))
    return min(candidates, key=lambda p: objective(p, models, refs))


def fit(rows, targets):
    """Small ridge least squares via pivoted Gaussian elimination (stdlib)."""
    n = len(rows[0])
    mat = [[sum(x[i] * x[j] for x in rows) + (1e-5 if i == j else 0)
            for j in range(n)] + [sum(x[i] * y for x, y in zip(rows, targets))]
           for i in range(n)]
    for i in range(n):
        pivot = max(range(i, n), key=lambda k: abs(mat[k][i]))
        mat[i], mat[pivot] = mat[pivot], mat[i]
        scale = mat[i][i]
        assert abs(scale) > 1e-12
        mat[i] = [v / scale for v in mat[i]]
        for j in range(n):
            if j != i:
                scale = mat[j][i]
                mat[j] = [a - scale * b for a, b in zip(mat[j], mat[i])]
    return tuple(row[-1] for row in mat)


def adapt(data):
    # SysID's explicit linear physical family omits nonlinear drag on purpose.
    linear_rows = [(u, -v, 1.0) for u, v, _ in data]
    g, d, bias = fit(linear_rows, [acc for _, _, acc in data])
    sysid = plant((g, d, 0.0, bias))
    # DR uses exactly those rows, residual spread -> lesson-defined wide bounds.
    rms = math.sqrt(sum((acc - sysid(u, v)) ** 2 for u, v, acc in data) / len(data))
    width = max(0.08, 2 * rms)
    dr = tuple(plant((max(0.2, g + sign * width), max(0.0, d - sign * width),
                      0.0, bias + other * width)) for sign, other in
               ((-1, -1), (-1, 1), (1, -1), (1, 1)))
    # Since g_A=1, delta-action target = observed acceleration - a_A(u,v).
    # This is an action correction, NOT an identified mass/friction parameter.
    features = [(u, v, v * abs(v), 1.0) for u, v, _ in data]
    coeff = fit(features, [acc - plant(A)(u, v) for u, v, acc in data])
    def aligned_sim(u, v):
        delta_action = sum(c * f for c, f in zip(coeff, (u, v, v * abs(v), 1.0)))
        return plant(A)(u + delta_action, v)
    return {"SysID-retune": (sysid,) * 4, "DR-widen": dr,
            "residual-action": (aligned_sim,) * 4}


def deploy(policy, real_proxy, eval_refs):
    # The signature deliberately has NO learned delta argument.
    results = [rollout(policy, real_proxy, refs) for refs in eval_refs]
    mae = statistics.mean(r[0] for r in results)
    fall_proxy = sum(r[1] > 0.35 for r in results) / len(results)
    return mae, fall_proxy


def main():
    train_refs = trajectories(71, 4)
    base = tune(None, (plant(A),) * 4, train_refs, pretrain=True)
    assert len(fit([(1, 0), (0, 1)], [2, 3])) == 2
    print(f"pretrained_A_policy={base} dt_s={DT:.2f} seeds={len(SEEDS)}")
    print(f"PlantB_samples_per_arm={BUDGET}/{BUDGET}/{BUDGET}; same_data_object=yes")
    print("finetune_sim_steps_per_arm=81_candidates*4_trajectories*100_steps")
    print("seed arm zero_shot_MAE_m tuned_MAE_m fall_proxy_episodes/5")
    summary = {name: [] for name in ("SysID-retune", "DR-widen", "residual-action")}
    for seed in SEEDS:
        rng = random.Random(seed)
        real = plant((rng.uniform(0.65, 0.95), rng.uniform(0.3, 0.8),
                      rng.uniform(0.8, 2.5), rng.uniform(-0.4, 0.4)))
        raw = []
        for refs in train_refs:
            remaining = BUDGET - len(raw)
            if remaining <= 0:
                break
            raw.extend(rollout(base, real, refs[:remaining])[2])
        data = tuple((u, v, a + rng.gauss(0, 0.025)) for u, v, a in raw)
        assert len(data) == BUDGET
        # Reserved seed namespace: neither calibration nor tuning sees eval refs.
        eval_refs = trajectories(seed + 10000, 5)
        zero, _ = deploy(base, real, eval_refs)
        for name, sims in adapt(data).items():
            policy = tune(base, sims, train_refs)
            mae, fall = deploy(policy, real, eval_refs)
            summary[name].append(mae)
            print(f"{seed} {name} {zero:.5f} {mae:.5f} {round(fall * 5)}/5")
    print("arm MAE_mean_m population_sd_m min_m max_m (across_5_seeds)")
    for name, values in summary.items():
        print(f"{name} {statistics.mean(values):.5f} {statistics.pstdev(values):.5f} "
              f"{min(values):.5f} {max(values):.5f}")
    print("deploy=policy_only_without_delta_model; residual_is_not_dynamics_SysID")
    print("model_families_differ; no_universal_winner; fall_proxy_not_humanoid_fall; Isaac_real_robot_not_run")


if __name__ == "__main__":
    main()
```
