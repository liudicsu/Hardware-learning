第 6 个月讲的是模仿学习：一张示范表、一个抄动作的策略，以及机器人离开数据之后的失败。<Term id="mdp">强化学习</Term>去掉示范。环境每一步返回一个标量 <Term id="reward">奖励</Term>；学习器最大化折扣 <Term id="return-discount">回报</Term>。本课点名 MDP 的零件，对照第 6 个月，在 1-D 到达玩具上用三个种子训练裁剪比率（<Term id="ppo">PPO</Term> 思路）的 <Term id="policy-rl">策略</Term>，并指向 Stable-Baselines3 作为真实库路径。

## 一页纸说清 MDP

- **状态** `s`：智能体被允许看到的东西（这里：[−1, 1] 上的标量位置）。
- **动作** `a`：它可以做什么（这里：[−0.2, 0.2] 的一步）。
- **奖励** `r`：每一步一个标量（这里：`r = −|x − 0.8|`）。
- **回报** `G_t = r_t + γ·r_(t+1) + γ²·r_(t+2) + …`。`γ = 0.99`、视野 20 时，后面的奖励仍算进回报。
- **策略** `π(a | s)`：选动作的规则。第 6 个月它拟合示范动作；这里它由回报更新。

模仿学习需要示范者，并在示范没去过的地方失败。RL 需要奖励，并在奖励目标写错时失败——这就是 **reward hacking**。真机上如果奖励只看距离，夹爪可能去蹭目标、削掉最后几毫米误差，而不是稳住抓取。

## 20 步的 1-D 到达

玩具：从原点左侧出发，目标在 +0.8，二十步，每步奖励 `−|x − goal|`。策略是线性高斯：均值 `= tanh(w x + b) · 0.2`。每次迭代收集轨迹、标准化优势，再做若干轮**裁剪概率比率**更新——这就是 Schulman 等 2017（arXiv 1707.06347）的 PPO 思路——不用 PyTorch，好让下表可用标准库复现。

```python
#!/usr/bin/env python3
"""ppo_reach.py: 1-D reach MDP trained with a clipped-surrogate (PPO-style) policy, 3 seeds.

Toy environment: state is position in [-1, 1], action is a step in [-0.2, 0.2], reward is
−|x − goal| each step, episode length 20, goal at +0.8. The policy is a linear Gaussian:
mean = tanh(w · x + b) · 0.2. We take several epochs of clipped-ratio updates on each
batch of rollouts (the PPO idea from Schulman et al. 2017), without a neural net or
PyTorch, so the expected table is reproducible with the standard library only.

On a learner machine with GPU optional: Stable-Baselines3's PPO is meant to run primarily
on the CPU when not using a CNN; see the SB3 PPO docs. That path is not run here.
usage: python3 ppo_reach.py
"""
from __future__ import annotations

import math
import random

GOAL = 0.8
HORIZON = 20
ACTION_SCALE = 0.2
GAMMA = 0.99
CLIP = 0.2
LR = 0.12
EPOCHS = 6
ROLLOUTS = 48
ITERS = 80
SEEDS = (0, 1, 2)


def step_env(x: float, a: float) -> tuple[float, float, bool]:
    a = max(-ACTION_SCALE, min(ACTION_SCALE, a))
    nx = max(-1.0, min(1.0, x + a))
    r = -abs(nx - GOAL)
    done = False
    return nx, r, done


def discount(rewards: list[float], gamma: float = GAMMA) -> list[float]:
    g = 0.0
    out = [0.0] * len(rewards)
    for t in range(len(rewards) - 1, -1, -1):
        g = rewards[t] + gamma * g
        out[t] = g
    return out


class Policy:
    """Linear Gaussian: a ~ N(tanh(w x + b) * ACTION_SCALE, sigma)."""

    def __init__(self, rng: random.Random):
        self.w = rng.uniform(-0.3, 0.3)
        self.b = rng.uniform(-0.3, 0.3)
        self.log_std = -1.0  # sigma ≈ 0.37 * ACTION_SCALE after tanh scale

    def mean(self, x: float) -> float:
        return math.tanh(self.w * x + self.b) * ACTION_SCALE

    def std(self) -> float:
        return math.exp(self.log_std) * ACTION_SCALE

    def act(self, x: float, rng: random.Random) -> tuple[float, float]:
        mu = self.mean(x)
        sig = self.std()
        a = mu + sig * rng.gauss(0.0, 1.0)
        a = max(-ACTION_SCALE, min(ACTION_SCALE, a))
        # log prob under unclipped Gaussian (for ratio); clamp a for env only
        logp = -0.5 * ((a - mu) / sig) ** 2 - math.log(sig) - 0.5 * math.log(2 * math.pi)
        return a, logp

    def logp(self, x: float, a: float) -> float:
        mu = self.mean(x)
        sig = self.std()
        return -0.5 * ((a - mu) / sig) ** 2 - math.log(sig) - 0.5 * math.log(2 * math.pi)


def collect(policy: Policy, rng: random.Random, n: int = ROLLOUTS):
    batch = []
    returns = []
    for _ in range(n):
        x = rng.uniform(-1.0, -0.2)
        traj = []
        rewards = []
        for _t in range(HORIZON):
            a, lp = policy.act(x, rng)
            nx, r, _ = step_env(x, a)
            traj.append((x, a, lp))
            rewards.append(r)
            x = nx
        Gs = discount(rewards)
        returns.append(sum(rewards))  # undiscounted episode return for the table
        for (x, a, lp), G in zip(traj, Gs):
            batch.append((x, a, lp, G))
    return batch, sum(returns) / len(returns)


def ppo_update(policy: Policy, batch):
    # Normalise advantages
    Gs = [b[3] for b in batch]
    mean_g = sum(Gs) / len(Gs)
    var = sum((g - mean_g) ** 2 for g in Gs) / len(Gs)
    std = math.sqrt(var) + 1e-8
    for _ in range(EPOCHS):
        for x, a, old_lp, G in batch:
            adv = (G - mean_g) / std
            new_lp = policy.logp(x, a)
            ratio = math.exp(max(-20.0, min(20.0, new_lp - old_lp)))
            unclipped = ratio * adv
            clipped = max(1.0 - CLIP, min(1.0 + CLIP, ratio)) * adv
            # surrogate to maximise; gradient via finite difference on w, b
            # analytical d logp / d mu * d mu / d param
            mu = policy.mean(x)
            sig = policy.std()
            dlogp_dmu = (a - mu) / (sig * sig)
            # mu = tanh(z)*ACTION_SCALE, z = w x + b
            z = policy.w * x + policy.b
            dmu_dz = (1.0 - math.tanh(z) ** 2) * ACTION_SCALE
            # Use clipped objective: if ratio outside clip and adv has the wrong sign, skip
            if (adv > 0 and ratio > 1.0 + CLIP) or (adv < 0 and ratio < 1.0 - CLIP):
                continue
            # ascend: param += LR * adv * dlogp/dparam  (on-policy PG inside the clip)
            policy.w += LR * adv * dlogp_dmu * dmu_dz * x
            policy.b += LR * adv * dlogp_dmu * dmu_dz


def train(seed: int) -> tuple[float, float, float]:
    rng = random.Random(seed)
    policy = Policy(rng)
    # initial return
    _, r0 = collect(policy, random.Random(seed + 1000), n=16)
    for it in range(ITERS):
        batch, _ = collect(policy, rng)
        ppo_update(policy, batch)
    _, rT = collect(policy, random.Random(seed + 2000), n=32)
    # final mean distance to goal over 32 eval rollouts (greedy mean action)
    dists = []
    for k in range(32):
        erng = random.Random(seed + 3000 + k)
        x = erng.uniform(-1.0, -0.2)
        for _ in range(HORIZON):
            a = policy.mean(x)
            x, _, _ = step_env(x, a)
        dists.append(abs(x - GOAL))
    return r0, rT, sum(dists) / len(dists)


def main() -> None:
    print(f"1-D reach  goal={GOAL}  horizon={HORIZON}  gamma={GAMMA}  clip={CLIP}")
    print("seed  return_init  return_final  mean_|x-goal|_greedy")
    finals = []
    for s in SEEDS:
        r0, rT, d = train(s)
        finals.append(rT)
        print(f"{s:4d}  {r0:11.3f}  {rT:12.3f}  {d:20.4f}")
    print(f"mean final return over {len(SEEDS)} seeds: {sum(finals)/len(finals):.3f}")


if __name__ == "__main__":
    main()
```

## 常见错误

- **只报最好的种子。** 表里种子 0 的结局不如种子 2；两者都是结果的一部分。
- **把 BC 损失当成奖励。** 对示范动作的平方误差是监督学习；奖励是环境发明的标量。
- **以为 PPO 一定要 GPU。** 对 MLP 策略，SB3 自己的文档指向 CPU。
- **只按距离设计奖励。** 就是上面的 reward hacking；上真机时要加抓取稳定或到达后保持静止的项。

## 残差动作迁移：等数据的两阶段实验

策略在合成 **Plant-A** 能跟踪，换到 **Plant-B** 却可能失配，即使任务和观测接口没有改变。本节只负责小型动力学失配实验，不重讲 PPO，不重做 7.4 接触参数 DR 或 5.1 PE。浮动基接触、LIPM 与人形背景见 [11.6 人形 WBC](/zh/month/month-11/humanoid-wbc-overview)；下面的标量对象**不是**行走机器人。

### 核对 ASAP 叙事，包括部署时不带什么

[ASAP 原论文 arXiv 2502.01143v3](https://arxiv.org/html/2502.01143v3) 的 Figure 2 与 III-B/C/D 描述：仿真预训练跟踪策略；收集真机 rollout；拟合 **delta-action 模型**，使 `f_sim(s, a + delta(s,a))` 匹配记录的真实转移；冻结该模型，在对齐后的仿真中微调策略；最后部署微调策略，**不带 delta 模型**。delta 不是随机器人上线的额外“修电机指令”网络。它的方向是让名义仿真模仿真实响应：真实执行器较弱时，仿真修正可能减小动作强度。这不同于在真机上叠加在线残差控制器抵消误差。

**残差不是动力学 SysID。** 动作坐标中的修正不等于识别出唯一的质量、摩擦系数或执行器常数，不同物理原因可能产生相同动作效果。本页的标量监督岭回归与有限策略搜索只模仿流程，不复现 ASAP 的神经 delta 策略/PPO 训练、重定向或真机成绩。所有论文数字均**以论文为准**。IsaacGym/IsaacSim/Genesis 训练、真实机器人与 G1 数据采集：**本页未运行 (not run on this page)**；不需要权重或硬件。

### 共用数据，不同模型假设

课程定义对象为 `qddot = g*u - d*v - c*v*abs(v) + b`，以 **0.04 s** 半隐式 Euler 积分。q 单位 m，v 单位 m/s，u 为加速度等效动作（偏置只是合成转矩偏置代理，不是实测电机转矩）。Plant-A 使用 `(g,d,c,b)=(1,0.5,0,0)`。预训练在四条合成参考轨迹上搜索 27 个简单跟踪器，选出 `(kp,kd,ff,bias)=(12,6,1,0)`；这是有限控制策略搜索，不声称新增 RL 训练。每个种子抽取隐藏的 Plant-B 增益、阻尼、非线性阻力与偏置。只有采集器/评估器访问 B；拟合和调参看不到其参数。

采集器运行同一个预训练策略，准确收集 **160** 条 B 转移后停止（第一条校准轨迹完整 100 步，再取下一条 60 步），统一加入一次加速度观测噪声。每臂得到**同一个不可变数据元组**：**160/160/160**，不是看结果后挑不同 rollout。四条调参参考只在仿真中复用，五条评估参考使用保留的 `seed + 10000` 命名空间。模型、超参数、策略均不根据评估轨迹或结果选择。

| 臂 | 从这 160 条数据学什么 | 微调仿真 |
|---|---|---|
| SysID-retune | 在线性模型族中显式拟合增益/阻尼/偏置，省略非线性阻力 | 一个拟合物理参数模型 |
| DR-widen | 相同线性拟合及其残差 RMS | 四个课程定义的加宽参数角点，不是校准过的置信区间 |
| residual-action | 用观测加速度减 A 加速度，在 `(u,v,v*abs(v),1)` 上岭回归 delta 动作 | 只在仿真内部给 Plant-A 加冻结的动作修正 |

每臂搜索相同 **81** 个跟踪器候选，每个候选用 **4 × 100** 个仿真步。DR 给四条轨迹分配不同角点，因此目标是四场景经验平均，不是最坏情况鲁棒性或完整分布积分。模型族容量不同：相同 B 数据量与仿真步数**不等于**同容量算法基准。这里 `g_A=1`，所以加速度差和 delta 动作数值尺度相同；对任意执行器一般不成立。

部署函数 `deploy(policy, real_proxy, eval_refs)` **没有 delta 参数**：B 只接收选出的跟踪器动作，所有臂使用相同饱和限制。这种分离是两阶段素养的关键。

```python
#!/usr/bin/env python3
"""Synthetic two-stage tracker transfer, NOT humanoid PPO or ASAP reproduction.
Equal B-data: all arms see the SAME 160 noisy (u,v,acceleration) transitions.
Fit a delta ACTION in nominal A action coordinates; freeze it in SIM tuning;
all real-proxy B evaluations call the tuned policy alone, never the delta model.
"""
from __future__ import annotations
import itertools
import math
import random
import statistics

SEED = 720
SEEDS = tuple(range(SEED, SEED + 5))
DT = 0.04
BUDGET = 160
STEPS = 100
A = (1.0, 0.5, 0.0, 0.0)  # gain, damping, nonlinear drag, torque-bias proxy


def plant(params):
    g, d, c, b = params
    return lambda u, v: g * u - d * v - c * v * abs(v) + b


def reference(phase, omega, amp):
    return tuple((amp * math.sin(omega * i * DT + phase),
                  amp * omega * math.cos(omega * i * DT + phase),
                  -amp * omega ** 2 * math.sin(omega * i * DT + phase))
                 for i in range(STEPS))


def trajectories(seed, count):
    rng = random.Random(seed)
    return tuple(reference(rng.uniform(-1, 1), rng.uniform(0.6, 1.8),
                           rng.uniform(0.3, 0.7)) for _ in range(count))


def action(policy, q, v, ref):
    kp, kd, ff, bias = policy
    x, vel, acc = ref
    return max(-5.0, min(5.0, kp * (x - q) + kd * (vel - v) + ff * acc + bias))


def rollout(policy, model, refs):
    q, v = refs[0][0], refs[0][1]
    errors, transitions = [], []
    for ref in refs:
        u = action(policy, q, v, ref)
        acc = model(u, v)
        transitions.append((u, v, acc))
        errors.append(abs(q - ref[0]))
        v += DT * acc
        q += DT * v
    return sum(errors) / len(errors), max(errors), transitions


def objective(policy, models, refs):
    # Equal 4 simulation trajectories/candidate for all arms; no B evaluation.
    return sum(rollout(policy, model, trajectory)[0]
               for model, trajectory in zip(models, refs)) / len(refs)


def tune(initial, models, refs, pretrain=False):
    if pretrain:
        candidates = itertools.product((4.0, 8.0, 12.0), (2.0, 4.0, 6.0),
                                       (0.6, 1.0, 1.4), (0.0,))
    else:
        kp, kd, ff, _ = initial
        candidates = itertools.product((kp * 0.75, kp, kp * 1.25),
                                       (kd * 0.75, kd, kd * 1.25),
                                       (ff * 0.75, ff, ff * 1.25), (-0.3, 0.0, 0.3))
    return min(candidates, key=lambda p: objective(p, models, refs))


def fit(rows, targets):
    """Small ridge least squares via pivoted Gaussian elimination (stdlib)."""
    n = len(rows[0])
    mat = [[sum(x[i] * x[j] for x in rows) + (1e-5 if i == j else 0)
            for j in range(n)] + [sum(x[i] * y for x, y in zip(rows, targets))]
           for i in range(n)]
    for i in range(n):
        pivot = max(range(i, n), key=lambda k: abs(mat[k][i]))
        mat[i], mat[pivot] = mat[pivot], mat[i]
        scale = mat[i][i]
        assert abs(scale) > 1e-12
        mat[i] = [v / scale for v in mat[i]]
        for j in range(n):
            if j != i:
                scale = mat[j][i]
                mat[j] = [a - scale * b for a, b in zip(mat[j], mat[i])]
    return tuple(row[-1] for row in mat)


def adapt(data):
    # SysID's explicit linear physical family omits nonlinear drag on purpose.
    linear_rows = [(u, -v, 1.0) for u, v, _ in data]
    g, d, bias = fit(linear_rows, [acc for _, _, acc in data])
    sysid = plant((g, d, 0.0, bias))
    # DR uses exactly those rows, residual spread -> lesson-defined wide bounds.
    rms = math.sqrt(sum((acc - sysid(u, v)) ** 2 for u, v, acc in data) / len(data))
    width = max(0.08, 2 * rms)
    dr = tuple(plant((max(0.2, g + sign * width), max(0.0, d - sign * width),
                      0.0, bias + other * width)) for sign, other in
               ((-1, -1), (-1, 1), (1, -1), (1, 1)))
    # Since g_A=1, delta-action target = observed acceleration - a_A(u,v).
    # This is an action correction, NOT an identified mass/friction parameter.
    features = [(u, v, v * abs(v), 1.0) for u, v, _ in data]
    coeff = fit(features, [acc - plant(A)(u, v) for u, v, acc in data])
    def aligned_sim(u, v):
        delta_action = sum(c * f for c, f in zip(coeff, (u, v, v * abs(v), 1.0)))
        return plant(A)(u + delta_action, v)
    return {"SysID-retune": (sysid,) * 4, "DR-widen": dr,
            "residual-action": (aligned_sim,) * 4}


def deploy(policy, real_proxy, eval_refs):
    # The signature deliberately has NO learned delta argument.
    results = [rollout(policy, real_proxy, refs) for refs in eval_refs]
    mae = statistics.mean(r[0] for r in results)
    fall_proxy = sum(r[1] > 0.35 for r in results) / len(results)
    return mae, fall_proxy


def main():
    train_refs = trajectories(71, 4)
```

## TD3：两个评论家、取最小值和延迟的演员

上面的 PPO 式 toy 从新采的轨迹学习随机策略。DDPG 这类离策略演员-评论家方法则反复使用回放缓冲区：评论家学 `Q(s, a)`，确定性的演员沿它往上爬。这种复用有一个已知的失败方式：演员专挑评论家恰好估高了的动作，自举目标把误差抄过去，价值估计逐渐往上漂。**TD3**（Twin Delayed Deep Deterministic policy gradient）在 DDPG 上加了三处修正。本节在一个很小的线性替身上检验它们。这**不是**把 PPO toy 再做一遍，**不是** HER，也**不是**对 MuJoCo 分数的断言。

<Resource id="td3-paper" />

**实际打开的来源（用 pdftotext 读了 arXiv 1802.09477 v3 PDF 正文，并核对了摘要页）。**

- **Fujimoto、van Hoof 和 Meger，《Addressing Function Approximation Error in Actor-Critic Methods》（ICML 2018）**，要点如下。
  - 论文表明 Q-learning 中已知的高估偏差在演员-评论家方法中依然存在，而 Double DQN 的演员-评论家版本并不能消除它。
  - **截断双 Q 学习**（Clipped Double Q-learning），式 (10)：两个评论家，目标取二者的最小值，`y1 = r + γ min over i of Qθi′(s′, πφ1(s′))`。论文指出这会偏向低估，而低估不会像高估那样经由策略更新传播。
  - **延迟策略更新**，每 `d` 次评论家更新才更新一次演员和目标网络（`d = 2`），“直到价值误差尽可能小”。
  - **目标策略平滑**，式 (15)：`ã = πφ′(s′) + ε`，其中 `ε ~ clip(N(0, 0.2), −0.5, 0.5)`。
  - 算法 1 把三者合在一起。实验用 `τ = 0.005`、探索噪声 `N(0, 0.1)`、400-300 网络、学习率 10⁻³ 的 Adam、批大小 100、10 个种子，在 OpenAI Gym 的 MuJoCo 上跑 100 万步。
- **论文报告的结果**（以论文为准）如下。
  - 表 1 的最大平均回报：HalfCheetah 上 TD3 为 9636.95 ± 859.065，DDPG 为 3305.60。InvertedPendulum 上列出的七种方法都是 1000.00，打平；InvertedDoublePendulum 上 DDPG 的 9355.52 高于 TD3 的 9337.47。
  - 表 2 的消融：去掉延迟（TD3 − DP）在 HalfCheetah（9590.65 对 9532.99）和 Walker2d（4695.50 对 4565.24）上比完整 TD3 分数更高。
  - 所以论文自己的表格里就有打平和落败。
- MuJoCo、Gym、SB3 和 GPU 训练均**本页未运行**。

### 为什么取最小值能对抗高估

<Formula label="DDPG 式目标（一个评论家）">y = r + γ · Q′(s′, π′(s′))</Formula>

<Formula label="TD3 目标（论文式 15）">y = r + γ · min(Q1′, Q2′)(s′, π′(s′) + ε),  ε ~ clip(N(0, 0.2), −0.5, 0.5)</Formula>

演员会去找高值，所以评论家在演员所选动作上的误差平均为正，由它构造的每个目标都会继承这个误差。在同样数据上训练的两个评论家取较小者，可以去掉一部分向上的误差；代价是两者都准确时会带来向下的偏差。

**算例。** 设下一状态的真值为 −2.0，两个评论家分别读出 −1.6 和 −2.3。DDPG 的目标用 −1.6，取 `r = −0.5`、`γ = 0.8`，得 `y = −0.5 + 0.8 · (−1.6) = −1.78`，而真值为 −2.10，高估 0.32。TD3 用 −2.3，得 `y = −0.5 + 0.8 · (−2.3) = −2.34`，低估 0.24。取最小值并不能让目标变对，只是让误差倾向于向下。

**这里没有积分步长。** 被控对象是一行映射 `s′ = clip(s + 0.5a, −2, 2)`，不是常微分方程。**稳定性判据。** 评论家用 SGD 学习平方 TD 误差。一步 SGD 把一个样本上的误差乘以 `1 − LR · |φ|²`，所以 `LR · |φ|² ≤ 2` 时稳定。toy 打印状态-动作盒子内最大的 `|φ|²`（27.0，对应上限 `LR = 0.0741`），并断言 `LR · max|φ|² = 0.2700 < 2`。测试检查在 `2 / |φ|²` 的 0.98× 处单样本更新使误差缩小，在 1.02× 处使误差变大。任何权重超出 ±10⁴ 就标记该次运行发散；没有一次运行被标记。

### 公平设置

`td3_clipped_double_q_toy.py` 是**合成**模型，只用标准库。**模型假设**如下。
- 一维连续任务，奖励为 `−s′² − 0.1a²` 加零均值噪声。每 10 步重置到新的初始状态，重置不视为终止。
- 评论家对六个二次特征 `(1, s, a, s², a², s·a)` 是线性的。演员为 `a = clip(k·s + b, −1, 1)`。
- 3000 步，其中前 200 步为预热。批大小 8，评论家 `LR = 0.01`，演员 `LR = 0.002`，`τ = 0.01`，探索噪声 `N(0, 0.3)`。
- **没有**神经网络，也没有 MuJoCo；超参数是 toy 自己的，不是论文的。

| 组 | 评论家和目标 | 演员和目标网络的更新 |
|---|---|---|
| A_ddpg | 一个评论家，`y = r + γ Q′(s′, π′(s′))` | 每步 |
| B_td3 | 两个评论家，取最小值，目标动作加平滑 | 每 `d = 2` 次评论家更新 |
| C_cdq_nodelay | 同 B | 每步（去掉延迟的 TD3） |

**共用随机数**（common random numbers）的做法是，每个种子的初始状态、探索噪声、奖励噪声、批次抽样、平滑噪声和评论家初始权重都来自各自独立的随机流，只抽一次，所有组共用。指标在 13 个固定评估状态上计算，不加探索。
- `bias`：评论家在最终策略动作上的值，减去该策略无噪声的折扣蒙特卡洛回报；
- `absbias`：`|bias|`；
- `ret`：最终策略的 10 步平均回报。

| 情形 | 奖励噪声标准差 | γ | 运行前的预期 |
|---|---|---|---|
| noisy_reward | 1.0 | 0.8 | TD3 高估更少 |
| easy | 0 | 0.8 | 接近打平 |
| bandit_no_bootstrap | 0 | 0 | 没有自举，就没有可修正的估计偏差：TD3 应打平或落败 |

**看过结果后做的选择。** 第一次运行后没有改任何参数或情形。事后只加了一行输出，就是 C−A 的比较；加它是因为第一次运行显示 `bandit_no_bootstrap` 中 C 的数字与 A 相同。这一行在每个情形都打印，其余都没变。

```python
#!/usr/bin/env python3
"""Synthetic DDPG-style single critic vs TD3 vs clipped double Q without delay. Stdlib only.

Plant (continuing, 1-D): s' = clip(s + 0.5 a, -2, 2), a in [-1, 1], reward r = -s'^2 - 0.1 a^2 + noise
(noise sd SIGMA_R per regime, zero mean). The agent resets to a fresh start state every EP steps; the reset is
not treated as terminal, so every target bootstraps with discount GAMMA.
Function approximation: each critic is linear in quadratic features phi(s, a) = (1, s, a, s^2, a^2, s a);
the actor is a = clip(k s + b, -1, 1). Critics learn by minibatch SGD on squared TD error; the actor follows the
deterministic policy gradient dQ1/da * da/d(k, b). Targets are Polyak copies (TAU). Arms:
  A_ddpg      one critic, target y = r + GAMMA Q'(s', pi'(s')), actor and targets updated every step
  B_td3       two critics, y = r + GAMMA min(Q1', Q2')(s', pi'(s') + eps), eps ~ clip(N(0, 0.2), -0.5, 0.5),
              actor and targets updated every D = 2 critic steps (Fujimoto et al. 2018, Algorithm 1)
  C_cdq_nodelay  B with D = 1 (clipped double Q and smoothing, no delay)
Common random numbers: per seed the start states, exploration noise, reward noise, minibatch picks, target
smoothing noise and initial critic weights are drawn once and shared by every arm (one stream per role).
No integration step: the plant is a one-line map, not an ODE. Stability criterion instead: SGD on the critic
is stable when LR * |phi|^2 < 2 (the toy prints the largest |phi|^2 over the feature box and asserts
LR * max|phi|^2 < 2; the tests check a single-sample update shrinks the error at 0.98x and grows it at 1.02x
of LR = 2/|phi|^2). A run is flagged "diverged" if any weight leaves [-1e4, 1e4].
Metrics (on fixed evaluation start states, no exploration): bias = mean of Q1(s, pi(s)) minus the
noise-free discounted Monte Carlo return of the final policy (60 steps); ret = mean undiscounted 10-step
return of the final policy; absbias = |bias|.
Labels: tie = identical on every seed; numerically zero = |paired mean| and sd both < 1e-9 (effectively a
tie, no sign count); |m| <= sd = near-tie; |m| > sd = consistent gap (small if |m| < 20 % of the smaller arm
mean, else clear), with a sign count.
Not PPO (the clipped-ratio toy above) and not HER. MuJoCo / Gym / SB3 / GPU training: 本页未运行.
Paper numbers: 以论文为准.
"""
import math
import random

SEEDS = (0, 1, 2, 3, 4)
GAMMA = 0.8
EP = 10
STEPS = 3000
WARMUP = 200
BATCH = 8
LR = 0.01
LR_ACTOR = 0.002
TAU = 0.01
EXPL = 0.3
D = 2
SMOOTH_SD, SMOOTH_CLIP = 0.2, 0.5
BOUND = 1e4
NUM_ZERO = 1e-9
ARMS = ("A_ddpg", "B_td3", "C_cdq_nodelay")
# name: (reward noise sd, discount); pre-declared before the first run
REGIMES = {
    "noisy_reward": (1.0, GAMMA),
    "easy": (0.0, GAMMA),
    "bandit_no_bootstrap": (0.0, 0.0),
}
EVAL = [-1.5 + 0.25 * i for i in range(13)]


def phi(s, a):
    return (1.0, s, a, s * s, a * a, s * a)


def q(w, s, a):
    return sum(x * y for x, y in zip(w, phi(s, a)))


def dq_da(w, s, a):
    return w[2] + 2.0 * w[4] * a + w[5] * s


def act(k, b, s):
    return max(-1.0, min(1.0, k * s + b))


def step(s, a):
    return max(-2.0, min(2.0, s + 0.5 * a))


def max_phi2():
    """Largest |phi|^2 over the box s in [-2, 2], a in [-1, 1] (attained at a corner)."""
    return max(sum(x * x for x in phi(s, a)) for s in (-2.0, 2.0) for a in (-1.0, 1.0))


def shared_draws(seed):
    """Common random numbers: one stream per role, shared by every arm."""
    rngs = [random.Random(seed * 100 + i) for i in range(6)]
    n = STEPS
    return {
        "start": [rngs[0].uniform(-1.5, 1.5) for _ in range(n // EP + 1)],
        "expl": [rngs[1].gauss(0.0, EXPL) for _ in range(n)],
        "rnoise": [rngs[2].gauss(0.0, 1.0) for _ in range(n)],
        "pick": [[rngs[3].random() for _ in range(BATCH)] for _ in range(n)],
        "smooth": [[max(-SMOOTH_CLIP, min(SMOOTH_CLIP, rngs[4].gauss(0.0, SMOOTH_SD))) for _ in range(BATCH)] for _ in range(n)],
        "w0": [[rngs[5].uniform(-0.1, 0.1) for _ in range(6)] for _ in range(2)],
    }


def mc_return(k, b, s, gamma):
    g, disc = 0.0, 1.0
    for _ in range(60):
        a = act(k, b, s)
        s = step(s, a)
        g += disc * (-s * s - 0.1 * a * a)
        disc *= gamma
        if gamma == 0.0:
            break
    return g


def run(arm, regime, dr):
    sigma_r, gamma = REGIMES[regime]
    twin = arm != "A_ddpg"
    d = D if arm == "B_td3" else 1
    w = [list(dr["w0"][0]), list(dr["w0"][1])]
    wt = [list(w[0]), list(w[1])]
    k, b = 0.0, 0.0
    kt, bt = k, b
    buf = []
    s = dr["start"][0]
    for t in range(STEPS):
        if t % EP == 0:
            s = dr["start"][t // EP]
        a = max(-1.0, min(1.0, act(k, b, s) + dr["expl"][t]))
        s2 = step(s, a)
        r = -s2 * s2 - 0.1 * a * a + sigma_r * dr["rnoise"][t]
        buf.append((s, a, r, s2))
        s = s2
        if t < WARMUP:
            continue
        batch = [buf[int(u * len(buf))] for u in dr["pick"][t]]
        for j, (bs, ba, br, bs2) in enumerate(batch):
            a2 = act(kt, bt, bs2)
            if twin:
                a2 = max(-1.0, min(1.0, a2 + dr["smooth"][t][j]))
                nxt = min(q(wt[0], bs2, a2), q(wt[1], bs2, a2))
            else:
                nxt = q(wt[0], bs2, a2)
            y = br + gamma * nxt
            f = phi(bs, ba)
            for c in ((0, 1) if twin else (0,)):
                err = y - q(w[c], bs, ba)
                w[c] = [wi + LR / BATCH * err * fi for wi, fi in zip(w[c], f)]
        if any(abs(x) > BOUND for c in w for x in c):
            return None
        if (t - WARMUP) % d == 0:
            gk = gb = 0.0
            for bs, _, _, _ in batch:
                a0 = k * bs + b
                if -1.0 < a0 < 1.0:
                    g = dq_da(w[0], bs, a0)
                    gk += g * bs
                    gb += g
            k += LR_ACTOR * gk / BATCH
            b += LR_ACTOR * gb / BATCH
            for c in range(2):
                wt[c] = [TAU * x + (1 - TAU) * y_ for x, y_ in zip(w[c], wt[c])]
            kt, bt = TAU * k + (1 - TAU) * kt, TAU * b + (1 - TAU) * bt
    bias = sum(q(w[0], s0, act(k, b, s0)) - mc_return(k, b, s0, gamma) for s0 in EVAL) / len(EVAL)
    ret = 0.0
    for s0 in EVAL:
        x = s0
        for _ in range(EP):
            a = act(k, b, x)
            x = step(x, a)
            ret += -x * x - 0.1 * a * a
    return {"bias": bias, "absbias": abs(bias), "ret": ret / len(EVAL), "k": k}


def mean_sd(xs):
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    if x != 0.0 and abs(x) < 5e-5:
        return f"{x:.3e}"
    return f"{x:.4f}"


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{num(m)}+-{num(s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{num(m)}+-{num(s)}: {label}; {sign_note(diffs, m)}"


KEYS = ("bias", "absbias", "ret")


def main():
    print("Synthetic DDPG-style vs TD3 vs clipped double Q without delay; linear critics on a 1-D plant; not MuJoCo / Gym")
    m2 = max_phi2()
```

## 广义优势估计：在蒙特卡洛与一步 TD 之间取一个 λ

<Resource id="gae-paper" />

上面的 PPO 风格玩具用折扣后的未来回报 `G` 给每个动作打分，并在整批上归一化：没有价值函数，所以每个动作都带着它之后发生的一切的噪声。TD3 一节用了评论家，但对应的是确定性的执行器。本节给出介于两者之间、用于随机策略梯度的估计器：**广义优势估计**（GAE）。

**实际打开的来源（已读摘要页和 PDF 正文；arXiv 1506.02438 v6，2018 年 10 月 20 日）。** **John Schulman、Philipp Moritz、Sergey Levine、Michael Jordan、Pieter Abbeel，《High-Dimensional Continuous Control Using Generalized Advantage Estimation》**（ICLR 2016；作者按摘要页顺序；PDF 署名写作 Michael I. Jordan，是同一个人，顺序也相同）。论文把 GAE(γ, λ) 定义为各个 k 步优势估计的**指数加权平均**，它化简为 TD 残差 `δ_t = r_t + γ V(s_{t+1}) − V(s_t)` 的折扣和（式 16）。由此得到两个特例：λ = 0 就是一步残差 `δ_t`（式 17），λ = 1 就是折扣回报减去价值基线（式 18）。按论文的说法，GAE(γ, 1) 不论 `V` 准不准都是 γ-just 的，但方差大；GAE(γ, 0) 只有在价值函数准确时才是 γ-just 的，否则会引入偏差，但方差通常小得多；0 < λ < 1 在偏差和方差之间折中。论文报告在 cart-pole 上，λ 取 0.92–0.98 之间的中间值时策略提升最快；在另一处又说一步回报（λ = 0）导致偏差过大、表现很差。这些是论文的陈述和数字，以原文为准；下面没有一条是玩具的输出，论文的双足和四足实验都**本页未运行**。

本节**不是**第二遍 PPO 推导：上面的截断比率保持不变，GAE 只改变送进去的优势。它也**不是** TD3，**不是**残差动作迁移。不涉及神经网络、MuJoCo 或 Stable-Baselines3 的运行。

<Formula label="GAE(γ, λ)，在一个回合内倒序计算">δ_t = r_t + γ V(t+1) − V(t),   V(H) = 0;   Â_t = δ_t + γλ Â_{t+1},   Â_H = 0;   λ = 0 → Â_t = δ_t;   λ = 1 → Â_t = Σ_l γ^l r_{t+l} − V(t)</Formula>

**算例（手算，玩具也会打印）。** 往后第 `l` 步残差的权重是 `(γλ)^l`，所以估计器大约看向未来 `1 / (1 − γλ)` 步。取 `γ = 0.9900`、λ = 0.95，是 `16.8067` 步；λ = 1 时是 `100.0000` 步，比 50 步的回合还长，所以 λ = 1 的求和一直加到回合结束。只有一步（`H = 1`）且 `V(1) = 0` 时，每个 λ 都给出同样的 `Â_0 = r_0 − V(0)`，三个估计器必然相同。对下面用的策略，真实梯度是 `Σ_t γ^t · P1(1 − P1)`，`P1 = 0.5000`；50 步时玩具打印 `9.8748`。

**积分步长。** 无：每个回合是一次倒序遍历算出的有限和；没有时间常数，也没有学习率，因为策略不更新。

### 公平设置

`gae_lambda_advantage_toy.py` 是**合成**的，只用标准库。**模型假设:** 一个 `H` 步的回合，状态就是步序号；两动作策略 `P(a = 1) = 0.5`（θ = 0 处的逻辑斯蒂策略，得分函数为 `a − P1`）；即时模式下奖励是 `a_t + 噪声`（sd 1.0），延迟模式下全部奖励在最后一步一次付清；真实优势和真实梯度都有闭式解，所以可以直接测偏差。价值替身要么是每一步的真实价值加一个固定误差（sd 0.3），要么是等于其均值的单个常数（粗糙）。所有数值都是**声明的玩具参数**；λ = 0.95 是在运行前从论文报告的区间里选定的。

```python
#!/usr/bin/env python3
"""Synthetic generalized advantage estimation (GAE) toy.

A scalar two-action logistic policy (theta = 0, P(a=1) = P1 = 0.5) on a
finite episode of H steps; not a neural network, not MuJoCo, not SB3.
Paper biped / quadruped locomotion results 以论文为准; GPU / SB3 are not
run here (本页未运行).

Arms: A_lam1 (LAM = 1), B_lam (LAM = 0.95, the declared value inside the
0.92-0.98 range the paper reports as fastest on cart-pole, 以论文为准),
C_lam0 (LAM = 0).  Regimes: long_noisy (immediate reward, H = 50, reward
noise sd 1.0, noisy value with per-state sd 0.3); one_step (immediate,
H = 1, same noise); delayed_coarse (all reward at the final step, H = 30,
value a single constant equal to the mean true value over time).

True advantages are known in closed form; the value stand-in is either
the true state value plus a fixed per-state error or a coarse constant.
Gradients are estimated over N_EP = 16 episodes per repetition and
M_DRAW = 40 repetitions per seed; metrics are the variance of the
per-repetition gradient estimate and its mean squared error to g_true.

Common random numbers: one random.Random(f"{regime}/{seed}") stream per
(regime, seed) draws the value errors first, then all actions and reward
noises for every repetition and episode, in a fixed order before any arm
is evaluated; all arms share the same episodes and the same V.

Regimes and parameters were written before the first run.  Declared
expectations are printed as they come, without tuning.
"""
import math
import random
from decimal import ROUND_HALF_UP, Decimal, getcontext

getcontext().rounding = ROUND_HALF_UP

SEEDS = (0, 1, 2, 3, 4)
NUM_ZERO = 1e-9

# Declared parameters, written before the first run.
GAMMA = 0.99
P1 = 0.5
C_R = 1.0
SIG_R = 1.0
N_EP = 16
M_DRAW = 40
LAM_DECL = 0.95

ARMS = ("A_lam1", "B_lam", "C_lam0")
ARM_LAM = {"A_lam1": 1.0, "B_lam": LAM_DECL, "C_lam0": 0.0}
METRICS = ("grad_var", "grad_mse")
PAIRS = (("B_lam", "A_lam1"), ("C_lam0", "B_lam"), ("C_lam0", "A_lam1"))

REGIMES = {
    "long_noisy": {"H": 50, "mode": "immediate", "value": "noisy", "SIG_V": 0.3},
    "one_step": {"H": 1, "mode": "immediate", "value": "noisy", "SIG_V": 0.3},
    "delayed_coarse": {"H": 30, "mode": "delayed", "value": "coarse", "SIG_V": 0.0},
}
DECLARED = {
    "long_noisy": "B consistently lower grad_var than A - keep; C as it comes",
    "one_step": "all three lambdas tie or near-tie - keep",
    "delayed_coarse": "C (lambda = 0) loses on grad_mse to B and to A - keep",
}


def mean_sd(xs):
    if all(x == xs[0] for x in xs):
        return xs[0], 0.0  # identical values: no floating residue in mean or sd
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    """Half-up rounding on the shortest decimal repr; tiny non-zero values in scientific form."""
    if x != 0.0 and abs(x) < 5e-4:
        return format(Decimal(repr(x)), ".3e")
    s = format(Decimal(repr(x)), ".4f")
    return "0.0000" if s == "-0.0000" else s


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def fmt_pair(m, s):
    return f"{num(m)}+-{num(s)}"


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{fmt_pair(m, s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{fmt_pair(m, s)}: {label}; {sign_note(diffs, m)}"


def true_value(regime, t):
    cfg = REGIMES[regime]
    H = cfg["H"]
    if cfg["mode"] == "immediate":
        return sum(GAMMA ** ell for ell in range(H - t)) * C_R * P1
    return (GAMMA ** (H - 1 - t)) * H * C_R * P1


def true_grad(regime):
    cfg = REGIMES[regime]
    H = cfg["H"]
    total = 0.0
    for t in range(H):
        if cfg["mode"] == "immediate":
            total += (GAMMA ** t) * C_R * P1 * (1.0 - P1)
        else:
            total += (GAMMA ** t) * (GAMMA ** (H - 1 - t)) * C_R * P1 * (1.0 - P1)
    return total


def build_data(regime, seed, value_mode=None, sig_v=None):
    cfg = REGIMES[regime]
    H = cfg["H"]
    mode = cfg["mode"]
    if value_mode is None:
        value_mode = cfg["value"]
    if sig_v is None:
        sig_v = cfg["SIG_V"]
    rng = random.Random(f"{regime}/{seed}")
    errs = [rng.gauss(0.0, 1.0) for _ in range(H)]
    raw = []
    for _ in range(M_DRAW):
        eps = []
        for _ in range(N_EP):
            actions = [1 if rng.random() < P1 else 0 for _ in range(H)]
            if mode == "immediate":
                rewards = [C_R * actions[t] + rng.gauss(0.0, SIG_R) for t in range(H)]
            else:
                total = C_R * sum(actions)
                noise = rng.gauss(0.0, SIG_R) * math.sqrt(H)
                rewards = [0.0] * H
                rewards[H - 1] = total + noise
            eps.append((actions, rewards))
        raw.append(eps)
    true_v = [true_value(regime, t) for t in range(H)]
    if value_mode == "noisy":
        V = [true_v[t] + sig_v * errs[t] for t in range(H)]
    else:
        m = sum(true_v) / H
        V = [m] * H
    V.append(0.0)
    data = []
    for eps in raw:
        draw = []
        for actions, rewards in eps:
            deltas = [0.0] * H
            for t in range(H):
                deltas[t] = rewards[t] + GAMMA * V[t + 1] - V[t]
            draw.append((actions, deltas))
        data.append(draw)
    return H, data


def seed_metrics(regime, seed, lams, value_mode=None, sig_v=None):
    H, data = build_data(regime, seed, value_mode, sig_v)
    target = true_grad(regime)
    out = {}
    for lam in lams:
        gs = []
        for draw in data:
            total = 0.0
            for actions, deltas in draw:
                adv = [0.0] * H
                running = 0.0
                for t in range(H - 1, -1, -1):
                    running = deltas[t] + GAMMA * lam * running
                    adv[t] = running
                disc = 1.0
                for t in range(H):
                    total += disc * adv[t] * (actions[t] - P1)
                    disc *= GAMMA
            gs.append(total / N_EP)
        var = mean_sd(gs)[1] ** 2
        mse = sum((g - target) ** 2 for g in gs) / len(gs)
        out[lam] = (var, mse)
    return out


def run_regime(regime):
    acc = {arm: {key: [] for key in METRICS} for arm in ARMS}
    for seed in SEEDS:

            acc[arm]["grad_var"].append(var)
            acc[arm]["grad_mse"].append(mse)
    return acc
```
