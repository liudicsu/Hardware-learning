上一课用传递函数描述机械臂关节：一个输入、一个输出。<Term id="state-space">状态空间</Term>（state space）把每个内部状态都放进向量 x，写成 `x' = A x + B u` 这样的矩阵方程。平衡车恰好需要这样：位置、速度、角度、角速度必须同时考虑。这里的对象先从[第 1 课](/zh/month/month-5/control-modelling)的关节开始，再换成第 2 个月的平衡车简化成的小车倒立摆（cart-pole）。

## 两个对象，一种形式

对关节取 x = [q, w]，q 是角度，w = q'，力矩是唯一的输入 u。代入 J = 0.004 kg·m²、B = 0.05 N·m/(rad/s)，方程 J q'' + B q' = u 就变成两行状态方程。第一行是 q' = w。第二行是 w' = −(B/J) w + (1/J) u = −12.5 w + 250 u，所以 A = [[0, 1], [0, −12.5]]，B = [[0], [250]]。一个方程现在同时带着角度和速度。

<Formula label="状态空间：把所有状态装进一个向量">x' = A x + B u</Formula>

小车倒立摆是第 2 个月平衡车（balancer）的极简版：M = 0.5 kg 的小车，m = 0.2 kg 的质点挂在 L = 0.3 m 处，输入是水平力 F。非线性方程（角度从直立量起）是小车加速度 `a = (F + m L ω² sinθ − m g sinθ cosθ)/(M + m − m cos²θ)` 与杆角加速度 `θ'' = (g sinθ − a cosθ)/L`。在直立处线性化：对 θ 求偏导，取 θ = 0、ω = 0 且 `a ≈ F/M`，得到 `−m g/M = −3.92`、`1/M = 2.00`、`(M+m)g/(M L) = 45.78` 和 `−1/(M L) = −6.67`。取 x = [position, speed, angle, rate]，模型打印：

```python
"""lqr_sim.py: state space, LQR on a cart-pole, and a small MPC on the Month 3 arm joint.

Part 1 writes a cart-pole (the Month 2 balancer reduced to its essentials: a cart that is
pushed by a force and a pole balanced on it) in state space, x = [position, speed, angle,
angular rate], linearised about upright, discretised at 100 Hz. LQR gains come from
iterating the discrete Riccati equation. The gains are then tested on the full nonlinear
cart-pole, with the force limited to 10 N.
Part 2 moves the arm joint from lesson 1 (J = 0.004, B = 0.05, gravity cancelled by
feedforward) through 1 rad with a torque limit, comparing LQR whose output is simply
clipped with an MPC that knows the limit. Standard library only.
"""
import math

# ---------------------------------------------------------------- small matrix helpers
def mul(a, b):
    return [[sum(a[i][k] * b[k][j] for k in range(len(b))) for j in range(len(b[0]))] for i in range(len(a))]


def add(a, b, s=1.0):
    return [[a[i][j] + s * b[i][j] for j in range(len(a[0]))] for i in range(len(a))]


def tr(a):
    return [list(r) for r in zip(*a)]


def eye(n, v=1.0):
    return [[v if i == j else 0.0 for j in range(n)] for i in range(n)]


def discretise(a, b, dt):
    """Zero-order-hold discretisation by a Taylor series of the matrix exponential."""
    n = len(a)
    phi, gam, term = eye(n), [[dt * x[0]] for x in b], eye(n)
    for k in range(1, 25):
        term = [[v * dt / k for v in r] for r in mul(term, a)]           # (A dt)^k / k!
        phi = add(phi, term)
        gam = add(gam, [[v * dt / (k + 1)] for v in [r[0] for r in mul(term, b)]])
    return phi, gam


def lqr(phi, gam, q, r, iters=5000):
    """Iterate P = Q + A'PA - A'PB (R + B'PB)^-1 B'PA; return K for u = -K x."""
    p = q
    for _ in range(iters):
        bp = mul(tr(gam), p)
        s = r + mul(bp, gam)[0][0]
        k = [[v / s for v in mul(bp, phi)[0]]]
        p = add(q, mul(tr(phi), mul(p, add(phi, mul(gam, k), -1.0))))
    return k[0]


# ---------------------------------------------------------------- part 1: cart-pole
M, m, L, G, DT, FMAX = 0.5, 0.2, 0.3, 9.81, 0.01, 10.0
A_CP = [[0, 1, 0, 0], [0, 0, -m * G / M, 0], [0, 0, 0, 1], [0, 0, (M + m) * G / (M * L), 0]]
B_CP = [[0], [1 / M], [0], [-1 / (M * L)]]


def cartpole(x, f, dt=0.001):
    """Full nonlinear cart-pole (point mass m at distance L), angle from upright."""
    p, v, th, w = x
    s, c = math.sin(th), math.cos(th)
    acc = (f + m * L * w * w * s - m * G * s * c) / (M + m - m * c * c)
    alpha = (G * s - acc * c) / L
    return [p + v * dt, v + acc * dt, th + w * dt, w + alpha * dt]


def balance(k, deg, seconds=4.0, show=False):
    x = [0.0, 0.0, math.radians(deg), 0.0]
    fpeak = 0.0
    for i in range(int(seconds / DT)):
        f = max(-FMAX, min(FMAX, -sum(ki * xi for ki, xi in zip(k, x))))
        fpeak = max(fpeak, abs(f))
        if show and i % 50 == 0:
            print(f"  t = {i * DT:3.1f} s  x = {x[0]:6.3f} m  angle = {math.degrees(x[2]):6.2f} deg  force = {f:6.2f} N")
        for _ in range(10):
            x = cartpole(x, f)
        if abs(x[2]) > math.pi / 2:
            return None, fpeak
    return x, fpeak


def part1():
    phi, gam = discretise(A_CP, B_CP, DT)
    print("Cart-pole, x = [position, speed, angle, rate], 100 Hz")
    print("  A (continuous) rows:", "  ".join("[" + ", ".join(f"{v:.2f}" for v in r) + "]" for r in A_CP))
    print("  B (continuous):     ", "[" + ", ".join(f"{r[0]:.2f}" for r in B_CP) + "]")
    for name, qd, r in (("Q = diag(1, 1, 10, 1), R = 1   ", (1, 1, 10, 1), 1.0),
                        ("Q = diag(100, 1, 10, 1), R = 1 ", (100, 1, 10, 1), 1.0),
                        ("Q = diag(1, 1, 10, 1), R = 0.01", (1, 1, 10, 1), 0.01)):
        q = [[qd[i] if i == j else 0.0 for j in range(4)] for i in range(4)]
        k = lqr(phi, gam, q, r)
        print(f"{name}  K = [" + ", ".join(f"{v:7.2f}" for v in k) + "]")
        if name.startswith("Q = diag(1, 1, 10, 1), R = 1 "):
            k_first = k
        x, fpeak = balance(k, 10)
        print(f"    from 10 deg: after 4 s x = {x[0]:6.3f} m, angle = {math.degrees(x[2]):5.2f} deg, peak force {fpeak:5.2f} N")
    print("Nonlinear run with the first K, starting 10 deg off upright:")
    balance(k_first, 10, seconds=3.0, show=True)
    biggest = max(d for d in range(1, 90) if balance(k_first, d, seconds=6.0)[0] is not None
                  and abs(balance(k_first, d, seconds=6.0)[0][2]) < 0.01)
    print(f"  largest start angle it still recovers from (force limit {FMAX:.0f} N): {biggest} deg")


# ---------------------------------------------------------------- part 2: MPC on the joint
J, BV, T, UMAX, N = 0.004, 0.05, 0.005, 0.6, 40
PHI_J, GAM_J = discretise([[0, 1], [0, -BV / J]], [[0], [1 / J]], T)


def jstep(x, u):
    return [PHI_J[0][0] * x[0] + PHI_J[0][1] * x[1] + GAM_J[0][0] * u,
            PHI_J[1][0] * x[0] + PHI_J[1][1] * x[1] + GAM_J[1][0] * u]


def mpc(x, target, kq, warm):
    """Minimise sum over N steps of kq*(q-target)^2 + 0.01*w^2 + u^2, |u| <= UMAX,
    by projected gradient descent on the N future torques; return them all."""
    u = warm[1:] + warm[-1:]
    lr = 2e-4
    for _ in range(150):
        xs = [x]
        for uk in u:
            xs.append(jstep(xs[-1], uk))
        lam = [0.0, 0.0]                  # backward pass: gradient of the cost wrt each u
        grad = [0.0] * N
        for k in range(N - 1, -1, -1):
            q, w = xs[k + 1]
            lam = [2 * kq * (q - target) + PHI_J[0][0] * lam[0] + PHI_J[1][0] * lam[1],
                   2 * 0.01 * w + PHI_J[0][1] * lam[0] + PHI_J[1][1] * lam[1]]
            grad[k] = 2 * u[k] + GAM_J[0][0] * lam[0] + GAM_J[1][0] * lam[1]
        u = [max(-UMAX, min(UMAX, uk - lr * g)) for uk, g in zip(u, grad)]
    return u


def move(kind, seconds=1.0):
    k = lqr(PHI_J, GAM_J, [[400.0, 0], [0, 0.01]], 1.0)
    x, target, peak, settle = [0.0, 0.0], 1.0, 0.0, None
    plan = [0.0] * N
    sat = 0
    for i in range(int(seconds / T)):
        if kind == "lqr":
            u = max(-UMAX, min(UMAX, -(k[0] * (x[0] - target) + k[1] * x[1])))
        else:
            plan = mpc(x, target, 400.0, plan)
            u = plan[0]
        sat += abs(u) >= UMAX - 1e-9
        x = jstep(x, u)
        peak = max(peak, x[0])
        if abs(x[0] - target) > math.radians(1):
            settle = None
        elif settle is None:
            settle = (i + 1) * T
    return k, peak, settle, sat * T


def part2():
    print(f"Arm joint, 1 rad move, |torque| <= {UMAX} N*m (on top of gravity feedforward), 200 Hz")
    for kind, label in (("lqr", "LQR, output clipped"), ("mpc", f"MPC, horizon {N} steps")):
        k, peak, settle, sat = move(kind)
        if kind == "lqr":
            print(f"  LQR gains: K = [{k[0]:.2f}, {k[1]:.3f}]  (kp in N*m/rad, kd in N*m/(rad/s))")
        s = f"{settle:.3f} s" if settle else "never"
        print(f"  {label:22s} overshoot {math.degrees(peak - 1.0):5.2f} deg, within 1 deg from {s}, at the limit for {sat:.3f} s")


if __name__ == "__main__":
    part1()
    print()
    part2()
```

## 一行增益，四个状态

线性二次调节器（<Term id="lqr">LQR</Term>，linear quadratic regulator）要的是让整段运行代价最小的输入序列：

<Formula label="LQR 代价：状态误差加控制用力">J = Σ (xᵀ Q x + uᵀ R u)</Formula>

Q 是 4 × 4 矩阵，惩罚状态误差；R 是正标量，惩罚力。单位要一致：位置以 m² 进入，角度以 rad² 进入，力的平方以 N² 进入。增益来自反复迭代离散 Riccati 方程（这里的 A、B 是零阶保持后的矩阵），直到 P 不再变化：

<Formula label="离散 Riccati 迭代和它给出的增益">P ← Q + Aᵀ P A − Aᵀ P B (R + Bᵀ P B)⁻¹ Bᵀ P A,   K = (R + Bᵀ P B)⁻¹ Bᵀ P A</Formula>

控制律是 u = −K x。K 每个状态一项，所以同一行同时用位置、速度、角度和角速度。第 3 个月的串级（cascade）用了两个先后调好的回路：内环管俯仰角，外环管轮速，并通过移动内环的设定值来起作用。LQR 用一行增益取代两者，让 Q 和 R 决定取舍。

<LqrWidget />

<strong>动手实验。</strong>本实验展示 LQR 权重如何在状态误差与控制力之间取舍，并展示 R 减小时峰值力如何接近 10 N 上限。

1. 保持 Q1 = 1.0、Q3 = 10、R = 1.0，读取 K = [−0.92, −1.75, −18.76, −3.03]、峰值力 = 3.28 N。
2. 将 R 设为 0.01，读取 K = [−6.86, −10.95, −59.07, −11.51]、峰值力 = 10.00 N（碰到 10 N 上限）。
3. 将 R 设为 1.0、Q1 设为 100，保持 Q3 = 10，读取 K = [−9.01, −6.54, −29.74, −4.91]、峰值力 = 5.19 N。
4. 将 Q1 设为 1.0、R 设为 10，读取 K = [−0.29, −0.78, −15.51, −2.37]、峰值力 = 2.71 N。
5. 结论：LQR 在状态误差与控制力之间取舍；R 越小，控制作用的代价越低，所有增益增大直到力饱和（R = 0.01 时达到 10 N 上限的 10.00 N，而 R = 1.0 时为 3.28 N）；Q1 越大，位置回正越强（Q1 = 100 使峰值力升到 5.19 N）。

注意位置增益的符号。位置增益是负的，和角度增益同号。按 u = −K x，正的位置误差（小车在目标右侧）会命令正的力，把它继续往右推。这是故意的：控制器先让杆向左倒，再向左追过去。小车倒立摆在位置上是非最小相位（non-minimum phase）的，直接朝目标推反而会把杆推向错误的一侧。

想用更小的力，就增大 R 或减小状态权重。反过来，增大 Q 里的位置权重会得到更紧的定位，也多花力。

<Quiz id="q4" question="哪种改动会让控制器用更小的力？" options={["增大 R", "增大 Q 里的位置权重", "增大 Q 里的角度权重", "缩短时域"]} answer={0} explanation="R 乘的是 u²，R 越大，力就越“贵”：增益变小，峰值力下降，代价是回得更慢或定位更松。增大 Q 的效果相反。" />

## MPC 与滚动时域

第 2 部分让机械臂关节转过 1 rad，力矩上限 0.6 N·m（重力前馈之外），频率 200 Hz，即 T = 5 ms。LQR 用同一套代价，但输出只是被削到上限。模型预测控制（MPC，model predictive control）向前预测 N = 40 步，40 × 5 ms = 0.2 s，在 `|u| ≤ 0.6 N·m` 的约束下优化未来 40 个力矩，只施加第一个，然后重复。这就是滚动时域（receding horizon）。模型用投影梯度下降（projected gradient descent）求解：每个控制周期 150 步，并用上一周期的计划平移一步作为热启动（warm start）。

</SimOutput>

LQR 增益是 K = [15.93, 0.323]，比第 1 课的 PD 增益 2.86 N·m/rad 和 0.115 N·m/(rad/s) 大得多，因为 Q = diag(400, 0.01)、R = 1 把 1 rad 的误差看得比 1 N·m 的力矩贵得多：1 rad 误差的代价是 `400 × 1² = 400`，与 `u² = 400` → `|u| = 20 N·m` 相当（约是 0.6 N·m 限幅的 33 倍、第 1 课 2.86 N·m/rad 量级的 7 倍），所以位置增益会落在 16 附近。增益这么大，动作的大部分时间都顶在上限上，而这恰恰是普通 LQR 不再成立的地方：K 是按输出不被削平算出来的，输出被削平后，实际上的回路增益已经不是设计时的那个。削平之后，LQR 超调 2.24°，0.210 s 起进入 1° 以内；MPC 超调 0.86°，0.160 s 起进入 1° 以内。MPC 在极限上待了 0.165 s，LQR 是 0.175 s。改善存在，但幅度有限；更大的差别在计算量。

MPC 每步的计算量大得多：每个 5 ms 周期要跑 150 次梯度迭代，每次都要沿 40 步做一遍前向推演（rollout）和一遍反向传递，合计约 12,000 次小的状态更新；LQR 只是两次乘法和一次加法。单片机上的 5 ms 截止时间不会变。MPC 算不完，就只能缩短时域、降低控制频率，或者换更快的芯片。降低频率会增加延迟，而延迟会吃掉相位裕度，第 1 课已经看过这件事。

## 模型误差下的观测器对照 oracle

第 5.2 课已经用离散 Riccati 设计 LQR。这一节问另一个问题：增益在**名义** `(A, B)` 上固定之后，若只能测量 `y = C x + 噪声`，在**扰动**对象上反馈会差多少？观测器是离散 Luenberger 拷贝，用极点配置（对本 2 状态对象用 Ackermann），结构指针见 UC Berkeley CE295 第 2 章状态估计。本页**不**为观测器声称离散 Riccati 最优，也**不是** H∞ 设计。

### 特权信息，命名要诚实

三臂在扰动对象上共用同一条过程噪声实现：

| 臂 | 看见什么 | 控制 |
|---|---|---|
| `oracle_fullstate` | 特权真状态 `x` | `u = −K x` |
| `observer_output_feedback` | 只有 `y = C x + v` | Luenberger `x̂`，再 `u = −K x̂` |
| `open_loop` | 无 | `u = 0` |

全状态反馈是 **oracle 上界参考**，不是同信息基线。不要把表读成「同样传感器下全状态更优」。

```python
#!/usr/bin/env python3
"""lqr_model_error_toy.py: observer vs oracle under model error (stdlib).

The observer sees ONLY the scalar measurement y = C x + v. Full-state feedback
is an ORACLE upper reference that reads the privileged true state x — it is
NOT an equal-information baseline. Arm names make that explicit:
  oracle_fullstate          u = -K x_true   (privileged)
  observer_output_feedback  Luenberger xhat from y only; u = -K xhat
  open_loop                 u = 0

Gain K and observer gain L are designed on the NOMINAL (A, B) by pole
placement (Ackermann for this 2-state plant). Evaluation runs on a PERTURBED
plant A_pert with the SAME process-noise realisation across arms. This is
model-error sensitivity awareness, NOT H∞ design, and NOT a discrete-Riccati
optimality claim (lesson 5.2's lqr_sim.py already covers Riccati LQR).

The scored cost is sum (x0² + 0.1 x1² + 0.01 u²): under model error the
observer can look similar on position RMS alone (its lag filters) while
spending far more control effort and leaving more velocity — the cost makes
the privileged-information gap visible without claiming equal-info superiority.

Luenberger / observability structure: UC Berkeley CE295 notes, Ch. 2 State
Estimation (https://ecal.studentorg.berkeley.edu/files/ce295/CH02-StateEstimation.pdf).
Simon, Optimal State Estimation (Wiley, 2006) is the standard KF book citation;
no free full-text URL is claimed.
usage: python3 lqr_model_error_toy.py
"""
from __future__ import annotations

import math
import random

SEED = 52
DT = 0.02                 # s
HORIZON = 250             # steps ≈ 5 s
M = 1.0                   # kg
B_NOM = 0.40              # N·s/m  (design model)
B_PERT = 0.10             # lighter damping on the evaluation plant
PROCESS_STD = 0.04        # shared process noise on velocity (discrete)
MEAS_STD = 0.05           # observer measurement noise on y = position (m)
K_POLES = (0.92, 0.88)
L_POLES = (0.85, 0.80)
# Running cost weights for the scored sum.
QX, QV, RU = 1.0, 0.1, 0.01


def mat_mul(A, B):
    return [[sum(A[i][k] * B[k][j] for k in range(len(B))) for j in range(len(B[0]))]
            for i in range(len(A))]


def eye(n):
    return [[1.0 if i == j else 0.0 for j in range(n)] for i in range(n)]


def scale_mat(s, A):
    return [[s * A[i][j] for j in range(len(A[0]))] for i in range(len(A))]


def cont_to_disc(A_c, B_c, dt):
    """Forward-Euler discretisation (teaching toy; not zoh-exact)."""
    I = eye(len(A_c))
    A = [[I[i][j] + dt * A_c[i][j] for j in range(2)] for i in range(2)]
    B = [[dt * B_c[i][0]] for i in range(2)]
    return A, B


def char_poly_poles(p1, p2):
    return [1.0, -(p1 + p2), p1 * p2]


def adjugate_2(M):
    return [[M[1][1], -M[0][1]], [-M[1][0], M[0][0]]]


def det_2(M):
    return M[0][0] * M[1][1] - M[0][1] * M[1][0]


def ackermann_K(A, B, poles):
    AB = mat_mul(A, B)
    Cc = [[B[0][0], AB[0][0]], [B[1][0], AB[1][0]]]
    assert abs(det_2(Cc)) > 1e-9, "not controllable"
    _, a1, a0 = char_poly_poles(*poles)
    A2 = mat_mul(A, A)
    alphaA = [[A2[i][j] + a1 * A[i][j] + a0 * (1.0 if i == j else 0.0)
               for j in range(2)] for i in range(2)]
    inv = scale_mat(1.0 / det_2(Cc), adjugate_2(Cc))
    eT = inv[1]
    return [eT[0] * alphaA[0][j] + eT[1] * alphaA[1][j] for j in range(2)]


def ackermann_L(A, C, poles):
    CA = mat_mul(C, A)
    Ob = [[C[0][0], C[0][1]], [CA[0][0], CA[0][1]]]
    assert abs(det_2(Ob)) > 1e-9, "not observable"
    _, a1, a0 = char_poly_poles(*poles)
    A2 = mat_mul(A, A)
    alphaA = [[A2[i][j] + a1 * A[i][j] + a0 * (1.0 if i == j else 0.0)
               for j in range(2)] for i in range(2)]
    inv = scale_mat(1.0 / det_2(Ob), adjugate_2(Ob))
    col = [inv[0][1], inv[1][1]]
    return [alphaA[i][0] * col[0] + alphaA[i][1] * col[1] for i in range(2)]


def build(b):
    A_c = [[0.0, 1.0], [0.0, -b / M]]
    B_c = [[0.0], [1.0 / M]]
    return cont_to_disc(A_c, B_c, DT)


def simulate(arm, A_nom, A_pert, B, K, L, C, noise_w, noise_v, x0):
    x = list(x0)
    xhat = [0.0, 0.0]
    xs, us = [], []
    for k in range(HORIZON):
        if arm == "oracle_fullstate":
            # Privileged true state — NOT the same information as the observer.
            u = -(K[0] * x[0] + K[1] * x[1])
        elif arm == "observer_output_feedback":
            u = -(K[0] * xhat[0] + K[1] * xhat[1])
        else:
            u = 0.0
        w = [0.0, noise_w[k]]
        x_next = [A_pert[i][0] * x[0] + A_pert[i][1] * x[1] + B[i][0] * u + w[i]
                  for i in range(2)]
        y = C[0][0] * x[0] + C[0][1] * x[1] + noise_v[k]
        if arm == "observer_output_feedback":
            yhat = C[0][0] * xhat[0] + C[0][1] * xhat[1]
            innov = y - yhat
            xhat = [A_nom[i][0] * xhat[0] + A_nom[i][1] * xhat[1]
                    + B[i][0] * u + L[i] * innov for i in range(2)]
        x = x_next
        xs.append(x[:])
        us.append(u)
    return xs, us


def rms(vals):
    return math.sqrt(sum(v * v for v in vals) / len(vals))


def cost(xs, us):
    return sum(QX * x[0] * x[0] + QV * x[1] * x[1] + RU * u * u
               for x, u in zip(xs, us))


def main() -> None:
    A_nom, B = build(B_NOM)
    A_pert, _ = build(B_PERT)
    C = [[1.0, 0.0]]
    K = ackermann_K(A_nom, B, K_POLES)
    L = ackermann_L(A_nom, C, L_POLES)

    rng = random.Random(SEED)
    noise_w = [rng.gauss(0.0, PROCESS_STD) for _ in range(HORIZON)]
    noise_v = [rng.gauss(0.0, MEAS_STD) for _ in range(HORIZON)]
    x0 = [1.0, 0.0]

    print(f"seed={SEED} dt={DT} horizon={HORIZON} m={M} b_nom={B_NOM} b_pert={B_PERT}")
    print(f"design: pole-placement on nominal (A,B); K_poles={list(K_POLES)} L_poles={list(L_POLES)}")
    print(f"K=[{K[0]:.4f}, {K[1]:.4f}]  L=[{L[0]:.4f}, {L[1]:.4f}]")
    print("oracle_fullstate uses privileged true x; observer uses only y=Cx+noise")
    print("NOT H-infinity; NOT equal-info baseline; shared process noise across arms")
    print(f"cost = sum(x_pos^2 + {QV}*x_vel^2 + {RU}*u^2) over the horizon")
    print()
    hdr = f"{'arm':<28} {'rms_pos':>10} {'rms_vel':>10} {'rms_u':>10} {'cost':>12} {'final_pos':>10}"
    print(hdr)
    results = {}
    for arm in ("open_loop", "oracle_fullstate", "observer_output_feedback"):
        xs, us = simulate(arm, A_nom, A_pert, B, K, L, C, noise_w, noise_v, x0)
        rp, rv, ru = rms([x[0] for x in xs]), rms([x[1] for x in xs]), rms(us)
        c = cost(xs, us)
        results[arm] = (rp, rv, ru, c, xs[-1][0])
        print(f"{arm:<28} {rp:10.4f} {rv:10.4f} {ru:10.4f} {c:12.2f} {xs[-1][0]:10.4f}")

    c_or = results["oracle_fullstate"][3]
    c_ob = results["observer_output_feedback"][3]
    c_ol = results["open_loop"][3]
    assert c_or < c_ob < c_ol, f"cost order oracle<observer<open failed: {c_or}, {c_ob}, {c_ol}"
    assert results["oracle_fullstate"][2] < results["observer_output_feedback"][2], "oracle should spend less |u|"
    print("ok: oracle_fullstate (privileged x) cost < observer_output_feedback cost < open_loop; "
          "model-error sweep on perturbed plant (real hardware 本页未运行)")


if __name__ == "__main__":
    main()
```

## 常见错误

- <strong>Q 和 R 的单位不一致。</strong>把位置从米换成厘米，该状态就乘了 100，同样的 Q 项会多惩罚 10,000 倍。状态保持 SI，单位一改就重新调。
- <strong>把线性模型用到远离工作点的地方。</strong>45.78 这一项是直立附近的线性化。到了 53°，真实动力学已经不同；在 10° 好用的增益，不能证明那么远也成立。
- <strong>设计 LQR 时忽略执行器上限。</strong>不削平的 LQR 假设任何力都出得来。R = 0.01 那组峰值达到 10 N 上限；仿真里要包含饱和，或者改用 MPC 设计。
- <strong>忘了状态的符号约定。</strong>x = [position, speed, angle, rate]，角度从竖直方向量起。把角度符号弄反，控制器就会把杆推倒。调参前先写清正方向。
- <strong>以为 MPC 总是更好。</strong>MPC 每个周期都要模型和求解器。如果约束没被激活、线性模型又足够准，LQR 用少得多的代码和时间就能做到同样的事。
- **把全状态反馈叫作同信息基线。** Oracle 读 `x`，观测器读 `y`，信息不同。
- **只看位置 RMS。** 模型误差下观测器的滞后可能滤了位置，却抬高速度与 `|u|`。用同时计价三者的代价，或并列报告。
- **一次 Δ 扫描就声称 H∞。** 名义极点配置加一个扰动对象是敏感性意识，不是 H∞ 综合。
