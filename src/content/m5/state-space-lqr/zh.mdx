上一课用传递函数描述机械臂关节：一个输入、一个输出。<Term id="state-space">状态空间</Term>（state space）把每个内部状态都放进向量 x，写成 `x' = A x + B u` 这样的矩阵方程。平衡车恰好需要这样：位置、速度、角度、角速度必须同时考虑。这里的对象先从[第 1 课](/zh/month/month-5/control-modelling)的关节开始，再换成第 2 个月的平衡车简化成的小车倒立摆（cart-pole）。

## 两个对象，一种形式

对关节取 x = [q, w]，q 是角度，w = q'，力矩是唯一的输入 u。代入 J = 0.004 kg·m²、B = 0.05 N·m/(rad/s)，方程 J q'' + B q' = u 就变成两行状态方程。第一行是 q' = w。第二行是 w' = −(B/J) w + (1/J) u = −12.5 w + 250 u，所以 A = [[0, 1], [0, −12.5]]，B = [[0], [250]]。一个方程现在同时带着角度和速度。

<Formula label="状态空间：把所有状态装进一个向量">x' = A x + B u</Formula>

小车倒立摆是第 2 个月平衡车（balancer）的极简版：M = 0.5 kg 的小车，m = 0.2 kg 的质点挂在 L = 0.3 m 处，输入是水平力 F。非线性方程（角度从直立量起）是小车加速度 `a = (F + m L ω² sinθ − m g sinθ cosθ)/(M + m − m cos²θ)` 与杆角加速度 `θ'' = (g sinθ − a cosθ)/L`。在直立处线性化：对 θ 求偏导，取 θ = 0、ω = 0 且 `a ≈ F/M`，得到 `−m g/M = −3.92`、`1/M = 2.00`、`(M+m)g/(M L) = 45.78` 和 `−1/(M L) = −6.67`。取 x = [position, speed, angle, rate]，模型打印：

```python
"""lqr_sim.py: state space, LQR on a cart-pole, and a small MPC on the Month 3 arm joint.

Part 1 writes a cart-pole (the Month 2 balancer reduced to its essentials: a cart that is
pushed by a force and a pole balanced on it) in state space, x = [position, speed, angle,
angular rate], linearised about upright, discretised at 100 Hz. LQR gains come from
iterating the discrete Riccati equation. The gains are then tested on the full nonlinear
cart-pole, with the force limited to 10 N.
Part 2 moves the arm joint from lesson 1 (J = 0.004, B = 0.05, gravity cancelled by
feedforward) through 1 rad with a torque limit, comparing LQR whose output is simply
clipped with an MPC that knows the limit. Standard library only.
"""
import math

# ---------------------------------------------------------------- small matrix helpers
def mul(a, b):
    return [[sum(a[i][k] * b[k][j] for k in range(len(b))) for j in range(len(b[0]))] for i in range(len(a))]


def add(a, b, s=1.0):
    return [[a[i][j] + s * b[i][j] for j in range(len(a[0]))] for i in range(len(a))]


def tr(a):
    return [list(r) for r in zip(*a)]


def eye(n, v=1.0):
    return [[v if i == j else 0.0 for j in range(n)] for i in range(n)]


def discretise(a, b, dt):
    """Zero-order-hold discretisation by a Taylor series of the matrix exponential."""
    n = len(a)
    phi, gam, term = eye(n), [[dt * x[0]] for x in b], eye(n)
    for k in range(1, 25):
        term = [[v * dt / k for v in r] for r in mul(term, a)]           # (A dt)^k / k!
        phi = add(phi, term)
        gam = add(gam, [[v * dt / (k + 1)] for v in [r[0] for r in mul(term, b)]])
    return phi, gam


def lqr(phi, gam, q, r, iters=5000):
    """Iterate P = Q + A'PA - A'PB (R + B'PB)^-1 B'PA; return K for u = -K x."""
    p = q
    for _ in range(iters):
        bp = mul(tr(gam), p)
        s = r + mul(bp, gam)[0][0]
        k = [[v / s for v in mul(bp, phi)[0]]]
        p = add(q, mul(tr(phi), mul(p, add(phi, mul(gam, k), -1.0))))
    return k[0]


# ---------------------------------------------------------------- part 1: cart-pole
M, m, L, G, DT, FMAX = 0.5, 0.2, 0.3, 9.81, 0.01, 10.0
A_CP = [[0, 1, 0, 0], [0, 0, -m * G / M, 0], [0, 0, 0, 1], [0, 0, (M + m) * G / (M * L), 0]]
B_CP = [[0], [1 / M], [0], [-1 / (M * L)]]


def cartpole(x, f, dt=0.001):
    """Full nonlinear cart-pole (point mass m at distance L), angle from upright."""
    p, v, th, w = x
    s, c = math.sin(th), math.cos(th)
    acc = (f + m * L * w * w * s - m * G * s * c) / (M + m - m * c * c)
    alpha = (G * s - acc * c) / L
    return [p + v * dt, v + acc * dt, th + w * dt, w + alpha * dt]


def balance(k, deg, seconds=4.0, show=False):
    x = [0.0, 0.0, math.radians(deg), 0.0]
    fpeak = 0.0
    for i in range(int(seconds / DT)):
        f = max(-FMAX, min(FMAX, -sum(ki * xi for ki, xi in zip(k, x))))
        fpeak = max(fpeak, abs(f))
        if show and i % 50 == 0:
            print(f"  t = {i * DT:3.1f} s  x = {x[0]:6.3f} m  angle = {math.degrees(x[2]):6.2f} deg  force = {f:6.2f} N")
        for _ in range(10):
            x = cartpole(x, f)
        if abs(x[2]) > math.pi / 2:
            return None, fpeak
    return x, fpeak


def part1():
    phi, gam = discretise(A_CP, B_CP, DT)
    print("Cart-pole, x = [position, speed, angle, rate], 100 Hz")
    print("  A (continuous) rows:", "  ".join("[" + ", ".join(f"{v:.2f}" for v in r) + "]" for r in A_CP))
    print("  B (continuous):     ", "[" + ", ".join(f"{r[0]:.2f}" for r in B_CP) + "]")
    for name, qd, r in (("Q = diag(1, 1, 10, 1), R = 1   ", (1, 1, 10, 1), 1.0),
                        ("Q = diag(100, 1, 10, 1), R = 1 ", (100, 1, 10, 1), 1.0),
                        ("Q = diag(1, 1, 10, 1), R = 0.01", (1, 1, 10, 1), 0.01)):
        q = [[qd[i] if i == j else 0.0 for j in range(4)] for i in range(4)]
        k = lqr(phi, gam, q, r)
        print(f"{name}  K = [" + ", ".join(f"{v:7.2f}" for v in k) + "]")
        if name.startswith("Q = diag(1, 1, 10, 1), R = 1 "):
            k_first = k
        x, fpeak = balance(k, 10)
        print(f"    from 10 deg: after 4 s x = {x[0]:6.3f} m, angle = {math.degrees(x[2]):5.2f} deg, peak force {fpeak:5.2f} N")
    print("Nonlinear run with the first K, starting 10 deg off upright:")
    balance(k_first, 10, seconds=3.0, show=True)
    biggest = max(d for d in range(1, 90) if balance(k_first, d, seconds=6.0)[0] is not None
                  and abs(balance(k_first, d, seconds=6.0)[0][2]) < 0.01)
    print(f"  largest start angle it still recovers from (force limit {FMAX:.0f} N): {biggest} deg")


# ---------------------------------------------------------------- part 2: MPC on the joint
J, BV, T, UMAX, N = 0.004, 0.05, 0.005, 0.6, 40
PHI_J, GAM_J = discretise([[0, 1], [0, -BV / J]], [[0], [1 / J]], T)


def jstep(x, u):
    return [PHI_J[0][0] * x[0] + PHI_J[0][1] * x[1] + GAM_J[0][0] * u,
            PHI_J[1][0] * x[0] + PHI_J[1][1] * x[1] + GAM_J[1][0] * u]


def mpc(x, target, kq, warm):
    """Minimise sum over N steps of kq*(q-target)^2 + 0.01*w^2 + u^2, |u| <= UMAX,
    by projected gradient descent on the N future torques; return them all."""
    u = warm[1:] + warm[-1:]
    lr = 2e-4
    for _ in range(150):
        xs = [x]
        for uk in u:
            xs.append(jstep(xs[-1], uk))
        lam = [0.0, 0.0]                  # backward pass: gradient of the cost wrt each u
        grad = [0.0] * N
        for k in range(N - 1, -1, -1):
            q, w = xs[k + 1]
            lam = [2 * kq * (q - target) + PHI_J[0][0] * lam[0] + PHI_J[1][0] * lam[1],
                   2 * 0.01 * w + PHI_J[0][1] * lam[0] + PHI_J[1][1] * lam[1]]
            grad[k] = 2 * u[k] + GAM_J[0][0] * lam[0] + GAM_J[1][0] * lam[1]
        u = [max(-UMAX, min(UMAX, uk - lr * g)) for uk, g in zip(u, grad)]
    return u


def move(kind, seconds=1.0):
    k = lqr(PHI_J, GAM_J, [[400.0, 0], [0, 0.01]], 1.0)
    x, target, peak, settle = [0.0, 0.0], 1.0, 0.0, None
    plan = [0.0] * N
    sat = 0
    for i in range(int(seconds / T)):
        if kind == "lqr":
            u = max(-UMAX, min(UMAX, -(k[0] * (x[0] - target) + k[1] * x[1])))
        else:
            plan = mpc(x, target, 400.0, plan)
            u = plan[0]
        sat += abs(u) >= UMAX - 1e-9
        x = jstep(x, u)
        peak = max(peak, x[0])
        if abs(x[0] - target) > math.radians(1):
            settle = None
        elif settle is None:
            settle = (i + 1) * T
    return k, peak, settle, sat * T


def part2():
    print(f"Arm joint, 1 rad move, |torque| <= {UMAX} N*m (on top of gravity feedforward), 200 Hz")
    for kind, label in (("lqr", "LQR, output clipped"), ("mpc", f"MPC, horizon {N} steps")):
        k, peak, settle, sat = move(kind)
        if kind == "lqr":
            print(f"  LQR gains: K = [{k[0]:.2f}, {k[1]:.3f}]  (kp in N*m/rad, kd in N*m/(rad/s))")
        s = f"{settle:.3f} s" if settle else "never"
        print(f"  {label:22s} overshoot {math.degrees(peak - 1.0):5.2f} deg, within 1 deg from {s}, at the limit for {sat:.3f} s")


if __name__ == "__main__":
    part1()
    print()
    part2()
```

## 一行增益，四个状态

线性二次调节器（<Term id="lqr">LQR</Term>，linear quadratic regulator）要的是让整段运行代价最小的输入序列：

<Formula label="LQR 代价：状态误差加控制用力">J = Σ (xᵀ Q x + uᵀ R u)</Formula>

Q 是 4 × 4 矩阵，惩罚状态误差；R 是正标量，惩罚力。单位要一致：位置以 m² 进入，角度以 rad² 进入，力的平方以 N² 进入。增益来自反复迭代离散 Riccati 方程（这里的 A、B 是零阶保持后的矩阵），直到 P 不再变化：

<Formula label="离散 Riccati 迭代和它给出的增益">P ← Q + Aᵀ P A − Aᵀ P B (R + Bᵀ P B)⁻¹ Bᵀ P A,   K = (R + Bᵀ P B)⁻¹ Bᵀ P A</Formula>

控制律是 u = −K x。K 每个状态一项，所以同一行同时用位置、速度、角度和角速度。第 3 个月的串级（cascade）用了两个先后调好的回路：内环管俯仰角，外环管轮速，并通过移动内环的设定值来起作用。LQR 用一行增益取代两者，让 Q 和 R 决定取舍。

<LqrWidget />

<strong>动手实验。</strong>本实验展示 LQR 权重如何在状态误差与控制力之间取舍，并展示 R 减小时峰值力如何接近 10 N 上限。

1. 保持 Q1 = 1.0、Q3 = 10、R = 1.0，读取 K = [−0.92, −1.75, −18.76, −3.03]、峰值力 = 3.28 N。
2. 将 R 设为 0.01，读取 K = [−6.86, −10.95, −59.07, −11.51]、峰值力 = 10.00 N（碰到 10 N 上限）。
3. 将 R 设为 1.0、Q1 设为 100，保持 Q3 = 10，读取 K = [−9.01, −6.54, −29.74, −4.91]、峰值力 = 5.19 N。
4. 将 Q1 设为 1.0、R 设为 10，读取 K = [−0.29, −0.78, −15.51, −2.37]、峰值力 = 2.71 N。
5. 结论：LQR 在状态误差与控制力之间取舍；R 越小，控制作用的代价越低，所有增益增大直到力饱和（R = 0.01 时达到 10 N 上限的 10.00 N，而 R = 1.0 时为 3.28 N）；Q1 越大，位置回正越强（Q1 = 100 使峰值力升到 5.19 N）。

注意位置增益的符号。位置增益是负的，和角度增益同号。按 u = −K x，正的位置误差（小车在目标右侧）会命令正的力，把它继续往右推。这是故意的：控制器先让杆向左倒，再向左追过去。小车倒立摆在位置上是非最小相位（non-minimum phase）的，直接朝目标推反而会把杆推向错误的一侧。

想用更小的力，就增大 R 或减小状态权重。反过来，增大 Q 里的位置权重会得到更紧的定位，也多花力。

<Quiz id="q4" question="哪种改动会让控制器用更小的力？" options={["增大 R", "增大 Q 里的位置权重", "增大 Q 里的角度权重", "缩短时域"]} answer={0} explanation="R 乘的是 u²，R 越大，力就越“贵”：增益变小，峰值力下降，代价是回得更慢或定位更松。增大 Q 的效果相反。" />

## MPC 与滚动时域

第 2 部分让机械臂关节转过 1 rad，力矩上限 0.6 N·m（重力前馈之外），频率 200 Hz，即 T = 5 ms。LQR 用同一套代价，但输出只是被削到上限。模型预测控制（MPC，model predictive control）向前预测 N = 40 步，40 × 5 ms = 0.2 s，在 `|u| ≤ 0.6 N·m` 的约束下优化未来 40 个力矩，只施加第一个，然后重复。这就是滚动时域（receding horizon）。模型用投影梯度下降（projected gradient descent）求解：每个控制周期 150 步，并用上一周期的计划平移一步作为热启动（warm start）。

LQR 增益是 K = [15.93, 0.323]，比第 1 课的 PD 增益 2.86 N·m/rad 和 0.115 N·m/(rad/s) 大得多，因为 Q = diag(400, 0.01)、R = 1 把 1 rad 的误差看得比 1 N·m 的力矩贵得多：1 rad 误差的代价是 `400 × 1² = 400`，与 `u² = 400` → `|u| = 20 N·m` 相当（约是 0.6 N·m 限幅的 33 倍、第 1 课 2.86 N·m/rad 量级的 7 倍），所以位置增益会落在 16 附近。增益这么大，动作的大部分时间都顶在上限上，而这恰恰是普通 LQR 不再成立的地方：K 是按输出不被削平算出来的，输出被削平后，实际上的回路增益已经不是设计时的那个。削平之后，LQR 超调 2.24°，0.210 s 起进入 1° 以内；MPC 超调 0.86°，0.160 s 起进入 1° 以内。MPC 在极限上待了 0.165 s，LQR 是 0.175 s。改善存在，但幅度有限；更大的差别在计算量。

MPC 每步的计算量大得多：每个 5 ms 周期要跑 150 次梯度迭代，每次都要沿 40 步做一遍前向推演（rollout）和一遍反向传递，合计约 12,000 次小的状态更新；LQR 只是两次乘法和一次加法。单片机上的 5 ms 截止时间不会变。MPC 算不完，就只能缩短时域、降低控制频率，或者换更快的芯片。降低频率会增加延迟，而延迟会吃掉相位裕度，第 1 课已经看过这件事。

## 模型误差下的观测器对照 oracle

第 5.2 课已经用离散 Riccati 设计 LQR。这一节问另一个问题：增益在**名义** `(A, B)` 上固定之后，若只能测量 `y = C x + 噪声`，在**扰动**对象上反馈会差多少？观测器是离散 Luenberger 拷贝，用极点配置（对本 2 状态对象用 Ackermann），结构指针见 UC Berkeley CE295 第 2 章状态估计。本页**不**为观测器声称离散 Riccati 最优，也**不是** H∞ 设计。

### 特权信息，命名要诚实

三臂在扰动对象上共用同一条过程噪声实现：

| 臂 | 看见什么 | 控制 |
|---|---|---|
| `oracle_fullstate` | 特权真状态 `x` | `u = −K x` |
| `observer_output_feedback` | 只有 `y = C x + v` | Luenberger `x̂`，再 `u = −K x̂` |
| `open_loop` | 无 | `u = 0` |

全状态反馈是 **oracle 上界参考**，不是同信息基线。不要把表读成「同样传感器下全状态更优」。

#!/usr/bin/env python3
"""lqr_model_error_toy.py: observer vs oracle under model error (stdlib).

The observer sees ONLY the scalar measurement y = C x + v. Full-state feedback
is an ORACLE upper reference that reads the privileged true state x — it is
NOT an equal-information baseline. Arm names make that explicit:
  oracle_fullstate          u = -K x_true   (privileged)
  observer_output_feedback  Luenberger xhat from y only; u = -K xhat
  open_loop                 u = 0

Gain K and observer gain L are designed on the NOMINAL (A, B) by pole
placement (Ackermann for this 2-state plant). Evaluation runs on a PERTURBED
plant A_pert with the SAME process-noise realisation across arms. This is
model-error sensitivity awareness, NOT H∞ design, and NOT a discrete-Riccati
optimality claim (lesson 5.2's lqr_sim.py already covers Riccati LQR).

The scored cost is sum (x0² + 0.1 x1² + 0.01 u²): under model error the
observer can look similar on position RMS alone (its lag filters) while
spending far more control effort and leaving more velocity — the cost makes
the privileged-information gap visible without claiming equal-info superiority.

Luenberger / observability structure: UC Berkeley CE295 notes, Ch. 2 State
Estimation (https://ecal.studentorg.berkeley.edu/files/ce295/CH02-StateEstimation.pdf).
Simon, Optimal State Estimation (Wiley, 2006) is the standard KF book citation;
no free full-text URL is claimed.
usage: python3 lqr_model_error_toy.py
"""
from __future__ import annotations

import math
import random

SEED = 52
DT = 0.02                 # s
HORIZON = 250             # steps ≈ 5 s
M = 1.0                   # kg
B_NOM = 0.40              # N·s/m  (design model)
B_PERT = 0.10             # lighter damping on the evaluation plant
PROCESS_STD = 0.04        # shared process noise on velocity (discrete)
MEAS_STD = 0.05           # observer measurement noise on y = position (m)
K_POLES = (0.92, 0.88)
L_POLES = (0.85, 0.80)
# Running cost weights for the scored sum.
QX, QV, RU = 1.0, 0.1, 0.01


def mat_mul(A, B):
    return [[sum(A[i][k] * B[k][j] for k in range(len(B))) for j in range(len(B[0]))]
            for i in range(len(A))]


def eye(n):
    return [[1.0 if i == j else 0.0 for j in range(n)] for i in range(n)]


def scale_mat(s, A):
    return [[s * A[i][j] for j in range(len(A[0]))] for i in range(len(A))]


def cont_to_disc(A_c, B_c, dt):
    """Forward-Euler discretisation (teaching toy; not zoh-exact)."""
    I = eye(len(A_c))
    A = [[I[i][j] + dt * A_c[i][j] for j in range(2)] for i in range(2)]
    B = [[dt * B_c[i][0]] for i in range(2)]
    return A, B


def char_poly_poles(p1, p2):
    return [1.0, -(p1 + p2), p1 * p2]


def adjugate_2(M):
    return [[M[1][1], -M[0][1]], [-M[1][0], M[0][0]]]


def det_2(M):
    return M[0][0] * M[1][1] - M[0][1] * M[1][0]


def ackermann_K(A, B, poles):
    AB = mat_mul(A, B)
    Cc = [[B[0][0], AB[0][0]], [B[1][0], AB[1][0]]]
    assert abs(det_2(Cc)) > 1e-9, "not controllable"
    _, a1, a0 = char_poly_poles(*poles)
    A2 = mat_mul(A, A)
    alphaA = [[A2[i][j] + a1 * A[i][j] + a0 * (1.0 if i == j else 0.0)
               for j in range(2)] for i in range(2)]
    inv = scale_mat(1.0 / det_2(Cc), adjugate_2(Cc))
    eT = inv[1]
    return [eT[0] * alphaA[0][j] + eT[1] * alphaA[1][j] for j in range(2)]


def ackermann_L(A, C, poles):
    CA = mat_mul(C, A)
    Ob = [[C[0][0], C[0][1]], [CA[0][0], CA[0][1]]]
    assert abs(det_2(Ob)) > 1e-9, "not observable"
    _, a1, a0 = char_poly_poles(*poles)
    A2 = mat_mul(A, A)
    alphaA = [[A2[i][j] + a1 * A[i][j] + a0 * (1.0 if i == j else 0.0)
               for j in range(2)] for i in range(2)]
    inv = scale_mat(1.0 / det_2(Ob), adjugate_2(Ob))
    col = [inv[0][1], inv[1][1]]
    return [alphaA[i][0] * col[0] + alphaA[i][1] * col[1] for i in range(2)]


def build(b):
    A_c = [[0.0, 1.0], [0.0, -b / M]]
    B_c = [[0.0], [1.0 / M]]
    return cont_to_disc(A_c, B_c, DT)


def simulate(arm, A_nom, A_pert, B, K, L, C, noise_w, noise_v, x0):
    x = list(x0)
    xhat = [0.0, 0.0]
    xs, us = [], []
    for k in range(HORIZON):
        if arm == "oracle_fullstate":
            # Privileged true state — NOT the same information as the observer.
            u = -(K[0] * x[0] + K[1] * x[1])
        elif arm == "observer_output_feedback":
            u = -(K[0] * xhat[0] + K[1] * xhat[1])
        else:
            u = 0.0
        w = [0.0, noise_w[k]]
        x_next = [A_pert[i][0] * x[0] + A_pert[i][1] * x[1] + B[i][0] * u + w[i]
                  for i in range(2)]
        y = C[0][0] * x[0] + C[0][1] * x[1] + noise_v[k]
        if arm == "observer_output_feedback":
            yhat = C[0][0] * xhat[0] + C[0][1] * xhat[1]
            innov = y - yhat
            xhat = [A_nom[i][0] * xhat[0] + A_nom[i][1] * xhat[1]
                    + B[i][0] * u + L[i] * innov for i in range(2)]
        x = x_next
        xs.append(x[:])
        us.append(u)
    return xs, us


def rms(vals):
    return math.sqrt(sum(v * v for v in vals) / len(vals))


def cost(xs, us):
    return sum(QX * x[0] * x[0] + QV * x[1] * x[1] + RU * u * u
               for x, u in zip(xs, us))


def main() -> None:
    A_nom, B = build(B_NOM)
    A_pert, _ = build(B_PERT)
    C = [[1.0, 0.0]]
    K = ackermann_K(A_nom, B, K_POLES)
    L = ackermann_L(A_nom, C, L_POLES)

    rng = random.Random(SEED)
    noise_w = [rng.gauss(0.0, PROCESS_STD) for _ in range(HORIZON)]
    noise_v = [rng.gauss(0.0, MEAS_STD) for _ in range(HORIZON)]
    x0 = [1.0, 0.0]

    print(f"seed={SEED} dt={DT} horizon={HORIZON} m={M} b_nom={B_NOM} b_pert={B_PERT}")
    print(f"design: pole-placement on nominal (A,B); K_poles={list(K_POLES)} L_poles={list(L_POLES)}")
    print(f"K=[{K[0]:.4f}, {K[1]:.4f}]  L=[{L[0]:.4f}, {L[1]:.4f}]")
    print("oracle_fullstate uses privileged true x; observer uses only y=Cx+noise")
    print("NOT H-infinity; NOT equal-info baseline; shared process noise across arms")
    print(f"cost = sum(x_pos^2 + {QV}*x_vel^2 + {RU}*u^2) over the horizon")
    print()
    hdr = f"{'arm':<28} {'rms_pos':>10} {'rms_vel':>10} {'rms_u':>10} {'cost':>12} {'final_pos':>10}"
    print(hdr)
    results = {}
    for arm in ("open_loop", "oracle_fullstate", "observer_output_feedback"):
        xs, us = simulate(arm, A_nom, A_pert, B, K, L, C, noise_w, noise_v, x0)
        rp, rv, ru = rms([x[0] for x in xs]), rms([x[1] for x in xs]), rms(us)
        c = cost(xs, us)
        results[arm] = (rp, rv, ru, c, xs[-1][0])
        print(f"{arm:<28} {rp:10.4f} {rv:10.4f} {ru:10.4f} {c:12.2f} {xs[-1][0]:10.4f}")

    c_or = results["oracle_fullstate"][3]
    c_ob = results["observer_output_feedback"][3]
    c_ol = results["open_loop"][3]
    assert c_or < c_ob < c_ol, f"cost order oracle<observer<open failed: {c_or}, {c_ob}, {c_ol}"
    assert results["oracle_fullstate"][2] < results["observer_output_feedback"][2], "oracle should spend less |u|"
    print("ok: oracle_fullstate (privileged x) cost < observer_output_feedback cost < open_loop; "
          "model-error sweep on perturbed plant (real hardware 本页未运行)")


if __name__ == "__main__":
    main()

## 常见错误

- <strong>Q 和 R 的单位不一致。</strong>把位置从米换成厘米，该状态就乘了 100，同样的 Q 项会多惩罚 10,000 倍。状态保持 SI，单位一改就重新调。
- <strong>把线性模型用到远离工作点的地方。</strong>45.78 这一项是直立附近的线性化。到了 53°，真实动力学已经不同；在 10° 好用的增益，不能证明那么远也成立。
- <strong>设计 LQR 时忽略执行器上限。</strong>不削平的 LQR 假设任何力都出得来。R = 0.01 那组峰值达到 10 N 上限；仿真里要包含饱和，或者改用 MPC 设计。
- <strong>忘了状态的符号约定。</strong>x = [position, speed, angle, rate]，角度从竖直方向量起。把角度符号弄反，控制器就会把杆推倒。调参前先写清正方向。
- <strong>以为 MPC 总是更好。</strong>MPC 每个周期都要模型和求解器。如果约束没被激活、线性模型又足够准，LQR 用少得多的代码和时间就能做到同样的事。
- **把全状态反馈叫作同信息基线。** Oracle 读 `x`，观测器读 `y`，信息不同。
- **只看位置 RMS。** 模型误差下观测器的滞后可能滤了位置，却抬高速度与 `|u|`。用同时计价三者的代价，或并列报告。
- **一次 Δ 扫描就声称 H∞。** 名义极点配置加一个扰动对象是敏感性意识，不是 H∞ 综合。

## 离散 LQR 与零阶保持

第 5.2 课已在倒立摆上迭代离散 Riccati。本节问更窄的采样问题：若设计**连续** LQR 增益并用**零阶保持（ZOH）**保持，与在同一采样周期设计的**离散** LQR 相比如何？这**不是** Bode / 裕度重做，也**不只是**再推一遍连续 Riccati。

- **Underactuated Robotics — LQR（Tedrake）：** 离散时间 Riccati 差分方程与无穷时域 **DARE** 不动点；离散与连续 LQR 对照。数字 以原文为准。

### 离散化与 Riccati 假设

**ZOH** 把连续对象在 `T_s` 上的输入当作分段常数。本玩具用增广块 `[[A_c, B_c],[0, 0]] * T_s` 的矩阵指数（scaling-and-squaring + Taylor）构造 `(A_d, B_d)`——不是前向欧拉。离散阶段权取 `Q_d ≈ Q_c T_s`、`R_d ≈ R_c T_s`，使求和近似连续积分代价。

离散增益来自迭代

<Formula label="离散 Riccati 迭代（DARE 映射）">P ← Q + AᵀPA − AᵀPB(R + BᵀPB)⁻¹BᵀPA</Formula>

直至 `‖ΔP‖_F < 1e-12`（显式容差）。**稳定性判据：** `A_d − B_d K` 的特征值须满足 `|λ| < 1`。连续 `K_c` 来自 Kleinman CARE 迭代；在 ZOH 下采样间隔内保持恒定。保持期内用 ≪ `T_s` 的子步积分（测试中检查步长稳定性）。

**指标盲区：** 无穷时域风格代价与沉降**不计**更细 `T_s` 的 CPU / 截止期代价——C 的细采样是**预算**，不是免费的。选速率时要带上这一限制。

### 公平设置

`discrete_lqr_zoh_toy.py` 是**合成**的，只用标准库。每个臂共享**同一**连续 `(A_c, B_c)`、**同一** `Q_c, R_c`，且每种种子下按绝对子步索引的过程噪声序列**相同**（公共随机数）。

| 臂 | 控制器 |
|---|---|
| A_cont_zoh | 连续 `K_c`，在制度 `T_s` 下 ZOH 保持 |
| B_disc | 在同一 `T_s` 设计的离散 `K_d` |
| C_disc_fine | 在细 `T_s` 设计并施加的离散 `K_d` |

制度：`coarse_Ts`（按保持周期设计的离散优于连续+ZOH）与 `fine_Ts`（`T_s→0` 时**接近打平 / 差距很小**——剩余差距来自保持 `K_c` 相对 DARE `K_d` 以及共享噪声；并非真正零差距）。指标：代价、沉降时间、`max|λ|`。五多种子 mean±sd。

Shared continuous plant (A_c, B_c) and shared Q, R meaning for every arm.
Process-noise draws are **common random numbers**: for a given seed, A/B/C
see the identical noise sequence (no arm gets a quieter plant).

Discretization: exact **zero-order hold (ZOH)** via the matrix exponential
of the augmented [[A_c, B_c],[0,0]] block (scaling-and-squaring + Taylor),
not forward-Euler.

Discrete Riccati: iterate the DARE map until ‖ΔP‖_F < RICCATI_TOL (explicit).
Continuous K_c from Kleinman iteration on the CARE (stabilizing Lyapunov steps).

Arms:
  A_cont_zoh   continuous LQR K_c held with ZOH at the regime T_s
  B_disc       discrete LQR K_d designed at that same T_s
  C_disc_fine  discrete LQR designed and applied at fine T_s (always)

  coarse_Ts — discrete B beats cont+ZOH A (design at the hold rate matters)
  fine_Ts   — A and B near-tie as T_s→0 (remaining small gap from hold + noise)

Stability criterion: closed-loop eigenvalues of (A_d − B_d K) have modulus < 1
(tested for B and C). Metrics: infinite-horizon-style cost, settling steps,
max |λ|. ≠ Bode redo; ≠ continuous Riccati-only lesson. Real plant: 本页未运行.

# Continuous plant: unit mass + damping  ẍ = −b ẋ + u
M = 1.0
B_DAMP = 0.40
A_C = [[0.0, 1.0], [0.0, -B_DAMP / M]]

# Shared continuous cost meaning; discrete stage cost ≈ integral ≈ Q*Ts, R*Ts

SETTLE_EPS = 0.05

RICCATI_MAX_ITER = 10_000
EXPM_SQUARINGS_CAP = 8

def mat_add(A, B):
    return [[A[i][j] + B[i][j] for j in range(len(A[0]))] for i in range(len(A))]


def mat_sub(A, B):
    return [[A[i][j] - B[i][j] for j in range(len(A[0]))] for i in range(len(A))]


def mat_scale(s, A):

    return [
        [sum(A[i][k] * B[k][j] for k in range(len(B))) for j in range(len(B[0]))]
        for i in range(len(A))
    ]


def mat_T(A):
    return [[A[j][i] for j in range(len(A))] for i in range(len(A[0]))]


def eye(n):
    return [[1.0 if i == j else 0.0 for j in range(n)] for i in range(n)]


def frobenius(A):
    return math.sqrt(sum(A[i][j] * A[i][j] for i in range(len(A)) for j in range(len(A[0]))))


def solve_2x2(M, v):
    det = M[0][0] * M[1][1] - M[0][1] * M[1][0]
    if abs(det) < 1e-18:
        raise ZeroDivisionError("singular 2x2")
    return [
        (M[1][1] * v[0] - M[0][1] * v[1]) / det,
        (-M[1][0] * v[0] + M[0][0] * v[1]) / det,
    ]


def mat_inv_2(M):
    det = M[0][0] * M[1][1] - M[0][1] * M[1][0]
    return [[M[1][1] / det, -M[0][1] / det], [-M[1][0] / det, M[0][0] / det]]


def expm(M):
    """Scaling-and-squaring + Taylor; M square. Stdlib-only ZOH building block."""
    n = len(M)
    # scale so ||M/2^s|| small
    norm = max(sum(abs(M[i][j]) for j in range(n)) for i in range(n))
    s = 0
    while norm > 0.5 and s < EXPM_SQUARINGS_CAP:
        M = mat_scale(0.5, M)
        norm *= 0.5
        s += 1
    # Taylor sum_k M^k / k!
    term = eye(n)
    total = eye(n)
    for k in range(1, 20):
        term = mat_scale(1.0 / k, mat_mul(term, M))
        total = mat_add(total, term)
        if frobenius(term) < 1e-16:
            break
    for _ in range(s):
        total = mat_mul(total, total)
    return total


def zoh_discretize(A_c, B_c, ts: float):
    """Exact ZOH: exp([[A,B],[0,0]]*ts) → (A_d, B_d)."""
    n = len(A_c)
    m = len(B_c[0])
    aug = [[0.0] * (n + m) for _ in range(n + m)]
    for i in range(n):
        for j in range(n):
            aug[i][j] = A_c[i][j] * ts
        for j in range(m):
            aug[i][n + j] = B_c[i][j] * ts
    E = expm(aug)
    A_d = [[E[i][j] for j in range(n)] for i in range(n)]
    B_d = [[E[i][n + j] for j in range(m)] for i in range(n)]
    return A_d, B_d


def dare_iterate(A, B, Q, R, tol=RICCATI_TOL):
    """Discrete Riccati iteration → (P, K, n_iter). Explicit convergence tol."""
    n = len(A)
    P = [row[:] for row in Q]
    K = [[0.0] * n]
    for it in range(1, RICCATI_MAX_ITER + 1):
        # K = (R + B'P B)^{-1} B' P A
        BtP = mat_mul(mat_T(B), P)
        S = mat_add([[R]], mat_mul(BtP, B))  # 1x1 for our SISO B
        BtPA = mat_mul(BtP, A)
        # S is 1x1
        invS = 1.0 / S[0][0]
        K_new = [[invS * BtPA[0][j] for j in range(n)]]
        # P = Q + A'P A - A'P B K
        AtP = mat_mul(mat_T(A), P)
        P_new = mat_add(Q, mat_sub(mat_mul(AtP, A), mat_mul(mat_mul(AtP, B), K_new)))
        if frobenius(mat_sub(P_new, P)) < tol:
            return P_new, K_new, it
        P, K = P_new, K_new
    return P, K, RICCATI_MAX_ITER


def care_kleinman(A, B, Q, R, tol=1e-12):
    """Continuous ARE via Kleinman iteration; returns (P, K)."""
    n = len(A)
    # Start from a stabilizing gain (LQR-ish PD on this plant)
    K = [[4.0, 2.0]]
    P = eye(n)
    for _ in range(200):
        # A_cl = A - B K
        A_cl = mat_sub(A, mat_mul(B, K))
        # Solve Lyapunov A_cl' P + P A_cl = -(Q + K' R K) for 2x2
        QK = mat_add(Q, mat_scale(R, mat_mul(mat_T(K), K)))
        P_new = solve_lyapunov_2(A_cl, QK)
        # K = R^{-1} B' P
        BtP = mat_mul(mat_T(B), P_new)
        K_new = [[BtP[0][0] / R, BtP[0][1] / R]]
        if frobenius(mat_sub(P_new, P)) < tol and frobenius(mat_sub(K_new, K)) < tol:
            return P_new, K_new
        P, K = P_new, K_new
    return P, K


def solve_lyapunov_2(A, Q):
    """Solve A'P + P A = -Q for 2×2 P (symmetric) via 3×3 linear system."""
    # P = [[p, q],[q, r]]; expand
    a, b = A[0][0], A[0][1]
    c, d = A[1][0], A[1][1]
    # coeffs for [p,q,r]
    # (1,1): 2a p + 2c q = -Q00
    # (1,2): b p + (a+d) q + c r = -Q01
    # (2,2): 2b q + 2d r = -Q11
    M = [
        [2 * a, 2 * c, 0.0],
        [b, a + d, c],
        [0.0, 2 * b, 2 * d],
    ]
    rhs = [-Q[0][0], -Q[0][1], -Q[1][1]]
    # Gaussian elimination 3x3
    A3 = [M[i][:] + [rhs[i]] for i in range(3)]
    for col in range(3):
        piv = max(range(col, 3), key=lambda r: abs(A3[r][col]))
        A3[col], A3[piv] = A3[piv], A3[col]
        div = A3[col][col]
        if abs(div) < 1e-18:
            raise ZeroDivisionError("Lyapunov singular")
        for j in range(col, 4):
            A3[col][j] /= div
        for r in range(3):
            if r == col:
                continue
            fac = A3[r][col]
            for j in range(col, 4):
                A3[r][j] -= fac * A3[col][j]
    p, q, r = A3[0][3], A3[1][3], A3[2][3]
    return [[p, q], [q, r]]


def eig_moduli_2(M):
    """|λ| of 2×2 real matrix via characteristic polynomial."""
    tr = M[0][0] + M[1][1]
    det = M[0][0] * M[1][1] - M[0][1] * M[1][0]
    disc = tr * tr - 4.0 * det
    if disc >= 0.0:
        s = math.sqrt(disc)
        return [abs(0.5 * (tr + s)), abs(0.5 * (tr - s))]
    # complex conjugate: |λ| = sqrt(det) when det>0 for stable oscillation
    return [math.sqrt(abs(det)), math.sqrt(abs(det))]


def stage_QR(ts: float):
    """Map continuous Q,R meaning to discrete stage weights ≈ ∫ over Ts."""
    Qd = mat_scale(ts, Q_C)
    Rd = R_C * ts
    return Qd, Rd


def shared_noise_series(seed: int) -> list[float]:
    """Common random numbers: one accel-noise sample per SUBSTEP for the horizon.

    Every arm indexes the **same** series by absolute substep, so changing the
    hold period does not give anyone a quieter plant.
    """
    n = int(round(HORIZON_S / SUBSTEP))
    rng = random.Random(50_000 + seed)
    return [rng.gauss(0.0, PROCESS_STD) for _ in range(n)]


def simulate(
    K, ts: float, seed: int, noise: list[float] | None = None
) -> tuple[float, float, list[float]]:
    """ZOH hold of u=-Kx at period ts; integrate continuous plant with SUBSTEP.

    Returns (cost, settle_time_s, noise_series). Cost ≈ Σ (x'Q_c x + R_c u²)·dt.
    """
    if noise is None:
        noise = shared_noise_series(seed)
    n_hold = int(round(HORIZON_S / ts))
    n_sub_per = max(1, int(round(ts / SUBSTEP)))
    dt = ts / n_sub_per
    x = [1.0, 0.0]
    cost = 0.0
    settled_at = None
    sub_i = 0
    for k in range(n_hold):
        u = -(K[0][0] * x[0] + K[0][1] * x[1])
        for _ in range(n_sub_per):
            w = noise[sub_i] if sub_i < len(noise) else 0.0
            sub_i += 1
            xdot0 = A_C[0][0] * x[0] + A_C[0][1] * x[1] + B_C[0][0] * (u + w)
            xdot1 = A_C[1][0] * x[0] + A_C[1][1] * x[1] + B_C[1][0] * (u + w)
            xQx = Q_C[0][0] * x[0] * x[0] + Q_C[1][1] * x[1] * x[1]
            cost += (xQx + R_C * u * u) * dt
            x = [x[0] + dt * xdot0, x[1] + dt * xdot1]
        if settled_at is None and abs(x[0]) < SETTLE_EPS and abs(x[1]) < SETTLE_EPS:
            settled_at = (k + 1) * ts
    if settled_at is None:
        settled_at = HORIZON_S
    return cost, settled_at, noise


def design_cont_K():
    _, K = care_kleinman(A_C, B_C, Q_C, R_C)
    return K


def design_disc_K(ts: float):
    A_d, B_d = zoh_discretize(A_C, B_C, ts)
    Qd, Rd = stage_QR(ts)
    P, K, n_it = dare_iterate(A_d, B_d, Qd, Rd)
    mods = eig_moduli_2(mat_sub(A_d, mat_mul(B_d, K)))
    return K, max(mods), n_it, A_d, B_d


def mean_sd(xs: list[float]) -> tuple[float, float]:
    n = len(xs)
    mu = sum(xs) / n
    var = sum((x - mu) ** 2 for x in xs) / max(1, n - 1)
    return mu, math.sqrt(var)

## LQG：卡尔曼滤波器配同一个 LQR 增益

<Resource id="wiki-lqg" />

上面的观测器一节手工放置 Luenberger 极点（Ackermann，极点 0.85 与 0.80），再拿它去对付阻尼写错的对象。本节问的是另一件事：如果对象模型**完全已知**，但噪声不为零，估计器增益应该从哪里来？维基百科 *Linear–quadratic–Gaussian control*（已读正文，rev 1337444202；以原文为准）给出了线性系统、加性高斯白噪声下的答案：最优输出反馈控制器**唯一**，就是一个**卡尔曼滤波器**（线性二次估计器）加上一个 **LQR** 增益。**分离原理**说估计器和状态反馈可以分开设计。卡尔曼增益来自过程噪声协方差 V 与测量噪声协方差 W，**而不是**手放的极点。条目也提醒：LQG 最优并不自动保证良好的鲁棒性。

本节**不是**又一个极点配置观测器，也**不是**上面模型误差玩具的重做：这里的对象**没有模型误差**。本节**不**作任何 H∞ 结论。所有臂用**同一个** K，只换估计器。

<Formula label="离散 LQG 稳态增益（维基百科的离散时间形式）">K = (BᵀSB + R)⁻¹BᵀSA;   L = PCᵀ(CPCᵀ + W)⁻¹;   x̂ ← (A x̂ + B u) + L (y_next − C (A x̂ + B u));   u = −K x̂</Formula>

**算例（手算，玩具也会打印）。** 卡尔曼递推从 P0 = diag(0.25, 0.01) 出发，用本课的 0.05 m 传感器噪声，所以 W = 0.0025。只测位置，第一步增益 L1 = [0.25 / (0.25 + 0.0025), 0] = [0.9901, 0.0000]：先验太不确定，滤波器几乎完全相信第一次读数。迭代到稳态，增益落在 [0.1576, 0.6712]。同一对象上的 LQR 增益 K = [9.5023, 5.0042]，max|eig(A − B K)| = 0.9464。本页的观测器极点给出 Lp = [0.3420, 1.3632]，和上面观测器一节的 L 相同，max|eig(A − Lp C)| = 0.8500。匹配的卡尔曼滤波器 max|eig(A − L C)| = 0.9215。这些特征值不是选出来的，而是由 V 和 W 算出来的。

**积分步长。** 两个 Riccati 映射都迭代到最大变化小于 1e-12。对象就是观测器玩具里前向欧拉离散的质量-阻尼系统，DT = 0.02 s 之外没有另外的步长判据。

### 公平设置

`lqg_kalman_lqr_separation_toy.py` 是**合成**的，只用标准库。**模型假设:** 上面观测器玩具的质量-阻尼对象（DT = 0.02 s，M = 1.0 kg，阻尼 0.40 N·s/m），完全已知。过程噪声只作用在速度上（SIG_W = 0.04，所以 V = diag(0, SIG_W²)），只测位置（W = SIG_V²），代价权重 Q = diag(1.0, 0.1)、R = 0.01，时长 500 步。卡尔曼臂用维基百科的离散形式，用最新一次测量做校正；Luenberger 臂保留观测器玩具的预测器形式，用上一次测量做校正。这个时序差别也是比较的一部分。所有数值都是**声明的玩具参数**。

| 臂 | 估计器 | 控制 |
|---|---|---|
| A_oracle | 无：直接读取特权真实状态 | u = −K x |
| B_lqg | 用该工况真实 V、W 的卡尔曼滤波器 | u = −K x̂ |
| C_luen_page | Luenberger，本页极点 0.85 / 0.80 | u = −K x̂ |
| D_luen_fast | Luenberger，随手取的快极点 0.20 / 0.20 | u = −K x̂ |
| E_kf_misspec | 按 V × 0.01、W × 100 设计的卡尔曼：过分相信模型 | u = −K x̂ |

"""LQG separation toy: a Kalman filter plus an LQR gain on a known linear plant.

Synthetic, stdlib only, deterministic.  Not H-infinity, not a real IMU or
plant (本页未运行).  Not a redo of the lesson's Luenberger-versus-oracle
model-error toy: there is NO model error here, the plant is known exactly.
The full-state arm is a privileged oracle, not an equal-information baseline.

Plant (same mass-damper as the lesson's observer toy, forward-Euler, declared):
DT = 0.02 s, M = 1.0 kg, B_DAMP = 0.40 N*s/m.  A = [[1, DT],
[0, 1 - B_DAMP*DT/M]], B = [[0], [DT/M]], C = [1, 0] (position measured).
x_{i+1} = A x_i + B u_i + v_i, y_i = C x_i + w_i, with v_i = [0, SIG_W*e1_i]
(process noise on velocity only, so V = diag(0, SIG_W^2)) and w_i =
SIG_V*e2_i (W = SIG_V^2).  SIG_W = 0.04 (declared).  Cost weights
Q = diag(1.0, 0.1), R = 0.01.  HORIZON = 500 steps.

Separation principle: the estimator and the state-feedback gain are designed
independently.  The Kalman gain comes from the noise covariances V and W, not
from placed poles.

LQR gain K: iterate the discrete Riccati map S <- A'SA - A'SB (B'SB + R)^-1
B'SA + Q from S = Q until the max abs change < 1e-12; K = (B'SB + R)^-1 B'SA.
Kalman gain (steady state, predictive form exactly as Wikipedia's discrete
LQG): iterate P <- A (P - P C' (C P C' + W)^-1 C P) A' + V from
P = diag(0.25, 0.01) until the max abs change < 1e-12; L = P C' (C P C' + W)^-1.
Estimator update each step: x_pred = A xhat + B u; after measuring y_{i+1}:
xhat = x_pred + L (y_{i+1} - C x_pred).  Control u_i = -K xhat_i.

Luenberger arms use the lesson's existing predictor form:
xhat_{i+1} = A xhat_i + B u_i + Lp (y_i - C xhat_i), with Lp from Ackermann
pole placement on (A, C).

Arms (all start from the same x0 and xhat0 = [0, 0] = E[x0]):
  A_oracle     : u = -K x (privileged true state).
  B_lqg        : Kalman with the TRUE V, W of the regime, then u = -K xhat.
  C_luen_page  : Luenberger with the lesson's observer poles (0.85, 0.80).
  D_luen_fast  : Luenberger with casually fast poles (0.20, 0.20).
  E_kf_misspec : Kalman designed with misspecified covariances
                 V_design = 0.01*V_true and W_design = 100*W_true (trusts the
                 model far too much).

Common random numbers: for each (regime, seed) draw ONCE from
random.Random(f"{regime}/{seed}"): x0 = [0.5*g, 0.1*g] (two gauss(0,1) draws,
so P0 = diag(0.25, 0.01)), then HORIZON values e1 and HORIZON+1 values e2
(standard normal), shared by all arms; each regime scales e2 by its SIG_V.

Regimes (true measurement noise SIG_V): "matched" 0.05 (the lesson's
MEAS_STD); "noisy_sensor" 0.25; "quiet_sensor" 0.005.

Metrics per seed: cost = mean over the horizon of (x0^2 + 0.1*x1^2 +
0.01*u^2); est_rms = rms of the position estimation error (0 for A_oracle);
u_rms = rms of u.

Pairs, every metric in every regime: B-A, B-C, B-D, E-C, E-B.

Regimes and parameters were written before the first run.  Declared
expectations (not tuned; print whatever comes):
  matched      : A is the oracle bound (B-A cost consistent > 0); B
                 consistently below D on cost; B vs C either a consistent B
                 win or a near-tie; E may lose to C on cost - keep.
  noisy_sensor : B consistently below C and D on cost (fast observers pass
                 sensor noise into u); E keep as it comes.
  quiet_sensor : B vs D may near-tie; B consistently below C or a near-tie;
                 E loses to C - keep.

The ordering of the arms follows from the declared noise covariances and pole
sets; the toy shows magnitudes and seed scatter, not evidence that one
estimator always wins.

Alternatives (same seeds, declared before the run, not adopted):
  matched with the opposite misspecification V_design = 100*V, W_design =
  0.01*W (paired E-C cost);
  matched with slower page poles (0.90, 0.85) for C (paired B-C cost);
  noisy_sensor with HORIZON = 1000 (paired B-C cost).

def mean_sd(xs):
    if all(x == xs[0] for x in xs):
        return xs[0], 0.0  # identical values: no floating residue in mean or sd
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    """Half-up rounding on the shortest decimal repr; tiny non-zero values in scientific form."""
    if x != 0.0 and abs(x) < 5e-4:
        return format(Decimal(repr(x)), ".3e")
    s = format(Decimal(repr(x)), ".4f")
    return "0.0000" if s == "-0.0000" else s


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def fmt_pair(m, s):
    return f"{num(m)}+-{num(s)}"


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{fmt_pair(m, s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{fmt_pair(m, s)}: {label}; {sign_note(diffs, m)}"


def mat_mul(X, Y):
    n, k, m = len(X), len(Y), len(Y[0])
    return [[sum(X[i][t] * Y[t][j] for t in range(k)) for j in range(m)]
            for i in range(n)]


def mat_add(X, Y):
    return [[X[i][j] + Y[i][j] for j in range(len(X[0]))] for i in range(len(X))]


def mat_sub(X, Y):
    return [[X[i][j] - Y[i][j] for j in range(len(X[0]))] for i in range(len(X))]


def mat_T(X):
    return [[X[j][i] for j in range(len(X))] for i in range(len(X[0]))]


def inv2(Mm):
    a, b = Mm[0]
    c, d = Mm[1]
    det = a * d - b * c
    return [[d / det, -b / det], [-c / det, a / det]]


def max_abs_eig(Mm):
    a, b = Mm[0]
    c, d = Mm[1]
    tr = (a + d) / 2.0
    disc = ((a - d) / 2.0) ** 2 + b * c
    if disc >= 0.0:
        r = math.sqrt(disc)
        return max(abs(tr + r), abs(tr - r))
    return math.sqrt(tr * tr - disc)


def lqr_gain(Aa, Bb, Qq, Rr):
    """Discrete Riccati map from S = Q; returns K = (B'SB + R)^-1 B'SA."""
    S = [row[:] for row in Qq]
    At, Bt = mat_T(Aa), mat_T(Bb)
    for _ in range(100000):
        BtS = mat_mul(Bt, S)
        BtSB = mat_mul(BtS, Bb)[0][0]
        BtSA = mat_mul(BtS, Aa)
        AtS = mat_mul(At, S)
        AtSA = mat_mul(AtS, Aa)
        AtSB = mat_mul(AtS, Bb)
        coef = 1.0 / (BtSB + Rr)
        corr = [[AtSB[i][0] * coef * BtSA[0][j] for j in range(2)] for i in range(2)]
        S_new = mat_add(mat_sub(AtSA, corr), Qq)
        delta = max(abs(S_new[i][j] - S[i][j]) for i in range(2) for j in range(2))
        S = S_new
        if delta < 1e-12:
            break
    BtS = mat_mul(Bt, S)
    BtSB = mat_mul(BtS, Bb)[0][0]
    BtSA = mat_mul(BtS, Aa)
    return [[BtSA[0][j] / (BtSB + Rr) for j in range(2)]]


def kalman_gain(Aa, Cc, Vv, Ww, P_init):
    """Steady-state predictive-form Kalman gain for the discrete LQG map."""
    P = [row[:] for row in P_init]
    At, Ct = mat_T(Aa), mat_T(Cc)
    for _ in range(100000):
        PCt = mat_mul(P, Ct)
        CP = mat_mul(Cc, P)
        S_meas = mat_mul(CP, Ct)[0][0] + Ww
        corr = [[PCt[i][0] * (1.0 / S_meas) * CP[0][j] for j in range(2)]
                for i in range(2)]
        Pbar = mat_sub(P, corr)
        P_new = mat_add(mat_mul(mat_mul(Aa, Pbar), At), Vv)
        delta = max(abs(P_new[i][j] - P[i][j]) for i in range(2) for j in range(2))
        P = P_new
        if delta < 1e-12:
            break
    PCt = mat_mul(P, Ct)
    CP = mat_mul(Cc, P)
    S_meas = mat_mul(CP, Ct)[0][0] + Ww
    return [[PCt[i][0] / S_meas] for i in range(2)]


def ackermann_observer(Aa, Cc, poles):
    """Lp with eig(A - Lp C) = poles, by Ackermann on the dual pair (A', C')."""
    At, Ct = mat_T(Aa), mat_T(Cc)
    ATCt = mat_mul(At, Ct)
    Ctrb = [[Ct[0][0], ATCt[0][0]], [Ct[1][0], ATCt[1][0]]]
    Ctrb_inv = inv2(Ctrb)
    At2 = mat_mul(At, At)
    tr = sum(poles)
    prod = poles[0] * poles[1]
    phi = mat_add(mat_sub(At2, [[tr * At[i][j] for j in range(2)]
                                for i in range(2)]),
                  [[prod * (1.0 if i == j else 0.0) for j in range(2)]
                   for i in range(2)])
    Kd = mat_mul(mat_mul([[0.0, 1.0]], Ctrb_inv), phi)
    return [[Kd[0][0]], [Kd[0][1]]]


def shared_draws(regime, seed, horizon):
    """Common random numbers: one x0, one e1 train, one e2 train per (regime, seed)."""
    rng = random.Random(f"{regime}/{seed}")
    g1 = rng.gauss(0.0, 1.0)
    g2 = rng.gauss(0.0, 1.0)
    x0 = [0.5 * g1, 0.1 * g2]
    e1 = [rng.gauss(0.0, 1.0) for _ in range(horizon)]
    e2 = [rng.gauss(0.0, 1.0) for _ in range(horizon + 1)]
    return x0, e1, e2


def run_arm(arm, regime, seed, K, L_kf, Lp, horizon=HORIZON, sig_v=0.05):
    x0, e1, e2 = shared_draws(regime, seed, horizon)
    x = [x0[0], x0[1]]
    xhat = [0.0, 0.0]
    cost_sum, est_sum, u_sum = 0.0, 0.0, 0.0
    for i in range(horizon):
        if arm == "A_oracle":
            u = -(K[0][0] * x[0] + K[0][1] * x[1])
        else:
            u = -(K[0][0] * xhat[0] + K[0][1] * xhat[1])
        cost_sum += x[0] * x[0] + 0.1 * x[1] * x[1] + 0.01 * u * u
        if arm != "A_oracle":
            est_sum += (xhat[0] - x[0]) ** 2
        u_sum += u * u
        x_next = [A[0][0] * x[0] + A[0][1] * x[1] + B[0][0] * u,
                  A[1][0] * x[0] + A[1][1] * x[1] + B[1][0] * u + SIG_W * e1[i]]
        if arm == "A_oracle":
            pass
        elif arm in ("B_lqg", "E_kf_misspec"):
            y_next = x_next[0] + sig_v * e2[i + 1]
            x_pred = [A[0][0] * xhat[0] + A[0][1] * xhat[1] + B[0][0] * u,
                      A[1][0] * xhat[0] + A[1][1] * xhat[1] + B[1][0] * u]
            innov = y_next - x_pred[0]
            xhat = [x_pred[0] + L_kf[0][0] * innov,
                    x_pred[1] + L_kf[1][0] * innov]
        else:
            y_cur = x[0] + sig_v * e2[i]
            innov = y_cur - xhat[0]
            xhat = [A[0][0] * xhat[0] + A[0][1] * xhat[1] + B[0][0] * u
                    + Lp[0][0] * innov,
                    A[1][0] * xhat[0] + A[1][1] * xhat[1] + B[1][0] * u
                    + Lp[1][0] * innov]
        x = x_next
    return {"cost": cost_sum / horizon,
            "est_rms": math.sqrt(est_sum / horizon),
            "u_rms": math.sqrt(u_sum / horizon)}


def collect(regime, K, L_true, L_miss, Lp_page, Lp_fast, horizon=HORIZON):
    sig_v = REGIMES[regime]
    gains = {"A_oracle": (None, None), "B_lqg": (L_true, None),
             "C_luen_page": (None, Lp_page), "D_luen_fast": (None, Lp_fast),
             "E_kf_misspec": (L_miss, None)}
    return {arm: [run_arm(arm, regime, s, K, gains[arm][0], gains[arm][1],
                          horizon, sig_v) for s in SEEDS] for arm in ARMS}


def arm_col(regime, arm, K, L_kf, Lp, horizon=HORIZON, sig_v=None):
    if sig_v is None:
        sig_v = REGIMES[regime]
    return [run_arm(arm, regime, s, K, L_kf, Lp, horizon, sig_v) for s in SEEDS]


def col(rows, key):
    return [x[key] for x in rows]


def report(regime, K, L_true, L_miss, Lp_page, Lp_fast):
    sig_v = REGIMES[regime]
    res = collect(regime, K, L_true, L_miss, Lp_page, Lp_fast)
    print(f"{regime} (SIG_V={num(sig_v)}):")
    print(f"  declared: {DECLARED[regime]}")
    for arm in ARMS:
        print(f"  {arm} " + " ".join(
            f"{key} {fmt_pair(*mean_sd(col(res[arm], key)))}" for key in METRICS))
    for a, b in PAIRS:
        for key in METRICS:
            print(f"  paired {a[0]}-{b[0]} {key}: "
                  f"{verdict(col(res[a], key), col(res[b], key))}")
    return res
