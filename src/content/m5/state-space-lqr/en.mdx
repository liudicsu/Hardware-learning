The last lesson described the arm joint with a transfer function: one input, one output. <Term id="state-space">State space</Term> keeps every internal state in a vector x and writes the dynamics as `x' = A x + B u`. That is exactly what a balancer needs, because position, speed, angle and angular rate all matter at once. The plant here starts as the joint from [Lesson 1](/en/month/month-5/control-modelling), then becomes the cart-pole from Month 2.

## Two plants, one form

For the joint, let x = [q, w], where q is the angle and w = q'. The torque is the single input u. With J = 0.004 kg·m² and B = 0.05 N·m per rad/s, the equation J q'' + B q' = u becomes a two-row state equation. The first row is q' = w. The second row is w' = −(B/J) w + (1/J) u = −12.5 w + 250 u, so A = [[0, 1], [0, −12.5]] and B = [[0], [250]]. One equation now carries both the angle and the speed.

<Formula label="State space: every state in one vector">x' = A x + B u</Formula>

The cart-pole is Month 2's balancer cut down to a cart of M = 0.5 kg and a point mass m = 0.2 kg at L = 0.3 m. The input is a horizontal force F. The nonlinear equations (angle from upright) are `a = (F + m L ω² sinθ − m g sinθ cosθ)/(M + m − m cos²θ)` for the cart and `θ'' = (g sinθ − a cosθ)/L` for the pole. Linearised about upright, take ∂/∂θ at θ = 0, ω = 0 with `a ≈ F/M` to obtain `−m g/M = −3.92`, `1/M = 2.00`, `(M+m)g/(M L) = 45.78` and `−1/(M L) = −6.67`. With x = [position, speed, angle, rate], the model prints:

```python
"""lqr_sim.py: state space, LQR on a cart-pole, and a small MPC on the Month 3 arm joint.

Part 1 writes a cart-pole (the Month 2 balancer reduced to its essentials: a cart that is
pushed by a force and a pole balanced on it) in state space, x = [position, speed, angle,
angular rate], linearised about upright, discretised at 100 Hz. LQR gains come from
iterating the discrete Riccati equation. The gains are then tested on the full nonlinear
cart-pole, with the force limited to 10 N.
Part 2 moves the arm joint from lesson 1 (J = 0.004, B = 0.05, gravity cancelled by
feedforward) through 1 rad with a torque limit, comparing LQR whose output is simply
clipped with an MPC that knows the limit. Standard library only.
"""
import math

# ---------------------------------------------------------------- small matrix helpers
def mul(a, b):
    return [[sum(a[i][k] * b[k][j] for k in range(len(b))) for j in range(len(b[0]))] for i in range(len(a))]


def add(a, b, s=1.0):
    return [[a[i][j] + s * b[i][j] for j in range(len(a[0]))] for i in range(len(a))]


def tr(a):
    return [list(r) for r in zip(*a)]


def eye(n, v=1.0):
    return [[v if i == j else 0.0 for j in range(n)] for i in range(n)]


def discretise(a, b, dt):
    """Zero-order-hold discretisation by a Taylor series of the matrix exponential."""
    n = len(a)
    phi, gam, term = eye(n), [[dt * x[0]] for x in b], eye(n)
    for k in range(1, 25):
        term = [[v * dt / k for v in r] for r in mul(term, a)]           # (A dt)^k / k!
        phi = add(phi, term)
        gam = add(gam, [[v * dt / (k + 1)] for v in [r[0] for r in mul(term, b)]])
    return phi, gam


def lqr(phi, gam, q, r, iters=5000):
    """Iterate P = Q + A'PA - A'PB (R + B'PB)^-1 B'PA; return K for u = -K x."""
    p = q
    for _ in range(iters):
        bp = mul(tr(gam), p)
        s = r + mul(bp, gam)[0][0]
        k = [[v / s for v in mul(bp, phi)[0]]]
        p = add(q, mul(tr(phi), mul(p, add(phi, mul(gam, k), -1.0))))
    return k[0]


# ---------------------------------------------------------------- part 1: cart-pole
M, m, L, G, DT, FMAX = 0.5, 0.2, 0.3, 9.81, 0.01, 10.0
A_CP = [[0, 1, 0, 0], [0, 0, -m * G / M, 0], [0, 0, 0, 1], [0, 0, (M + m) * G / (M * L), 0]]
B_CP = [[0], [1 / M], [0], [-1 / (M * L)]]


def cartpole(x, f, dt=0.001):
    """Full nonlinear cart-pole (point mass m at distance L), angle from upright."""
    p, v, th, w = x
    s, c = math.sin(th), math.cos(th)
    acc = (f + m * L * w * w * s - m * G * s * c) / (M + m - m * c * c)
    alpha = (G * s - acc * c) / L
    return [p + v * dt, v + acc * dt, th + w * dt, w + alpha * dt]


def balance(k, deg, seconds=4.0, show=False):
    x = [0.0, 0.0, math.radians(deg), 0.0]
    fpeak = 0.0
    for i in range(int(seconds / DT)):
        f = max(-FMAX, min(FMAX, -sum(ki * xi for ki, xi in zip(k, x))))
        fpeak = max(fpeak, abs(f))
        if show and i % 50 == 0:
            print(f"  t = {i * DT:3.1f} s  x = {x[0]:6.3f} m  angle = {math.degrees(x[2]):6.2f} deg  force = {f:6.2f} N")
        for _ in range(10):
            x = cartpole(x, f)
        if abs(x[2]) > math.pi / 2:
            return None, fpeak
    return x, fpeak


def part1():
    phi, gam = discretise(A_CP, B_CP, DT)
    print("Cart-pole, x = [position, speed, angle, rate], 100 Hz")
    print("  A (continuous) rows:", "  ".join("[" + ", ".join(f"{v:.2f}" for v in r) + "]" for r in A_CP))
    print("  B (continuous):     ", "[" + ", ".join(f"{r[0]:.2f}" for r in B_CP) + "]")
    for name, qd, r in (("Q = diag(1, 1, 10, 1), R = 1   ", (1, 1, 10, 1), 1.0),
                        ("Q = diag(100, 1, 10, 1), R = 1 ", (100, 1, 10, 1), 1.0),
                        ("Q = diag(1, 1, 10, 1), R = 0.01", (1, 1, 10, 1), 0.01)):
        q = [[qd[i] if i == j else 0.0 for j in range(4)] for i in range(4)]
        k = lqr(phi, gam, q, r)
        print(f"{name}  K = [" + ", ".join(f"{v:7.2f}" for v in k) + "]")
        if name.startswith("Q = diag(1, 1, 10, 1), R = 1 "):
            k_first = k
        x, fpeak = balance(k, 10)
        print(f"    from 10 deg: after 4 s x = {x[0]:6.3f} m, angle = {math.degrees(x[2]):5.2f} deg, peak force {fpeak:5.2f} N")
    print("Nonlinear run with the first K, starting 10 deg off upright:")
    balance(k_first, 10, seconds=3.0, show=True)
    biggest = max(d for d in range(1, 90) if balance(k_first, d, seconds=6.0)[0] is not None
                  and abs(balance(k_first, d, seconds=6.0)[0][2]) < 0.01)
    print(f"  largest start angle it still recovers from (force limit {FMAX:.0f} N): {biggest} deg")


# ---------------------------------------------------------------- part 2: MPC on the joint
J, BV, T, UMAX, N = 0.004, 0.05, 0.005, 0.6, 40
PHI_J, GAM_J = discretise([[0, 1], [0, -BV / J]], [[0], [1 / J]], T)


def jstep(x, u):
    return [PHI_J[0][0] * x[0] + PHI_J[0][1] * x[1] + GAM_J[0][0] * u,
            PHI_J[1][0] * x[0] + PHI_J[1][1] * x[1] + GAM_J[1][0] * u]


def mpc(x, target, kq, warm):
    """Minimise sum over N steps of kq*(q-target)^2 + 0.01*w^2 + u^2, |u| <= UMAX,
    by projected gradient descent on the N future torques; return them all."""
    u = warm[1:] + warm[-1:]
    lr = 2e-4
    for _ in range(150):
        xs = [x]
        for uk in u:
            xs.append(jstep(xs[-1], uk))
        lam = [0.0, 0.0]                  # backward pass: gradient of the cost wrt each u
        grad = [0.0] * N
        for k in range(N - 1, -1, -1):
            q, w = xs[k + 1]
            lam = [2 * kq * (q - target) + PHI_J[0][0] * lam[0] + PHI_J[1][0] * lam[1],
                   2 * 0.01 * w + PHI_J[0][1] * lam[0] + PHI_J[1][1] * lam[1]]
            grad[k] = 2 * u[k] + GAM_J[0][0] * lam[0] + GAM_J[1][0] * lam[1]
        u = [max(-UMAX, min(UMAX, uk - lr * g)) for uk, g in zip(u, grad)]
    return u


def move(kind, seconds=1.0):
    k = lqr(PHI_J, GAM_J, [[400.0, 0], [0, 0.01]], 1.0)
    x, target, peak, settle = [0.0, 0.0], 1.0, 0.0, None
    plan = [0.0] * N
    sat = 0
    for i in range(int(seconds / T)):
        if kind == "lqr":
            u = max(-UMAX, min(UMAX, -(k[0] * (x[0] - target) + k[1] * x[1])))
        else:
            plan = mpc(x, target, 400.0, plan)
            u = plan[0]
        sat += abs(u) >= UMAX - 1e-9
        x = jstep(x, u)
        peak = max(peak, x[0])
        if abs(x[0] - target) > math.radians(1):
            settle = None
        elif settle is None:
            settle = (i + 1) * T
    return k, peak, settle, sat * T


def part2():
    print(f"Arm joint, 1 rad move, |torque| <= {UMAX} N*m (on top of gravity feedforward), 200 Hz")
    for kind, label in (("lqr", "LQR, output clipped"), ("mpc", f"MPC, horizon {N} steps")):
        k, peak, settle, sat = move(kind)
        if kind == "lqr":
            print(f"  LQR gains: K = [{k[0]:.2f}, {k[1]:.3f}]  (kp in N*m/rad, kd in N*m/(rad/s))")
        s = f"{settle:.3f} s" if settle else "never"
        print(f"  {label:22s} overshoot {math.degrees(peak - 1.0):5.2f} deg, within 1 deg from {s}, at the limit for {sat:.3f} s")


if __name__ == "__main__":
    part1()
    print()
    part2()
```

## One gain row, four states

<Term id="lqr">LQR</Term> asks for the input sequence that minimises a cost over the whole run:

<Formula label="LQR cost: state error plus control effort">J = Σ (xᵀ Q x + uᵀ R u)</Formula>

Q is a 4 × 4 matrix that penalises state error; R is a positive scalar that penalises force. Units matter: position enters in m², angle in rad², force squared in N². The gains come from iterating the discrete Riccati equation (A and B here are the zero-order-hold matrices) until P stops changing:

<Formula label="Discrete Riccati iteration and the gain it gives">P ← Q + Aᵀ P A − Aᵀ P B (R + Bᵀ P B)⁻¹ Bᵀ P A,   K = (R + Bᵀ P B)⁻¹ Bᵀ P A</Formula>

The control law is u = −K x. K has one entry per state, so the same row uses position, speed, angle and rate together. Month 3's cascade used two loops tuned one after the other: an inner loop on the pitch angle and an outer loop on wheel speed that moved the inner loop's setpoint. LQR replaces both with one gain row and lets Q and R decide the balance.

<Quiz id="q3" question="With K = [−0.92, −1.75, −18.76, −3.03] and u = −K x, what force does a 0.1 rad lean command when the other states are zero?" answer={1.88} unit="N" explanation="Only the angle term acts: u = −(−18.76 × 0.1) = 1.876 N, about 1.88 N. The positive force pushes the cart under the leaning pole." />

## Tuning Q and R

Three runs show the trade-off. All start 10° off upright and use the same 10 N force limit. Q = diag(1, 1, 10, 1), R = 1 is the baseline. Raising the position weight to 100 asks for the cart to return: the final position falls from 0.028 m to 0.000 m, but peak force rises from 3.28 N to 5.19 N. 1. Leave Q1 = 1.0, Q3 = 10 and R = 1.0 and read K = [−0.92, −1.75, −18.76, −3.03] and peak force = 3.28 N.
2. Set R = 0.01 and read K = [−6.86, −10.95, −59.07, −11.51] and peak force = 10.00 N (at the 10 N limit).
3. Set R = 1.0 and Q1 = 100, keep Q3 = 10, and read K = [−9.01, −6.54, −29.74, −4.91] and peak force = 5.19 N.
4. Set Q1 = 1.0 and R = 10 and read K = [−0.29, −0.78, −15.51, −2.37] and peak force = 2.71 N.
5. Conclusion: LQR trades state error against force; a smaller R lowers the cost of control effort so all gains grow until the force saturates (R = 0.01 reaches 10.00 N at the 10 N limit against 3.28 N at R = 1.0), and a larger Q1 tightens the position return (Q1 = 100 raises peak force to 5.19 N).

Note the sign of the position gain. The position gain is negative, like the angle gain. With u = −K x, a positive position error, meaning the cart is to the right of the target, commands a positive force and pushes it further right. That is deliberate: the controller first tips the pole left, then drives left to catch it. A cart-pole is non-minimum phase in position, so an immediate push toward the target would lean the pole the wrong way.

To use less force, raise R or lower the state weights. Raising the position weight in Q instead gives a tighter return and spends more force.

## MPC and the receding horizon

Part 2 moves the arm joint through 1 rad with a torque limit of 0.6 N·m on top of gravity feedforward, at 200 Hz, so T = 5 ms. LQR uses the same cost but its output is simply clipped at the limit. MPC predicts N = 40 steps ahead, 40 × 5 ms = 0.2 s, optimises all 40 future torques subject to `|u| ≤ 0.6 N·m`, applies only the first, then repeats. That is the receding horizon. The model solves it by projected gradient descent: 150 steps per control period, warm-started with the previous plan shifted by one.

<SimOutput source="model">

The LQR gains are K = [15.93, 0.323], much larger than lesson 1's PD gains 2.86 N·m/rad and 0.115 N·m/(rad/s), because Q = diag(400, 0.01) with R = 1 prices a 1 rad error far above 1 N·m of torque: a 1 rad error costs `400 × 1² = 400`, matched by `u² = 400` → `|u| = 20 N·m` (about 33× the 0.6 N·m limit and about 7× lesson 1's 2.86 N·m/rad scale), which is why the position gain lands near 16. Gains that large spend most of the move pinned at the limit, which is exactly where plain LQR stops holding: K is computed for an unclipped output, and once the command is clipped the effective loop gain is no longer the designed one. With clipping, LQR overshoots 2.24° and is within 1° from 0.210 s. MPC overshoots 0.86° and is within 1° from 0.160 s. It is at the limit for 0.165 s, LQR for 0.175 s. The improvement exists but is limited; the bigger difference is computation.

MPC costs far more computation per step: every 5 ms period it runs 150 gradient steps, each a forward pass and a backward pass over 40 steps, about 12,000 small state updates, while LQR is two multiplications and an addition. On a microcontroller the 5 ms deadline is fixed. If the MPC does not finish in time, you must shorten the horizon, lower the control rate, or move to a faster chip. A lower rate adds delay, and delay eats phase margin, as lesson 1 showed.

## Observer vs oracle under model error

Lesson 5.2 already designs LQR with a discrete Riccati equation. This section asks a different question: once the gain is fixed on a **nominal** `(A, B)`, how does feedback degrade on a **perturbed** plant when you only measure `y = C x + noise`? The observer is a discrete Luenberger copy with pole placement (Ackermann on this 2-state plant), following the structure in UC Berkeley CE295 Ch. 2 State Estimation. This page does **not** claim discrete-Riccati optimality for the observer, and it is **not** an H∞ design.

### Privileged information, named honestly

Three arms share one process-noise realisation on the perturbed plant:

| Arm | What it sees | Control |
|---|---|---|
| `oracle_fullstate` | privileged true `x` | `u = −K x` |
| `observer_output_feedback` | only `y = C x + v` | Luenberger `x̂`, then `u = −K x̂` |
| `open_loop` | nothing | `u = 0` |

Full-state feedback is an **oracle upper reference**, not an equal-information baseline. Never read the table as "same sensors, full state wins."

```python
#!/usr/bin/env python3
"""lqr_model_error_toy.py: observer vs oracle under model error (stdlib).

The observer sees ONLY the scalar measurement y = C x + v. Full-state feedback
is an ORACLE upper reference that reads the privileged true state x — it is
NOT an equal-information baseline. Arm names make that explicit:
  oracle_fullstate          u = -K x_true   (privileged)
  observer_output_feedback  Luenberger xhat from y only; u = -K xhat
  open_loop                 u = 0

Gain K and observer gain L are designed on the NOMINAL (A, B) by pole
placement (Ackermann for this 2-state plant). Evaluation runs on a PERTURBED
plant A_pert with the SAME process-noise realisation across arms. This is
model-error sensitivity awareness, NOT H∞ design, and NOT a discrete-Riccati
optimality claim (lesson 5.2's lqr_sim.py already covers Riccati LQR).

The scored cost is sum (x0² + 0.1 x1² + 0.01 u²): under model error the
observer can look similar on position RMS alone (its lag filters) while
spending far more control effort and leaving more velocity — the cost makes
the privileged-information gap visible without claiming equal-info superiority.

Luenberger / observability structure: UC Berkeley CE295 notes, Ch. 2 State
Estimation (https://ecal.studentorg.berkeley.edu/files/ce295/CH02-StateEstimation.pdf).
Simon, Optimal State Estimation (Wiley, 2006) is the standard KF book citation;
no free full-text URL is claimed.
usage: python3 lqr_model_error_toy.py
"""
from __future__ import annotations

import math
import random

SEED = 52
DT = 0.02                 # s
HORIZON = 250             # steps ≈ 5 s
M = 1.0                   # kg
B_NOM = 0.40              # N·s/m  (design model)
B_PERT = 0.10             # lighter damping on the evaluation plant
PROCESS_STD = 0.04        # shared process noise on velocity (discrete)
MEAS_STD = 0.05           # observer measurement noise on y = position (m)
K_POLES = (0.92, 0.88)
L_POLES = (0.85, 0.80)
# Running cost weights for the scored sum.
QX, QV, RU = 1.0, 0.1, 0.01


def mat_mul(A, B):
    return [[sum(A[i][k] * B[k][j] for k in range(len(B))) for j in range(len(B[0]))]
            for i in range(len(A))]


def eye(n):
    return [[1.0 if i == j else 0.0 for j in range(n)] for i in range(n)]


def scale_mat(s, A):
    return [[s * A[i][j] for j in range(len(A[0]))] for i in range(len(A))]


def cont_to_disc(A_c, B_c, dt):
    """Forward-Euler discretisation (teaching toy; not zoh-exact)."""
    I = eye(len(A_c))
    A = [[I[i][j] + dt * A_c[i][j] for j in range(2)] for i in range(2)]
    B = [[dt * B_c[i][0]] for i in range(2)]
    return A, B


def char_poly_poles(p1, p2):
    return [1.0, -(p1 + p2), p1 * p2]


def adjugate_2(M):
    return [[M[1][1], -M[0][1]], [-M[1][0], M[0][0]]]


def det_2(M):
    return M[0][0] * M[1][1] - M[0][1] * M[1][0]


def ackermann_K(A, B, poles):
    AB = mat_mul(A, B)
    Cc = [[B[0][0], AB[0][0]], [B[1][0], AB[1][0]]]
    assert abs(det_2(Cc)) > 1e-9, "not controllable"
    _, a1, a0 = char_poly_poles(*poles)
    A2 = mat_mul(A, A)
    alphaA = [[A2[i][j] + a1 * A[i][j] + a0 * (1.0 if i == j else 0.0)
               for j in range(2)] for i in range(2)]
    inv = scale_mat(1.0 / det_2(Cc), adjugate_2(Cc))
    eT = inv[1]
    return [eT[0] * alphaA[0][j] + eT[1] * alphaA[1][j] for j in range(2)]


def ackermann_L(A, C, poles):
    CA = mat_mul(C, A)
    Ob = [[C[0][0], C[0][1]], [CA[0][0], CA[0][1]]]
    assert abs(det_2(Ob)) > 1e-9, "not observable"
    _, a1, a0 = char_poly_poles(*poles)
    A2 = mat_mul(A, A)
    alphaA = [[A2[i][j] + a1 * A[i][j] + a0 * (1.0 if i == j else 0.0)
               for j in range(2)] for i in range(2)]
    inv = scale_mat(1.0 / det_2(Ob), adjugate_2(Ob))
    col = [inv[0][1], inv[1][1]]
    return [alphaA[i][0] * col[0] + alphaA[i][1] * col[1] for i in range(2)]


def build(b):
    A_c = [[0.0, 1.0], [0.0, -b / M]]
    B_c = [[0.0], [1.0 / M]]
    return cont_to_disc(A_c, B_c, DT)


def simulate(arm, A_nom, A_pert, B, K, L, C, noise_w, noise_v, x0):
    x = list(x0)
    xhat = [0.0, 0.0]
    xs, us = [], []
    for k in range(HORIZON):
        if arm == "oracle_fullstate":
            # Privileged true state — NOT the same information as the observer.
            u = -(K[0] * x[0] + K[1] * x[1])
        elif arm == "observer_output_feedback":
            u = -(K[0] * xhat[0] + K[1] * xhat[1])
        else:
            u = 0.0
        w = [0.0, noise_w[k]]
        x_next = [A_pert[i][0] * x[0] + A_pert[i][1] * x[1] + B[i][0] * u + w[i]
                  for i in range(2)]
        y = C[0][0] * x[0] + C[0][1] * x[1] + noise_v[k]
        if arm == "observer_output_feedback":
            yhat = C[0][0] * xhat[0] + C[0][1] * xhat[1]
            innov = y - yhat
            xhat = [A_nom[i][0] * xhat[0] + A_nom[i][1] * xhat[1]
                    + B[i][0] * u + L[i] * innov for i in range(2)]
        x = x_next
        xs.append(x[:])
        us.append(u)
    return xs, us


def rms(vals):
    return math.sqrt(sum(v * v for v in vals) / len(vals))


def cost(xs, us):
    return sum(QX * x[0] * x[0] + QV * x[1] * x[1] + RU * u * u
               for x, u in zip(xs, us))


def main() -> None:
    A_nom, B = build(B_NOM)
    A_pert, _ = build(B_PERT)
    C = [[1.0, 0.0]]
    K = ackermann_K(A_nom, B, K_POLES)
    L = ackermann_L(A_nom, C, L_POLES)

    rng = random.Random(SEED)
    noise_w = [rng.gauss(0.0, PROCESS_STD) for _ in range(HORIZON)]
    noise_v = [rng.gauss(0.0, MEAS_STD) for _ in range(HORIZON)]
    x0 = [1.0, 0.0]

    print(f"seed={SEED} dt={DT} horizon={HORIZON} m={M} b_nom={B_NOM} b_pert={B_PERT}")
    print(f"design: pole-placement on nominal (A,B); K_poles={list(K_POLES)} L_poles={list(L_POLES)}")
    print(f"K=[{K[0]:.4f}, {K[1]:.4f}]  L=[{L[0]:.4f}, {L[1]:.4f}]")
    print("oracle_fullstate uses privileged true x; observer uses only y=Cx+noise")
    print("NOT H-infinity; NOT equal-info baseline; shared process noise across arms")
    print(f"cost = sum(x_pos^2 + {QV}*x_vel^2 + {RU}*u^2) over the horizon")
    print()
    hdr = f"{'arm':<28} {'rms_pos':>10} {'rms_vel':>10} {'rms_u':>10} {'cost':>12} {'final_pos':>10}"
    print(hdr)
    results = {}
    for arm in ("open_loop", "oracle_fullstate", "observer_output_feedback"):
        xs, us = simulate(arm, A_nom, A_pert, B, K, L, C, noise_w, noise_v, x0)
        rp, rv, ru = rms([x[0] for x in xs]), rms([x[1] for x in xs]), rms(us)
        c = cost(xs, us)
        results[arm] = (rp, rv, ru, c, xs[-1][0])
        print(f"{arm:<28} {rp:10.4f} {rv:10.4f} {ru:10.4f} {c:12.2f} {xs[-1][0]:10.4f}")

    c_or = results["oracle_fullstate"][3]
    c_ob = results["observer_output_feedback"][3]
    c_ol = results["open_loop"][3]
    assert c_or < c_ob < c_ol, f"cost order oracle<observer<open failed: {c_or}, {c_ob}, {c_ol}"
    assert results["oracle_fullstate"][2] < results["observer_output_feedback"][2], "oracle should spend less |u|"
    print("ok: oracle_fullstate (privileged x) cost < observer_output_feedback cost < open_loop; "
          "model-error sweep on perturbed plant (real hardware 本页未运行)")


if __name__ == "__main__":
    main()
```

## Common mistakes

- <strong>Q and R in inconsistent units.</strong> Changing position from metres to centimetres multiplies that state by 100, so the same Q entry penalises 10,000 times more. Keep states in SI and re-tune after a unit change.
- <strong>Using the linear model far from upright.</strong> The 45.78 term is a linearisation about upright. At 53° the true dynamics are different; a gain that works at 10° is not a proof that far out.
- <strong>Ignoring actuator limits in the LQR design.</strong> An unclipped LQR assumes it can deliver any force. The R = 0.01 run reaches the 10 N limit; include saturation in the simulation, or design with MPC.
- <strong>Forgetting the sign conventions of the state.</strong> x = [position, speed, angle, rate] with angle measured from upright. Flip the angle sign and the controller pushes the pole over. Write the positive direction down before tuning.
- <strong>Believing MPC is always better.</strong> MPC needs a model and a solver every period. If the limit is not active and the linear model is good, LQR can match it with a fraction of the code and time.
- **Calling full-state feedback a same-info baseline.** The oracle reads `x`. The observer reads `y`. Different information.
- **Reading position RMS alone.** Under model error the observer's lag can filter position while inflating velocity and `|u|`. Use a cost that prices all three, or report them side by side.
- **Claiming H∞ from a one-Δ sweep.** Pole placement on nominal + one perturbed plant is sensitivity awareness, not H∞ synthesis.

## Discrete LQR and zero-order hold

Lesson 5.2 already iterates a discrete Riccati on a cart-pole. This section asks a narrower sampling question: if you design a **continuous** LQR gain and hold it with a **zero-order hold (ZOH)**, how does that compare with a **discrete** LQR designed at the same sample period? It is **not** a Bode / margins redo, and it is **not** only re-deriving continuous Riccati.

### Discretization and Riccati assumptions

**ZOH** maps the continuous plant with a piecewise-constant input over `T_s`. This toy builds `(A_d, B_d)` from the matrix exponential of the augmented block `[[A_c, B_c],[0, 0]] * T_s` (scaling-and-squaring + Taylor) — not forward-Euler. Discrete stage weights use `Q_d ≈ Q_c T_s`, `R_d ≈ R_c T_s` so the sum approximates the continuous integral cost.

The discrete gain comes from iterating

<Formula label="Discrete Riccati iteration (DARE map)">P ← Q + AᵀPA − AᵀPB(R + BᵀPB)⁻¹BᵀPA</Formula>

until `‖ΔP‖_F < 1e-12` (explicit tolerance). **Stability criterion:** eigenvalues of `A_d − B_d K` must satisfy `|λ| < 1`. Continuous `K_c` comes from a Kleinman CARE iteration; under ZOH it is held constant between samples. Integration under the hold uses a substep ≪ `T_s` (step-size stability check in the tests).

**Metric blindness:** infinite-horizon-style cost and settling do **not** price the CPU / deadline cost of a finer `T_s` — C’s fine rate is a **budget**, not free. Judge rate choices with that limit in mind.

### Fair setup

`discrete_lqr_zoh_toy.py` is **synthetic** and stdlib only. Every arm shares the **same** continuous `(A_c, B_c)`, **same** `Q_c, R_c`, and for each seed the **identical** process-noise series indexed by absolute substep (common random numbers).

| Arm | Controller |
|---|---|
| A_cont_zoh | continuous `K_c`, held with ZOH at the regime `T_s` |
| B_disc | discrete `K_d` designed at that same `T_s` |
| C_disc_fine | discrete `K_d` designed and applied at fine `T_s` |

Regimes: `coarse_Ts` (discrete-at-hold beats cont+ZOH) and `fine_Ts` (**near-tie / small gap** as `T_s→0` — remaining gap from holding `K_c` vs DARE `K_d` plus shared noise; not a true zero gap). Metrics: cost, settle time, `max|λ|`. Five-seed mean±sd.

```python
#!/usr/bin/env python3
"""Synthetic discrete LQR vs continuous+ZOH literacy toy. Stdlib only.

Shared continuous plant (A_c, B_c) and shared Q, R meaning for every arm.
Process-noise draws are **common random numbers**: for a given seed, A/B/C
see the identical noise sequence (no arm gets a quieter plant).

Discretization: exact **zero-order hold (ZOH)** via the matrix exponential
of the augmented [[A_c, B_c],[0,0]] block (scaling-and-squaring + Taylor),
not forward-Euler.

Discrete Riccati: iterate the DARE map until ‖ΔP‖_F < RICCATI_TOL (explicit).
Continuous K_c from Kleinman iteration on the CARE (stabilizing Lyapunov steps).

Arms:
  A_cont_zoh   continuous LQR K_c held with ZOH at the regime T_s
  B_disc       discrete LQR K_d designed at that same T_s
  C_disc_fine  discrete LQR designed and applied at fine T_s (always)

Regimes (≥5 seeds on shared process noise):
  coarse_Ts — discrete B beats cont+ZOH A (design at the hold rate matters)
  fine_Ts   — A and B near-tie as T_s→0 (remaining small gap from hold + noise)

Stability criterion: closed-loop eigenvalues of (A_d − B_d K) have modulus < 1
(tested for B and C). Metrics: infinite-horizon-style cost, settling steps,
max |λ|. ≠ Bode redo; ≠ continuous Riccati-only lesson. Real plant: 本页未运行.
"""
from __future__ import annotations

import math
import random

SEEDS = (0, 1, 2, 3, 4)
ARMS = ("A_cont_zoh", "B_disc", "C_disc_fine")
# Continuous plant: unit mass + damping  ẍ = −b ẋ + u
M = 1.0
B_DAMP = 0.40
A_C = [[0.0, 1.0], [0.0, -B_DAMP / M]]
B_C = [[0.0], [1.0 / M]]
# Shared continuous cost meaning; discrete stage cost ≈ integral ≈ Q*Ts, R*Ts
Q_C = [[25.0, 0.0], [0.0, 1.0]]
R_C = 0.05
TS_COARSE = 0.10
TS_FINE = 0.01
HORIZON_S = 6.0
PROCESS_STD = 0.05  # shared accel noise, continuous-time intensity proxy
SETTLE_EPS = 0.05
RICCATI_TOL = 1e-12
RICCATI_MAX_ITER = 10_000
EXPM_SQUARINGS_CAP = 8
SUBSTEP = 0.001  # continuous integration substep under ZOH (<< fine Ts)


def mat_add(A, B):
    return [[A[i][j] + B[i][j] for j in range(len(A[0]))] for i in range(len(A))]


def mat_sub(A, B):
    return [[A[i][j] - B[i][j] for j in range(len(A[0]))] for i in range(len(A))]


def mat_scale(s, A):
    return [[s * A[i][j] for j in range(len(A[0]))] for i in range(len(A))]


def mat_mul(A, B):
    return [
        [sum(A[i][k] * B[k][j] for k in range(len(B))) for j in range(len(B[0]))]
        for i in range(len(A))
    ]


def mat_T(A):
    return [[A[j][i] for j in range(len(A))] for i in range(len(A[0]))]


def eye(n):
    return [[1.0 if i == j else 0.0 for j in range(n)] for i in range(n)]


def frobenius(A):
    return math.sqrt(sum(A[i][j] * A[i][j] for i in range(len(A)) for j in range(len(A[0]))))


def solve_2x2(M, v):
    det = M[0][0] * M[1][1] - M[0][1] * M[1][0]
    if abs(det) < 1e-18:
        raise ZeroDivisionError("singular 2x2")
    return [
        (M[1][1] * v[0] - M[0][1] * v[1]) / det,
        (-M[1][0] * v[0] + M[0][0] * v[1]) / det,
    ]


def mat_inv_2(M):
    det = M[0][0] * M[1][1] - M[0][1] * M[1][0]
    return [[M[1][1] / det, -M[0][1] / det], [-M[1][0] / det, M[0][0] / det]]


def expm(M):
    """Scaling-and-squaring + Taylor; M square. Stdlib-only ZOH building block."""
    n = len(M)
    # scale so ||M/2^s|| small
    norm = max(sum(abs(M[i][j]) for j in range(n)) for i in range(n))
    s = 0
    while norm > 0.5 and s < EXPM_SQUARINGS_CAP:
        M = mat_scale(0.5, M)
        norm *= 0.5
        s += 1
    # Taylor sum_k M^k / k!
    term = eye(n)
    total = eye(n)
    for k in range(1, 20):
        term = mat_scale(1.0 / k, mat_mul(term, M))
        total = mat_add(total, term)
        if frobenius(term) < 1e-16:
            break
    for _ in range(s):
        total = mat_mul(total, total)
    return total


def zoh_discretize(A_c, B_c, ts: float):
    """Exact ZOH: exp([[A,B],[0,0]]*ts) → (A_d, B_d)."""
    n = len(A_c)
    m = len(B_c[0])
    aug = [[0.0] * (n + m) for _ in range(n + m)]
    for i in range(n):
        for j in range(n):
            aug[i][j] = A_c[i][j] * ts
        for j in range(m):
            aug[i][n + j] = B_c[i][j] * ts
    E = expm(aug)
    A_d = [[E[i][j] for j in range(n)] for i in range(n)]
    B_d = [[E[i][n + j] for j in range(m)] for i in range(n)]
    return A_d, B_d


def dare_iterate(A, B, Q, R, tol=RICCATI_TOL):
    """Discrete Riccati iteration → (P, K, n_iter). Explicit convergence tol."""
    n = len(A)
    P = [row[:] for row in Q]
    K = [[0.0] * n]
    for it in range(1, RICCATI_MAX_ITER + 1):
        # K = (R + B'P B)^{-1} B' P A
        BtP = mat_mul(mat_T(B), P)
        S = mat_add([[R]], mat_mul(BtP, B))  # 1x1 for our SISO B
        BtPA = mat_mul(BtP, A)
        # S is 1x1
        invS = 1.0 / S[0][0]
        K_new = [[invS * BtPA[0][j] for j in range(n)]]
        # P = Q + A'P A - A'P B K
        AtP = mat_mul(mat_T(A), P)
        P_new = mat_add(Q, mat_sub(mat_mul(AtP, A), mat_mul(mat_mul(AtP, B), K_new)))
        if frobenius(mat_sub(P_new, P)) < tol:
            return P_new, K_new, it
        P, K = P_new, K_new
    return P, K, RICCATI_MAX_ITER


def care_kleinman(A, B, Q, R, tol=1e-12):
    """Continuous ARE via Kleinman iteration; returns (P, K)."""
    n = len(A)
    # Start from a stabilizing gain (LQR-ish PD on this plant)
    K = [[4.0, 2.0]]
    P = eye(n)
    for _ in range(200):
        # A_cl = A - B K
        A_cl = mat_sub(A, mat_mul(B, K))
        # Solve Lyapunov A_cl' P + P A_cl = -(Q + K' R K) for 2x2
        QK = mat_add(Q, mat_scale(R, mat_mul(mat_T(K), K)))
        P_new = solve_lyapunov_2(A_cl, QK)
        # K = R^{-1} B' P
        BtP = mat_mul(mat_T(B), P_new)
        K_new = [[BtP[0][0] / R, BtP[0][1] / R]]
        if frobenius(mat_sub(P_new, P)) < tol and frobenius(mat_sub(K_new, K)) < tol:
            return P_new, K_new
        P, K = P_new, K_new
    return P, K


def solve_lyapunov_2(A, Q):
    """Solve A'P + P A = -Q for 2×2 P (symmetric) via 3×3 linear system."""
    # P = [[p, q],[q, r]]; expand
    a, b = A[0][0], A[0][1]
    c, d = A[1][0], A[1][1]
    # coeffs for [p,q,r]
    # (1,1): 2a p + 2c q = -Q00
    # (1,2): b p + (a+d) q + c r = -Q01
    # (2,2): 2b q + 2d r = -Q11
    M = [
        [2 * a, 2 * c, 0.0],
        [b, a + d, c],
        [0.0, 2 * b, 2 * d],
    ]
    rhs = [-Q[0][0], -Q[0][1], -Q[1][1]]
    # Gaussian elimination 3x3
    A3 = [M[i][:] + [rhs[i]] for i in range(3)]
    for col in range(3):
        piv = max(range(col, 3), key=lambda r: abs(A3[r][col]))
        A3[col], A3[piv] = A3[piv], A3[col]
        div = A3[col][col]
        if abs(div) < 1e-18:
            raise ZeroDivisionError("Lyapunov singular")
        for j in range(col, 4):
            A3[col][j] /= div
        for r in range(3):
            if r == col:
                continue
            fac = A3[r][col]
            for j in range(col, 4):
                A3[r][j] -= fac * A3[col][j]
    p, q, r = A3[0][3], A3[1][3], A3[2][3]
    return [[p, q], [q, r]]


def eig_moduli_2(M):
    """|λ| of 2×2 real matrix via characteristic polynomial."""
    tr = M[0][0] + M[1][1]
    det = M[0][0] * M[1][1] - M[0][1] * M[1][0]
    disc = tr * tr - 4.0 * det
    if disc >= 0.0:
        s = math.sqrt(disc)
        return [abs(0.5 * (tr + s)), abs(0.5 * (tr - s))]
    # complex conjugate: |λ| = sqrt(det) when det>0 for stable oscillation
    return [math.sqrt(abs(det)), math.sqrt(abs(det))]


def stage_QR(ts: float):
    """Map continuous Q,R meaning to discrete stage weights ≈ ∫ over Ts."""
    Qd = mat_scale(ts, Q_C)
    Rd = R_C * ts
    return Qd, Rd


def shared_noise_series(seed: int) -> list[float]:
    """Common random numbers: one accel-noise sample per SUBSTEP for the horizon.

    Every arm indexes the **same** series by absolute substep, so changing the
    hold period does not give anyone a quieter plant.
    """
    n = int(round(HORIZON_S / SUBSTEP))
    rng = random.Random(50_000 + seed)
    return [rng.gauss(0.0, PROCESS_STD) for _ in range(n)]


def simulate(
    K, ts: float, seed: int, noise: list[float] | None = None
) -> tuple[float, float, list[float]]:
    """ZOH hold of u=-Kx at period ts; integrate continuous plant with SUBSTEP.

    Returns (cost, settle_time_s, noise_series). Cost ≈ Σ (x'Q_c x + R_c u²)·dt.
    """
    if noise is None:
        noise = shared_noise_series(seed)
    n_hold = int(round(HORIZON_S / ts))
    n_sub_per = max(1, int(round(ts / SUBSTEP)))
    dt = ts / n_sub_per
    x = [1.0, 0.0]
    cost = 0.0
    settled_at = None
    sub_i = 0
    for k in range(n_hold):
        u = -(K[0][0] * x[0] + K[0][1] * x[1])
        for _ in range(n_sub_per):
            w = noise[sub_i] if sub_i < len(noise) else 0.0
            sub_i += 1
            xdot0 = A_C[0][0] * x[0] + A_C[0][1] * x[1] + B_C[0][0] * (u + w)
            xdot1 = A_C[1][0] * x[0] + A_C[1][1] * x[1] + B_C[1][0] * (u + w)
            xQx = Q_C[0][0] * x[0] * x[0] + Q_C[1][1] * x[1] * x[1]
            cost += (xQx + R_C * u * u) * dt
            x = [x[0] + dt * xdot0, x[1] + dt * xdot1]
        if settled_at is None and abs(x[0]) < SETTLE_EPS and abs(x[1]) < SETTLE_EPS:
            settled_at = (k + 1) * ts
    if settled_at is None:
        settled_at = HORIZON_S
    return cost, settled_at, noise


def design_cont_K():
    _, K = care_kleinman(A_C, B_C, Q_C, R_C)
    return K


def design_disc_K(ts: float):
    A_d, B_d = zoh_discretize(A_C, B_C, ts)
    Qd, Rd = stage_QR(ts)
    P, K, n_it = dare_iterate(A_d, B_d, Qd, Rd)
    mods = eig_moduli_2(mat_sub(A_d, mat_mul(B_d, K)))
    return K, max(mods), n_it, A_d, B_d


def mean_sd(xs: list[float]) -> tuple[float, float]:
    n = len(xs)
    mu = sum(xs) / n
    var = sum((x - mu) ** 2 for x in xs) / max(1, n - 1)
    return mu, math.sqrt(var)
```
