## 组合契约四条

1. **坐标系：** `map` → `base_link` → `arm_base` → `ee`；导航目标与抓取位姿要说清落在哪一层。
2. **互斥：** 底盘未 `base_settled` 之前，臂不许动（避免边走边伸）。
3. **超时：** 导航与抓取各自有上限（玩具里 `timeout_nav_s=30`、`timeout_grasp_s=10`）。
4. **失败回传：** 臂技能失败要回到导航层——常见恢复是「再靠近一点再抓」（`approach_closer_then_grasp`）。

对照第 4 个月 `slam-nav` 的 Nav2 行为树恢复（清代价地图 / 旋转）；本课不重讲导航调参，只接「导航 + 操作」这一层。

```python
#!/usr/bin/env python3
"""nav_pick_bt.py: navigate-then-pick compose contract on a grid (stdlib).

Compose contract (printed): frames map>base_link>arm_base>ee; mutex =
base settled before arm; nav/grasp timeouts; failure feedback from grasp
to nav ("approach closer then grasp").

Grid A* to a far desk approach, grasp fails (outside arm workspace),
Fallback recovers by navigating one cell closer, then grasp succeeds.

Fixed geometry + seed=9 → expected/. No real LeKiwi / Nav2 / ROS runtime.
usage: python3 nav_pick_bt.py
"""
from __future__ import annotations

import math
from heapq import heappop, heappush
from typing import Dict, List, Optional, Tuple

SEED = 9
W, H = 7, 7
START = (1, 1)
DESK = (6, 3)  # grasp target cell (desk edge object)
APPROACH_FAR = (4, 3)
APPROACH_NEAR = (5, 3)
ARM_WORKSPACE_R = 1.5  # Euclidean cells from base to desk
OBSTACLES = {(3, 1), (3, 2), (3, 4), (3, 5)}  # vertical wall with a gap at y=3
TIMEOUT_NAV_S = 30
TIMEOUT_GRASP_S = 10

Pos = Tuple[int, int]


def in_bounds(p: Pos) -> bool:
    x, y = p
    return 0 <= x < W and 0 <= y < H


def neighbours(p: Pos) -> List[Pos]:
    x, y = p
    out: List[Pos] = []
    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
        q = (x + dx, y + dy)
        if in_bounds(q) and q not in OBSTACLES:
            out.append(q)
    return out


def astar(start: Pos, goal: Pos) -> List[Pos]:
    """4-connected A*; returns path including start and goal."""
    if start == goal:
        return [start]
    open_h: List[Tuple[int, int, Pos]] = []
    counter = 0
    heappush(open_h, (0, counter, start))
    came: Dict[Pos, Optional[Pos]] = {start: None}
    gscore: Dict[Pos, int] = {start: 0}

    def h(p: Pos) -> int:
        return abs(p[0] - goal[0]) + abs(p[1] - goal[1])

    while open_h:
        _, _, cur = heappop(open_h)
        if cur == goal:
            path = [cur]
            while came[cur] is not None:
                cur = came[cur]  # type: ignore[assignment]
                path.append(cur)
            path.reverse()
            return path
        for nxt in neighbours(cur):
            tentative = gscore[cur] + 1
            if tentative < gscore.get(nxt, 10**9):
                came[nxt] = cur
                gscore[nxt] = tentative
                counter += 1
                heappush(open_h, (tentative + h(nxt), counter, nxt))
    raise RuntimeError(f"no path {start} -> {goal}")


def dist(a: Pos, b: Pos) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def in_workspace(base: Pos) -> bool:
    return dist(base, DESK) <= ARM_WORKSPACE_R + 1e-9


def fmt_path(path: List[Pos]) -> str:
    return "->".join(f"({x},{y})" for x, y in path)


def main() -> None:
    print(f"seed={SEED}")
    print(
        f"map={W}x{H} start={START} desk={DESK} "
        f"approach_far={APPROACH_FAR} approach_near={APPROACH_NEAR}"
    )
    print(f"arm_workspace_r={ARM_WORKSPACE_R}")
    print(
        "compose: frames=map>base_link>arm_base>ee "
        "mutex=base_settled_before_arm "
        f"timeout_nav_s={TIMEOUT_NAV_S} timeout_grasp_s={TIMEOUT_GRASP_S}"
    )

    # --- primary: nav to far approach, try grasp ---
    path_far = astar(START, APPROACH_FAR)
    path_length_far = len(path_far) - 1
    print(f"nav_path_far={fmt_path(path_far)} path_length_far={path_length_far}")
    print("base_settled=True")  # mutex: arm may move only after this
    d1 = dist(APPROACH_FAR, DESK)
    ws1 = in_workspace(APPROACH_FAR)
    print(
        f"grasp_attempt=1 at_base={APPROACH_FAR} dist_to_desk={d1:.3f} "
        f"in_workspace={ws1} -> FAILURE"
    )
    assert not ws1, "far approach must be outside workspace for the demo"

    # --- recovery: approach closer then grasp ---
    print("recovery=approach_closer_then_grasp")
    path_near = astar(APPROACH_FAR, APPROACH_NEAR)
    path_length_near = len(path_near) - 1
    print(f"nav_path_near={fmt_path(path_near)} path_length_near={path_length_near}")
    print("base_settled=True")
    d2 = dist(APPROACH_NEAR, DESK)
    ws2 = in_workspace(APPROACH_NEAR)
    print(
        f"grasp_attempt=2 at_base={APPROACH_NEAR} dist_to_desk={d2:.3f} "
        f"in_workspace={ws2} -> SUCCESS"
    )
    assert ws2, "near approach must be inside workspace"

    path_length_total = path_length_far + path_length_near
    approach_closer_fired = True
    root_status = "SUCCESS"
    print(f"path_length_total={path_length_total}")
    print(f"grasp_in_workspace_final={ws2}")
    print(f"approach_closer_fired={approach_closer_fired}")
    print(f"root_status={root_status}")
    ok = (
        root_status == "SUCCESS"
        and approach_closer_fired
        and ws2
        and not ws1
        and path_length_total > 0
    )
    print(
        f"ok: compose navigate-then-pick with approach-closer recovery ({ok})"
    )


if __name__ == "__main__":
    main()
```

### 相同动力学，披露信息差异

种子 **926–930** 抽样双方起点和终点，再预生成每个 tick、每坐标±0.005m 内均匀分布的邻居位置噪声。所有臂复用不可变的场景 / 噪声元组；路径不同会让有噪观测状态不同，但加性噪声实现和时间索引不变。各臂同样知道自身位置、静态地图、路线和目标；每个机器人共用 **vmax=0.6 m/s**、**amax=0.8 m/s²**、dt=0.05s 与 **24 s** 截止。代码断言向量加速度和速度，包括到达刹停，不靠墙面投影或瞬移绕过限制。这里 amax 指每 tick 的 norm(v_next−v)/dt；运动段为分段线性，不是连续时间执行器模型。

| 臂 | 信息与动作 |
|---|---|
| `ignore_others` | 只按自身路线，故意忽略邻居，朝目标运动。 |
| `priority_mutex_extra_info` | **额外可靠通信 / 共享预约**：双方在0时刻请求，ID0先拿令牌，ID1留在等待区；实际到达消息再释放 / 授予下一令牌。不使用未来真值或全局最优调度器。 |
| `ORCA_lite_reciprocal_stop` | 3m 内有噪邻居位置、全向速度控制；互惠停车分担间距责任，不用预约通道或全局退让路线。 |

互斥臂的通信是**额外信息**，不是等信息条件下击败局部避障；它仍遵守同样速度 / 加速度。固定路线是共享地图知识，不是偷偷只给某一臂的中央真值规划。

### 这里的“ORCA-lite”究竟是什么

已打开的 ORCA 一手页面说明各方承担一半责任，用低维 LP 求解避碰半平面。**本臂只是受 ORCA 启发的互惠停车简化**，不是完整 ORCA 的速度障碍构造 / LP，不继承它的形式化无碰撞保证。它假设 **holonomic discs（全向圆盘）**，不是差速底盘实现。明确地说，这不是原始 RVO/ORCA 的线性规划。

观测中心距 D 对应余隙 `max(0,D−2r−0.08)`m；τ=2s 预测窗内各方分一半，因此自身朝向对方的首选速度分量不超过 `clearance/(2τ)`。代码按比例缩小整个首选速度，不选择会车侧、不构造 ORCA 直线，也不倒退出走廊；噪声和加速度限幅还会削弱理想速度空间论证。因此只**测量**重叠，不称为安全证书。这种小型简化用于说明：对称局部让行可以停止碰撞，却形成需要路线级调度 / 回退才能解决的死锁。

### 不隐藏未完成机器人的指标

- **碰撞**：统计进入圆盘重叠（`中心距 <0.44m`）的事件，不按重叠 tick 重复计数。最小中心距在每段完整线性运动上求，不只采 tick 端点。运动学重叠只记账，不模拟接触冲量；穿过对方后的快速到达不是安全完成。
- **到达时间**：距目标0.06m 内，且能按共享加速度上限刹停。打印各自实际到达时间或 `timeout`；**截尾均值**把每个未完成者代入24s。24s 不是观测到达时间，也不是最终均值估计。
- **截止未完成**：24s 时尚未到达的数量，分母2。
- **死锁代理**：双方未完成，且最后3s 合计位移低于0.02m。
- **饥饿代理**：未完成且累计速度低于0.06m/s 的时间至少12s。可以与死锁代理重叠，两个有限测试都不能证明无限期饥饿 / 死锁。

15行种子 / 臂结果保留所有失败。5个种子汇总，忽略他人臂有 **5次碰撞事件**，截尾均值 **12.740±0.355 s**，最小中心距 **0.004718 m**；互斥臂 **0次碰撞**，均值 **18.295±0.319 s**，但有 **5个截止未完成**、**4个饥饿代理**，最小中心距 **1.096968 m**。互惠停车 **0次碰撞**、**10个截止未完成**、**5次死锁代理**、**10个饥饿代理**，最小中心距 **0.518211 m**。它的截尾均值24s、标准差0来自全超时，不是成功性能相同。互斥超时不是死锁：第二个机器人已获准通行，但顺序调度超过此处紧预算。不能预设普遍优胜；三者都未完整满足这个“无碰撞且双方截止前到达”的任务。

```python
#!/usr/bin/env python3
"""Two holonomic discs, single-lane corridor with holding rooms (stdlib).
Shared starts/goals and pre-generated neighbor-noise draws across all arms.
Ignore sees own route only. Mutex uses reliable request/grant/release messages
and a shared reservation: EXTRA information, not a local-only competitor.
ORCA-lite is an ORCA-INSPIRED reciprocal STOP sketch, NOT the ORCA velocity-
obstacle construction/LP. Each approaching disc takes half the available
clearance budget over horizon TAU, slowing BOTH. No passing-side or back-out
planner: local collision avoidance can deadlock in a single-lane corridor.
For clarity, this is not the linear program of the original RVO/ORCA.
Holonomic vector velocities, same speed/acceleration limits in ALL arms.
Acceleration/noise can invalidate ideal reciprocal-avoidance guarantees;
we measure collisions instead of claiming a certificate or real ORCA results.
Kinematic overlaps are logged, not resolved with fictional contact dynamics.
Real fleet/communications 本页未运行; IL/VMAS paper is reading only, 以论文为准.
"""
import math
import random
import statistics

DT, LIMIT_S = .05, 24.
VMAX, AMAX, RADIUS = .6, .8, .22
TAU, MARGIN, SENSOR_RANGE = 2., .08, 3.
SEEDS = (926, 927, 928, 929, 930)
ARMS = ("ignore_others", "priority_mutex_extra_info", "ORCA_lite_reciprocal_stop")


def norm(v):
    return math.hypot(*v)


def sub(a, b):
    return (a[0]-b[0], a[1]-b[1])


def cap(v, length):
    scale = min(1., length/max(norm(v), 1e-15))
    return (v[0]*scale, v[1]*scale)


def scenario(seed):
    rng = random.Random(seed)
    starts = ((-rng.uniform(2.3,3.2), -rng.uniform(.65,.85)),
              (rng.uniform(2.3,3.2), rng.uniform(.65,.85)))
    goals = ((rng.uniform(2.3,3.2), -rng.uniform(.65,.85)),
             (-rng.uniform(2.3,3.2), rng.uniform(.65,.85)))
    # Same indexed perturbations, not the same measured state after paths diverge.
    noise = tuple(tuple((rng.uniform(-.005,.005), rng.uniform(-.005,.005))
                        for _ in range(2)) for _ in range(round(LIMIT_S/DT)))
    return starts, goals, noise


def segment_distance(old, new):
    # Exact closest relative point over one linear integration segment.
    r, dr = sub(old[0],old[1]), sub(sub(new[0],new[1]),sub(old[0],old[1]))
    t = max(0., min(1., -(r[0]*dr[0]+r[1]*dr[1])/max(norm(dr)**2,1e-30)))
    return norm((r[0]+t*dr[0], r[1]+t*dr[1]))


def simulate(arm, data):
    starts, goals, noise = data
    p, v = list(starts), [(0.,0.),(0.,0.)]
    routes = (((-1.4,0.),(1.4,0.),goals[0]), ((1.4,0.),(-1.4,0.),goals[1]))
    stage, arrival, waited = [0,0], [None,None], [0.,0.]
    # Both request at t=0; fixed ID priority, token released by arrival message.
    token = 0
    collisions, overlapping, minsep = 0, False, norm(sub(*starts))
    positions = [tuple(p)]
    for k, eps in enumerate(noise):
        desired = []
        for i in range(2):
            if arrival[i] is not None:
                desired.append((0.,0.)); continue
            if stage[i] < 2 and norm(sub(routes[i][stage[i]],p[i])) < .04:
                stage[i] += 1
            delta = sub(routes[i][stage[i]],p[i])
            pref = cap((2*delta[0],2*delta[1]),VMAX)
            if arm == "priority_mutex_extra_info" and i != token:
                pref = (0.,0.)
            elif arm == "ORCA_lite_reciprocal_stop":
                j = 1-i
                rel = (p[j][0]-p[i][0]+eps[i][0], p[j][1]-p[i][1]+eps[i][1])
                dist = norm(rel)
                approach = (pref[0]*rel[0]+pref[1]*rel[1])/max(dist,1e-15)
                if dist < SENSOR_RANGE and approach > 0:
                    clearance = max(0.,dist-2*RADIUS-MARGIN)
                    # Equal half of pair clearance/T; stop only, no side-step LP.
                    scale = min(1.,clearance/(2*TAU*approach))
                    pref = (pref[0]*scale,pref[1]*scale)
            desired.append(pref)
        old = tuple(p)
        for i in range(2):
            dv = cap(sub(desired[i],v[i]),AMAX*DT)
            nextv = (v[i][0]+dv[0],v[i][1]+dv[1])
            if stage[i] == 2 and norm(sub(goals[i],p[i])) < .06 and norm(v[i]) <= AMAX*DT:
                nextv = (0.,0.)
                if arrival[i] is None:
                    arrival[i] = (k+1)*DT
                    if arm == "priority_mutex_extra_info" and token == i:
                        token = 1-i  # release/grant; never pre-grants future clearance
            assert norm(nextv) <= VMAX+1e-12 and norm(sub(nextv,v[i])) <= AMAX*DT+1e-12
            v[i] = nextv
            p[i] = (p[i][0]+v[i][0]*DT,p[i][1]+v[i][1]*DT)
            # Conservative inflated single-lane rectangle; no wall projection/teleport.
            assert abs(p[i][0]) > 1.22 or abs(p[i][1]) <= .08+1e-12
            if arrival[i] is None and norm(v[i]) < .06:
                waited[i] += DT
        sep = segment_distance(old,p)
        minsep = min(minsep,sep)
        inside = sep < 2*RADIUS
        collisions += inside and not overlapping
        overlapping = inside
        positions.append(tuple(p))
    unfinished = sum(t is None for t in arrival)
    # Finite-time liveness proxies, not a proof of infinite starvation/deadlock.
    tail_start = positions[-1-round(3/DT)]
    progress = sum(norm(sub(p[i],tail_start[i])) for i in range(2) if arrival[i] is None)
    deadlock = int(unfinished == 2 and progress < .02)
```
