```python
#!/usr/bin/env python3
"""latent_predict_toy.py: tiny discrete latent imagination (stdlib).

Hand-written latent×action transition dict. From a fixed seed start state,
roll H steps under a fixed action schedule, print predicted latent sequence
and a proxy reward. Proves "imagine in latent space" needs no pixel replay
and no neural net training / weight download.
usage: python3 latent_predict_toy.py
"""
from __future__ import annotations

import random

SEED = 99
H = 8
# Discrete latents and actions.
LATENTS = ("idle", "reach", "grasp", "lift", "fail")
ACTIONS = ("wait", "move", "close", "up")
# (z, a) → next_z
TRANSITIONS: dict[tuple[str, str], str] = {
    ("idle", "wait"): "idle",
    ("idle", "move"): "reach",
    ("idle", "close"): "idle",
    ("idle", "up"): "idle",
    ("reach", "wait"): "reach",
    ("reach", "move"): "reach",
    ("reach", "close"): "grasp",
    ("reach", "up"): "fail",
    ("grasp", "wait"): "grasp",
    ("grasp", "move"): "fail",
    ("grasp", "close"): "grasp",
    ("grasp", "up"): "lift",
    ("lift", "wait"): "lift",
    ("lift", "move"): "lift",
    ("lift", "close"): "lift",
    ("lift", "up"): "lift",
    ("fail", "wait"): "idle",
    ("fail", "move"): "idle",
    ("fail", "close"): "fail",
    ("fail", "up"): "fail",
}
REWARD = {
    "idle": 0.0,
    "reach": 0.1,
    "grasp": 0.4,
    "lift": 1.0,
    "fail": -0.5,
}
# Fixed action schedule for the imagined rollout.
SCHEDULE = ("move", "move", "close", "up", "wait", "move", "wait", "wait")


def main() -> None:
    rng = random.Random(SEED)
    # Start latent drawn once from seed (pinned).
    start = LATENTS[rng.randrange(0, 3)]  # idle / reach / grasp
    assert len(SCHEDULE) == H
    print(
        f"seed={SEED} H={H} start={start} "
        f"n_latents={len(LATENTS)} n_actions={len(ACTIONS)}"
    )
    print(f"{'t':>3} {'z':<8} {'a':<8} {'z_next':<8} {'r':>6}")
    z = start
    total_r = 0.0
    path = [z]
    for t in range(H):
        a = SCHEDULE[t]
        z_next = TRANSITIONS[(z, a)]
        r = REWARD[z_next]
        total_r += r
        print(f"{t:3d} {z:<8} {a:<8} {z_next:<8} {r:6.2f}")
        z = z_next
        path.append(z)
    print(
        f"ok: path={'→'.join(path)} total_r={total_r:.2f} "
        f"(latent imagination; no pixels / no NN weights)"
    )


if __name__ == "__main__":
    main()
```

## DreamGen：用视频世界模型生成神经轨迹

上面的阅读地图里，世界模型是用来**在想象中规划或学习**的。DreamGen（Jang 等，arXiv **2505.12705**）把视频世界模型用在另一件事上：当作行为克隆的**合成数据生成器**。论文把生成结果叫作*神经轨迹*（neural trajectories）：生成的机器人视频，配上伪动作（pseudo-actions）。本节先读懂这套流程，再在 CPU 上跑一个替身玩具，每个阶段只有几行最小二乘。视频模型、GPU 和人形机器人**本页未运行**。

<Resource id="dreamgen-paper" />

### 四个阶段

| 阶段（以论文为准） | 这一步做什么 | `neural_traj_idm_toy.py` 里的替身 |
| --- | --- | --- |
| 1. 微调视频世界模型 | 用一台机器人的遥操作视频适配一个图生视频模型（论文多用 WAN2.1 + LoRA），让它学到这个本体怎么动 | 用 10 条真实轨迹拟合帧步映射 `df = [f_goal − f, 1] · W_wm`（OLS） |
| 2. 滚动世界模型 | 给它初始帧 + 语言指令，生成视频，包括它没见过的行为 | 从 40 组（起始帧, 目标帧）生成 15 步帧序列，**不带动作** |
| 3. 伪动作标注 | 视频里没有动作，用逆动力学模型（IDM）或潜动作模型（LAPA）恢复 | 在**真实带标签帧对**上拟合 IDM `df → a`，再用到**生成的**帧上 |
| 4. 训练策略 | 用视频/伪动作对训练视觉运动策略 | 用 OLS 做行为克隆：`a = [f_goal − f, 1] · W_pi` |

为什么需要第 3 步：生成的视频能看出机器人做了什么，却看不出是哪些电机指令做到的。**IDM** 看两帧，预测两帧之间的动作；论文的 IDM 是以两帧为条件的扩散 Transformer，在这台机器人自己的带标签数据上训练，不输入语言和本体感觉。**潜动作模型**则从无标签视频里学一个表示「两帧之间变了什么」的离散码（Ye 等，LAPA，arXiv **2410.11758**）。两种做法给出的标签都是模型输出，不是测量值，所以都带着 IDM 的误差。

<Resource id="lapa-paper" />

### 两个误差源，两个旋钮

一条神经轨迹可能在两个不同的地方出错。**世界模型**生成的帧可能不符合真实物理（漂移、抖动、终点不对）。**IDM** 可能从完全正确的帧上读出错误的动作。两个同时改，就分不清是哪一个拖累了策略，所以玩具一次只扫一个：

- `WM_ERR`：在每个生成帧步上加固定方向的漂移和同等大小的抖动；`IDM_ERR = 0`。
- `IDM_ERR`：在 IDM 拟合用的真实帧对上加同等大小的帧噪声；`WM_ERR = 0`。

公平性规则，全部写在代码里：真实数据策略和每个神经轨迹策略拿到**相同数量的轨迹**（`K_TRAJ = 40`），对应**同一份指令列表**；所有策略都从**同样 60 个起点**、用**同一条评估噪声流**评估；每个旋钮的效果都实测，不靠假设：`wm_frame_rmse` 比较同一条指令下生成帧与真实帧，`idm_rmse` 在 IDM 从没见过的 30 条留出真实轨迹上给 IDM 打分，`idm_gain` 是拟合出的 IDM 对角元除以帧映射精确逆的对角元。

```python
#!/usr/bin/env python3
"""neural_traj_idm_toy.py: DreamGen-style neural trajectories on 2-D frames (stdlib).

DreamGen (arXiv 2505.12705) is a 4-stage recipe; this toy keeps the stages and
replaces video pixels with synthetic 2-D "frames" f = R s of a point robot:
  1. fine-tune WM  : fit the world model's frame-step map on a few real teleop
                     trajectories (OLS of df on (f_goal - f, 1))
  2. roll out WM   : from (initial frame, goal frame) pairs, generate frame
                     sequences -- no actions in them, like generated videos
  3. pseudo-label  : an inverse dynamics model (IDM), fitted on REAL labelled
                     pairs (df -> a), labels the GENERATED frames
  4. train policy  : behaviour cloning (OLS) on the neural trajectories
Two error sources are separate knobs, swept one at a time:
  WM_ERR   generated frames drift / jitter away from real physics
  IDM_ERR  the IDM sees noisy frames when it is fitted -> attenuated labels
Fair comparison: matched trajectory count (K_TRAJ real vs K_TRAJ neural, same
instruction list), the same shared evaluation starts and evaluation noise for
every arm. Measured knob effect is printed (wm_frame_rmse, idm_rmse, and
idm_gain = fitted IDM diagonal / exact inverse; below 1 = shrunk labels) next to
the policy outcome. No pixels, no video model, no GPU; WAN/Cosmos fine-tuning,
video generation and GR1 robots are not run on this page (本页未运行).
DreamGen Bench's correlation with policy success is a paper claim only; this
toy measures nothing about it.
usage: python3 neural_traj_idm_toy.py
"""
from __future__ import annotations

import math
import random

SEED = 2505
T_STEPS = 15            # steps per trajectory
K_TRAJ = 40             # matched count: real and neural policies each get 40
K_WM_FIT = 10           # real trajectories used to fine-tune the WM and fit the IDM
N_EVAL = 60             # shared evaluation starts
GAIN = 0.3              # expert: a = GAIN * (g - s) + noise
EXPERT_NOISE = 0.01
PROCESS_NOISE = 0.005
SUCCESS_R = 0.05        # success radius in state units
# Frame map (stand-in for "camera"): rotation + scale, frames are 2-D.
R = ((1.2, -0.5), (0.5, 1.2))
DRIFT_DIR = (0.6, -0.8)  # fixed direction of WM drift (lesson-chosen)
WM_SWEEP = (0.0, 0.005, 0.01, 0.02, 0.04)
IDM_SWEEP = (0.0, 0.02, 0.05, 0.1, 0.2)


def matvec(m, v):
    return (m[0][0] * v[0] + m[0][1] * v[1], m[1][0] * v[0] + m[1][1] * v[1])


def frame(s):
    return matvec(R, s)


def solve(a, b):
    """Gaussian elimination with partial pivoting; a is n x n, b is n x k."""
    n = len(a)
    m = [list(a[i]) + list(b[i]) for i in range(n)]
    for c in range(n):
        p = max(range(c, n), key=lambda r: abs(m[r][c]))
        m[c], m[p] = m[p], m[c]
        for r in range(n):
            if r != c:
                f = m[r][c] / m[c][c]
                m[r] = [x - f * y for x, y in zip(m[r], m[c])]
    return [[m[i][j] / m[i][i] for j in range(n, len(m[0]))] for i in range(n)]


def ols(xs, ys):
    """Least squares W (d_in x d_out) minimising sum ||x W - y||^2."""
    d, k = len(xs[0]), len(ys[0])
    xtx = [[sum(x[i] * x[j] for x in xs) for j in range(d)] for i in range(d)]
    xty = [[sum(x[i] * y[j] for x, y in zip(xs, ys)) for j in range(k)] for i in range(d)]
    return solve(xtx, xty)


def apply(w, x):
    return tuple(sum(x[i] * w[i][j] for i in range(len(x))) for j in range(len(w[0])))


def instructions(rng, n):
    """(start state, goal state) pairs = 'initial frame + language' stand-in."""
    out = []
    for _ in range(n):
        s = (rng.uniform(-1, 1), rng.uniform(-1, 1))
        g = (rng.uniform(-1, 1), rng.uniform(-1, 1))
        out.append((s, g))
    return out


def real_rollout(s, g, rng):
    """Teleop on the real robot: frames AND true action labels."""
    frames, acts = [frame(s)], []
    for _ in range(T_STEPS):
        a = (GAIN * (g[0] - s[0]) + rng.gauss(0, EXPERT_NOISE),
             GAIN * (g[1] - s[1]) + rng.gauss(0, EXPERT_NOISE))
        s = (s[0] + a[0] + rng.gauss(0, PROCESS_NOISE), s[1] + a[1] + rng.gauss(0, PROCESS_NOISE))
        acts.append(a)
        frames.append(frame(s))
    return frames, acts


def fit_wm(real):
    """Stage 1: frame-step map df = [f_goal - f, 1] W_wm, fitted on real data."""
    xs, ys = [], []
    for frames, _acts, fg in real:
        for t in range(T_STEPS):
            f, f2 = frames[t], frames[t + 1]
            xs.append((fg[0] - f[0], fg[1] - f[1], 1.0))
            ys.append((f2[0] - f[0], f2[1] - f[1]))
    return ols(xs, ys)


def wm_rollout(w_wm, f0, fg, wm_err, rng):
    """Stage 2: generated frames only (no actions), with drift + jitter."""
    frames, f = [f0], f0
    for _ in range(T_STEPS):
        df = apply(w_wm, (fg[0] - f[0], fg[1] - f[1], 1.0))
        f = (f[0] + df[0] + wm_err * DRIFT_DIR[0] + rng.gauss(0, wm_err),
             f[1] + df[1] + wm_err * DRIFT_DIR[1] + rng.gauss(0, wm_err))
        frames.append(f)
    return frames


def fit_idm(real, idm_err, rng):
    """Stage 3 model: IDM (df -> a) fitted on REAL labelled pairs; idm_err = frame noise it sees."""
    xs, ys = [], []
    for frames, acts, _fg in real:
        for t in range(T_STEPS):
            f = [(x + rng.gauss(0, idm_err), y + rng.gauss(0, idm_err)) for x, y in frames[t:t + 2]]
            xs.append((f[1][0] - f[0][0], f[1][1] - f[0][1], 1.0))
            ys.append(acts[t])
    return ols(xs, ys)


def idm_label(w_idm, frames):
    return [apply(w_idm, (frames[t + 1][0] - frames[t][0], frames[t + 1][1] - frames[t][1], 1.0))
            for t in range(T_STEPS)]


def fit_policy(trajs):
    """Stage 4: BC policy a = [f_goal - f, 1] W_pi (OLS)."""
    xs, ys = [], []
    for frames, acts, fg in trajs:
        for t in range(T_STEPS):
            xs.append((fg[0] - frames[t][0], fg[1] - frames[t][1], 1.0))
            ys.append(acts[t])
    return ols(xs, ys)


def evaluate(w_pi, evals):
    """Shared starts and a re-seeded noise stream for every arm."""
    rng = random.Random(SEED + 1)
    wins, dist = 0, 0.0
    for s, g in evals:
        fg = frame(g)
        for _ in range(T_STEPS):
            f = frame(s)
            a = apply(w_pi, (fg[0] - f[0], fg[1] - f[1], 1.0))
            s = (s[0] + a[0] + rng.gauss(0, PROCESS_NOISE), s[1] + a[1] + rng.gauss(0, PROCESS_NOISE))
        d = math.hypot(s[0] - g[0], s[1] - g[1])
        wins += d < SUCCESS_R
        dist += d
    return wins, dist / len(evals)


def rmse(pairs):
    return math.sqrt(sum((a - b) ** 2 for a, b in pairs) / len(pairs))


def main() -> None:
    rng = random.Random(SEED)
    instr = instructions(rng, K_TRAJ)
    evals = instructions(random.Random(SEED + 2), N_EVAL)
    real = []
    for s, g in instr:
        frames, acts = real_rollout(s, g, rng)
        real.append((frames, acts, frame(g)))
    fit_set = real[:K_WM_FIT]
    w_wm = fit_wm(fit_set)
    held = real[K_WM_FIT:]

    print(f"seed={SEED} T={T_STEPS} K_TRAJ={K_TRAJ} (matched) K_WM_FIT={K_WM_FIT} "
          f"N_EVAL={N_EVAL} success_r={SUCCESS_R}")
    print("stages: 1 fit WM on real -> 2 WM rollout frames -> 3 IDM pseudo-labels -> 4 BC policy")
    print(f"stage1 WM step gain diag={w_wm[0][0]:.3f},{w_wm[1][1]:.3f} (expert in frames: {GAIN})")
    k, d = evaluate(fit_policy(real), evals)
    print(f"real_traj  policy (true labels, K={K_TRAJ})   success={k}/{N_EVAL} mean_final_dist={d:.4f}")

    def neural(wm_err, idm_err):
        r = random.Random(SEED + 3)  # same generation noise stream for every grid point
        w_idm = fit_idm(fit_set, idm_err, random.Random(SEED + 4))
        idm_pairs = []
        for frames, acts, _fg in held:  # IDM error measured on held-out REAL pairs
            idm_pairs += [(p, q) for pa, ta in zip(idm_label(w_idm, frames), acts) for p, q in zip(pa, ta)]
        wm_pairs, trajs = [], []
        for (s, g), (rframes, _a, fg) in zip(instr, real):
            gen = wm_rollout(w_wm, frame(s), fg, wm_err, r)
            wm_pairs += [(p, q) for gf, rf in zip(gen, rframes) for p, q in zip(gf, rf)]
            trajs.append((gen, idm_label(w_idm, gen), fg))
        k, d = evaluate(fit_policy(trajs), evals)
        ideal = R[0][0] / (R[0][0] * R[1][1] - R[0][1] * R[1][0])  # diag of the exact inverse map
        gain = (w_idm[0][0] + w_idm[1][1]) / 2 / ideal
        return rmse(wm_pairs), rmse(idm_pairs), gain, k, d

    curves = {}
    for name, sweep in (("wm_sweep", WM_SWEEP), ("idm_sweep", IDM_SWEEP)):
        knob = "WM_ERR" if name == "wm_sweep" else "IDM_ERR"
        print(f"{name}: {knob} varies, other knob = 0, same eval starts")
        print(f"  {'knob':>6} {'wm_frame_rmse':>13} {'idm_rmse':>9} {'idm_gain':>8} {'success':>8} {'mean_final_dist':>15}")
        rows = []
        for v in sweep:
            wm_r, idm_r, gain, k, d = neural(v, 0.0) if name == "wm_sweep" else neural(0.0, v)
            rows.append(k)
            print(f"  {v:6.3f} {wm_r:13.4f} {idm_r:9.4f} {gain:8.3f} {k:5d}/{N_EVAL} {d:15.4f}")
        curves[name] = rows
    for name, rows in curves.items():
        assert rows[0] >= rows[-1] and rows[-1] < rows[0], f"{name}: error must cost success"
    print(f"ok: two separate drop curves (wm {curves['wm_sweep'][0]}->{curves['wm_sweep'][-1]}, "
          f"idm {curves['idm_sweep'][0]}->{curves['idm_sweep'][-1]} of {N_EVAL}); "
          "IDM fitted on real pairs, applied to WM frames; DreamGen Bench correlation = paper claim only; "
          "video WM / GPU 本页未运行")


if __name__ == "__main__":
    main()
```

## 常见错误

- **以为读完就可以复现 DreamerV3 训练曲线。** 本课是阅读图 + 字典玩具。
- **用世界模型课替代 8.2 微调或 7.2 RL 基础。** 它们互补。
- **把 DayDreamer 摘要小时数写成你必须达到的作业指标。** 那是论文动机数字。
- **把伪动作当真值。** IDM 或潜动作给出的标签是模型输出；在带噪帧上拟合的 IDM 会把它们压小（`IDM_ERR = 0.05` 时 `idm_gain` 0.744），策略会继承这个偏差。
- **同时改世界模型和 IDM。** 那样成功率下降就无法归因。一次只扫一个旋钮，评估起点和轨迹数保持不变。
- **把 DreamGen 或 DreamGen Bench 的数字当成本站结果。** 22 种新行为和基准相关性以论文为准；玩具两者都没测。

## 感知–认知记忆库（MemoryVLA）

上面的 DreamGen 替身给**生成**帧打 IDM 伪标签。MemoryVLA 问的是另一类时间问题：任务**非马尔可夫**时，工作记忆加上巩固后的记忆库，能否找回已经离开当前帧的线索？

<Resource id="memoryvla-paper" />

**实际打开的来源（arXiv abs + HTML，curl 200）。**

- **MemoryVLA（Shi 等，arXiv 2508.19236）：** Cognition-Memory-Action 框架。VLM 把观测编成感知与认知词元形成**工作记忆**；**感知–认知记忆库**存下由其巩固的低层细节与高层语义；工作记忆**检索**决策相关条目，与当前词元融合，并经合并冗余**更新**记忆库；再由记忆条件扩散动作专家输出动作。论文报告的 SimplerEnv-Bridge / Fractal / LIBERO / Mikasa-Robo 与真机 %（如 71.9 %、84.0 %、长时程 +26）以论文为准。GPU / MemoryVLA 权重 本页未运行。

### 共享延迟线索 POMDP，三个制度

`memoryvla_bank_retrieve_toy.py` 是**合成**的，只用标准库。每个臂在相同移动预算、**无未来观测**下走同一长时程走廊；臂之间只差记忆：

| 臂 | 记忆 |
|---|---|
| A_markov | 仅当前观测 |
| B_window | 最近 `W = 6` 帧拼接 |
| C_memory_bank | 工作记忆检索巩固库条目再融合 |

结果按**三个制度**分开报告，共享输入与预算（≥5 种子）：

| 制度 | 线索时刻 | 要展示的 |
|---|---|---|
| (a) `a_no_delay` | 决策步出现线索 | A/B/C 应**打平** |
| (b) `b_beyond_window` | 线索 ∈ `[2, 9]`，按键在 `t = 15` | 延迟**故意**超过 B 的窗口（`W = 6` ⇒ B 只看得到 `cue_t ≥ 10`）；**B 与 C 信息不对等** |
| (c) `c_stale_switch` | 中途线索翻转 + 干扰库条目 + 紧 `BANK_CAP` | 记忆库必须**明显变差** |

指标：各制度成功率。五多种子 mean±sd。

```python
#!/usr/bin/env python3
"""Synthetic MemoryVLA-style perceptual-cognitive memory-bank literacy toy.

NOT MemoryVLA weights, NOT MTC stage-prefix memory, NOT DreamGen IDM/LAPA,
NOT TraceVLA. Stdlib only.

Paper (Shi et al., arXiv 2508.19236; numbers 以论文为准): long-horizon
manipulation is non-Markovian; working memory + a Perceptual-Cognitive Memory
Bank that retrieves/consolidates past tokens, then conditions the action
expert. MemoryVLA++ (2606.09827) is reading-only here — not a second toy.

Shared long-horizon POMDP for every arm (equal action budget; no future leak).
Three delay regimes are reported separately so fairness is visible:
  (a) no_delay       cue shown at the decision step — A/B/C should tie
  (b) beyond_window  cue early enough that a fixed window of W misses it
                     (delay chosen on purpose to exceed B's window; B and C
                     therefore get unequal information)
  (c) stale_switch   mid-episode cue flip + distractor bank entries + small
                     BANK_CAP so wrong retrieval visibly degrades C

Arms (same 1-D mover; differ only in memory):
  A_markov          current observation only
  B_window          concat of the last W observations
  C_memory_bank     working memory retrieves top-1 consolidated bank entry

Metrics: success rate per regime. ≥5 seeds. GPU / MemoryVLA weights 本页未运行.
"""
from __future__ import annotations

import math
import random

T = 16
BUTTON_STEP = T - 1  # 15
W = 6
# beyond_window: cue must satisfy cue_t <= BUTTON_STEP - W (= 9) so B misses it
BEYOND_CUE_HI = BUTTON_STEP - W  # 9
BANK_CAP = 8
BANK_CAP_STALE = 3  # tight capacity under distractors
SEEDS = (0, 1, 2, 3, 4)
N_EP = 40
ARMS = ("A_markov", "B_window", "C_memory_bank")
REGIMES = ("a_no_delay", "b_beyond_window", "c_stale_switch")


def obs_vector(pos: int, cue_visible: int | None, distractor: int) -> tuple:
    cue = -1 if cue_visible is None else cue_visible
    return (pos, cue, distractor)


def run_episode(arm: str, rng: random.Random, regime: str) -> bool:
    if regime == "a_no_delay":
        cue_bit = rng.randint(0, 1)
        cue_t = BUTTON_STEP  # visible at decision
        switch_t = None
        final_bit = cue_bit
        bank_cap = BANK_CAP
        inject_distractors = False
    elif regime == "b_beyond_window":
        cue_bit = rng.randint(0, 1)
        # deliberately earlier than the window can reach at decision time
        cue_t = rng.randint(2, BEYOND_CUE_HI)  # 2..9 inclusive
        switch_t = None
        final_bit = cue_bit
        bank_cap = BANK_CAP
        inject_distractors = False
    else:
        # c_stale_switch: early cue, then flip; distractors fill a tiny bank
        cue_bit = rng.randint(0, 1)
        cue_t = rng.randint(2, 5)
        switch_t = rng.randint(6, 9)
        final_bit = 1 - cue_bit
        bank_cap = BANK_CAP_STALE
        inject_distractors = True

    pos = 0
    hist: list[tuple] = []
    bank: list[tuple[tuple, float]] = []

    for t in range(T):
        distractor = rng.randint(0, 1)
        cue_vis: int | None = None
        if t == cue_t:
            cue_vis = cue_bit
        elif switch_t is not None and t == switch_t:
            cue_vis = final_bit
        o = obs_vector(pos, cue_vis, distractor)
        hist.append(o)

        if arm == "A_markov":
            cue_est = o[1]
        elif arm == "B_window":
            window = hist[-W:]
            while len(window) < W:
                window = [(-1, -1, -1)] + window
            cue_est = -1
            for w in window:
                if w[1] != -1:
                    cue_est = w[1]
        else:
            if o[1] != -1:
                key = (o[0], o[2])
                bank.append((key, float(o[1])))
                if inject_distractors:
                    # distractor consolidations with the opposite bit
                    for _ in range(2):
                        dpos = rng.randint(0, BUTTON_STEP)
                        ddis = rng.randint(0, 1)
                        bank.append(((dpos, ddis), float(1 - int(o[1]))))
                if len(bank) > bank_cap:
                    bank = bank[-bank_cap:]
            cue_est = -1.0
            if bank:
                best = min(
                    bank,
                    key=lambda kv: abs(kv[0][0] - o[0]) + abs(kv[0][1] - o[2]),
                )
                cue_est = best[1] if o[1] == -1 else float(o[1])
            elif o[1] != -1:
                cue_est = float(o[1])

        if t < BUTTON_STEP:
            pos = min(pos + 1, BUTTON_STEP)
        else:
            if cue_est == -1 or cue_est == -1.0:
                guess = rng.randint(0, 1)
            else:
                guess = int(round(cue_est))
            return guess == final_bit

    return False


_REGIME_SALT = {"a_no_delay": 11, "b_beyond_window": 22, "c_stale_switch": 33}


def run_arm(arm: str, seed: int, regime: str) -> float:
    shared = random.Random(60_000 + seed * 17 + _REGIME_SALT[regime])
    episodes = [shared.randint(0, 10**9) for _ in range(N_EP)]
    succ = []
    for ep_seed in episodes:
        ok = run_episode(arm, random.Random(ep_seed), regime)
        succ.append(1.0 if ok else 0.0)
    return sum(succ) / len(succ)


def mean_sd(xs: list[float]) -> tuple[float, float]:
    mu = sum(xs) / len(xs)
    var = sum((x - mu) ** 2 for x in xs) / (len(xs) - 1)
    return mu, math.sqrt(var)


def main() -> None:
    print(
        "Synthetic MemoryVLA memory-bank toy; shared delayed-cue POMDP; "
        "three delay regimes; not MTC stage memory; not DreamGen IDM"
    )
    print(
        f"T={T} button_t={BUTTON_STEP} window={W} "
        f"beyond_cue_t∈[2,{BEYOND_CUE_HI}] (chosen to exceed B's window; "
        f"B and C get unequal information) "
        f"bank_cap={BANK_CAP} stale_cap={BANK_CAP_STALE} "
        f"episodes/seed={N_EP} seeds={len(SEEDS)}; "
        f"no future leak; equal move budget; paper % 以论文为准"
    )
    print("regime arm success (mean+-sd over seeds)")
    for regime in REGIMES:
        for arm in ARMS:
            vals = [run_arm(arm, seed, regime) for seed in SEEDS]
            m, s = mean_sd(vals)
            print(f"{regime} {arm} {m:.4f}+-{s:.4f}")
    print(
        "Regime note: (a) cue at decision → A/B/C should tie; "
        "(b) delay beyond W → B misses, C can keep the cue (unequal info); "
        "(c) cue switch + distractors + tight bank → C degrades"
    )
    print(
        "GPU / MemoryVLA weights / robot: 本页未运行; Shi et al. 2508.19236 以论文为准; "
        "MemoryVLA++ 2606.09827 reading only"
    )


if __name__ == "__main__":
    main()
```

### 读三个制度

**(a) 无延迟。** 线索在决策步：A/B/C 都打印 **1.0000±0.0000**——线索仍在当前帧时三者打平。

**(b) 延迟超出窗口。** 线索时刻取自 `[2, 9]`，按键在 `t = 15`、`W = 6`，因此延迟是**故意**超过 B 的窗口（B 需要 `cue_t ≥ 10` 才看得到）。A 与 B 都接近随机（**0.4700±0.0481**）；C 靠记忆库保住线索（**1.0000±0.0000**）。这是 B 与 C 的**信息不对等**，不是同信息下的公平对决——按此报告，不要写成普遍赢家。A 与 B 在每个种子上打印的数值**完全相同**：线索从未落入 B 的窗口，因此 B 看不到它，退化为与马尔可夫基线相同的随机猜测。

**(c) 陈旧 / 干扰记忆。** 中途线索翻转，再加上干扰巩固与紧容量，使 C 降到 **0.2850±0.0978**，低于 A/B 的 **0.5200±0.0818**——检索出错时记忆库**明显变差**。A 与 B 在每个种子上同样打印**完全相同**的数值：早期线索与翻转后的线索在决策时都落在 B 的窗口之外，因此 B 看不到它们，与马尔可夫的随机猜测一致。
