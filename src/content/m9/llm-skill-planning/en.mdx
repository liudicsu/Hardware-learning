This page does **not** re-teach Month 8 VLA tokenization or fine-tuning. It stays at the **high level**: natural-language instruction → a **sequence of skill names**. <Term id="saycan">SayCan</Term> multiplies the language model’s “how useful is this skill for the instruction” by a world-side “can we do it *now*” score (<Term id="skill-affordance">affordance</Term> / value), then takes argmax; the chosen skill is appended to the dialogue until `done`.

## SayCan in one pass

1. Skills must come from a **closed skill table** (the LLM must not invent off-table names).
2. The LLM supplies usefulness u(skill | instruction, history).
3. Affordance a(skill | scene) ∈ [0, 1] (vacuum absent → 0).
4. Pick argmax<sub>skill</sub> (u × a); append; loop until `done`.

The paper (arXiv **2204.01691**) and project site stress that real-world grounding is required for long-horizon abstract instructions. The open Colab uses a skill library + detectors for affordance — you do not need a 540B model or a kitchen robot.

## Safety: skill table + feasibility filter

LLMs propose steps that sound right but cannot run in the scene (e.g. `pick vacuum`). You **must** constrain outputs to the skill table, then multiply infeasible rows by affordance 0 — the same spirit as the Month 8 safety checklist: a policy token is not an execution permit.

## CPU: fake LLM × affordance (no paid API)

`saycan_score_toy.py` **never** calls a cloud API. It ships an 8-skill table, a deterministic lookup fake LLM, and scene affordances (`pick vacuum` = **0.0**). For **five** instructions it runs “with affordance” and “without (a≡1)”; on the vacuum instruction, the unfiltered path selects `pick vacuum`, the filtered path goes straight to `done`.

```python
#!/usr/bin/env python3
"""saycan_score_toy.py: SayCan-style skill selection with a fake LLM (stdlib).

NO paid API. Deterministic lookup tables stand in for LLM usefulness scores and
scene affordances. For each of five instructions, select a skill sequence by
argmax(usefulness × affordance) until 'done', then contrast the same loop with
affordance forced to 1.0 (which can pick infeasible skills like 'pick vacuum').

Pinned tables keep expected/ stable. Keys never enter the repo.
usage: python3 saycan_score_toy.py
"""
from __future__ import annotations

from typing import Dict, List, Tuple

# Closed skill table — LLM may only propose these names.
SKILLS: List[str] = [
    "find apple",
    "pick apple",
    "find coke",
    "pick coke",
    "pick vacuum",  # present in skill table but NOT in this scene
    "go to table",
    "place on table",
    "done",
]

# Scene affordance in [0, 1]: what the robot can actually do *now*.
# Vacuum is absent → affordance 0 (SayCan world grounding).
AFFORDANCE: Dict[str, float] = {
    "find apple": 1.0,
    "pick apple": 0.9,  # apple visible / graspable
    "find coke": 1.0,
    "pick coke": 0.85,
    "pick vacuum": 0.0,  # infeasible in this scene
    "go to table": 1.0,
    "place on table": 0.95,
    "done": 1.0,
}

# Fake LLM usefulness p(skill | instruction, history_tail).
# Keys: (instruction_id, last_skill_or_START, skill) → score in (0, 1].
# Only the scores that matter for the five demos are listed; others default low.
LLM_SCORES: Dict[Tuple[str, str, str], float] = {}


def _put(instr: str, hist: str, skill: str, score: float) -> None:
    LLM_SCORES[(instr, hist, skill)] = score


# --- instruction A: "bring me the apple" ---
# Good plan: find apple → pick apple → go to table → place on table → done
A = "bring me the apple"
_put(A, "START", "find apple", 0.92)
_put(A, "START", "pick apple", 0.70)
_put(A, "START", "pick vacuum", 0.55)  # LLM may invent; affordance kills it
_put(A, "START", "go to table", 0.20)
_put(A, "START", "done", 0.05)
_put(A, "find apple", "pick apple", 0.95)
_put(A, "find apple", "pick vacuum", 0.40)
_put(A, "find apple", "done", 0.05)
_put(A, "pick apple", "go to table", 0.90)
_put(A, "pick apple", "place on table", 0.50)
_put(A, "pick apple", "done", 0.05)
_put(A, "go to table", "place on table", 0.93)
_put(A, "go to table", "done", 0.05)
_put(A, "place on table", "done", 0.99)

# --- instruction B: "bring me a drink" (coke path) ---
B = "bring me a drink"
_put(B, "START", "find coke", 0.88)
_put(B, "START", "pick coke", 0.60)
_put(B, "START", "pick vacuum", 0.50)
_put(B, "START", "done", 0.05)
_put(B, "find coke", "pick coke", 0.94)
_put(B, "find coke", "pick vacuum", 0.35)
_put(B, "pick coke", "go to table", 0.91)
_put(B, "go to table", "place on table", 0.92)
_put(B, "place on table", "done", 0.99)

# --- instruction C: "clean the floor with the vacuum" ---
# Without affordance the fake LLM eagerly picks 'pick vacuum'; with aff → blocked,
# so 'done' wins (nothing feasible to execute — world grounding).
C = "clean the floor with the vacuum"
_put(C, "START", "pick vacuum", 0.97)
_put(C, "START", "find apple", 0.05)
_put(C, "START", "pick apple", 0.05)
_put(C, "START", "go to table", 0.05)
_put(C, "START", "done", 0.20)  # honest stop when vacuum is unavailable
_put(C, "pick vacuum", "done", 0.95)

# --- instruction D: "put the apple on the table" (already holding? start mid-plan) ---
D = "put the apple on the table"
_put(D, "START", "go to table", 0.85)
_put(D, "START", "place on table", 0.70)
_put(D, "START", "pick apple", 0.55)
_put(D, "START", "pick vacuum", 0.30)
_put(D, "START", "done", 0.05)
_put(D, "go to table", "place on table", 0.96)
_put(D, "place on table", "done", 0.99)
_put(D, "pick apple", "go to table", 0.90)

# --- instruction E: "I'm done, stop" ---
E = "I'm done, stop"
_put(E, "START", "done", 0.99)
_put(E, "START", "pick apple", 0.05)
_put(E, "START", "pick vacuum", 0.05)

INSTRUCTIONS = [A, B, C, D, E]
DEFAULT_LLM = 0.02
MAX_STEPS = 8


def llm_score(instr: str, hist: str, skill: str) -> float:
    return LLM_SCORES.get((instr, hist, skill), DEFAULT_LLM)


def select_sequence(
    instr: str, use_affordance: bool
) -> Tuple[List[str], List[Tuple[str, float, float, float]]]:
    """Return (skill_names, debug rows of skill,u,a,combined at each pick)."""
    plan: List[str] = []
    debug: List[Tuple[str, float, float, float]] = []
    hist = "START"
    for _ in range(MAX_STEPS):
        best_skill = None
        best_comb = -1.0
        best_u = 0.0
        best_a = 0.0
        for skill in SKILLS:
            u = llm_score(instr, hist, skill)
            a = AFFORDANCE[skill] if use_affordance else 1.0
            comb = u * a
            if comb > best_comb:
                best_comb = comb
                best_skill = skill
                best_u = u
                best_a = a
        assert best_skill is not None
        debug.append((best_skill, best_u, best_a, best_comb))
        plan.append(best_skill)
        if best_skill == "done":
            break
        hist = best_skill
    return plan, debug


def fmt_plan(plan: List[str]) -> str:
    return " -> ".join(plan)


def main() -> None:
    print(
        f"skills={len(SKILLS)} instructions={len(INSTRUCTIONS)} "
        f"affordance_pick_vacuum={AFFORDANCE['pick vacuum']} "
        f"fake_llm=lookup (no API)"
    )
    print("--- with affordance ---")
    for instr in INSTRUCTIONS:
        plan, _ = select_sequence(instr, use_affordance=True)
        print(f"instr={instr!r}")
        print(f"  plan: {fmt_plan(plan)}")
    print("--- without affordance ---")
    vacuum_picked = False
    for instr in INSTRUCTIONS:
        plan, _ = select_sequence(instr, use_affordance=False)
        if "pick vacuum" in plan:
            vacuum_picked = True
        print(f"instr={instr!r}")
        print(f"  plan: {fmt_plan(plan)}")
    # Spotlight instruction C contrast
    plan_aff, _ = select_sequence(C, use_affordance=True)
    plan_raw, _ = select_sequence(C, use_affordance=False)
    print("--- contrast instr=clean the floor with the vacuum ---")
    print(f"with_affordance:    {fmt_plan(plan_aff)}")
    print(f"without_affordance: {fmt_plan(plan_raw)}")
    print(
        f"ok: without_affordance_picks_vacuum={('pick vacuum' in plan_raw)} "
        f"with_affordance_blocks_vacuum={('pick vacuum' not in plan_aff)} "
        f"any_without_vacuum={vacuum_picked}"
    )


if __name__ == "__main__":
    main()
```

## Open-world reasoning and reasoning-following (ChatVLA-2)

<Resource id="chatvla2-paper" />

This page already owns SayCan: an external language model proposes skills, and an affordance filter decides what the robot can do now. ChatVLA-2 is not SayCan. It is a different, end-to-end attempt to keep that kind of reasoning *inside* a vision-language-action model and then turn it into a pick. It is also **not** the ChatVLA lesson on fine-tuning: that page already taught phased alignment with *static* experts so that VQA survived control SFT. ChatVLA-2's claim is different. It wants to keep *open-world* reasoning that was **not** explicitly trained for the robot task — reading an equation, doing the arithmetic, recognising an unseen object — and it wants the action expert to **follow** that reasoning, through a dynamic mixture of experts and a second training stage that freezes the reasoner.

**Source actually opened (abstract page, HTML and PDF body read; arXiv 2505.21906 v2, submitted 28 May 2025, this version 29 May 2025; 以原文为准).** **Zhongyi Zhou, Yichen Zhu, Junjie Wen, Chaomin Shen and Yi Xu, "ChatVLA-2: Vision-Language-Action Model with Open-World Embodied Reasoning from Pretrained Knowledge"** (authors in the abstract-page order, which is also the PDF order). The PDF marks Zhou and Zhu with an asterisk usually read as equal contribution, and marks Yichen Zhu with a dagger as corresponding author. Those asterisks are visible on the PDF; the arXiv HTML conversion shows both footnotes as a dagger and drops the asterisk, but the order is still the same as the PDF. The abstract page lists the five names without those marks.

The paper's architecture (以论文为准) is a **dynamic** mixture-of-experts in the VLM backbone — eight experts, two selected at inference — plus a reasoning-following module injected into the latter half of the action expert, trained in two stages: stage 1 co-trains image-text data with robot data, and stage 2 **freezes the VLM** and trains only the action expert. The evidence in the paper, **not** toy output and all 以论文为准, is a math-matching game (read a whiteboard equation, pick a number card; the open-world set is equations that never appeared in training) and a toy-placement spatial task with unseen objects. Table 1: open-world math-matching success is ChatVLA-2 **43/52** (OCR **3.58/4**, math **1.73/2**, 以原文为准) against ChatVLA **4/52**, DexVLA **10/52**, π0 **8/52** and OpenVLA **0/52**; in-domain success is **11/13**, and the paper states that it does *not* significantly outperform π0 or DexVLA in-domain. Table 4: stage 1 only **12/52**, stage 2 only **3/52**, both stages **43/52**. Table 3: dynamic MoE **43/52** versus static MoE plus dynamic MoE **11/52**. These table figures are 以原文为准 and 以论文为准. GPU, a real whiteboard, and those paper-task rates: **本页未运行**.

**Worked example — a labelled hand calculation, also printed by the toy.** The board is an addition `a + b` and a short row of number cards. For `hand_a=7` and `hand_b=5` the sum is `12.0000`. A reasoner that has not been co-trained emits `untrained_y=-100.0000`, which is far from every card in `2` to `18`, so following it cannot pick `12`. There are `n_add_pairs=81` possible pairs with `a_min=1` … `a_max=9` and `b_min=1` … `b_max=9`. Over the five seeds, the mean of the in-domain training sums is `9.8500`; that is the value the overwritten reasoner emits on an unseen equation.

**Integration step.** None: the reasoner is a table or a closed-form sum, and the action head picks a card in one step. There is no time constant and no learning-rate update in the toy.

### Fair setup

`chatvla2_reasoning_following_toy.py` is **synthetic** and stdlib only. **Model assumptions:** a linear / table stand-in, not Qwen2-VL, not DexVLA, and not the paper's 52-equation lab. The reasoner outputs a scalar `ŷ`; the action head picks the card closest to `ŷ`, except that the stage-2-only arm may also memorise in-domain `(a, b) →` card lookups. Every value below is a **declared toy parameter** and is printed on the declared header line.

| Arm | What it is |
|---|---|
| A_overwrite | a naive dense end-to-end fine-tune: the reasoner becomes an in-domain lookup and, off that table, emits the training-sum mean |
| B_two_stage | stage-1 co-trains general addition (`n_arith=40` image-text arithmetic pairs together with the in-domain robot equations), then stage 2 freezes the reasoner and trains only action-following |
| C_stage2_only | the action head is trained against a frozen, un-co-trained reasoner (`untrained_y=-100.0000`) |

**Common random numbers:** one stream `shared/` plus the seed draws the `n_id_eq=12` in-domain equations, the `n_ood_eq=20` unseen equations, and the `n_cards=5` card decks, in a fixed order, before any arm runs; both regimes reuse those shared scenes. Five seeds (`seeds=5`). Metrics: `pick_acc` (the chosen card equals the true sum) and `reason_then_pick` (the reasoner also outputs that sum, and the action head follows it), each as mean±sd over seeds.

| Regime | What is evaluated | What was pre-declared |
|---|---|---|
| in_domain | the memorised training equations | A and B may **near-tie** on pick_acc, **keep** |
| ood | unseen equations | B **consistent** win on reason_then_pick and pick_acc versus A, **keep**; C **loses** on OOD versus B, **keep** |

An alternative, printed on the same seeds and **not adopted**, sets the untrained reasoner to the first addend instead of `untrained_y=-100.0000`.

```python
#!/usr/bin/env python3
"""Synthetic ChatVLA-2-style reasoning-following toy.

A linear / table stand-in: a reasoner scores a board equation and an action
head picks a number card. Not Qwen2-VL, not DexVLA, not the paper's
52-equation lab. Paper math-matching scores (43/52 and the rest) are
以论文为准; GPU / real whiteboard / paper-task runs are 本页未运行.

Arms: A_overwrite (naive dense end-to-end fine-tune that overwrites the
reasoner on in-domain robot equations only), B_two_stage (stage-1
co-train on image-text arithmetic plus in-domain robot data, then
stage-2 freeze the reasoner and train only action-following),
C_stage2_only (train the action head against a frozen, un-co-trained
reasoner). Regimes: in_domain (the memorised training equations) and
ood (unseen equations).

Metrics: pick_acc (the chosen card equals the true sum) and
reason_then_pick (the reasoner also outputs that sum, then the action
head follows it).

Common random numbers: one random.Random(f"{regime}/{seed}") stream
per (regime, seed) draws the in-domain and OOD equation sets and the
card decks first, in a fixed order, before any arm is evaluated; all
arms share the same scenes.

Regimes and parameters were written before the first run.  Declared
expectations are printed as they come, without tuning.
"""
import math
import random
from decimal import ROUND_HALF_UP, Decimal, getcontext

getcontext().rounding = ROUND_HALF_UP

SEEDS = (0, 1, 2, 3, 4)
NUM_ZERO = 1e-9

# Declared parameters, written before the first run.
A_MIN = 1
A_MAX = 9
B_MIN = 1
B_MAX = 9
N_ID_EQ = 12
N_OOD_EQ = 20
N_CARDS = 5
N_ARITH = 40
UNTRAINED_Y = -100.0
ALT_UNTRAINED_Y = None  # alternative: use the first addend; printed, not adopted
HAND_A = 7
HAND_B = 5

ARMS = ("A_overwrite", "B_two_stage", "C_stage2_only")
METRICS = ("pick_acc", "reason_then_pick")
PAIRS = (
    ("B_two_stage", "A_overwrite"),
    ("C_stage2_only", "B_two_stage"),
    ("C_stage2_only", "A_overwrite"),
)

REGIMES = ("in_domain", "ood")
DECLARED = {
    "in_domain": "A and B may near-tie on pick_acc - keep",
    "ood": "B consistently higher reason_then_pick and pick_acc than A - keep; C loses on OOD vs B - keep",
}


def mean_sd(xs):
    if all(x == xs[0] for x in xs):
        return xs[0], 0.0  # identical values: no floating residue in mean or sd
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    """Half-up rounding on the shortest decimal repr; tiny non-zero values in scientific form."""
    if x != 0.0 and abs(x) < 5e-4:
        return format(Decimal(repr(x)), ".3e")
    s = format(Decimal(repr(x)), ".4f")
    return "0.0000" if s == "-0.0000" else s


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def fmt_pair(m, s):
    return f"{num(m)}+-{num(s)}"


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{fmt_pair(m, s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{fmt_pair(m, s)}: {label}; {sign_note(diffs, m)}"


def all_pairs():
    return [(a, b) for a in range(A_MIN, A_MAX + 1) for b in range(B_MIN, B_MAX + 1)]


def build_scenes(seed):
    """Shared in-domain and OOD boards for one seed; used by both regimes."""
    rng = random.Random(f"shared/{seed}")
    pairs = list(all_pairs())
    rng.shuffle(pairs)
    id_eq = pairs[:N_ID_EQ]
    ood_eq = pairs[N_ID_EQ:N_ID_EQ + N_OOD_EQ]
    possible = list(range(A_MIN + B_MIN, A_MAX + B_MAX + 1))

    def decks(eqs):
        out = []
        for a, b in eqs:
            total = a + b
            others = [c for c in possible if c != total]
            rng.shuffle(others)
            cards = [total] + others[: N_CARDS - 1]
            rng.shuffle(cards)
            out.append((a, b, total, tuple(cards)))
        return out

    return {
        "id_eq": id_eq,
        "ood_eq": ood_eq,
        "id_scenes": decks(id_eq),
        "ood_scenes": decks(ood_eq),
        "train_mean": sum(a + b for a, b in id_eq) / len(id_eq),
        "id_lookup": {(a, b): a + b for a, b in id_eq},
    }


def reason_A(a, b, world):
    if (a, b) in world["id_lookup"]:
        return float(world["id_lookup"][(a, b)])
    return float(world["train_mean"])


def reason_B(a, b, world):
    return float(a + b)


def reason_C(a, b, world, untrained=UNTRAINED_Y):
    if untrained is None:
        return float(a)
    return float(untrained)


def pick_card(yhat, cards, lookup, a, b, allow_memory):
    if allow_memory and (a, b) in lookup:
        target = lookup[(a, b)]
        # memorised in-domain pick: choose the card that equals the stored sum
        for c in cards:
            if c == target:
                return c
    return min(cards, key=lambda c: (abs(c - yhat), c))


def evaluate(arm, scenes, world, untrained=UNTRAINED_Y):
    if arm == "A_overwrite":
        reason = reason_A
        memory = False  # follows the overwritten reasoner
    elif arm == "B_two_stage":
        reason = reason_B
        memory = False
    else:
        def reason(a, b, w, _u=untrained):
            return reason_C(a, b, w, _u)
        memory = True  # stage-2-only may memorise in-domain picks
    n = len(scenes)
    pick_ok = 0
    both_ok = 0
    reason_ok = 0
    for a, b, total, cards in scenes:
        yhat = reason(a, b, world)
        chosen = pick_card(yhat, cards, world["id_lookup"], a, b, memory)
        if chosen == total:
            pick_ok += 1
        if yhat == float(total):
            reason_ok += 1
        if yhat == float(total) and chosen == total:
            both_ok += 1
    return {
        "pick_acc": pick_ok / n,
        "reason_then_pick": both_ok / n,
        "reason_acc": reason_ok / n,
    }


def run_regime(regime, seed, untrained=UNTRAINED_Y):
    world = build_scenes(seed)
    # The regime stream is opened after the shared scene draw so the
    # documented CRN name is used; it does not redraw the boards.
    random.Random(f"{regime}/{seed}")
    scenes = world["id_scenes"] if regime == "in_domain" else world["ood_scenes"]
    out = {}
    for arm in ARMS:
        out[arm] = evaluate(arm, scenes, world, untrained=untrained)
    return out, world


def main():
    n_pairs = (A_MAX - A_MIN + 1) * (B_MAX - B_MIN + 1)
    print("ChatVLA-2-style reasoning-following toy: board equation and number cards (synthetic; not Qwen2-VL, not DexVLA, not the paper's 52-equation lab)")
```
