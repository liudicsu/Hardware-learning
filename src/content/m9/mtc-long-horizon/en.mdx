Month 5's MoveIt pick-and-place plans **one** trajectory well; a real task is “open → connect to grasp → approach → IK → allow collision → close → attach → lift → … → home.” <Term id="task-constructor">MoveIt Task Constructor (MTC)</Term> splits that long horizon into composable <Term id="mtc-stage">stages</Term>. This lesson is only about stage types and dependencies — **not** a learned policy; it complements Month 8 VLAs that emit action chunks from one policy.

## A task is a chain of stages

The official tutorial decomposes hard planning into subproblems. Each stage, by **result flow**, is one of:

1. **generator** — produces solutions on its own (e.g. `CurrentState`, `GenerateGraspPose`).
2. **propagator** — consumes upstream solutions and pushes downstream (approach, close, lift).
3. **connector** — **connects** a feasible motion between two existing solutions (`Connect`).

Three containers:

- **serial** — children in order.
- **parallel** — try or merge in parallel.
- **wrapper** — wrap one child (collision edits, timeouts, …).

When you read the official pick-place skeleton, keep this spine: `CurrentState` → open hand → **Connect** → SerialContainer(approach, GenerateGraspPose + ComputeIK, allow collision, close, attach, lift) → symmetric place stages → return home. Drop that Connect and the serial grasp container has nothing to hang from.

```python
#!/usr/bin/env python3
"""stage_graph_toy.py: MoveIt Task Constructor–style stage DAG checks (stdlib).

Models a pick-place task as stages with types (generator / propagator /
connector) and container roles (serial / parallel / wrapper). Detects cycles,
disconnected stages, and a missing Connect between open-hand and the grasp
serial container — the classic MTC wiring mistake.

Prints a fixed stage table (SEED pins any mock cost noise) plus diagnostics.
Not a MoveIt / ROS 2 run — no URDF required.
usage: python3 stage_graph_toy.py
"""
from __future__ import annotations

from collections import defaultdict, deque

SEED = 9
# Mock planning costs (seconds); fixed table, no RNG draws so expected/ is stable.
STAGES = [
    # name, type, container, depends_on (comma-separated predecessors)
    ("current_state", "generator", "root", ""),
    ("open_hand", "propagator", "serial", "current_state"),
    ("connect_to_grasp", "connector", "root", "open_hand"),
    ("approach", "propagator", "serial_grasp", "connect_to_grasp"),
    ("generate_grasp_pose", "generator", "serial_grasp", "approach"),
    ("compute_ik", "propagator", "serial_grasp", "generate_grasp_pose"),
    ("allow_collision", "propagator", "serial_grasp", "compute_ik"),
    ("close_hand", "propagator", "serial_grasp", "allow_collision"),
    ("attach_object", "propagator", "serial_grasp", "close_hand"),
    ("lift", "propagator", "serial_grasp", "attach_object"),
    ("connect_to_place", "connector", "root", "lift"),
    ("place_approach", "propagator", "serial_place", "connect_to_place"),
    ("generate_place_pose", "generator", "serial_place", "place_approach"),
    ("place_ik", "propagator", "serial_place", "generate_place_pose"),
    ("open_place", "propagator", "serial_place", "place_ik"),
    ("detach", "propagator", "serial_place", "open_place"),
    ("retreat", "propagator", "serial_place", "detach"),
    ("return_home", "propagator", "serial", "retreat"),
]

# Deterministic mock costs keyed by name (seconds of "planning effort").
MOCK_COST = {
    "current_state": 0.01,
    "open_hand": 0.05,
    "connect_to_grasp": 0.40,
    "approach": 0.20,
    "generate_grasp_pose": 0.15,
    "compute_ik": 0.25,
    "allow_collision": 0.02,
    "close_hand": 0.05,
    "attach_object": 0.02,
    "lift": 0.18,
    "connect_to_place": 0.45,
    "place_approach": 0.20,
    "generate_place_pose": 0.12,
    "place_ik": 0.22,
    "open_place": 0.05,
    "detach": 0.02,
    "retreat": 0.15,
    "return_home": 0.30,
}


def parse_stages():
    names = []
    types = {}
    containers = {}
    preds = {}
    for name, typ, container, dep in STAGES:
        names.append(name)
        types[name] = typ
        containers[name] = container
        preds[name] = [p for p in dep.split(",") if p]
    return names, types, containers, preds


def successors(preds):
    succ = defaultdict(list)
    for n, ps in preds.items():
        for p in ps:
            succ[p].append(n)
    return succ


def find_cycle(names, preds):
    """Return one cycle as a list of names, or None."""
    color = {n: 0 for n in names}  # 0=white 1=grey 2=black
    stack = []

    def dfs(u):
        color[u] = 1
        stack.append(u)
        for v in successors(preds)[u]:
            if color[v] == 1:
                i = stack.index(v)
                return stack[i:] + [v]
            if color[v] == 0:
                cyc = dfs(v)
                if cyc:
                    return cyc
        stack.pop()
        color[u] = 2
        return None

    for n in names:
        if color[n] == 0:
            cyc = dfs(n)
            if cyc:
                return cyc
    return None


def reachable_from(root, names, preds):
    succ = successors(preds)
    seen = set()
    q = deque([root])
    while q:
        u = q.popleft()
        if u in seen:
            continue
        seen.add(u)
        for v in succ[u]:
            q.append(v)
    return seen


def missing_connect(names, types, preds):
    """Flag if open_hand has no outgoing connector before the grasp serial chain."""
    issues = []
    if "open_hand" not in names:
        return issues
    succ = successors(preds)
    outs = succ.get("open_hand", [])
    if not any(types.get(s) == "connector" for s in outs):
        issues.append(
            "missing_connect: open_hand has no connector successor "
            "(need Connect before serial_grasp)"
        )
    # Also: every generator that is not the root current_state should eventually
    # feed a connector or be inside a container that a connector reaches.
    return issues


def broken_graph():
    """A deliberately broken DAG: drop connect_to_grasp and add a cycle edge."""
    names, types, containers, preds = parse_stages()
    # remove the connect stage
    names = [n for n in names if n != "connect_to_grasp"]
    for d in (types, containers, preds):
        d.pop("connect_to_grasp", None)
    # approach now hangs off open_hand directly (no connector)
    preds["approach"] = ["open_hand"]
    # cycle: return_home -> current_state
    preds["current_state"] = ["return_home"]
    return names, types, containers, preds


def diagnose(names, types, containers, preds, label):
    print(f"=== {label} ===")
    print(f"{'stage':<22} {'type':<12} {'container':<14} {'cost_s':>7} deps")
    total = 0.0
    for n in names:
        cost = MOCK_COST.get(n, 0.0)
        total += cost
        deps = ",".join(preds[n]) if preds[n] else "-"
        print(
            f"{n:<22} {types[n]:<12} {containers[n]:<14} {cost:7.2f} {deps}"
        )
    print(f"stages={len(names)} total_mock_cost_s={total:.2f} seed={SEED}")

    cyc = find_cycle(names, preds)
    if cyc:
        print(f"FAIL cycle: {' -> '.join(cyc)}")
    else:
        print("ok: no cycle")

    root = "current_state" if "current_state" in names else names[0]
    seen = reachable_from(root, names, preds)
    disconnected = [n for n in names if n not in seen]
    if disconnected:
        print(f"FAIL disconnected_from_{root}: {','.join(disconnected)}")
    else:
        print(f"ok: all stages reachable from {root}")

    for issue in missing_connect(names, types, preds):
        print(f"FAIL {issue}")
    if not missing_connect(names, types, preds):
        print("ok: open_hand has a connector successor")

    n_gen = sum(1 for n in names if types[n] == "generator")
    n_prop = sum(1 for n in names if types[n] == "propagator")
    n_conn = sum(1 for n in names if types[n] == "connector")
    print(f"counts generators={n_gen} propagators={n_prop} connectors={n_conn}")


def main() -> None:
    names, types, containers, preds = parse_stages()
    diagnose(names, types, containers, preds, "good_pick_place")
    print()
    b_names, b_types, b_containers, b_preds = broken_graph()
    diagnose(b_names, b_types, b_containers, b_preds, "broken_no_connect_plus_cycle")


if __name__ == "__main__":
    main()
```

## Successful-prefix memory is conditional on the current world

The original18-stage DAG checks stay unchanged. `mtc_stage_memory_toy.py` studies **planning recovery**, not wiring or policy learning. The opened official MTC tutorial describes stage result flow, containers, `CurrentState` capturing the planning scene, and separate planning/execution failures. **This cache policy is our own search sketch**, not automatic cross-request caching supplied by that tutorial or an MTC API. ### One task stream, three strategies

All arms receive the **same eight ordered planning requests**, root robot/object context, candidate order0,1 and **same current-world feasibility predicate**. These are planning-only requests from a shared start, with world changes between requests: no arm executes a prefix and then gets magically reset. Only the current request is passed to search; decisions cannot look ahead in the stream.

Five stages: **connect → grasp → ik → place → retreat**, each with two candidates. Grasp0 can be proposed but both IK alternatives fail; grasp1/IK0 works. Place needs the current slot; `place_blocked` rejects both. `world_change` blocks route0, leaving route1; `both_routes_blocked` blocks both. Feasibility is discovered by charged candidate checks, not free solution labels.

| Arm | Memory and budget |
|---|---|
| `full_replan` | Deterministic depth-first search from root each request; no cross-request cache. |
| `cached_prefix` | Save an actually accepted first-three-stage prefix plus root-context key from a past full/partial search. Revalidate every edge before suffix recovery. A stale edge discards dependents and triggers root replanning. If a valid prefix's suffix fails, charge exhaustive root fallback too; no free global infeasibility oracle. |
| `bounded_8` | Same root search/information as full replanning, capped at8 expansions per request. Exhaustion means unknown, not infeasible. |

Cache contains **past successful planning results**, not future ground truth or memorized stream answers. Real validity keys/checks need robot start, object/goal dependencies, scene/configuration and attachment/collision settings. Here prefix stages do not depend on place slot, root context is fixed and current feasibility is always rechecked. Negative results are printed, not silently reused forever. Candidate ordering and predicate access are the same for all arms.

### Charge revalidation; retain stale failure and repair

One **expansion** here is one candidate-prefix predicate evaluation, including rejected candidates. Revalidation uses the **same unit**. `total_nodes = search_nodes + revalidation_nodes` is only a **node-expansion-count proxy**, not seconds, latency, a speedup ratio or measured wall-clock time. Real IK/collision queries need not cost the same. Old `stage_graph_toy.py` mock-cost seconds are not converted into these counts.

Stable repeat: full8, cache2 new +3 revalidation = **5**. But `place_blocked` costs cache **20+3=23** vs full18: suffix failure and global fallback are not free. On `world_change`, cached route0 **fails current-world revalidation**:1 charged check, discard dependents,10 search expansions to repair via route1, **11 total vs full10**. This is a genuinely failed stale candidate, not just comparing scene-version strings; final planning can still recover. No invalid prefix is executed. Both routes blocked causes another stale failure.

Totals over eight requests: full **6/8** success, **73** expansions; cache **6/8**, **56 search +14 revalidation =70**, **2 stale-prefix failures**; bounded **2/8**, **58**, **5 budget exhaustions**. Truly infeasible requests stay failures; aborted bounded work is not cheap successful planning. No universal winner: stable repeats can benefit, world changes/failures can cost more, and caps trade completeness for counted work.

```python
#!/usr/bin/env python3
"""Planning-only stage search, successful-prefix memory and expansion cap.
Same ordered requests, candidate order, current-world predicate and start state.
A node expansion means ONE candidate-prefix predicate evaluation; revalidation
is charged identically and included in total. Counts only, NOT wall-clock time.
Cache stores only an actually accepted first-three-stage prefix from the PAST,
with its root context; never future worlds, task outcomes or a solution oracle.
A stale prefix FAILS validation after a world change, is discarded and replanned.
No robot execution/reset trick: these are planning requests from the same root.
This cache policy is our sketch, not a claim of automatic MTC cache support.
Real Panda/SO-101/MoveIt planning 本页未运行.
"""
from dataclasses import dataclass

STAGES = ("connect", "grasp", "ik", "place", "retreat")
PREFIX_LEN, CAP = 3, 8
ARMS = ("full_replan", "cached_prefix", "bounded_8")


@dataclass(frozen=True)
class Request:
    name: str
    routes: tuple
    slot: object  # None: both place alternatives blocked.
    context: str = "same_start_object"


STREAM = (
    Request("warmup", (0,1), 0),
    Request("repeat", (0,1), 0),
    Request("place_blocked", (0,1), None),
    Request("recover_new_slot", (0,1), 1),
    Request("world_change", (1,), 1),
    Request("repeat_B", (1,), 0),
    Request("both_routes_blocked", (), 0),
    Request("restored_B", (1,), 0),
)


def feasible(prefix, request):
    """Current-world query, same access for all arms; no stream look-ahead."""
    stage, choice = len(prefix)-1, prefix[-1]
    if stage == 0:
        return choice in request.routes
    if stage == 1:
        return True  # IK feasibility is not known until the next stage is expanded.
    if stage == 2:
        return prefix[1] == 1 and choice == 0
    if stage == 3:
        return request.slot is not None and choice == request.slot
    return choice == 0


class Search:
    def __init__(self, request, cap=None):
        self.request, self.cap = request, cap
        self.search_nodes = self.revalidation_nodes = 0
        self.best_prefix, self.last_failure = (), "none"

    def check(self, path, revalidate=False):
        if self.cap is not None and self.search_nodes+self.revalidation_nodes >= self.cap:
            raise OverflowError("expansion_budget")
        if revalidate:
            self.revalidation_nodes += 1
        else:
            self.search_nodes += 1
        valid = feasible(path,self.request)
        if valid and min(PREFIX_LEN,len(path)) > len(self.best_prefix):
            self.best_prefix = path[:PREFIX_LEN]
        if not valid:
            self.last_failure = STAGES[len(path)-1]
        return valid

    def dfs(self, prefix):
        if len(prefix) == len(STAGES):
            return prefix
        for candidate in (0,1):
            path = prefix+(candidate,)
            if self.check(path):
                result = self.dfs(path)
                if result is not None:
                    return result
        return None


def run(arm, stream=STREAM):
    cache, cache_context = (), None
    rows = []
    for req in stream:  # Only the current request is passed to Search.
        search = Search(req,CAP if arm=="bounded_8" else None)
        prefix, stale = (), False
        if arm=="cached_prefix" and cache and cache_context==req.context:
            for end in range(1,len(cache)+1):
                if not search.check(cache[:end],revalidate=True):
                    stale = True  # Cached candidate really failed the current predicate.
                    search.best_prefix = ()  # Drop all dependent cached results.
                    break
            else:
                prefix = cache
        try:
            solution = search.dfs(prefix)
            if solution is None and prefix:
                solution = search.dfs(())  # Charge exhaustive root fallback, not a free global no-solution oracle.
            status = "success" if solution is not None else "no_solution"
        except OverflowError:
            solution, status = None, "budget_exhausted"
        if arm=="cached_prefix":
            cache = solution[:PREFIX_LEN] if solution else search.best_prefix
            cache_context = req.context
        total = search.search_nodes+search.revalidation_nodes
        if arm=="bounded_8":
            assert total <= CAP
        rows.append((req.name,status,search.search_nodes,search.revalidation_nodes,total,
                     int(stale),search.last_failure,solution))
    return rows


def main():
    print("shared stream=8 planning requests; stages=connect/grasp/ik/place/retreat candidates=0,1; prefix=3 budget=8")
    print("one expansion=one current-world prefix predicate check; total=search+revalidation; counts only, NOT wall-clock")
    print("request arm result search_nodes revalidation_nodes total_nodes stale_prefix_failed last_rejected_stage solution")
    summaries = []
    for arm in ARMS:
        rows = run(arm)
        for name,status,n,v,total,stale,reason,solution in rows:
            path = "-" if solution is None else "".join(map(str,solution))
            print(f"{name} {arm} {status} {n} {v} {total} {stale} {reason} {path}")
        summaries.append((arm,rows))
    print("arm successes/8 search_nodes revalidation_nodes total_nodes stale_prefix_failures budget_exhaustions")
    for arm,rows in summaries:
        print(f"{arm} {sum(r[1]=='success' for r in rows)}/8 {sum(r[2] for r in rows)} "
              f"{sum(r[3] for r in rows)} {sum(r[4] for r in rows)} {sum(r[5] for r in rows)} "
              f"{sum(r[1]=='budget_exhausted' for r in rows)}")
    changed = next(r for r in run("cached_prefix") if r[0]=="world_change")
    assert changed[5] == 1 and changed[3] > 0
    print("world_change: cached route0 FAILS revalidation; discard dependents, then repair via route1 (not an unsafe execution)")
    print("cache contains past accepted prefixes only; bounded exhaustion is not proof of infeasibility; no universal winner")
    print("new toy counts are not old stage_graph_toy mock seconds; real MoveIt/Panda/SO-101 本页未运行")


if __name__ == "__main__":
    main()
```
