## 为什么桌面臂也需要整机配电

六只总线舵机同时堵转时，峰值电流不是「一只舵机 × 6」那么简单——导线电阻会把负载端电压拉下去，逻辑板却还想要稳定的 5 V / 3.3 V。M2 已经见过「电池掉电压 → 开环转速变」；这里把它写成一张可算的清单。

硬规则三条（EMC 卫生，阅读 Novanta / Roboteq / ARK 布线指南后归纳，本页未实测）：

1. **电机相线远离编码器 / I2C。** 大电流边沿是噪声源；信号线与电机线分开走，交叉尽量 90°。
2. **公共地单点汇合。** 逻辑地与电机地在电源入口附近星形汇合，避免大电流在信号回路里回流。
3. **分清四类「地」与屏蔽端接（依设备规范，不能一刀切）。** 信号地（GND_D）、功率回流（GND_P / VBAT−）、机壳·PE、电缆屏蔽是不同导体。Ethernet / EtherCAT（100Base-Tx）等高频屏蔽应在**两端**形成连续低阻抗端接，让屏蔽电流能流通；**不要用 GND_D 作屏蔽回流**（噪声会直接进逻辑参考）。单端接地只适用于特定低频方案，且须按该设备规范——不能概括成「所有线缆默认源端接地」。

## 电池包：只读，不拆

串联抬电压、并联抬容量；C 率说明电池能持续/脉冲放出多大电流。BMS 负责过放、过流、短路切断——**本课不教改 BMS 参数，不给拆包步骤**。工作台上的安全习惯：

- 锂电有专用充电器，不要用可调电源「随便充一下」。
- 充电时有人在场；鼓包、异味、发热立刻停。
- 急停切的是电机总线，不是只切逻辑（见 [8.5 安全与可靠性](/zh/month/month-8/safety-reliability)）。

```python
"""power_budget_toy.py: IR drop and fuse margin for a desktop-arm bus (stdlib).

Illustrative SO-101-like bus-servo currents — NOT measured on this site.
Fixed seed for reproducibility; change n_servos / wire length to explore.

# SO-101-like bus servos: illustrative currents, not measured
V_nom = 12.0
I_idle = 0.35  # A per servo idle
I_peak = 1.8   # A stall-ish peak
```

## 往返压降、I²R 与熔断器数据门禁

`power_roundtrip_loss_toy.py` 在同一组示例电流下比较：

1. **单程 vs 往返**压降（R_go + R_ret，对称回程 = 2×单程）。
2. **I²R** 导线损耗（空载与峰值）——热**卫生**，不是绕组 MTPA。
3. **连续电流与峰值电流分开**：`I_cont_total`（每只 0.55 A × 6 = 3.30 A）与 `I_peak_total`（1.8 A × 6 = 10.80 A）是两套示例假设；载流量门禁只用连续，峰值不能冒充连续。
4. **保护状态**：默认缺厂商时间–电流曲线 / DC 分断能力 / 故障电流 → `insufficient_data`；仅当**显式给定**的数值约束被违反时才给 `reject_known_constraint`（例如 `V_at_load < V_floor`，或 `I_continuous > wire_ampacity`）。**从不**输出 `ok_slow_blow`。

选型前你必须从熔断器 + 导线数据手册拿到至少：DC 电压额定、分断能力（DC vs AC）、导线载流量与降额、预期故障电流、制造商时间–电流 / I²t，以及冲击形状。本课**不提供**这些曲线；玩具不算认证选型；更换熔断器本页未运行。

```python
#!/usr/bin/env python3
"""power_roundtrip_loss_toy.py: round-trip IR drop, I^2R loss, fuse data gates (stdlib).

Illustrative SO-101-like bus-servo currents — NOT measured on this site.
Continuous and peak currents are INDEPENDENT example assumptions (do not
pretend peak is continuous). Ampacity gate uses continuous only; peak without
a time–current / I^2t curve stays insufficient_data.
Protection outcomes are ONLY insufficient_data or reject_known_constraint.
Never emits ok_slow_blow / "suitable fuse" from a peak-current ratio alone.
"""
SEED = 7

# Illustrative currents / geometry (labelled not measured)
V_nom = 12.0
I_idle = 0.35    # A per servo idle (example assumption)
I_cont = 0.55    # A per servo continuous under load (example assumption; NOT peak)
I_peak = 1.8     # A stall-ish peak (example assumption; duration unknown)
n_servos = 6
wire_mohm_per_m = 50.0  # 22 AWG ballpark illustrative
len_short_m = 0.8
len_long_m = 1.5
V_floor = 11.0
# Illustrative continuous ampacity — deliberately below I_cont_total for reject demo
wire_ampacity_A = 3.0

I_idle_total = n_servos * I_idle
I_cont_total = n_servos * I_cont   # continuous — used for ampacity gate
I_peak_total = n_servos * I_peak   # peak — NOT passed as continuous


def one_way_R(length_m):
    return wire_mohm_per_m * 1e-3 * length_m


def round_trip_drop(I, length_m):
    R_rt = 2.0 * one_way_R(length_m)
    return I * R_rt, R_rt


def i2r_loss(I, R):
    return I * I * R


def protection_status(*, v_at_load=None, v_floor=None, i_continuous=None, wire_ampacity=None,
                      have_fuse_curve=False, have_dc_interrupt=False, have_fault_current=False):
    """Data-gated protection literacy. Never certifies a fuse as suitable."""
    if v_at_load is not None and v_floor is not None and v_at_load < v_floor:
        return "reject_known_constraint"
    if (i_continuous is not None and wire_ampacity is not None
            and i_continuous > wire_ampacity):
        return "reject_known_constraint"
    if not (have_fuse_curve and have_dc_interrupt and have_fault_current):
        return "insufficient_data"
    return "insufficient_data"


def undervoltage_label(v_short, v_long, v_floor):
    short_uv = v_short < v_floor
    long_uv = v_long < v_floor
    if short_uv and long_uv:
        return "both"
    if long_uv:
        return "long_only"
    if short_uv:
        return "short_only"
    return "neither"


# Round-trip IR / I^2R at peak (thermal hygiene numbers); idle I^2R for contrast
drop_short, R_rt_short = round_trip_drop(I_peak_total, len_short_m)
V_load_short = V_nom - drop_short
P_idle_short = i2r_loss(I_idle_total, R_rt_short)
P_peak_short = i2r_loss(I_peak_total, R_rt_short)
R_ow_short = one_way_R(len_short_m)
drop_ow_short = I_peak_total * R_ow_short

drop_long, R_rt_long = round_trip_drop(I_peak_total, len_long_m)
V_load_long = V_nom - drop_long

status_default = protection_status()  # missing fuse datasheet fields
status_long_v = protection_status(v_at_load=V_load_long, v_floor=V_floor)
# Ampacity gate uses CONTINUOUS only — never I_peak_total as i_continuous
status_ampacity = protection_status(
    i_continuous=I_cont_total, wire_ampacity=wire_ampacity_A
)
# Peak alone without curve/duration → still insufficient_data
status_peak_only = protection_status()  # same missing fields; peak not a reject input

hits_V_floor = undervoltage_label(V_load_short, V_load_long, V_floor)

assert status_default == "insufficient_data"
assert status_peak_only == "insufficient_data"
assert status_ampacity == "reject_known_constraint"
assert "ok" not in status_default
assert I_cont_total != I_peak_total

print(f"seed={SEED}")
print(f"V_nom={V_nom:.1f}")
print(f"n_servos={n_servos}")
print(f"I_idle_total={I_idle_total:.2f}")
print(f"I_cont_total={I_cont_total:.2f}")  # continuous example assumption
print(f"I_peak_total={I_peak_total:.2f}")  # peak example assumption (not continuous)
print(f"len_short_m={len_short_m:.1f}")
print(f"len_long_m={len_long_m:.1f}")
print(f"R_oneway_short_ohm={R_ow_short:.4f}")
print(f"R_roundtrip_short_ohm={R_rt_short:.4f}")
print(f"V_drop_oneway_short={drop_ow_short:.3f}")
print(f"V_drop_roundtrip_short={drop_short:.3f}")
print(f"V_at_load_roundtrip_short={V_load_short:.3f}")
print(f"P_i2r_idle_short_W={P_idle_short:.4f}")
print(f"P_i2r_peak_short_W={P_peak_short:.4f}")
print(f"V_drop_roundtrip_long={drop_long:.3f}")
print(f"V_at_load_roundtrip_long={V_load_long:.3f}")
print(f"V_floor={V_floor:.1f}")
print(f"hits_V_floor={hits_V_floor}")
print(f"protection_status_default={status_default}")
print(f"protection_status_long_wire={status_long_v}")
print(f"protection_status_ampacity={status_ampacity}")
print(f"protection_status_peak_only={status_peak_only}")
print(f"wire_ampacity_A={wire_ampacity_A:.1f}")
print("currents_labelled=example_assumptions")
print("fuse_certification=never")
```

## 库仑计 SoC 与课内定义的热降额

前面的玩具管的是**压降**，并拒绝给保险丝做认证；SoC 还停在「读 C 数、别拆包」。这一节补一个只读的 SoC 积分器和温度降额——仍然**不**刷 BMS、**不**拆包、**不**教充电流程。

### 符号约定与 η 只作用在充电

本课固定（玩具也会打印）：

- `I > 0` — **放电**（SoC 下降）
- `I < 0` — **充电**（SoC 上升）
- 库仑效率 `η` **只作用在充电**；放电用 `η = 1`。

<Formula label="库仑计">SoC_(k+1) = SoC_k − η_eff(I_k)·I_k·Δt / Q</Formula>

四个估计器共用**同一条**电流剖面（放电 → 静置 → 充电 → 更大电流放电）：

| 臂 | 错在哪里 |
|---|---|
| `ideal` | 无 —— 真值参考 |
| `shunt_bias` | 测得电流 = I + 0.05 A |
| `q_error` | 容量 Q̂ = 0.92 Q |
| `soc0_error` | 初始 SoC 低 5 个百分点 |

恒定分流偏置在均值非零电流下使 SoC 误差**近似随时间线性增长**（放电时斜率 ≈ −bias / Q）。容量误差与初始 SoC 误差是另两轴，不要揉成一句「电池漂了」。**OCV 静置重锚**只作概念点名：本页**不**伪造一张 OCV–SoC 数据手册曲线。

### 课内定义的连续降额

`I_allow(T) = I_cont_nom · max(0, 1 − K·max(0, T − T_ref))`，其中 `I_cont_nom`、`K`、`T_ref` 为课内取值。当 `|I|` 超过 `I_allow` 时玩具升起 `derate`。数字**不是**电芯数据手册。真 LiPo / BMS / 温箱：本页未运行。

```python
#!/usr/bin/env python3
"""soc_coulomb_thermal_toy.py: coulomb-counting SoC drift + thermal derate (stdlib).

Sign convention (stated, fixed for this toy):
  I > 0  discharge (SoC falls)
  I < 0  charge    (SoC rises)
Coulombic efficiency η applies ONLY on charge (I < 0). Discharge uses η = 1.
Integrator (discrete, Δt seconds):
  SoC_{k+1} = SoC_k - η_eff(I_k) * I_k * Δt / Q
  η_eff = η if I_k < 0 else 1.0

Four estimators share ONE current profile (same I[k] sequence):
  ideal          true I, true Q, true SoC0
  shunt_bias     I_meas = I + BIAS_A  (constant current-shunt offset)
  q_error        Q_hat = Q * Q_SCALE   (capacity mis-estimate)
  soc0_error     SoC_hat0 = SoC0 + SOC0_BIAS

Truth SoC uses ideal parameters. Drift from a constant shunt bias grows
~linearly in time under a non-zero mean current. Capacity and initial-SoC
errors are separate sources; OCV rest re-anchoring is named as a concept
only (no OCV curve is invented as a datasheet).

Returned SoC values are clipped to [0,1] for display only, compressing reported
error near the bounds; the internal integrator remains unclipped. This is not
physical saturation or an estimator correction, nor proof of bounded error.

Thermal hygiene: lesson-defined continuous derate
  I_allow(T) = I_cont_nom * max(0, 1 - K_DERATE * max(0, T - T_REF))
A derate_flag rises when |I| would exceed I_allow at the current T proxy.
Not a fuel-gauge IC; not a BMS flash; no charging procedure taught.
Real LiPo / BMS / thermal chamber: 本页未运行.
usage: python3 soc_coulomb_thermal_toy.py
"""
from __future__ import annotations

SEED = 41
DT = 1.0                 # s
N = 600                  # 10 min profile
Q_AS = 3600.0            # 1 A·h in ampere-seconds
ETA_CHARGE = 0.98        # coulombic efficiency on charge only
SOC0 = 0.80
BIAS_A = 0.05            # shunt offset (A), discharge-positive convention
Q_SCALE = 0.92           # capacity underestimated by 8 %
SOC0_BIAS = -0.05        # started 5 points low
I_CONT_NOM = 2.5         # A continuous rating at T_REF (lesson-defined)
T_REF = 25.0             # °C
K_DERATE = 0.02          # per °C above T_REF (lesson-defined curve)
T_COOL = 0.01            # 1/s cool toward T_REF
T0 = 27.0                # start temperature proxy °C
T_RISE_PER_W = 0.004     # toy thermal rise vs |I|*V_nom proxy
V_NOM = 14.8             # V, example assumption only


def eta_eff(i: float) -> float:
    return ETA_CHARGE if i < 0.0 else 1.0


def current_profile() -> list[float]:
    """Shared profile: discharge, rest, charge, discharge. Fixed, not random."""
    i: list[float] = []
    # 0..199: discharge 1.5 A
    i += [1.5] * 200
    # 200..249: rest
    i += [0.0] * 50
    # 250..399: charge -1.0 A
    i += [-1.0] * 150
    # 400..599: discharge 2.2 A (will trip derate as T rises)
    i += [2.2] * 200
    assert len(i) == N
    return i


def integrate(i_seq: list[float], q: float, soc0: float) -> list[float]:
    soc = soc0
    out = [soc]
    for i in i_seq:
        soc = soc - eta_eff(i) * i * DT / q
        # Display-only clip [0,1] compresses errors near the bounds.
        # Keep raw soc for the next step; clipping is not estimator correction.
        out.append(max(0.0, min(1.0, soc)))
    return out  # length N+1


def thermal_series(i_seq: list[float]) -> tuple[list[float], list[bool]]:
    t = T0
    temps: list[float] = []
    flags: list[bool] = []
    for i in i_seq:
        # Lesson-defined proxy: heat ~ |I| * V_nom; cool toward T_REF.
        # Not a cell datasheet; numbers are lesson-chosen.
        t = t + T_RISE_PER_W * abs(i) * V_NOM * DT - T_COOL * (t - T_REF) * DT
        i_allow = I_CONT_NOM * max(0.0, 1.0 - K_DERATE * max(0.0, t - T_REF))
        temps.append(t)
        flags.append(abs(i) > i_allow + 1e-12)
    return temps, flags


def main() -> None:
    i_true = current_profile()
    soc_true = integrate(i_true, Q_AS, SOC0)

    i_bias = [i + BIAS_A for i in i_true]
    soc_bias = integrate(i_bias, Q_AS, SOC0)
    soc_q = integrate(i_true, Q_AS * Q_SCALE, SOC0)
    soc_s0 = integrate(i_true, Q_AS, SOC0 + SOC0_BIAS)

    temps, derate = thermal_series(i_true)

    def err(est: list[float]) -> list[float]:
        return [a - b for a, b in zip(est, soc_true)]

    e_bias, e_q, e_s0 = err(soc_bias), err(soc_q), err(soc_s0)

    # Report at a few anchors + end
    anchors = [0, 200, 250, 400, N]
    print(f"seed={SEED}")

    print("same current profile for all arms")
    print("t_s  SoC_true  err_shunt  err_Q  err_SoC0  T_C  derate")
    for t in anchors:
        print(
            f"{t:4d}  {soc_true[t]:8.4f}  {e_bias[t]:+8.4f}  {e_q[t]:+7.4f}  "
            f"{e_s0[t]:+8.4f}  {temps[t-1] if t else T0:5.2f}  "
            f"{int(derate[t-1]) if t else 0}"
        )

    # Linear-growth check for shunt bias over the first discharge block (mean I > 0)
    # Expected rough slope ≈ -BIAS_A / Q  (SoC/s) under discharge with eta=1
    slope_est = (e_bias[200] - e_bias[0]) / 200.0
    slope_theory = -BIAS_A / Q_AS
    print(f"shunt_err_slope_per_s={slope_est:+.6e}  theory_neg_bias_over_Q={slope_theory:+.6e}")
```

## 共享回流地环与星形接地

前面的 EMC 要点已经说过：逻辑地与电机地应在电源入口附近星形汇合。本节把它做成共享脉冲的 IR 实验：电机回流走共享传感地导体，对比星形（以及开尔文传感）。它**不是** SoC 分流偏置库仑玩具，也**不是**只谈屏蔽端接的空话。

### 公平设定

| 臂 | 回流 / 传感安排 |
|---|---|

```python
#!/usr/bin/env python3
"""Synthetic shared-return ground-loop IR toy. Stdlib only.

Shared motor pulse current waveform and shared resistance map for every arm
(TI SLYT512: large return loops / star-ground joining). Arms differ only in
how the sense/ADC reference returns to the supply entry:
  A_shared_return   motor return shares the sense ground conductor (loop IR)
  B_star            star/single-point at supply entry (sense return separate)
  C_star_kelvin     star + kelvin sense pair (sense wires carry ADC bias only)

Literacy: A vs B is V = I·R on the chosen return R (shared vs star), not a
chassis measurement. R_MOTOR_FEED drop is reported separately and does NOT
enter the sense bounce (feed IR ≠ return IR into the ADC). Kelvin residual is
physical I_ADC_BIAS × R_SENSE_LEAD (no fudge factor). L·di/dt is not modelled.

Metrics: peak sense-node bounce (mV), false ADC counts (10-bit, 5 V, LSB),
peak feed drop (mV, informational). ≥5 seeds with pulse-amplitude jitter.
Not an EMC chamber certification. Chassis wiring 本页未运行.
"""
from __future__ import annotations

import math
import random

# Shared resistance map (synthetic lesson ohms, not a chassis measurement)
R_MOTOR_FEED = 0.020  # feed-side IR; reported, not added to sense bounce
R_SHARED_RETURN = 0.045  # shared segment used by A
R_STAR_RETURN = 0.008  # short star return at entry
R_SENSE_LEAD = 0.120  # sense wire resistance
I_ADC_BIAS = 50e-6  # physical ADC/sense bias through kelvin leads (A)
# Shared motor current pulse (A), triangular-ish burst
I_PEAK0 = 8.0
PULSE_SAMPLES = 40
SEEDS = (0, 1, 2, 3, 4)
JITTER = 0.10
VREF = 5.0
N_BITS = 10
LSB = VREF / (1 << N_BITS)
ARMS = ("A_shared_return", "B_star", "C_star_kelvin")


def pulse(i_peak: float) -> list[float]:
    """Deterministic shared shape: rise, hold, fall."""
    rise, hold, fall = 8, 16, 16
    assert rise + hold + fall == PULSE_SAMPLES
    out = []
    for k in range(rise):
        out.append(i_peak * (k + 1) / rise)
    out.extend([i_peak] * hold)
    for k in range(fall):
        out.append(i_peak * (fall - k) / fall)
    return out


def run(arm: str, i_peak: float) -> tuple[float, float, int, float]:
    """Return peak |bounce| V, mean |bounce| V, false ADC counts, peak feed drop V."""
    i_wave = pulse(i_peak)
    peaks = []
    false = 0
    feed_peaks = []
    # Ideal sense voltage is mid-rail stub; bounce adds series IR on the return
    v_true = 2.5
    for i in i_wave:
        # Feed-side drop is always I * R_MOTOR_FEED; never added to sense bounce.
        v_feed = i * R_MOTOR_FEED
        feed_peaks.append(abs(v_feed))
        if arm == "A_shared_return":
            # motor current through shared return; sense references that node
            v_bounce = i * R_SHARED_RETURN
        elif arm == "B_star":
            # motor return goes to star; sense still shares a tiny star impedance
            v_bounce = i * R_STAR_RETURN
        else:
            # kelvin: sense pair carries ADC bias only (physical I_bias * R_lead)
            v_bounce = I_ADC_BIAS * R_SENSE_LEAD
        peaks.append(abs(v_bounce))
        v_sense = v_true + v_bounce
        code = int(min(max(v_sense / LSB, 0), (1 << N_BITS) - 1))
        true_code = int(v_true / LSB)
        if code != true_code:
            false += 1
    peak = max(peaks)
    mean = sum(peaks) / len(peaks)
    return peak, mean, false, max(feed_peaks)


def mean_sd(xs: list[float]) -> tuple[float, float]:
    mu = sum(xs) / len(xs)
    var = sum((x - mu) ** 2 for x in xs) / (len(xs) - 1)
    return mu, math.sqrt(var)


def main() -> None:
    print(
        "Synthetic ground-loop return IR toy; shared motor pulses + R map; "
        "not an EMC certification"
    )
    print(
        f"R_shared={R_SHARED_RETURN*1e3:.1f}mohm R_star={R_STAR_RETURN*1e3:.1f}mohm "
        f"R_feed={R_MOTOR_FEED*1e3:.1f}mohm (reported, not in bounce) "
        f"R_sense={R_SENSE_LEAD*1e3:.1f}mohm I_bias={I_ADC_BIAS*1e6:.0f}uA "
        f"I_peak0={I_PEAK0:.1f}A samples={PULSE_SAMPLES} "
        f"ADC {N_BITS}-bit Vref={VREF:.1f}V LSB={LSB*1e3:.4f}mV "
        f"seeds={len(SEEDS)} amp_jitter=±{JITTER:.0%}; "
        f"A vs B = V=I*R on chosen return R (literacy, not a measurement); "
        f"L*di/dt not modelled"
    )
    print(
        "arm peak_bounce_mV mean_bounce_mV false_adc_counts peak_feed_mV "
        "(mean+-sd over seeds)"
    )
    for arm in ARMS:
        peaks, means, falses, feeds = [], [], [], []
        for seed in SEEDS:
            rng = random.Random(40_000 + seed)
            ip = I_PEAK0 * (1.0 + rng.uniform(-JITTER, JITTER))
            peak, mean, false, feed = run(arm, ip)
            peaks.append(peak * 1e3)
            means.append(mean * 1e3)
            falses.append(float(false))
            feeds.append(feed * 1e3)
        cols = [mean_sd(peaks), mean_sd(means), mean_sd(falses), mean_sd(feeds)]
        print(f"{arm} " + " ".join(f"{m:.4f}+-{s:.4f}" for m, s in cols))
    print(
        "No winner declared beyond the defined metric: A vs B is literacy V=I*R "
        "on chosen R; kelvin residual is I_bias*R_sense; feed IR stays off the "
        "sense node; L*di/dt not modelled"
    )
    print("Chassis wiring / scope-on-a-loop: 本页未运行; TI SLYT512 以原文为准")


if __name__ == "__main__":
    main()
```

## 浪涌电流与 NTC 限流

<Resource id="wiki-inrush" />
<Resource id="wiki-inrush-limiter" />

上面的熔断器清单要求拿到**浪涌波形**，但没有给出任何曲线。本节为机器人直流母线上最常见的情形补上一条：接通电源给大电容充电。Wikipedia 的 *Inrush current*（已读正文，rev 1326751079；以原文为准）说，放电的电容对电源**看起来像短路**；从直流源充电时，电流只受电源内阻和电容 ESR 限制。配套的 *Inrush current limiter*（已读正文，rev 1312732367；以原文为准）介绍了常用的解决办法。串联一只 **NTC 热敏电阻**，它**冷态阻值高**，能限制接通电流；之后**自热**，阻值下降。它必须持续耗散功率才能保持高温，所以"被保护电路不可能得到 100 % 的电源电压"。**固定电阻**同样能限制峰值，但阻值从不下降，所以一直在浪费功率。浪涌结束后，可以用继电器或 MOSFET 把两者切出电路。*Inrush current* 页面还说，刚断电时 NTC 仍然是热的，"冷却超过 1 分钟"之前限不住浪涌；限流器页面没有这样说。

本节**不是**熔断器选型，也不作任何认证：上面的清单仍然适用。它**不是**库仑计数 SoC 与热降额 toy，也**不是**元器件一课里的 Peukert 放电率–容量 toy。

<Formula label="经串联电阻 R 给电容充电">i(0) = V_S / R；   单看充电电流，∫ i² dt = V_S² C / (2 R)；   NTC：R(T) = R25 · exp(BETA · (1/T − 1/T25))</Formula>

**算例（标明的手算，toy 也会打印）。** 不加限流时，接通瞬间电流是 24 V / 0.05 Ω = 480.0000 A，给 2200 µF 充电存入 0.6336 J，按公式 I²t 为 12.6720 A²s。冷态 5 Ω 的 NTC 把串联电阻提高到 5.05 Ω：接通瞬间电流降到 4.7525 A，充电 I²t 降到 0.1255 A²s。不加限流时的稳态负载电流是 1.9917 A。这里所有数值都是**声明的玩具参数**，不是规格书数值。

**积分步长。** 每个 2 µs 步长内把限流电阻看作常数，用 RC 的精确解更新，所以电气部分不会失稳。NTC 温度在同一步长上用欧拉法推进。它的热时间常数 R_TH × C_TH 为 30 s（手算），远长于 0.05 s 的窗口，所以浪涌期间 NTC 几乎没有变热；热稳态单独按热平衡求解。

### 公平设置

`ntc_inrush_limiter_toy.py` 是**合成**的，只用标准库。它不是 SPICE，不是厂家的 NTC 曲线，也不是 IEC 61000 实验室。**模型假设:** 理想 24 V 直流源在 t = 0 接通，这是声明的替身，与上面 12 V 的预算 toy 无关。电源、导线和 ESR 合计 0.05 Ω，母线电容 2200 µF、初始放电，负载 12 Ω。NTC 在 25 °C 时 R25 = 5 Ω，BETA = 3500 K；它的热节点对环境 R_TH = 60 K/W，C_TH = 0.5 J/K。熔断预算是 0.05 s 窗口内 1.0 A²s，这是声明的数，不是熔断器曲线。这些都是**声明的玩具参数**。

| 组 | 限流 |
|---|---|
| A_none | 无 |
| B_ntc | 串联 NTC，阻值由温度决定 |

| D_bypass | 同一只 NTC，启动后由继电器短接 |

| 情形 | 起始状态 | 事先声明的内容 |
|---|---|---|
| cold_start | NTC 处于环境温度 | A 的峰值**稳定**高于 B 且超出预算；C 对 B 的峰值**可能**接近打平；C 的 p_ss 比 B 浪费更多，**保留**；B 相对 D 留有残余压降和功耗，**保留** |

```python
#!/usr/bin/env python3
"""NTC thermistor as an inrush-current limiter on a robot's 24 V DC bus.

Synthetic lumped electrical + one-node thermal model, stdlib only,
deterministic.  Not a manufacturer NTC curve, not SPICE, not an IEC 61000 lab.
Every value is a declared toy parameter, not a datasheet value.  A real PSU /
NTC disc / fuse: 本页未运行.  Not the lesson's fuse I2t certification text and
not the coulomb SoC + derate toy.  Wikipedia "Inrush current" and "Inrush
current limiter" statements are 以原文为准 and are not toy outputs.

Model.  V_S = 24.0 V ideal DC source switched on at t = 0; R_SRC = 0.05 ohm
(source + wiring + capacitor ESR); C_BUS = 2200e-6 F initially discharged; a
resistive load R_LOAD = 12.0 ohm across the capacitor.  The limiter resistance
R_lim is in series between source and capacitor.  Each electrical step holds
R_lim constant: R_s = R_SRC + R_lim; v_inf = V_S*R_LOAD/(R_s + R_LOAD);
tau = C_BUS*R_s*R_LOAD/(R_s + R_LOAD); v_c <- v_inf + (v_c - v_inf) *
exp(-DT/tau); line current i = (V_S - v_c)/R_s.  The NTC is
R(T) = R25*exp(BETA*(1/T - 1/T25)) with R25 = 5.0 ohm, BETA = 3500 K,
T25 = 298.15 K.  One thermal node: C_TH dT/dt = i^2 R(T) - (T - T_amb)/R_TH
with R_TH = 60.0 K/W, C_TH = 0.5 J/K, Euler on the same DT.  DT = 2e-6 s for
WINDOW = 0.05 s.  Peak current is the largest line current in the window and
the first sample at t = 0 counts.  I2t is the trapezoid integral of i^2 over
the window; a declared fuse budget I2T_BUDGET = 1.0 A^2 s gives fuse_ok = 1 if
I2t <= budget else 0.  Steady state solves T = T_amb + I^2 R(T) R_TH with
I = V_S/(R_SRC + R(T) + R_LOAD) by bisection on T in [T_amb, 800 K], then
v_drop_ss = I*R_lim_ss and p_ss = I^2*R_lim_ss.

Arms:
  A_none     : no limiter (R_lim = 0).
  B_ntc      : series NTC, R(T) from its temperature.
  C_fixed    : fixed resistor equal to the NTC cold resistance (R25 times the
               same tolerance factor), never falls.
  D_bypass   : the same NTC, switched out by a relay after start (ideal short
               in steady state: v_drop_ss = 0, p_ss = 0).  Declared model
               assumption: the bypassed NTC carries no current and has cooled
               to ambient before any restart.

Common random numbers: for each (regime, seed) draw ONCE from
random.Random(f"{regime}/{seed}") a capacitor factor uniform(0.9, 1.1), a
source-resistance factor uniform(0.8, 1.2), an NTC/fixed-resistor tolerance
factor uniform(0.9, 1.1) (applied to R25 for B, C and D) and an ambient
temperature T_amb = 298.15 + uniform(-3, 3) K; shared by all arms.

Regimes: "cold_start" (NTC starts at T_amb); "hot_restart" (declared model
assumption, not a Wikipedia value: supply off and on again with the capacitor
discharged but no cooling time, so B's NTC starts at its steady-state
temperature, while D's NTC starts at T_amb because it was bypassed and C has no
temperature dependence); "heavy_load" (cold start with R_LOAD = 6.0 ohm).

Metrics per seed: peak_a, i2t, fuse_ok, v_drop_ss, p_ss.

Pairs, every metric in every regime: A-B, C-B, B-D.

Regimes and parameters were written before the first run.  Pre-declared
expectations (not tuned; print whatever comes):
  cold_start  : A consistently higher peak than B and A fails the fuse budget;
                C vs B peak may near-tie; C consistently wastes more p_ss than
                B - keep; B keeps a residual v_drop_ss and p_ss above D's
                bypass - keep.
  hot_restart : B's peak rises toward A and B may fail the fuse budget - keep
                as it comes; D keeps its cold-start peak.
  heavy_load  : A fails the fuse budget; C consistently wastes more p_ss than
                B - keep; B residual above D - keep.

Alternatives (same seeds, declared before the run, not adopted):
  cold_start with R25 = 10.0 ohm (paired C-B p_ss);
  hot_restart with R_TH = 30.0 K/W (paired A-B peak_a);
  cold_start with I2T_BUDGET = 0.25 (paired A-B fuse_ok).
"""
import math
import random
from decimal import ROUND_HALF_UP, Decimal, getcontext

getcontext().rounding = ROUND_HALF_UP

SEEDS = (0, 1, 2, 3, 4)
NUM_ZERO = 1e-9

# Declared parameters, written before the first run.
V_S = 24.0
R_SRC = 0.05
C_BUS = 2200e-6
R_LOAD = 12.0
R_LOAD_HEAVY = 6.0
R25 = 5.0
BETA = 3500.0
T25 = 298.15
T_AMB_NOM = 298.15
R_TH = 60.0
C_TH = 0.5
DT = 2e-6
WINDOW = 0.05
I2T_BUDGET = 1.0
STEPS = int(WINDOW / DT)

REGIMES = ("cold_start", "hot_restart", "heavy_load")
ARMS = ("A_none", "B_ntc", "C_fixed", "D_bypass")

DECLARED = {
    "cold_start": "A consistently higher peak than B and A fails the fuse "
                  "budget; C vs B peak may near-tie; C consistently wastes "
                  "more p_ss than B - keep; B keeps a residual v_drop_ss and "
                  "p_ss above D's bypass - keep",
    "hot_restart": "B's peak rises toward A and B may fail the fuse budget - "
                   "keep as it comes; D keeps its cold-start peak",
    "heavy_load": "A fails the fuse budget; C consistently wastes more p_ss "
                  "than B - keep; B residual above D - keep",
}

METRICS = ("peak_a", "i2t", "fuse_ok", "v_drop_ss", "p_ss")
PAIRS = (("A_none", "B_ntc"),
         ("C_fixed", "B_ntc"),
         ("B_ntc", "D_bypass"))


def shared_draws(regime, seed):
    """Common random numbers: one shared draw set per (regime, seed)."""
    rng = random.Random(f"{regime}/{seed}")
    cap_f = rng.uniform(0.9, 1.1)
    src_f = rng.uniform(0.8, 1.2)
    tol = rng.uniform(0.9, 1.1)
    t_amb = T_AMB_NOM + rng.uniform(-3.0, 3.0)
    return cap_f, src_f, tol, t_amb


def ntc_r(t, r25_eff):
    return r25_eff * math.exp(BETA * (1.0 / t - 1.0 / T25))


def steady_ntc(t_amb, r25_eff, r_load, r_src, r_th):
    lo, hi = t_amb, 800.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        r = ntc_r(mid, r25_eff)
        i = V_S / (r_src + r + r_load)
        if mid - t_amb - i * i * r * r_th > 0.0:
            hi = mid
        else:
            lo = mid
    t_eq = 0.5 * (lo + hi)
    r_eq = ntc_r(t_eq, r25_eff)
    i_eq = V_S / (r_src + r_eq + r_load)
    return t_eq, r_eq, i_eq


def simulate(kind, r25_eff, t_start, t_amb, r_load, cap, r_src, r_th):
    v_c = 0.0
    t = t_start
    peak = 0.0
    i2t = 0.0
    for _ in range(STEPS):
        if kind == "ntc":
            r_lim = ntc_r(t, r25_eff)
        elif kind == "fixed":
            r_lim = r25_eff
        else:
            r_lim = 0.0
        r_s = r_src + r_lim
        i_start = (V_S - v_c) / r_s
        if i_start > peak:
            peak = i_start
        v_inf = V_S * r_load / (r_s + r_load)
        tau = cap * r_s * r_load / (r_s + r_load)
        v_next = v_inf + (v_c - v_inf) * math.exp(-DT / tau)
        i_next = (V_S - v_next) / r_s
        if i_next > peak:
            peak = i_next
        i2t += 0.5 * DT * (i_start * i_start + i_next * i_next)
        if kind == "ntc":
            t += DT * (i_start * i_start * r_lim - (t - t_amb) / r_th) / C_TH
        v_c = v_next
    return peak, i2t


def run_arm(arm, regime, seed, r25=R25, r_th=R_TH, budget=I2T_BUDGET):
    cap_f, src_f, tol, t_amb = shared_draws(regime, seed)
    r_load = R_LOAD_HEAVY if regime == "heavy_load" else R_LOAD
    cap = C_BUS * cap_f
    r_src = R_SRC * src_f
    r25_eff = r25 * tol
    if arm == "A_none":
        peak, i2t = simulate("none", r25_eff, t_amb, t_amb,
                             r_load, cap, r_src, r_th)
        v_drop, p_ss = 0.0, 0.0
    elif arm == "C_fixed":
        peak, i2t = simulate("fixed", r25_eff, t_amb, t_amb,
                             r_load, cap, r_src, r_th)
        i_ss = V_S / (r_src + r25_eff + r_load)
        v_drop = i_ss * r25_eff
        p_ss = i_ss * i_ss * r25_eff
    else:
        if regime == "hot_restart" and arm == "B_ntc":
            t_start = steady_ntc(t_amb, r25_eff, r_load, r_src, r_th)[0]
        else:
            t_start = t_amb
        peak, i2t = simulate("ntc", r25_eff, t_start, t_amb,
                             r_load, cap, r_src, r_th)
        if arm == "B_ntc":
            _, r_eq, i_eq = steady_ntc(t_amb, r25_eff, r_load, r_src, r_th)
            v_drop = i_eq * r_eq
            p_ss = i_eq * i_eq * r_eq
        else:
            v_drop, p_ss = 0.0, 0.0
    fuse_ok = 1 if i2t <= budget else 0
    return {"peak_a": peak, "i2t": i2t, "fuse_ok": fuse_ok,
            "v_drop_ss": v_drop, "p_ss": p_ss}


def mean_sd(xs):
    if all(x == xs[0] for x in xs):
        return xs[0], 0.0  # identical values: no floating residue in mean or sd
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    """Half-up rounding on the shortest decimal repr; tiny non-zero values in scientific form."""
    if x != 0.0 and abs(x) < 5e-4:
        return format(Decimal(repr(x)), ".3e")
    s = format(Decimal(repr(x)), ".4f")
    return "0.0000" if s == "-0.0000" else s


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def fmt_pair(m, s):
    return f"{num(m)}+-{num(s)}"


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{fmt_pair(m, s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{fmt_pair(m, s)}: {label}; {sign_note(diffs, m)}"


def collect(regime, **kw):
    return {arm: [run_arm(arm, regime, s, **kw) for s in SEEDS] for arm in ARMS}


def arm_col(regime, arm, **kw):
    return [run_arm(arm, regime, s, **kw) for s in SEEDS]


def col(rows, key):
    return [x[key] for x in rows]


def report(regime):
    res = collect(regime)
    print(f"{regime}:")
    print(f"  declared: {DECLARED[regime]}")
    for arm in ARMS:
        print(f"  {arm} " + " ".join(
            f"{key} {fmt_pair(*mean_sd(col(res[arm], key)))}" for key in METRICS))
    for a, b in PAIRS:
        for key in METRICS:
            print(f"  paired {a[0]}-{b[0]} {key}: {verdict(col(res[a], key), col(res[b], key))}")
    return res
```
