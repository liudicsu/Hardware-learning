A VLA rarely emits one joint target per forward pass. It emits an <Term id="action-chunk">action chunk</Term> — a short open-loop sequence. How you schedule the next chunk decides whether the arm freezes between inferences. LeRobot's async docs split the system into a **PolicyServer** (runs the model) and a **RobotClient** (talks to the arm). They may share a machine or span a network. This lesson contrasts sync idle frames with <Term id="async-inference">async inference</Term> on a CPU discrete-time toy. A timeline widget (`ChunkTimelineWidget`) is deferred; read the table until it ships.

## Sync idle vs async overlap

**Sync:** client waits for the whole inference, then plays `actions_per_chunk` steps, then waits again. Every inference window is idle time on the arm.

**Async:** client keeps playing the current chunk while the server computes the next. When remaining actions drop to `chunk_size_threshold`, it kicks off the next inference. Swapping in a fresh chunk can jump the commanded action — that is the trade for fewer idle frames. LeRobot's `chunk_size_threshold` is a fraction in [0,1] (send when the action queue is ≤ that fill). This CPU toy uses an absolute remaining-action count (8 of 16 ≈ 0.5) under the same name.

Pinned toy constants: control **50 Hz**, inference **80 ms** → **4** control ticks, `actions_per_chunk=16`, `chunk_size_threshold=8` (toy absolute remaining-action count; LeRobot's official knob is a [0,1] fraction — 8 of 16 ≈ 0.5), horizon **4 s**.

```python
#!/usr/bin/env python3
"""chunk_latency_sim.py: sync vs async action-chunk latency (discrete time, stdlib).

Models a control loop at CONTROL_HZ. The policy returns ACTION_CHUNK steps per
inference that costs INFER_MS of wall time (converted to control ticks).

Sync: robot is idle for the whole inference; then plays the chunk.
Async: robot keeps playing the previous chunk while the next inference runs;
       a new chunk is swapped in when ready (may jump).

Reports idle-frame ratio and mean |Δa| at chunk boundaries for both modes.
Pinned constants keep expected/ stable. Not run on a real arm or GPU here.
usage: python3 chunk_latency_sim.py
"""
from __future__ import annotations

CONTROL_HZ = 50
INFER_MS = 80.0
ACTION_CHUNK = 16  # actions_per_chunk
THRESHOLD = 8  # toy absolute remaining-action count (not LeRobot's [0,1] fraction);
                 # async starts next infer when remaining <= this (8 of 16 ≈ 0.5 fill)
HORIZON_S = 4.0
SEED_ACTION = 0.02  # per-step drift inside a chunk


def ticks(ms: float) -> int:
    return max(1, int(round(ms / 1000.0 * CONTROL_HZ)))


def make_chunk(start_a: float, n: int, drift: float) -> list[float]:
    return [start_a + drift * i for i in range(n)]


def sim_sync(total_ticks: int, infer_ticks: int, chunk: int) -> tuple[float, float, int]:
    t = 0
    idle = 0
    jumps = []
    a_prev = None
    base = 0.0
    while t < total_ticks:
        # infer (idle)
        for _ in range(infer_ticks):
            if t >= total_ticks:
                break
            idle += 1
            t += 1
        actions = make_chunk(base, chunk, SEED_ACTION)
        base += SEED_ACTION * chunk + 0.05  # next chunk starts a bit higher
        for i, a in enumerate(actions):
            if t >= total_ticks:
                break
            if a_prev is not None and i == 0:
                jumps.append(abs(a - a_prev))
            a_prev = a
            t += 1
    idle_ratio = idle / total_ticks
    mean_jump = sum(jumps) / len(jumps) if jumps else 0.0
    return idle_ratio, mean_jump, len(jumps)


def sim_async(total_ticks: int, infer_ticks: int, chunk: int, threshold: int) -> tuple[float, float, int]:
    t = 0
    idle = 0
    jumps = []
    a_prev = None
    # bootstrap: must wait for first chunk
    for _ in range(infer_ticks):
        idle += 1
        t += 1
    base = 0.0
    queue = make_chunk(base, chunk, SEED_ACTION)
    base += SEED_ACTION * chunk + 0.05
    infer_remaining = 0
    pending = None
    while t < total_ticks:
        if infer_remaining > 0:
            infer_remaining -= 1
            if infer_remaining == 0 and pending is not None:
                # swap
                if queue:
                    # jump from current head to new chunk start
                    pass
                queue = pending
                pending = None
        if not queue:
            # underrun — idle one tick and kick an infer if needed
            idle += 1
            t += 1
            if pending is None and infer_remaining == 0:
                pending = make_chunk(base, chunk, SEED_ACTION)
                base += SEED_ACTION * chunk + 0.05
                infer_remaining = infer_ticks
            continue
        a = queue.pop(0)
        if a_prev is not None and len(queue) == chunk - 1:
            # just started a fresh chunk (queue was refilled to chunk then popped once → chunk-1)
            # detect swap: when previous queue emptied and we took from new — handled below
            pass
        if a_prev is not None:
            da = abs(a - a_prev)
            # count only large boundary jumps (> 2 * drift)
            if da > SEED_ACTION * 2.5:
                jumps.append(da)
        a_prev = a
        t += 1
        # start next infer when remaining <= threshold
        if pending is None and infer_remaining == 0 and len(queue) <= threshold:
            pending = make_chunk(base, chunk, SEED_ACTION)
            base += SEED_ACTION * chunk + 0.05
            infer_remaining = infer_ticks
    idle_ratio = idle / total_ticks
    mean_jump = sum(jumps) / len(jumps) if jumps else 0.0
    return idle_ratio, mean_jump, len(jumps)


def main() -> None:
    total = int(HORIZON_S * CONTROL_HZ)
    it = ticks(INFER_MS)
    # print uses same name as LeRobot; value here is toy absolute count, not official fraction
    print(
        f"control_hz={CONTROL_HZ} infer_ms={INFER_MS:g} infer_ticks={it} "
        f"actions_per_chunk={ACTION_CHUNK} chunk_size_threshold={THRESHOLD} horizon_s={HORIZON_S:g}"
    )
    s_idle, s_jump, s_n = sim_sync(total, it, ACTION_CHUNK)
    a_idle, a_jump, a_n = sim_async(total, it, ACTION_CHUNK, THRESHOLD)
    print(f"{'mode':<8} {'idle_ratio':>11} {'mean_jump':>10} {'n_jumps':>8}")
    print(f"{'sync':<8} {s_idle:11.3f} {s_jump:10.4f} {s_n:8d}")
    print(f"{'async':<8} {a_idle:11.3f} {a_jump:10.4f} {a_n:8d}")
    print(f"ok: async_idle < sync_idle ({a_idle:.3f} < {s_idle:.3f})")


if __name__ == "__main__":
    main()
```

## Multi-stream alignment without looking ahead

### Three different clocks in a packet

A **capture timestamp** says when the sensor observed the scene. **Transport delay** is arrival time minus capture time in a common clock. A **clock offset** changes the sensor's reported stamp relative to that clock; it is not network delay. Jitter changes delay from packet to packet. Matching receipt order alone does not make image, encoder and tactile observations simultaneous.

This synthetic evaluator has a target-camera stream at 30 Hz, an encoder at 100 Hz and a gap-proxy channel at 60 Hz (metres, **not a real tactile force**). Nominal delays are 0.065, 0.015 and 0.035 s, with bounded jitter of ±0.010, ±0.004 and ±0.008 s. Sensor clock offsets are 0.012, −0.006 and 0.004 s. We explicitly subtract these **known synthetic offsets** before comparing event times; the toy does not estimate or synchronise real clocks. Record capture convention, clock domain and offset calibration in a real log. Camera exposure midpoint vs end time is another convention you must choose. **A fixed clock offset ≠ long-term drift**: this toy has constant offsets only; real clocks drift over time, so a one-off offset calibration cannot be reused indefinitely.

### An arrival-time causal contract

Process packets in **arrival-time order**. At each arrival `now`, first append that packet to its stream history, then define the common query `q=now−0.14 s`. The buffered policy is allowed to interpolate between samples on either side of q **only when both have already arrived by now**. Samples newer than q are not automatically forbidden; future-arriving samples are. Example: at now=1.00 s, q=0.86 s; a bracket captured at 0.85/0.88 s and received at 0.92/0.95 s is usable. A bracket endpoint received at 1.02 s is not, even if its capture stamp is 0.88 s.

Histories are sorted by corrected capture time so out-of-order delivery is handled. No bracket → skip, not extrapolate. Both conditions use the same evaluated arrival events, exact packets and noise. The aligned condition interpolates all three channels at q; the baseline takes the newest-capture sample **among already arrived** data in each stream. Nothing gets to index an offline full dataset from inside interpolation.

This buys coherence by buffering: it does **not** erase the 0.14 s age. The latest baseline is fresher but temporally inconsistent. Both are intentionally evaluated against the same **lagged query**, so the score penalises baseline phase mismatch as well as inter-stream skew; it is not evidence that buffering beats a predictor on a live current-time task. A real robot needs a separate current-state prediction / stale-data policy and an availability budget.

### Metrics and a causal exercise

The target moves as `0.30 sin(2.2t)+0.10`, the encoder as `0.22 sin(2.2t−0.3)`. Channel three is their synthetic gap. Compare camera-minus-encoder gap and gap-proxy value with true gap at q; RMSE pools those two errors. The **catch-window proxy** passes when both absolute errors are within 0.012 m. It is a retrospective tolerance test, **not actual catch success** and not a closed-loop policy evaluation. Every channel has pre-generated noise with standard deviation 0.001 m shared by both methods.

```python
#!/usr/bin/env python3
"""Causal buffered alignment vs latest-arrived multi-stream fusion (stdlib).
Capture time != arrival time != sensor clock stamp. Known synthetic clock
biases are corrected, not magically estimated. At each arrival, the query
is now-LAG. Interpolation can use samples after the query ONLY if they have
already ARRIVED by now. Never inspect future-arriving packets.
Both modes share packets/noise and are scored at the SAME lagged query.
Catch-window hit is a retrospective synthetic proxy, not live robot success.
"""
import bisect
import math
import random

SEED = 41
LAG = 0.14
CLOCK_OFFSETS = (0.012, -0.006, 0.004)
RATES = (30, 100, 60)
DELAYS = (0.065, 0.015, 0.035)
JITTER = (0.010, 0.004, 0.008)
TOL = 0.012


def truth(t):
    target = 0.30 * math.sin(2.2 * t) + 0.10
    encoder = 0.22 * math.sin(2.2 * t - 0.3)
    # Third channel is a synthetic gap proxy in metres, not a real force sensor.
    return target, encoder, target - encoder


def make_packets():
    rng = random.Random(SEED)
    packets = []
    for stream, rate in enumerate(RATES):
        for k in range(5 * rate + 1):
            capture = k / rate
            arrival = capture + DELAYS[stream] + rng.uniform(-JITTER[stream], JITTER[stream])
            stamp = capture + CLOCK_OFFSETS[stream]
            value = truth(capture)[stream] + rng.gauss(0, 0.001)
            packets.append((arrival, stream, stamp, value))
    return sorted(packets)


def interpolate(history, query, now):
    # history contains ONLY packets appended by the arrival-event loop.
    times = [p[0] for p in history]
    i = bisect.bisect_left(times, query)
    if i == 0 or i == len(history):
        return None  # need an arrived bracket; no extrapolation
    a, b = history[i - 1], history[i]
    assert a[2] <= now and b[2] <= now  # arrival, NOT capture, is availability
    assert a[0] <= query <= b[0]
    w = (query - a[0]) / (b[0] - a[0])
    return a[1] + w * (b[1] - a[1])


def main():
    # Regression: a captured endpoint is not usable before it arrives.
    assert interpolate([(0.85, 1.0, 0.92)], 0.86, 1.0) is None
    arrived = [(0.85, 1.0, 0.92), (0.88, 4.0, 0.95)]
    assert abs(interpolate(arrived, 0.86, 1.0) - 2.0) < 1e-12
    try:
        interpolate([(0.85, 1.0, 0.92), (0.88, 4.0, 1.02)], 0.86, 1.0)
    except AssertionError:
        pass
    else:
        raise AssertionError("future-arriving bracket accepted")
    histories = [[], [], []]
    errors = {mode: [] for mode in ("latest", "stamp_aligned")}
    hits = {mode: 0 for mode in errors}
    for now, stream, stamp, value in make_packets():
        capture = stamp - CLOCK_OFFSETS[stream]
        # Insert by corrected capture time, handling out-of-order arrivals.
        bisect.insort(histories[stream], (capture, value, now))
        query = now - LAG
        if query < 0.2 or query > 4.8 or not all(histories):
            continue
        aligned = [interpolate(h, query, now) for h in histories]
        if any(x is None for x in aligned):
            continue  # matched coverage for both conditions
        latest = [h[-1][1] for h in histories]  # newest capture among arrived data
        true_gap = truth(query)[2]
        for mode, values in (("latest", latest), ("stamp_aligned", aligned)):
            gap_error = values[0] - values[1] - true_gap
            proxy_error = values[2] - true_gap
            errors[mode].append((gap_error, proxy_error))
            hits[mode] += int(abs(gap_error) <= TOL and abs(proxy_error) <= TOL)
    count = len(errors["latest"])
    assert count > 0 and count == len(errors["stamp_aligned"])
    print(f"seed={SEED} lag_s={LAG:.2f} shared_packets_and_noise=True")
    print("clock_offsets_s=0.012,-0.006,0.004 known_correction=True")
    print(f"matched_arrival_events={count} catch_tolerance_m={TOL:.3f}")
    for mode, e in errors.items():
        rmse = math.sqrt(sum(a*a + b*b for a, b in e) / (2 * count))
        print(f"{mode} misalignment_rmse_m={rmse:.6f} catch_proxy={hits[mode]}/{count}")
    print("arrived_brackets_only=True extrapolation=False live_success_claim=False")


if __name__ == "__main__":
    main()
```

## Knobs

- **`actions_per_chunk`:** how many actions one inference returns. Longer chunks amortise inference; stale open-loop drift grows.
- **`chunk_size_threshold`:** LeRobot's official value is a fraction in [0,1] (send when the action queue is ≤ that fill). This CPU toy uses an absolute remaining-action count under the same name (8 of 16 ≈ 0.5): when remaining actions in the current chunk fall to this value, async starts the next inference. Too high → you infer early and waste compute; too low → you go idle again.

Change `ACTION_CHUNK` to 8 locally and re-run; watch idle ratio and jump count move.

Async removes the idle frames but not the jump where a new chunk takes over; [10.2 real-time chunking](/en/month/month-10/real-time-chunking) smooths that seam.

## Common mistakes

- **Treating async as "no latency".** Inference still takes time; you only hide it behind motion.
- **Ignoring chunk-boundary jumps.** A success-rate drop after enabling async is often a splice, not "the model got worse".
- **Running server and client without a clock plan.** 50 Hz control with 80 ms inference is already 4 ticks — write the numbers down.
- **Skipping the sync baseline.** Always compare both modes on the same starts (Month 6 protocol).

## NORA: a small VLA with FAST+ action tokens

The sections above schedule chunks so the arm does not idle. A second way to cut deployment latency is to make each inference cheaper: a **smaller backbone** and **fewer action tokens per chunk**. **NORA** does both. It is a 3B-parameter VLA that generates FAST+ action tokens autoregressively. This section tests those two cost levers, and the risk that comes with a smaller model, on a synthetic stand-in. It is **not** VLA-Adapter's 0.5B Bridge Attention policy (Month 10, edge-deploy-checklist), **not** SmolVLA (Month 6, vla-models) and **not** the π0-FAST line in the Month 6 model catalogue.

### Fewer parameters, fewer tokens

<Formula label="Decode-cost proxy">cost = parameters (B) × generated action tokens per chunk</Formula>

<Formula label="Success">success iff chunk RMSE < TOL</Formula>

The large stand-in writes one token per joint per step (256 bins, the per-step scheme from the action-token section of the vla-models lesson in Month 6). The small stand-in writes FAST-style tokens: per joint, a DCT over the chunk's time axis, scaled by GAMMA = 10.0 and rounded, keeping only nonzero integers. Byte-pair encoding is **not** modelled, so the toy undercounts FAST+'s compression. NORA's §2.3 words its DCT as running across the action dimensions; the toy uses the time-axis DCT of the FAST section in the Month 6 vla-models lesson, a stand-in, said so here.

**Worked example — labelled hand calculations, also printed by the toy.** The per-step head needs D × H = 7 × 10 = 70 tokens on a 10-step chunk and 7 × 30 = 210 on a 30-step chunk. Its bin width is 2 / 256 = 0.007813, so the binning error is at most half a bin, 0.003906. Rounding a coefficient scaled by GAMMA = 10.0 leaves an RMS error of 1 / (GAMMA √12) = 0.028868 per coefficient if the error is uniform. At equal token counts the cost ratio is P_SMALL / P_LARGE = 0.4286. Success is **strict**: RMSE 0.0600 fails, 0.0599 succeeds.

**No integration step here.** Chunks are scored, not simulated; there is no ODE.

```python
#!/usr/bin/env python3
"""NORA-style small VLA with FAST+-style action tokens vs a large per-step-binning head.

Synthetic action chunks, stdlib only.  Not the NORA 3B checkpoint, not
Qwen-2.5-VL, not the real FAST+ tokenizer (no BPE here), not LIBERO / WidowX
and not a GPU: 本页未运行.  Paper numbers (3B, 970k demonstrations, success
rates) 以论文为准 and are not toy outputs.  Not VLA-Adapter's Bridge Attention,
not SmolVLA and not the pi0-FAST catalogue line.

What the NORA paper (arXiv 2504.19854) says, and what this toy keeps:
  NORA is a 3B VLA on a Qwen-2.5-VL-3B backbone that autoregressively
  generates FAST+ action tokens (2048 extra vocabulary tokens) and decodes them
  into actions; the motivation is lower overhead than 7B-class VLAs.  This toy
  keeps two cost levers - fewer parameters and fewer action tokens per chunk -
  and one risk - a smaller backbone may fit long horizons worse.

Per (regime, seed) the chunks are smooth multi-joint trajectories; a policy's
prediction is the true chunk plus model error SIG * z, where z is one shared
standard-normal draw per value (common random numbers: the same z for every
arm, only the declared SIG differs).  Arms:
  A_large_bins : P_LARGE=7.0 B-parameter stand-in, one token per joint per step
                 (256 bins over -1..1, OpenVLA-style), SIG_LARGE
  B_small_fast : P_SMALL=3.0 B-parameter stand-in, FAST-style tokens: per joint
                 a DCT over the chunk's time axis, scaled by GAMMA and rounded;
                 tokens = nonzero rounded coefficients (BPE not modelled), SIG_SMALL
  C_shuffled   : B's model and tokens, but the decoder reads them through a
                 shuffled codebook (a fixed random permutation of token values)
Success: chunk RMSE against the true chunk strictly below TOL.  Cost proxy:
parameters x generated tokens (B-param x token), a stand-in for decode work.
No integration step: chunks are evaluated, not simulated.

Regimes and parameters were written before the first run:
  short_in_dist : H=10 steps, SIG_LARGE=SIG_SMALL=0.030 - A and B may near-tie
                  on success; B consistent lower cost; C consistent lose
  long_horizon  : H=30 steps, SIG_LARGE=0.040, SIG_SMALL=0.055 (the small
                  backbone fits long horizons worse - a declared assumption)
                  - B may LOSE success to A; keep and print whatever happens
"""
import math
import random
from decimal import Decimal, ROUND_HALF_UP

SEEDS = (0, 1, 2, 3, 4)
NUM_ZERO = 1e-9

# Declared model parameters, written before the first run (toy values).
D = 7                 # joints / action dimensions
N_EP = 200            # chunks per seed
BINS = 256            # A: per-step bins over -1..1
GAMMA = 10.0          # B/C: DCT coefficient scale before rounding (FAST-style)
P_LARGE = 7.0         # B-parameter stand-in for a 7B-class head
P_SMALL = 3.0         # B-parameter stand-in for a 3B head
TOL = 0.060           # success iff chunk RMSE < TOL (strict)
VOCAB = 64            # C: shuffled codebook over integer values -VOCAB..VOCAB

REGIMES = {
    "short_in_dist": {"H": 10, "SIG_LARGE": 0.030, "SIG_SMALL": 0.030},
    "long_horizon": {"H": 30, "SIG_LARGE": 0.040, "SIG_SMALL": 0.055},
}
ARMS = ("A_large_bins", "B_small_fast", "C_shuffled")


_COS = {}



def fmt6(x):
    """Six decimals with ROUND_HALF_UP (2/256 = 0.0078125 -> 0.007813, not the
    round-half-even 0.007812 of a plain :.6f). Batch-19 polish; helper drafted by
    opencode DeepSeek."""
    return f"{Decimal(str(x)).quantize(Decimal('0.000001'), rounding=ROUND_HALF_UP):.6f}"

def cos_table(n):
    """cos(pi*(t+0.5)*k/n) for every (k, t), cached per chunk length."""
    if n not in _COS:
        _COS[n] = [[math.cos(math.pi * (t + 0.5) * k / n) for t in range(n)] for k in range(n)]
    return _COS[n]


def dct(x):
    n = len(x)
    cs = cos_table(n)
    out = []
    for k in range(n):
        s = sum(x[t] * cs[k][t] for t in range(n))
        out.append(s * math.sqrt((1.0 if k == 0 else 2.0) / n))
    return out


def idct(c):
    n = len(c)
    cs = cos_table(n)
    w = [math.sqrt((1.0 if k == 0 else 2.0) / n) for k in range(n)]
    out = []
    for t in range(n):
        s = sum(c[k] * w[k] * cs[k][t] for k in range(n))
        out.append(s)
    return out


def shared_chunks(regime, seed):
    """Common random numbers: chunks, model-error draws and the codebook permutation."""
    h = REGIMES[regime]["H"]
    rng = random.Random(f"{regime}/{seed}")
    eps = []
    for _ in range(N_EP):
        truth, z = [], []
        for _ in range(D):
            a, f, ph, off = rng.uniform(0.2, 0.6), rng.uniform(0.2, 1.0), rng.uniform(0.0, 2 * math.pi), rng.uniform(-0.2, 0.2)
            truth.append([off + a * math.sin(2 * math.pi * f * t / h + ph) for t in range(h)])
            z.append([rng.gauss(0.0, 1.0) for _ in range(h)])
        eps.append({"truth": truth, "z": z})
    vals = list(range(-VOCAB, VOCAB + 1))
    perm = vals[:]
    rng.shuffle(perm)
    return eps, dict(zip(vals, perm))


def clip(v):
    return max(-1.0, min(1.0, v))


def bins_roundtrip(row):
    w = 2.0 / BINS
    return [-1.0 + (min(int((clip(v) + 1.0) / w), BINS - 1) + 0.5) * w for v in row]


def fast_tokens(row, gamma):
    return [int(round(gamma * c)) for c in dct([clip(v) for v in row])]


def run_seed(regime, seed, arm, gamma=GAMMA):
    cfg = REGIMES[regime]
    sig = cfg["SIG_LARGE"] if arm == "A_large_bins" else cfg["SIG_SMALL"]
    params = P_LARGE if arm == "A_large_bins" else P_SMALL
    eps, book = shared_chunks(regime, seed)
    succ = 0
    tok_sum = rmse_sum = 0.0
    for ep in eps:
        se = 0.0
        tokens = 0
        for d in range(D):
            truth = ep["truth"][d]
            pred = [x + sig * e for x, e in zip(truth, ep["z"][d])]
            if arm == "A_large_bins":
                rec = bins_roundtrip(pred)
                tokens += len(pred)
            else:
                q = fast_tokens(pred, gamma)
                tokens += sum(1 for v in q if v != 0)
                if arm == "C_shuffled":
                    q = [book.get(v, v) if v != 0 else 0 for v in q]
                rec = idct([v / gamma for v in q])
            se += sum((r - x) ** 2 for r, x in zip(rec, truth))
        rmse = math.sqrt(se / (D * len(ep["truth"][0])))
        rmse_sum += rmse
        tok_sum += tokens
        succ += int(rmse < TOL)
    tok = tok_sum / len(eps)
    return {"success_pct": 100.0 * succ / len(eps), "rmse": rmse_sum / len(eps),
            "tokens": tok, "cost": params * tok}


def mean_sd(xs):
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    if x != 0.0 and abs(x) < 5e-4:
        return f"{x:.3e}"
    return f"{x:.4f}"


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def fmt_pair(m, s):
    return f"{num(m)}+-{num(s)}"


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{fmt_pair(m, s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{fmt_pair(m, s)}: {label}; {sign_note(diffs, m)}"


_TABLES = {}


def table(regime, gamma=GAMMA):
    if (regime, gamma) not in _TABLES:
        _TABLES[(regime, gamma)] = {arm: [run_seed(regime, seed, arm, gamma) for seed in SEEDS] for arm in ARMS}
    return _TABLES[(regime, gamma)]


def col(res, arm, key):
    return [float(r[key]) for r in res[arm]]
```
