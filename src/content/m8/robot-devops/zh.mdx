```python
#!/usr/bin/env python3
"""bag_stats.py: topic rates and drop gaps from a synthetic rosbag2/MCAP fixture.

Real Jazzy rosbag2 defaults to the mcap storage plugin
(rosbag2_storage::get_default_storage_id() returns \"mcap\" on the jazzy branch).
This toy reads a JSON timeline that mirrors per-topic stamp sequences extracted
from such a bag — no libmcap required, so expected/ stays CPU-reproducible.

Reports Hz estimates and inter-arrival gaps that exceed 1.5× the nominal period.
usage: python3 bag_stats.py
"""
from __future__ import annotations

import json
import os

HERE = os.path.dirname(os.path.abspath(__file__))
FIX = os.path.join(HERE, "sample_bag", "timeline.json")


def main() -> None:
    with open(FIX, encoding="utf-8") as f:
        bag = json.load(f)
    assert bag.get("storage_id") == "mcap"
    print(
        f"bag={bag['uri']} storage_id={bag['storage_id']} "
        f"duration_s={bag['duration_s']:g} (synthetic fixture; not a binary .mcap)"
    )
    print(f"{'topic':<28} {'n':>5} {'hz_est':>8} {'nominal':>8} {'gaps':>5}")
    total_gaps = 0
    for topic in bag["topics"]:
        name = topic["name"]
        stamps = topic["stamps_ns"]
        nominal_hz = float(topic["nominal_hz"])
        n = len(stamps)
        if n >= 2:
            dur = (stamps[-1] - stamps[0]) / 1e9
            hz = (n - 1) / dur if dur > 0 else float("nan")
        else:
            hz = 0.0
        period = 1e9 / nominal_hz
        gaps = 0
        for a, b in zip(stamps, stamps[1:]):
            if (b - a) > 1.5 * period:
                gaps += 1
        total_gaps += gaps
        print(f"{name:<28} {n:5d} {hz:8.2f} {nominal_hz:8.1f} {gaps:5d}")
    print(f"total_gap_events={total_gaps}")
    print("ok: fixture storage_id is mcap (Jazzy rosbag2 default)")


if __name__ == "__main__":
    main()
```

## Bag 验收 ≠ 文件夹存在

磁盘上有个文件夹 ≠ 合格录包。`bag_qos_accept_toy.py` 用三条门槛闸合成时间线（须全部通过）：

1. **必需 topic 存在且 `n > 0`。** 缺少 `/camera/color/image_raw` 且 `n==0` → **拒收**，即使文件夹在。
2. **缺口预算。** 到达间隔超过名义周期的 `1.5×`；某必需 topic 缺口数超过 `max_gaps_allowed=2` → **拒收**。
3. **消息预算。** 每个必需 topic 须满足钉死的 `budget_min`（`n >= budget_min`）。固定示例行：`n=100` 对 `budget_min=90` → 通过；`n=10` 对 `budget_min=50` → 失败。**不要对调这两行比较。**

### Request vs Offered、录包适配与持久性（F1）

ROS 2 QoS 采用 **Request vs Offered（请求 vs 提供）** 模型（[About QoS Settings — Jazzy](https://docs.ros.org/en/jazzy/Concepts/Intermediate/About-Quality-of-Service-Settings.html)；RST 镜像：`ros2/ros2_documentation` jazzy 分支 `About-Quality-of-Service-Settings.rst`）。订阅方 **request** 可接受的最低质量；发布方 **offer** 能提供的最高质量。仅当**每一项**被请求的策略都不比对方提供的更严时才建连。可靠性或持久性不兼容 ⇒ 无匹配 ⇒ 无消息（**覆盖**请求过严的录包端会像「该 topic 空的」）。

**录包默认 vs 覆盖**（Jazzy [rosbag2 README — Overriding QoS Profiles](https://github.com/ros2/rosbag2/blob/jazzy/README.md)；raw：`https://raw.githubusercontent.com/ros2/rosbag2/jazzy/README.md`）：

- **默认（无覆盖）：** rosbag2 会按现有发布方的 *offered* 配置，**适配**自己的 *requested* 订阅 QoS。源码：`Rosbag2QoS::adapt_request_to_offers`（[`rosbag2_storage/.../qos.cpp`](https://github.com/ros2/rosbag2/blob/jazzy/rosbag2_storage/src/rosbag2_storage/qos.cpp)），在无覆盖时由 recorder 的 `subscription_qos_for_topic` 调用（[`recorder.cpp`](https://github.com/ros2/rosbag2/blob/jazzy/rosbag2_transport/src/rosbag2_transport/recorder.cpp)）。Jazzy `qos.cpp` 的规则：
  - **History：** 若发布方 offered 的最大 depth > 1，则 `keep_last(该 depth)`；否则（例如所有发布方都 offer `keep_last` depth=1）回退为 **`keep_all`**（DEBUG 日志：“Falling back to the RMW_QOS_POLICY_HISTORY_KEEP_ALL”）。
  - **可靠性：** 仅当**全部**发布方 offer reliable 时才请求 `reliable`；reliable/best_effort 混合时回退 **`best_effort`** 并打 WARN——仍连上所有发布方，只是来自 reliable 发布方的部分消息可能丢。
  - **持久性：** 仅当**全部**发布方 offer transient_local 时才请求 `transient_local`；transient_local/volatile 混合时回退 **`volatile`** 并打 WARN——仍建连，只是拿不到此前已发布的 latch 消息。
  - 所以默认适配不会造成「无连接」；多发布方混合只损失可靠 / 保留语义。
- **覆盖：** `--qos-profile-overrides-path` 对具名 topic 强制一份 **手动 requested** 配置（同一 README；`recorder.cpp` 直接返回覆盖值、跳过适配）。默认适配不适合你的晚加入 / latch 场景时再用。**RxO 匹配是按发布方–订阅方逐对进行的**（About QoS：同一 topic 上其他发布方不影响某一对的兼容性）。过严的覆盖**只**与不兼容的发布方失配；兼容发布方的消息仍会到。**该 topic 零消息，只有在所有相关发布方都不匹配时才可能**（例如请求 `reliable` 而每个发布方都是 best_effort，或请求 `transient_local` 而每个发布方都是 volatile）。反例：两个发布方分别 offer reliable 与 best_effort，覆盖请求 `reliable` → 仍能收到 reliable 那一路（见 `bag_qos_accept_toy.py` 的 `override_match_demo`）。默认适配不会造成「无连接」；只有过严覆盖才可能，且仅当全部发布方都不匹配。

**持久性与晚加入者**（About QoS 的 durability + 兼容表）：

- **Volatile：** 只投递订阅建立**之后**发布的样本。录包器晚加入 volatile topic 会 **错过先前样本**。
- **Transient local：** 在发布方仍存活且仍 offer transient_local、历史缓存（`keep_last` depth 或 `keep_all`）里确有样本时，可向晚加入订阅投递保留/历史样本。这**不是**发明超出 transient_local 的进程外 DDS 持久化。
- 保留样本前提：存活发布方 offer transient_local；历史缓存非空；订阅请求兼容（request transient_local 需要 offer transient_local）。Pub Volatile + Sub Transient local ⇒ **无通信**。

**持久性兼容表**（About QoS — Compatibility of durability QoS policies）：

| 发布方 | 订阅方 | 兼容 | 结果 |
|---|---|---|---|
| Volatile | Volatile | 是 | 仅新消息 |
| Volatile | Transient local | **否** | 无通信 |
| Transient local | Volatile | 是 | 仅新消息 |
| Transient local | Transient local | 是 | 新消息 + 保留消息 |

要让晚加入者看到「锁存」效果，**双方**都必须同意 transient local。

**其他兼容边界**（同一 About QoS — Jazzy RST 表；history/depth **不**在匹配表里）：

- **History / depth 不决定是否匹配。** `keep_last` 与 `keep_all` 以及数值 depth 影响缓存，不决定发布方–订阅方是否建连。
- **Deadline：** offered deadline 必须 **≤** requested（发布方承诺更紧的节拍，能满足更松的请求）。发布方 `Default` + 订阅方有限 deadline ⇒ **不兼容**。
- **Lease duration：** 与 deadline 同向 — offered lease ≤ requested。
- **Liveliness：** Automatic < Manual by topic。发布方 Automatic + 订阅方 Manual by topic ⇒ **否**；发布方 Manual by topic + 订阅方 Automatic ⇒ **是**。

**练习（无真机 / 无真 bag）：** 给定合成 *offered* 列表——

| Topic | Offered |
|---|---|
| `/tf_static` | reliable + transient_local，keep_last depth=1（单发布方） |
| `/joint_states` | best_effort + volatile，keep_last depth=10（单发布方） |
| `/camera/color/image_raw` | reliable + volatile，keep_last depth=5（单发布方） |

只用上面的 `qos.cpp` 规则（最大 depth ≤ 1 ⇒ `keep_all`；仅当每个发布方都 offer 时才请求 reliable / transient_local）。请推导：（1）默认 `adapt_request_to_offers` 在各 topic 上选出的 *requested* history / 可靠性 / 持久性（注意 `/tf_static` depth=1 ⇒ `keep_all`）；（2）若再出现一个 offer reliable 的 `/joint_states` 发布方会怎样（混合 ⇒ best_effort + WARN，仍建连），以及何时改写 override YAML；（3）覆盖两问：（a）若 override YAML 对 `/camera/color/image_raw` 在其**唯一**发布方 offer volatile 时请求 `transient_local`——RxO 不匹配 ⇒ 必需 topic `n==0` ⇒ `missing_topic.json` **REJECT**（`missing_required_topic`）；（b）若 `/joint_states` 有**两个**发布方分别 offer reliable 与 best_effort，覆盖请求 `reliable`，该 topic 的 bag 会是空的吗？（**不会**——仍能收到 reliable 那一路；见 `override_match_demo`。）对照 `bag_qos_accept_toy.py` 与 `accept_fixtures/`——仅 JSON 时间线；CI 不进二进制 MCAP。

```python
#!/usr/bin/env python3
"""bag_qos_accept_toy.py: synthetic rosbag2/MCAP *acceptance* gate (stdlib).

Acceptance ≠ "the bag folder exists". A recording is accepted only when:
  1. every required topic is present with n > 0 messages
  2. inter-arrival gaps stay within max_gaps_allowed (1.5× nominal period)
  3. per-topic message counts meet a fixed message budget (n >= budget_min)

QoS override YAML keys are taken from the ros2/rosbag2 **jazzy** README
section "Overriding QoS Profiles":
  https://github.com/ros2/rosbag2/blob/jazzy/README.md
  (raw: https://raw.githubusercontent.com/ros2/rosbag2/jazzy/README.md)
That README verifies the YAML *keys* (example uses durability: volatile).
The Request vs Offered model and durability compatibility table live in
About QoS Settings (Jazzy):
  https://docs.ros.org/en/jazzy/Concepts/Intermediate/About-Quality-of-Service-Settings.html
Recorder default adapt: Rosbag2QoS::adapt_request_to_offers
  (rosbag2_storage/.../qos.cpp; called from recorder.cpp when no override).
  Mixed reliable/best_effort -> best_effort, mixed transient_local/volatile
  -> volatile (WARN, still connects); max offered depth <= 1 -> keep_all.
RxO matching is per publisher–subscriber pair: an override that is too
strict fails to match only the incompatible publishers; zero messages on
the topic are only possible when EVERY relevant publisher mismatches.

MCAP ROS 2 getting-started: https://mcap.dev/guides/getting-started/ros-2
docs.ros.org QoS How-To for record/playback: optional when About QoS +
Jazzy README + recorder source already cover the teaching.

No binary .mcap / real bags in CI — JSON timeline fixtures only.
Live bag record, Foxglove, Rerun, wandb = 本页未运行.

usage: python3 bag_qos_accept_toy.py
"""
from __future__ import annotations

import json
import os

HERE = os.path.dirname(os.path.abspath(__file__))
FIX = os.path.join(HERE, "accept_fixtures")
QOS_YAML = os.path.join(FIX, "qos_override.yaml")

# Keys from jazzy README example under "Overriding QoS Profiles".
JAZZY_README_QOS_KEYS = (
    "history",
    "depth",
    "reliability",
    "durability",
    "deadline",
    "lifespan",
    "liveliness",
    "liveliness_lease_duration",
    "avoid_ros_namespace_conventions",
)

REQUIRED_TOPICS = (
    "/joint_states",
    "/camera/color/image_raw",
    "/tf",
)

# Fixed message-budget table — do NOT swap the pass/fail comparisons.
# pass example: n=100 >= budget_min=90 → PASS
# fail example: n=10  >= budget_min=50 → FAIL (reject)
FIXED_BUDGET_PASS = {"label": "fixed_ok", "n": 100, "budget_min": 90}
FIXED_BUDGET_FAIL = {"label": "fixed_fail", "n": 10, "budget_min": 50}

# Per-topic budgets applied to fixture bags (messages over the recording).
TOPIC_BUDGET_MIN = {
    "/joint_states": 90,
    "/camera/color/image_raw": 20,
    "/tf": 50,
}

MAX_GAPS_ALLOWED = 2
GAP_FACTOR = 1.5


def count_gaps(stamps_ns: list[int], nominal_hz: float) -> int:
    if len(stamps_ns) < 2 or nominal_hz <= 0:
        return 0
    period = 1e9 / nominal_hz
    gaps = 0
    for a, b in zip(stamps_ns, stamps_ns[1:]):
        if (b - a) > GAP_FACTOR * period:
            gaps += 1
    return gaps


def load_qos_keys(path: str) -> list[str]:
    """Parse top-level per-topic keys from the override YAML (no PyYAML)."""
    keys: list[str] = []
    with open(path, encoding="utf-8") as f:
        for line in f:
            raw = line.rstrip("\n")
            if not raw or raw.lstrip().startswith("#"):
                continue
            # topic header: "/name:" at column 0
            if raw.startswith("/") and raw.endswith(":"):
                continue
            # policy line: two-space indent, "key:" or "key: value"
            if raw.startswith("  ") and not raw.startswith("    "):
                key = raw.strip().split(":", 1)[0]
                if key not in keys:
                    keys.append(key)
    return keys


def accept_bag(bag: dict) -> tuple[str, str, dict]:
    """Return (decision, reason, detail). folder_exists alone never accepts."""
    detail: dict = {
        "folder_exists": bool(bag.get("folder_exists")),
        "uri": bag.get("uri"),
    }
    if not bag.get("folder_exists"):
        return "REJECT", "folder_missing", detail

    by_name = {t["name"]: t for t in bag.get("topics", [])}

    for req in REQUIRED_TOPICS:
        topic = by_name.get(req)
        n = 0 if topic is None else len(topic.get("stamps_ns", []))
        detail[f"n:{req}"] = n
        if topic is None or n == 0:
            detail["missing_topic"] = req
            return "REJECT", "missing_required_topic", detail

    for req in REQUIRED_TOPICS:
        topic = by_name[req]
        gaps = count_gaps(topic["stamps_ns"], float(topic["nominal_hz"]))
        detail[f"gaps:{req}"] = gaps
        if gaps > MAX_GAPS_ALLOWED:
            detail["gap_topic"] = req
            return "REJECT", "gap_exceeded", detail

    for req in REQUIRED_TOPICS:
        n = len(by_name[req]["stamps_ns"])
        budget = TOPIC_BUDGET_MIN[req]
        detail[f"budget:{req}"] = budget
        if n < budget:
            detail["budget_topic"] = req
            return "REJECT", "below_message_budget", detail

    return "ACCEPT", "ok", detail


def fixed_budget_demo() -> None:
    """Pinned comparisons — swapping pass/fail numbers must not be done."""
    ok = FIXED_BUDGET_PASS
    bad = FIXED_BUDGET_FAIL
    ok_pass = ok["n"] >= ok["budget_min"]
    bad_pass = bad["n"] >= bad["budget_min"]
    print(
        f"fixed_budget_examples "
        f"{ok['label']}: n={ok['n']} budget_min={ok['budget_min']} "
        f"cmp=ge → {'PASS' if ok_pass else 'FAIL'}; "
        f"{bad['label']}: n={bad['n']} budget_min={bad['budget_min']} "
        f"cmp=ge → {'PASS' if bad_pass else 'FAIL'} "
        f"(do not swap comparisons)"
    )
    assert ok_pass and not bad_pass
    # Guard against accidental swap of the two example rows.
    assert ok["n"] > bad["n"]
    assert ok["budget_min"] > bad["budget_min"]


def override_match_demo() -> None:
    """Per-publisher RxO matching under a too-strict override (stdlib demo).

    Matching is pairwise. An override requests one profile; each publisher
    is checked independently. Zero messages require ALL publishers to
    mismatch — a mixed reliable+best_effort topic with override=reliable
    still receives from the reliable publisher.
    """
    # reliability offered rank: reliable > best_effort (request must not
    # be more stringent than offer).
    def reliability_ok(offered: str, requested: str) -> bool:
        if requested == "best_effort":
            return True
        return offered == "reliable"

    cases = [
```

## 常见错误

- **把开发容器 CI 塞进本站仓库。** 学员功能包 CI ≠ Hardware-learning 网站 CI。
- **仍假定默认是 sqlite3。** Jazzy 上默认 storage_id 是 mcap。
- **无视缺口计数。** 「50 Hz」关节话题实际 49 Hz 还带缺口，已是日志事故。
- **把「文件夹存在」当成 bag 验收。** 缺必需 topic（`n==0`）、缺口超限或低于消息预算必须拒收。
- **臆造 QoS YAML 键，或拿 rosbag2 README 当 `transient_local` 兼容性出处。** 只用 README 里的键；持久性兼容（Pub volatile + Sub transient_local ⇒ 无连接）以 About QoS 为准；并记住只有过严的覆盖（而非默认适配）才可能无连接，且仅当该 topic 上**所有**发布方都不匹配。
- **以为一个发布方失配就会让整个 topic 变成零消息。** RxO 是逐对的；reliable+best_effort 混合且覆盖=`reliable` 时，仍会录到 reliable 那一路。
- **永远把 wandb 当可选项。** 微调（8.2）时不记超参，毕业项目 README 很难站得住。

## 官方 rosbag2 仿真时间时钟契约

前面的章节保持不动：开发容器与 action-ros-ci、Jazzy rosbag2 默认 `storage_id` 为 `"mcap"`、`bag_stats.py` 在合成 mcap 时间线上的统计，以及 `bag_qos_accept_toy.py` 的验收（缺 topic / 缺口 / 消息预算）和已经教过的 Request vs Offered / YAML 键措辞。本节只讲官方**仿真时间**时钟契约（`--use-sim-time` / `--clock` / `/clock`）。这不是第二遍 mcap、bag 统计或 bag-QoS 课。

此处引用的正文在 rosbag2 提交 `12b5e8ba4eb5665501a16f0d9e050552a1149f56` 上重新读过（Jazzy tip，以原文为准）：软件包 [README](https://raw.githubusercontent.com/ros2/rosbag2/12b5e8ba4eb5665501a16f0d9e050552a1149f56/README.md)、[`record.py`](https://raw.githubusercontent.com/ros2/rosbag2/12b5e8ba4eb5665501a16f0d9e050552a1149f56/ros2bag/ros2bag/verb/record.py) 与 [`play.py`](https://raw.githubusercontent.com/ros2/rosbag2/12b5e8ba4eb5665501a16f0d9e050552a1149f56/ros2bag/ros2bag/verb/play.py)。现场 `ros2 bag`、真实仿真器、Foxglove / Rerun 或硬件时钟：本页未运行。

### 官方仿真时间是什么

README 的 "Simulation time"，以原文为准：In ROS 2, "simulation time" refers to publishing a clock value on the `/clock` topic, instead of using the system clock to tell time. 向 `ros2 bag record` 传入 `--use-sim-time` 即为录制节点打开该选项。写入包中的消息用最近收到的 `/clock` 值作为记录时间戳。

`record.py` 的帮助（`store_true`，`default=False`），以原文为准："Use simulation time for message timestamps by subscribing to the /clock topic. Until first /clock message is received, no messages will be written to bag." README 用同样的等待重述："Until the first `/clock` message is received, the recorder will not write any messages." 在那条消息到达之前，"the time is 0, which leads to a significant time jump once simulation time begins, making the bag essentially unplayable if messages are written first with time 0 and then time N from `/clock`."

`--use-sim-time` 与 `--no-discovery` 不兼容。`validate_parsed_arguments` 返回的错误，以原文为准："--use-sim-time and --no-discovery both set, but are incompatible settings. The /clock topic needs to be discovered to record with sim time." 两旗同时置位时该拒绝是无条件的；本页不发明「若启动时 `/clock` 已存在便可例外」的说法。

### Play `--clock`

`play.py` 把 `--clock` 放在与 `--clock-topics`、`--clock-topics-all` 互斥的一组里。官方 argparse，以原文为准：`type=positive_float`，`metavar='Hz'`，`nargs='?'`，`const=40`，`default=0`。帮助按源码拼接引用，保留 "publishing." 后缺失的空格："Publish to /clock at a specific frequency in Hz, to act as a ROS Time Source. Value must be positive. Defaults to not publishing.If specified, /clock topic in the bag file is excluded to publish."

本页只用这些官方频率：默认 `0` 不发布；裸 `--clock` 使用官方 `const=40`。不另造赫兹。当 `clock_publish_frequency > 0` 时，play 的接线把 `/clock` 追加进排除列表。README 的 "Publishing simulation time"，以原文为准："When fixed-frequency `/clock` publication is enabled with `--clock` or the `play.clock_publish_frequency` node parameter, `/clock` is published only while a playback session is active. If the Player node remains alive after playback finishes or is stopped, clock publication stops. Starting another playback session restarts it. Pausing playback does not end the session; fixed-frequency updates continue with the paused time."

### 公平设置

`rosbag2_sim_time_clock_toy.py` 是**合成的**，只用标准库。**模型假定:** 五个种子 `(0, 1, 2, 3, 4)` 每个种子共享一份到达时间表（公共随机数）：12 条数据消息网格，间隔 `DATA_PERIOD_MS = 50`，再加上在 `CLOCK_FIRST_MS = 150` 之前抽出的 1–3 条额外早到消息，以及一条在该墙钟时刻才开始、仿真时间为 `SIM_T0_MS = 1000`、随后按 `CLOCK_PERIOD_MS = 50` 步进的 `/clock` 流（`N_CLOCK = 10`）。因此每个种子在首条 `/clock` 之前至少有一条数据消息。小差距与明显差距的因子打印为 `0.2000`（较小臂均值的 20 %）。房屋标签顺序是：每一种子都相同的打平；数值为零；近打平（`|m| <= sd`）；再用较小臂均值的 20 % 和符号计数区分 small 与 clear。工作安排 `30+25+50+40+35=180`。现场 ros2 bag / 真实仿真器 / Foxglove / Rerun / 硬件时钟：本页未运行。

| 臂 | 计算内容 |
| --- | --- |
| A_official | 官方 `--use-sim-time` 录制：在首条 `/clock` 到达前不写入，之后用最近的 `/clock` 给每条已写消息打戳。该臂上 `zero_then_n_jump` 为假。 |
| C_write_0_then_N | 在 `/clock` 到达前把早到消息写成时间戳 0，之后用来自 `/clock` 的时间 N 给后续消息打戳。预先声明为仍然可播放 / 不会引入官方的 0-then-N 跳变。预先声明的**失败**臂，保留。只有这一种失败模式，在第一次运行前声明。 |

`--use-sim-time` 与 `--no-discovery` 的拒绝按官方理由打印；它不是第二次定量比较的失败情形。Play `--clock` 只用官方 `default=0` 与 `const=40`：手算 `1000/40=25` ms 周期，手算 `250/25=10` 次发布（`SESSION_ACTIVE_MS = 250` 的一次会话）；默认 0 不发布；仅当频率 > 0 时排除包内 `/clock`；暂停并不结束会话（更新继续使用暂停时的时间）；停止后，即使播放「节点」仍存活，后续发布为 0；再开一次会话会重新开始发布。

**情形（第一次运行前写定）:** 在共享的晚到 `/clock` 时间表上比较官方录制臂与预先声明的 0-then-N 失败臂；命名的替代 `clock_first_ms=250`（同一种子族，CRN：相同数据到达、更晚的 `/clock`）只打印、不采用。

```python
#!/usr/bin/env python3
"""Official rosbag2 simulation-time clock contract, stdlib only.

Synthetic arrival schedules, not a live ros2 bag, not a real simulator,
and not a second mcap / bag-stats / bag-QoS lecture.  Live ros2 bag /
real simulator / Foxglove / Rerun / hardware clock: 本页未运行.
rosbag2 statements 以原文为准.

In ROS 2, "simulation time" refers to publishing a clock value on the
/clock topic, instead of using the system clock to tell time.

Common random numbers: one shared synthetic arrival schedule per seed
(data messages plus a /clock stream that starts late, so every seed has
at least one data message before the first /clock).  Arms, shared
arrivals (per seed):
  A_official        : --use-sim-time record: do not write until the
                      first /clock; thereafter stamp each written
                      message with the latest /clock.  zero_then_n_jump
                      is false on this arm.
  C_write_0_then_N  : write early messages with timestamp 0 until
                      /clock arrives, then stamp later messages with
                      time N from /clock.  Pre-declared as remaining
                      playable / not introducing the official 0-then-N
                      jump.  Pre-declared lose, kept.  One fail mode,
                      declared before the first run.
The arms differ only in how record timestamps are formed.

Play --clock facts in this file use official argparse const=40 and
default=0 only.  They are not a second pre-declared lose claim.
"""

from __future__ import annotations

import random

# Declared before the first official run.
SEEDS = (0, 1, 2, 3, 4)
N_DATA = 12  # grid data messages, plus per-seed early extras
DATA_PERIOD_MS = 50  # grid spacing, ms
CLOCK_FIRST_MS = 150  # first /clock wall arrival; starts late
CLOCK_FIRST_ALT_MS = 250  # named alternative late start, not adopted
CLOCK_PERIOD_MS = 50  # /clock stream period after the first, ms
N_CLOCK = 10  # /clock messages in the stream
SIM_T0_MS = 1000  # first /clock sim-time value N, ms (N > 0)
CLEAR_FRAC = 0.2000  # small-vs-clear factor; printed before the first official run
NUM_ZERO = 1e-9
CLOCK_HZ_DEFAULT = 0  # official argparse default
CLOCK_HZ_BARE = 40  # official argparse const for bare --clock
PLAY_PERIOD_MS = 25  # 手算 1000/40 = 25
SESSION_ACTIVE_MS = 250  # 手算 250/25 = 10 publishes at 40 Hz
IDLE_AFTER_STOP_MS = 200
RUN_BEFORE_PAUSE_MS = 100
PAUSE_MS = 100
RESTART_SESSION_MS = 100

ARMS = ("A_official", "C_write_0_then_N")


def shared_schedule(seed: int, clock_first_ms: int) -> tuple[list[int], list[tuple[int, int]]]:
    """One shared arrival schedule for this seed (CRN).

    Data grid plus extra early messages drawn before CLOCK_FIRST_MS, so
    every seed has at least one data message before the first /clock.
    The /clock stream starts at clock_first_ms (late) with sim time
    SIM_T0_MS, then steps by CLOCK_PERIOD_MS.
    """
    rng = random.Random(seed)
    data = [k * DATA_PERIOD_MS for k in range(N_DATA)]
    n_extra = 1 + rng.randrange(0, 3)  # 1, 2, or 3 extras, all early
    extras = [rng.randrange(0, CLOCK_FIRST_MS) for _ in range(n_extra)]
    data.extend(extras)
    data.sort()
    clocks = [
        (clock_first_ms + i * CLOCK_PERIOD_MS, SIM_T0_MS + i * CLOCK_PERIOD_MS)
        for i in range(N_CLOCK)
    ]
    return data, clocks


def record_arm(data: list[int], clocks: list[tuple[int, int]], official: bool) -> dict[str, float]:
    """Stamp data messages under the official or the 0-then-N record rule."""
    events: list[tuple[int, str, int]] = []
    for t in data:
        events.append((t, "data", 0))
    for t, sim in clocks:
        events.append((t, "clock", sim))
    # At equal wall time, apply /clock before data so a message arriving
    # with the first /clock is stamped with that clock, not dropped.
    events.sort(key=lambda e: (e[0], 0 if e[1] == "clock" else 1))

    latest: int | None = None
    written_stamps: list[int] = []
    dropped = 0
    for _wall, kind, sim in events:
        if kind == "clock":
            latest = sim
            continue
        if official:
            if latest is None:
                dropped += 1
                continue
            written_stamps.append(latest)
        else:
            if latest is None:
                written_stamps.append(0)
            else:
                written_stamps.append(latest)

    jump = 0
    zero_then_n = 0
    if written_stamps:
        has_zero = any(s == 0 for s in written_stamps)
        later_n = [s for s in written_stamps if s > 0]
        if has_zero and later_n:
            zero_then_n = 1
            jump = later_n[0] - 0
    return {
        "written": float(len(written_stamps)),
        "dropped": float(dropped),
        "first_stamp_ms": float(written_stamps[0]) if written_stamps else 0.0,
        "last_stamp_ms": float(written_stamps[-1]) if written_stamps else 0.0,
        "jump_from_zero_ms": float(jump),
        "zero_then_n_jump": float(zero_then_n),
        "playable": 0.0 if zero_then_n else 1.0,
    }


def mean_sd(xs: list[float]) -> tuple[float, float]:
    mu = sum(xs) / len(xs)
    if len(xs) == 1:
        return mu, 0.0
    return mu, (sum((x - mu) ** 2 for x in xs) / (len(xs) - 1)) ** 0.5


def num(x: float) -> str:
    if x != 0.0 and abs(x) < 5e-5:
        return f"{x:.3e}"
    return f"{x:.4f}"


def sign_note(diffs: list[float], m: float) -> str:
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def fmt_pair(m: float, s: float) -> str:
    return f"{num(m)}+-{num(s)}"


def verdict(xa: list[float], xb: list[float]) -> str:
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{fmt_pair(m, s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < CLEAR_FRAC * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{fmt_pair(m, s)}: {label}; {sign_note(diffs, m)}"


def collect(clock_first_ms: int) -> dict[str, dict[str, list[float]]]:
    keys = (
        "written",
        "dropped",
        "first_stamp_ms",
        "last_stamp_ms",
        "jump_from_zero_ms",
        "zero_then_n_jump",
        "playable",
    )
    out = {arm: {k: [] for k in keys} for arm in ARMS}
    for seed in SEEDS:
        data, clocks = shared_schedule(seed, clock_first_ms)
        for arm, official in (("A_official", True), ("C_write_0_then_N", False)):
            vals = record_arm(data, clocks, official)
            for k in keys:
                out[arm][k].append(vals[k])
    return out


def print_arm(tag: str, res: dict[str, list[float]]) -> None:
    print(
        f"  {tag} written {fmt_pair(*mean_sd(res['written']))}  "
        f"dropped {fmt_pair(*mean_sd(res['dropped']))}  "
        f"first_stamp_ms {fmt_pair(*mean_sd(res['first_stamp_ms']))}  "
        f"last_stamp_ms {fmt_pair(*mean_sd(res['last_stamp_ms']))}"
    )
    print(
        f"    jump_from_zero_ms {fmt_pair(*mean_sd(res['jump_from_zero_ms']))}  "
        f"zero_then_n_jump {fmt_pair(*mean_sd(res['zero_then_n_jump']))}  "
        f"playable {fmt_pair(*mean_sd(res['playable']))}"
    )


def play_clock_publishes(frequency_hz: float, active_ms: int, period_ms: int) -> int:
    """Fixed-frequency /clock count while a session is active (t < active_ms)."""
    if frequency_hz <= 0:
        return 0
    n = 0
    t = 0
    while t < active_ms:
        n += 1
        t += period_ms
    return n


def play_pause_demo() -> tuple[int, int, int, bool, int]:
    """Running then paused ticks; pause keeps publishing the paused time."""
    period = PLAY_PERIOD_MS
    play_head = 0
    running_times: list[int] = []
    pause_times: list[int] = []
    t = 0
    while t < RUN_BEFORE_PAUSE_MS:
        running_times.append(play_head)
        play_head += period
        t += period
    paused_time = play_head
    t = 0
    while t < PAUSE_MS:
        pause_times.append(paused_time)
        t += period
    all_paused = bool(pause_times) and all(x == paused_time for x in pause_times)
    further_after_stop = 0  # node may stay alive; session has ended
    return len(running_times), len(pause_times), paused_time, all_paused, further_after_stop

    print("Synthetic rosbag2 simulation-time clock toy; shared arrivals; not a live ros2 bag")
```
