```python
#!/usr/bin/env python3
"""suite_score_table.py: aggregate fake LIBERO-style suite scores (stdlib).

Pinned fake episode outcomes (seed=88) for five named suites. Prints per-suite
mean success, optional Wilson 95 % intervals (Month 6 formula, not re-derived),
and a grand mean. No lerobot-eval / GPU / MuJoCo.
usage: python3 suite_score_table.py
"""
from __future__ import annotations

import math
import random

SEED = 88
Z95 = 1.959964
# Five suites (docs: 5 suites / 130 tasks total). Toy uses fixed episode counts.
SUITES = (
    ("libero_spatial", 40),
    ("libero_object", 40),
    ("libero_goal", 30),
    ("libero_10", 20),
    ("libero_90", 50),
)
# Success rates used to sample fixed-seed Bernoulli outcomes (not real eval).
P_SUCCESS = {
    "libero_spatial": 0.70,
    "libero_object": 0.55,
    "libero_goal": 0.60,
    "libero_10": 0.45,
    "libero_90": 0.35,
}


def wilson(k: int, n: int, z: float = Z95) -> tuple[float, float]:
    """Wilson score interval (same intuition as Month 6 eval_report)."""
    if n <= 0:
        return 0.0, 0.0
    p = k / n
    centre = (p + z * z / (2 * n)) / (1 + z * z / n)
    half = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / (1 + z * z / n)
    return centre - half, centre + half


def main() -> None:
    rng = random.Random(SEED)
    print(f"seed={SEED} suites={len(SUITES)} (fake episode outcomes; not real LIBERO)")
    print(
        f"{'suite':<16} {'n':>4} {'k':>4} {'mean':>8} "
        f"{'wilson_lo':>10} {'wilson_hi':>10}"
    )
    total_k = 0
    total_n = 0
    means: list[float] = []
    for name, n in SUITES:
        p = P_SUCCESS[name]
        outcomes = [1 if rng.random() < p else 0 for _ in range(n)]
        k = sum(outcomes)
        mean = k / n
        lo, hi = wilson(k, n)
        means.append(mean)
        total_k += k
        total_n += n
        print(
            f"{name:<16} {n:4d} {k:4d} {mean:8.4f} "
            f"{lo:10.4f} {hi:10.4f}"
        )
    suite_mean = sum(means) / len(means)
    overall = total_k / total_n
    lo_o, hi_o = wilson(total_k, total_n)
    print(
        f"ok: n_suites={len(SUITES)} suite_mean={suite_mean:.4f} "
        f"overall={overall:.4f} ({total_k}/{total_n}) "
        f"wilson=[{lo_o:.4f},{hi_o:.4f}]"
    )
    print(
        "note: graduation bar remains Month 8.7 (safety/CI/20-start Wilson/"
        "portfolio 8/8); LIBERO is bonus reporting only"
    )


if __name__ == "__main__":
    main()
```

## 在想象中给策略排名

在真机起始前挑选检查点，仿真分数是一种办法，学习得到的**世界模型**是另一种。Ctrl-World（Guo、Shi、Chen、Finn，arXiv **2510.10125**）就是为此做的可控世界模型。本段事实均**以论文为准**，论文里的内容本页都未运行。它是一个在 DROID（9.5 万条轨迹、564 个场景）上训练的多视角视频世界模型：用以位姿为条件的记忆检索在长时间滚动中保持一致，用逐帧的动作条件让动作控制预测结果。策略在它*内部*运行：策略发出一个动作块，模型返回预测的后续帧，循环从与真机滚动相同的初始观测开始。作者评估了三个公开策略（π0、π0-FAST、π0.5），报告想象中的指令遵循与真机高度吻合；同时也报告模型往往**低估**执行成功率，会漏掉一些物理现象（碰撞、物体滑走、旋转），也漏掉真实策略失败后反复重试的习惯。用想象中成功的轨迹做微调，策略平均提升 44.7 %，同样以论文为准。

<Resource id="ctrl-world-paper" />

要决定哪个检查点上真机，需要的是**顺序**，不是绝对成功率。有偏的世界模型也可能排对——关键在于它的误差落在*哪些*策略身上。衡量排名一致的一个数是 n 个策略上的 **Kendall τ**：

<Formula label="Kendall τ（无并列）">τ = (同序对 − 逆序对) / (n(n − 1) / 2)</Formula>

5 个策略共 10 对：翻转 1 对 τ = 0.8，2 对 0.6，3 对 0.4。上面的 SimplerEnv 用的是它自己的排名指标 MMRV；τ 是更简单的替身。

`imagine_rank_toy.py` 里没有视频模型：每个回合只是几个数，每个「世界模型」都是 oracle 结果规则加上一种偏差。五个合成策略的区别在于：多常按指令选对物体、执行多精确、放好的物体会不会滑走、失手后会不会重试一次。每个策略在 oracle 和每个世界模型里都拿到**同样的 4 个任务 × 60 个初始条件**和同一组逐回合随机数。

```python
#!/usr/bin/env python3
"""imagine_rank_toy.py: rank policies in a biased world model vs the oracle env (stdlib).

Reading aid for the Ctrl-World idea (arXiv 2510.10125): roll each policy out
inside a learned world model instead of on the robot, and use the imagined
success rates to RANK policies. Here there is no video model: an episode is a
few numbers, and a "world model" is the oracle outcome rule with one bias.

Five synthetic policies (lesson-chosen, not real checkpoints) differ in
  follow  probability of picking the instructed object,
  sigma   execution error scale (m); success needs error < TOL,
  slide   chance a successful placement slides off afterwards (real physics),
  retry   whether the policy retries once after an execution miss.
Every policy, in the oracle and in every world model, sees the SAME tasks and
initial conditions and the SAME per-episode uniforms / normals (shared ICs).
World models:
  calibrated     same outcome rule as the oracle (sanity row)
  uniform_drop   imagines 15 % of successes as failures, the same episodes
                 for every policy (underestimates success, policy-blind)
  tight_tol      imagines execution as harder (TOL x 0.7): looks uniform, but
                 hurts sloppy (large-sigma) policies more
  no_slide       does not model objects sliding off after placement
  no_retry       ends the episode after the first miss (no retry)
Metric: Kendall tau between imagined and oracle success rates over the
policies (+1 same order, -1 reversed) and the list of flipped pairs.
The paper's numbers (e.g. its 44.7 % improvement) are per the paper; nothing
here is a DROID / pi0 / pi0.5 rollout (本页未运行).
usage: python3 imagine_rank_toy.py
"""
from __future__ import annotations

import random
from itertools import combinations

SEED = 10125
TOL = 0.020                # m
TASKS = ("pick_place", "drawer", "towel_fold", "stack")
N_IC = 60                  # initial conditions per task -> 240 shared episodes
POLICIES = (               # name, follow, sigma, slide, retry
    ("P1", 0.90, 0.012, 0.00, False),
    ("P2", 0.85, 0.010, 0.30, False),
    ("P3", 0.70, 0.018, 0.00, True),
    ("P4", 0.75, 0.016, 0.00, False),
    ("P5", 0.50, 0.010, 0.00, False),
)
WORLDS = ("oracle", "calibrated", "uniform_drop", "tight_tol", "no_slide", "no_retry")


def shared_episodes():
    """Per (task, ic): (u_follow, z1, z2, u_slide, u_wm); identical for every policy and world."""
    rng = random.Random(SEED)
    return [(task, ic, rng.random(), abs(rng.gauss(0, 1)), abs(rng.gauss(0, 1)), rng.random(), rng.random())
            for task in TASKS for ic in range(N_IC)]


def outcome(policy, ep, world):
    _name, follow, sigma, slide, retry = policy
    _task, _ic, u_f, z1, z2, u_s, u_wm = ep
    tol = TOL * (0.7 if world == "tight_tol" else 1.0)
    if u_f >= follow:                       # wrong object: instruction not followed
        return False
    placed = sigma * z1 < tol
    if not placed and retry and world != "no_retry":
        placed = sigma * z2 < tol
    if placed and world != "no_slide" and u_s < slide:
        return False                        # slid off after placement (real physics)
    if placed and world == "uniform_drop" and u_wm < 0.15:
        return False                        # imagined failure, same episodes for all policies
    return placed


def kendall_tau(a, b):
    conc = disc = 0
    flips = []
    for i, j in combinations(range(len(a)), 2):
        s = (a[i] - a[j]) * (b[i] - b[j])
        if s > 0:
            conc += 1
        elif s < 0:
            disc += 1
            flips.append((i, j))
    n = len(a) * (len(a) - 1) // 2
    return (conc - disc) / n, flips


def main() -> None:
    eps = shared_episodes()
    n = len(eps)
    print(f"seed={SEED} tasks={len(TASKS)} x ics={N_IC} = {n} shared episodes per policy and world; tol={TOL} m")
    rates = {w: [sum(outcome(p, e, w) for e in eps) / n for p in POLICIES] for w in WORLDS}
    print(f"{'policy':<6} {'follow':>6} {'sigma':>6} {'slide':>5} {'retry':>5} " + " ".join(f"{w:>12}" for w in WORLDS))
    for k, p in enumerate(POLICIES):
        print(f"{p[0]:<6} {p[1]:6.2f} {p[2]:6.3f} {p[3]:5.2f} {str(p[4]):>5} "
              + " ".join(f"{rates[w][k]:12.3f}" for w in WORLDS))
    oracle = rates["oracle"]
    order = sorted(range(len(POLICIES)), key=lambda k: -oracle[k])
    print("oracle ranking: " + " > ".join(POLICIES[k][0] for k in order))
    taus = {}
    for w in WORLDS[1:]:
        tau, flips = kendall_tau(oracle, rates[w])
        taus[w] = tau
        mean_gap = sum(rates[w]) / len(POLICIES) - sum(oracle) / len(POLICIES)
        names = ", ".join(f"{POLICIES[i][0]}/{POLICIES[j][0]}" for i, j in flips) or "none"
        rank = " > ".join(POLICIES[k][0] for k in sorted(range(len(POLICIES)), key=lambda k: -rates[w][k]))
        print(f"{w:<12} kendall_tau={tau:+.2f} mean_rate_shift={mean_gap:+.3f} flipped_pairs={names}  ranking: {rank}")
    assert taus["calibrated"] == 1.0
    assert taus["uniform_drop"] == 1.0, "a policy-blind bias should keep the ranking"
    assert taus["tight_tol"] < 1.0 and taus["no_slide"] < 1.0 and taus["no_retry"] < 1.0
    print("ok: a policy-blind underestimate lowers every rate but keeps the order; biases that hit "
          "some policies' failure modes flip pairs; Ctrl-World figures 以论文为准, not run here")


if __name__ == "__main__":
    main()
```

怎么读：

- **oracle 的顺序**是 P1 > P4 > P3 > P2 > P5（0.817、0.688、0.679、0.604、0.496）。`calibrated` 完全复现它，是一行自检。
- **与策略无关的偏差保持顺序。** `uniform_drop` 把 15 % 的成功判成失败，而且对每个策略都是同一批回合。所有成功率都下降（平均 −0.101），τ 仍为 +1.00——本种子下保持；与策略无关的同类偏差往往保序。像论文报告的那样低估成功率，就是这类偏差之一。
- **看起来一视同仁的偏差也可能翻转。** `tight_tol` 让所有执行都更难，但对更粗糙的策略（P4 σ = 0.016、P3 σ = 0.018）伤得更重：τ = +0.60，P2/P4 和 P3/P4 翻转。P3 和 P4 在 oracle 里只差 2 个回合（240 次中 163 对 165），它们的先后本来就在抽样噪声之内。相信一对接近的排名之前，先看区间（6.4、8.7）。
- **漏掉某种失败模式，就会抬高有这种问题的策略。** `no_slide` 不模拟物体滑走，P2 从 0.604 跳到 0.850，排到第一：τ = +0.40，翻转 3 对。`no_retry` 在第一次失手后就结束回合，唯一会重试的 P3 从 0.679 掉到 0.571：τ = +0.80，P2/P3 翻转。这正是论文列出的「滑走」和「重试」差距的玩具版。

相信想象中的排名之前，先问世界模型的误差是不是**与策略无关**。想象排名只用来缩小候选检查点，不能代替 8.7 的 20 次真机起始。

## 常见错误

- **用 LIBERO 顶替 8.7 统一门。** 基准是加分，不是毕业标准。
- **只报一个总成功率，不写套件与协议。** 至少写清评了哪些套件、每任务几集。
- **把仿真分直接写成「真机已经有多好」。** 先读 SimplerEnv 相关性叙事再下结论。
- **把想象中的成功率当成真实成功率。** Ctrl-World 的作者报告其模型往往低估执行成功率（以论文为准）；玩具里 `uniform_drop` 让所有成功率平均降低 0.101。用顺序，别用水平。
- **不问世界模型模拟不了什么就相信它的排名。** 漏掉的失败模式（滑走、重试）恰好抬高或压低有这种问题的策略：玩具里 τ = +0.40 和 +0.80。
- **不看区间就给一对接近的策略排名。** P3 和 P4 在 240 个回合里只差 2 个；这对翻不翻转都说明不了什么。

## LIBERO-X：分层鲁棒性阶梯

页上已有的五套件 LIBERO 假分数表、SimplerEnv 相关性阅读，以及想象中给策略排名的玩具，都保持不动。本节不改写它们，也不是 LIBERO-Plus、LIBERO-PRO 或 SafeLIBERO（论文相关工作表里点过名；不要把那些协议并进这里）。

LIBERO-X（Wang、Zhang、Liu 等，arXiv 2602.06556 v1，提交于 2026-02-06；以论文为准）是一个操作基准，要点是：单套件成功率可能掩盖**累积**分布偏移下的失败。它增加了五级分层评估（摘要 / PDF「Evaluation Framework」；以论文为准）：

- L1 局部空间扰动
- L2 扩展空间扰动
- L3 场景拓扑重组
- L4 视觉属性变化（包括未见与干扰物体）
- L5 语义等价的指令改写

七位作者按摘要顺序是：Guodong Wang、Chenkai Zhang、Qingjie Liu、Jinjin Zhang、Jiancheng Cai、Junjie Liu、Xinmin Liu。不带脚注标记的摘要作者行本身没有星号或剑号。PDF 扉页（不要抄 HTML 转换里乱掉的标记）把 Wang 与 Zhang 标为同等贡献，把 Junjie Liu 标为通讯作者，把 Xinmin Liu 标为项目负责人，单位为 Meituan 与 Beihang University。摘要 Comments 字段是 “19 pages, 14 figures and 8 tables”（以论文为准）；该字段没有会议，本课也不发明会议。引用 v1。

论文还报告了遥操作训练集的 2,520 条演示、600 个任务和 100 个场景（PDF；以论文为准，不是玩具输出）。从第 1 级到第 2 级平均成功率下降 9.8 %（Figure 5）也以论文为准：这是**降幅**，不是套件均值，也不是 L1 到 L5 的降幅，更不是玩具输出。

### 公平设定

`libero_x_robustness_ladder_toy.py` 是一个**标准库**策略替身。它不是 `lerobot-eval`，也不是真正的 LIBERO-X 运行。事先写明的参数是 `P_BASE = 0.8000`、`N_EP = 40` 个共享回合、`N_SCENES = 8`、`N_SLOTS = 4`，以及打印成 0.2000 的百分之二十「小间隙 / 明显间隙」因子。五种子 `0,1,2,3,4`。正式的、按级累积的惩罚是 `L1=0.0200`、`L2=0.1200`、`L3=0.1800`、`L4=0.1600`、`L5=0.1400`。打印的手算累积惩罚是 `0.0200`、`0.1400`、`0.3200`、`0.4800`、`0.6200`。共同随机数：每个（情形，种子）只抽一次场景、物体槽位、指令、能力扰动和伯努利均匀数，三个臂共用。玩具打印的论文数字标为 `paper_figure`，以论文为准，不是玩具实测：2,520 条演示、600 个任务、100 个场景，以及 Figure 5 中 9.8 % 的 L1 到 L2 降幅。

| 臂 | 报告 |
|---|---|
| A_unperturbed | 只报未扰动 / 分布内的套件均值 |
| B_ladder | 五级累积阶梯；逐级成功率以及 L1 到 L5 的降幅 |
| C_leak | 失败臂：训练–测试泄漏，把 L1 的结果当成 L5 来报告 |

| 情形 | 焦点 | 第一次运行前的声明 |
|---|---|---|
| l1_jitter | A 与 B 在 L1 | 类似 L1 的轻微抖动：A 与 B 在成功率上可能接近打平——保留 |
| cumulative_shift | B 在 L3、L4、L5 对照 A 的未扰动均值 | 累积的 L3–L5 偏移：B 相对 A 的未扰动均值有一致更低的成功率——保留 |
| leak_misreport | C 泄漏的「L5」对照 B 的真 L5 | C 把 L1 当成 L5 来误报——保留 |

**看过结果后做的选择**。正式参数在第一次运行之后都没有改过。课文是按冻结的正式输出写的。那次正式运行之后，论文数字 9.8 % 按站点体例在百分号前加了空格；正式 expected 输出没有重调。同一组种子上还跑了非结构化过苛扰动 `HARSH_PENALTY=0.7000`，**没有采用**。

```python
#!/usr/bin/env python3
"""Stdlib stand-in of a five-level cumulative robustness ladder.

Stdlib only. Not lerobot-eval, not a GPU LIBERO-X run, and not a redo of
the five-suite LIBERO fake-score table, SimplerEnv, or imagination
ranking already on this page. Real GPU lerobot-eval / LIBERO-X:
本页未运行.

LIBERO-X (Wang, Zhang, Liu, et al., arXiv 2602.06556 v1) is a
manipulation benchmark whose point is that a single suite success rate
can hide failure under cumulative distribution shifts. Paper figures
(2,520 demonstrations, 600 tasks, 100 scenes, and the 9.8 % drop in
average success rate from Level 1 to Level 2 in Figure 5) are
以论文为准: the 9.8 % figure is a drop, not a suite mean, not the
L1-to-L5 decline, and not a toy output.

Shared base scenes, object slots and instructions, and a shared policy
stand-in, are generated once per (regime, seed) and then read by every
arm (common random numbers).

Arms:
  A_unperturbed  in-distribution / unperturbed suite mean only
  B_ladder       five-level cumulative ladder (per-level success and
                 the drop from L1 to L5)
  C_leak         lose arm: a train-test leak that reuses the L1
                 outcomes and reports them as if they were L5

Regimes and parameters were written before the first run.

Labels: tie = identical on every seed; numerically zero = |paired mean|
and sd both < 1e-9; |m| <= sd = near-tie; |m| > sd = consistent gap
(small if |m| < 20 % of the smaller arm mean, else clear), with a
sign count. ROUND_HALF_UP output.

LIBERO-X paper: body read; 以论文为准.
"""
from __future__ import annotations

import math
import random
from decimal import ROUND_HALF_UP, Decimal, getcontext

getcontext().rounding = ROUND_HALF_UP

SEEDS = (0, 1, 2, 3, 4)
NUM_ZERO = 1e-9
SMALL_FRAC = 0.2000  # 20 % of the smaller-arm mean; printed as 0.2000

# Declared stand-in parameters. Written before the first run.
# These are toy penalties, not paper success rates.
N_EP = 40
N_SCENES = 8
N_SLOTS = 4
P_BASE = 0.8000
# Cumulative penalties added as each level is stacked on the previous.
# L1 is a small local jitter; later levels are larger stacked shifts.
PENALTY = {
    1: 0.0200,  # local spatial perturbation (minor jitter)
    2: 0.1200,  # extended spatial perturbation
    3: 0.1800,  # scene topology reconstruction
    4: 0.1600,  # visual attribute variation (unseen / confounding)
    5: 0.1400,  # semantic-equivalent instruction reformulation
}
# Unstructured harsh alternative (same seeds, not adopted).
HARSH_PENALTY = 0.7000

LEVELS = (1, 2, 3, 4, 5)
ARMS = ("A_unperturbed", "B_ladder", "C_leak")
# Official paired metrics: unperturbed mean vs ladder at the regime's
# focus levels, plus C's leaked "L5" report against B's true L5.
PAIRS = (("B", "A"), ("C", "B"), ("C", "A"))

REGIMES = {
    "l1_jitter": (
        "L1-like minor jitter; A and B may near-tie on success - keep"
    ),
    "cumulative_shift": (
        "cumulative L3-L5 shifts; B consistent lower success than A's "
        "unperturbed mean - keep"
    ),
    "leak_misreport": (
        "C mis-reports by reusing L1 outcomes as if they were L5 - keep"
    ),
}


def mean_sd(xs):
    if all(x == xs[0] for x in xs):
        return xs[0], 0.0
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    if x != 0.0 and abs(x) < 5e-4:
        return format(Decimal(repr(x)), ".3e")
    s = format(Decimal(repr(x)), ".4f")
    return "0.0000" if s == "-0.0000" else s


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def fmt_pair(m, s):
    return f"{num(m)}+-{num(s)}"


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{fmt_pair(m, s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < SMALL_FRAC * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{fmt_pair(m, s)}: {label}; {sign_note(diffs, m)}"


def rng_for(*parts):
    material = "libero-x-ladder|" + "|".join(str(p) for p in parts)
    return random.Random(material)


def clip01(p):
    return 0.0 if p < 0.0 else 1.0 if p > 1.0 else p


def cumulative_penalty(level, penalty=PENALTY):
    return sum(penalty[k] for k in LEVELS if k <= level)


def shared_bank(regime, seed, n_ep=N_EP):
    """Shared scenes, slots, instructions, competence and uniforms (CRN)."""
    rng = rng_for(regime, seed, "bank")
    episodes = []
    for i in range(n_ep):
        scene = i % N_SCENES
        slot = (i * 3) % N_SLOTS
        instruction = (i * 5 + scene) % 16
        competence = rng.uniform(-0.0400, 0.0400)
        u = rng.random()
        episodes.append(
            {
                "scene": scene,
                "slot": slot,
                "instruction": instruction,
                "competence": competence,
                "u": u,
            }
        )
    return episodes


def success_prob(episode, stacked_penalty):
    return clip01(P_BASE + episode["competence"] - stacked_penalty)


def roll_success(episode, stacked_penalty):
    return 1 if episode["u"] < success_prob(episode, stacked_penalty) else 0


def evaluate(regime, seed, penalty=None, harsh=None, n_ep=N_EP):
    if penalty is None:
        penalty = PENALTY
    episodes = shared_bank(regime, seed, n_ep=n_ep)
    # A: unperturbed / in-distribution, one suite mean on the shared bank.
    a_hits = [roll_success(ep, 0.0) for ep in episodes]
    a_mean = sum(a_hits) / n_ep
    # B: cumulative ladder, per level, same bank and same uniforms.
    b_level = {}
    b_hits = {}
    for lv in LEVELS:
        hits = [roll_success(ep, cumulative_penalty(lv, penalty)) for ep in episodes]
        b_hits[lv] = hits
        b_level[lv] = sum(hits) / n_ep
    b_drop_l1_l5 = b_level[1] - b_level[5]
    # C official: leak L1 outcomes and report them as L5.
    c_l5_leak = b_level[1]
    # Optional unstructured harsh report (used only in the named alternative).
    if harsh is None:
        c_harsh = None
    else:
        c_harsh = sum(roll_success(ep, harsh) for ep in episodes) / n_ep
    return {
        "n_ep": n_ep,
        "A_unperturbed": a_mean,
        "B_level": b_level,
        "B_drop_l1_l5": b_drop_l1_l5,
        "C_leak_l5": c_l5_leak,
        "C_harsh": c_harsh,
        "penalty": dict(penalty),
    }


def official_rows(name, penalty=None, harsh=None, n_ep=N_EP):
    return [evaluate(name, seed, penalty=penalty, harsh=harsh, n_ep=n_ep) for seed in SEEDS]


def print_regime(title, rows, focus):
    print(f"regime {title}  (mean+-sd over seeds)")
    print(f"  n_ep={rows[0]['n_ep']} n_scenes={N_SCENES} n_slots={N_SLOTS}")
    a = [r["A_unperturbed"] for r in rows]
    print(f"  A_unperturbed success {num(mean_sd(a)[0])}+-{num(mean_sd(a)[1])}")
    b_levels = {}
    for lv in LEVELS:
        xs = [r["B_level"][lv] for r in rows]
        b_levels[lv] = xs
        print(f"  B_ladder L{lv} success {num(mean_sd(xs)[0])}+-{num(mean_sd(xs)[1])}")
    drop = [r["B_drop_l1_l5"] for r in rows]
    print(f"  B_ladder drop_L1_to_L5 {num(mean_sd(drop)[0])}+-{num(mean_sd(drop)[1])}")
    c = [r["C_leak_l5"] for r in rows]
    print(f"  C_leak reported_as_L5 {num(mean_sd(c)[0])}+-{num(mean_sd(c)[1])}")
    if rows[0]["C_harsh"] is not None:
        h = [r["C_harsh"] for r in rows]
        print(f"  C_harsh unstructured success {num(mean_sd(h)[0])}+-{num(mean_sd(h)[1])}")

    # Focused official pairs for this regime, plus the leak comparison.
    if focus == "l1":
        print(f"  paired B-A L1_success: {verdict(b_levels[1], a)}")
        print(f"  paired C-B leaked_L5_minus_true_L5: {verdict(c, b_levels[5])}")
        print(f"  paired C-A leaked_L5_minus_unperturbed: {verdict(c, a)}")
    elif focus == "cumulative":
        for lv in (3, 4, 5):
            print(f"  paired B-A L{lv}_success: {verdict(b_levels[lv], a)}")
        print(f"  paired C-B leaked_L5_minus_true_L5: {verdict(c, b_levels[5])}")
        print(f"  paired C-A leaked_L5_minus_unperturbed: {verdict(c, a)}")
    else:
        # leak_misreport: C vs true L5 and vs A; also show L1 near A for context
        print(f"  paired B-A L1_success: {verdict(b_levels[1], a)}")
        print(f"  paired B-A L5_success: {verdict(b_levels[5], a)}")
        print(f"  paired C-B leaked_L5_minus_true_L5: {verdict(c, b_levels[5])}")
        print(f"  paired C-A leaked_L5_minus_unperturbed: {verdict(c, a)}")


def main():
    print("Stdlib stand-in of a five-level cumulative robustness ladder; not a LIBERO-X eval")
    print(
        f"declared P_BASE={num(P_BASE)} N_EP={N_EP} N_SCENES={N_SCENES} "
        f"N_SLOTS={N_SLOTS} SMALL_FRAC={num(SMALL_FRAC)} "
        f"seeds={','.join(str(s) for s in SEEDS)}"
    )
    print(
        "declared arms: A_unperturbed in-distribution suite mean; "
        "B_ladder cumulative L1-L5; "
        "C_leak reuses L1 outcomes as if they were L5"
    )
    print(
        "declared penalties L1={p1} L2={p2} L3={p3} L4={p4} L5={p5}".format(
            p1=num(PENALTY[1]),
            p2=num(PENALTY[2]),
            p3=num(PENALTY[3]),
            p4=num(PENALTY[4]),
            p5=num(PENALTY[5]),
        )
    )
    for lv in LEVELS:
        print(f"hand_calc cumulative_penalty L{lv} = {num(cumulative_penalty(lv))}")
    print(f"hand_calc 20 percent small-vs-clear factor = {num(SMALL_FRAC)}")
    # Paper figures, labelled as such, not toy outputs.
    print("paper_figure demonstrations = 2,520 (以论文为准, not a toy output)")
    print("paper_figure tasks = 600 (以论文为准, not a toy output)")
    print("paper_figure scenes = 100 (以论文为准, not a toy output)")
    print("paper_figure L1_to_L2_average_success_drop_pct = 9.8 (Figure 5; a drop, not a suite mean, not L1-to-L5, 以论文为准, not a toy output)")
    print("Regimes and parameters were written before the first run")
    print("Common random numbers: scenes, slots, instructions, competence and uniforms are shared across arms at each regime and seed")
    print("Assumptions: stdlib policy stand-in, not lerobot-eval, not LIBERO-Plus, not the five-suite table")
    print("Paper dataset sizes and the 9.8 % L1 to L2 drop 以论文为准")
    print("GPU lerobot-eval / real LIBERO-X: 本页未运行")
    print("declared l1_jitter: A and B may near-tie on success - keep")
    print("declared cumulative_shift: B consistent lower success than A's unperturbed mean - keep")
    print("declared leak_misreport: C mis-reports (leak) - keep")
    print(f"declared expectation l1_jitter: {REGIMES['l1_jitter']}")
    print_regime("l1_jitter", official_rows("l1_jitter"), "l1")
    print(f"declared expectation cumulative_shift: {REGIMES['cumulative_shift']}")
    print_regime("cumulative_shift", official_rows("cumulative_shift"), "cumulative")
    print(f"declared expectation leak_misreport: {REGIMES['leak_misreport']}")
    print_regime("leak_misreport", official_rows("leak_misreport"), "leak")

    print("alternative same seeds HARSH_PENALTY=0.7000 unstructured (not adopted)")
    print(f"hand_calc harsh residual success floor if competence=0 = {num(clip01(P_BASE - HARSH_PENALTY))}")
    rows = official_rows("cumulative_shift", harsh=HARSH_PENALTY)
    print_regime("cumulative_shift alt_harsh=0.7000", rows, "cumulative")

    print("No GPU eval was started")
    print("No winner is universal: L1-like jitter can near-tie; the hidden drop is the cumulative ladder, not a retune toward the unperturbed mean")


if __name__ == "__main__":
    main()
```

本页标签的用法，按这个顺序，**tie: identical on every seed** 只在每个种子上的成对差都恰好为零时使用；**numerically zero (floating-point residue), effectively a tie** 是成对均值的绝对值与标准差都小于 1e-9，那一行不算符号；**near-tie** 是 `|m| <= sd`；然后用较小臂均值的百分之二十（打印为 0.2000）以及同号计数（k/N 同号）区分 **small** 与 **clear**。

- **l1_jitter**。 A 的未扰动成功率是 `0.8100+-0.0285`；B 在 L1 是 `0.7950+-0.0371`。B-A 在 L1 成功率上是 `-0.0150+-0.0224: near-tie / small gap, within seed noise (|m| <= sd); no seed has the opposite sign, 3 of 5 equal`。事先声明的 A/B 接近打平**成立**，并**保留**。这一格并不是在说 L1 与未扰动均值相同。
- **cumulative_shift**。 A 是 `0.8100+-0.0675`。B 在 L3 是 `0.4200+-0.1483`，L4 是 `0.3100+-0.0859`，L5 是 `0.1850+-0.0576`，都更低。B-A 在 L3 是 `-0.3900+-0.1009`，在 L4 是 `-0.5000+-0.0530`，在 L5 是 `-0.6250+-0.0250`，都是**明显**间隙，全部 5 个种子同号。事先声明的一致下降**成立**，并**保留**。B 自己打印的 L1 到 L5 降幅是 `0.6150+-0.0224`。这个玩具降幅不是论文 Figure 5 里 9.8 % 的 L1 到 L2 降幅，也不是套件均值。
- **leak_misreport**。 C 把 L1 成功率 `0.7800+-0.0737` 当成 L5 来报告，而 B 的真 L5 是 `0.1500+-0.0559`。C-B（`leaked_L5_minus_true_L5`）是 `0.6300+-0.0671: clear gap, not a tie (|m| > sd); same sign on all 5 seeds`。事先声明的误报**成立**，并**保留**。泄漏数字与 A 接近打平（`-0.0250+-0.0250`），这正是泄漏会把阶梯藏起来的方式。

**事后备选，同一组种子**。 非结构化过苛扰动 `HARSH_PENALTY=0.7000` 在 `cumulative_shift` 的样本库上跑过，**没有采用**。它的成功率是 `0.0750+-0.0177`，比 B 的 L5 `0.1850+-0.0576` 还低。打印的手算下限（能力扰动为零时）是 `0.1000`。单个非结构化过苛数字可能过苛；正式失败臂仍是把 L1 当成 L5 的泄漏。没有改正式惩罚，让阶梯去和未扰动均值对齐。
