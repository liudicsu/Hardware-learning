```python
#!/usr/bin/env python3
"""compute_budget.py: estimate VRAM from params/precision/LoRA and compare official floors.

Stdlib toy for lesson 8.2. Does NOT load weights or run fine-tuning.
Official floors are pinned from primary docs (re-checked 2026-11-04 Asia/Shanghai):
  - openpi README: infer >8 GB; LoRA FT >22.5 GB; full FT >70 GB
  - Isaac-GR00T README: infer 16 GB+; fine-tune 40 GB+ recommended
  - LeRobot SmolVLA: smolvla_base 450M; 20k steps ~4 h on one A100
  - LeRobot π0.5: fine-tune command sized for a single 80 GB GPU

Rough estimator (teaching only, not a profiler):
  weight_GB ≈ n_params * bytes_per_param / 1e9
  train_GB  ≈ weight_GB * optimizer_factor   (AdamW-ish ~2–3× weights in fp16)
  lora_GB   ≈ weight_GB + adapter_GB * optimizer_factor
usage: python3 compute_budget.py
"""
from __future__ import annotations

BYTES = {"fp32": 4, "bf16": 2, "fp16": 2, "int8": 1}
# Adam-like: weights + moments; pinned for expected/
OPT_FACTOR_FULL = 2.5
OPT_FACTOR_LORA = 2.0
LORA_RANK_FRAC = 0.02  # adapters ~2 % of dense params (order-of-magnitude toy)


def weight_gb(n_params: float, precision: str) -> float:
    return n_params * BYTES[precision] / 1e9


def estimate(n_params: float, precision: str, mode: str) -> float:
    w = weight_gb(n_params, precision)
    if mode == "infer":
        return round(w * 1.2, 2)  # small activation overhead
    if mode == "full":
        return round(w * OPT_FACTOR_FULL, 2)
    if mode == "lora":
        adapters = w * LORA_RANK_FRAC
        return round(w + adapters * OPT_FACTOR_LORA, 2)
    raise ValueError(mode)


# Official floors table (source → number). "pass" if estimate sits under floor is NOT
# required — we only report estimate vs floor so students see the gap.
OFFICIAL = [
    # name, params, precision, mode, floor_GB, note
    ("smolvla_base", 450e6, "bf16", "full", 40.0, "LeRobot: ~4h/20k on A100; floor=illustrative A100-class"),
    ("openpi_infer", 3.3e9, "bf16", "infer", 8.0, "openpi README: Inference > 8 GB"),
    ("openpi_lora", 3.3e9, "bf16", "lora", 22.5, "openpi README: Fine-Tuning (LoRA) > 22.5 GB"),
    ("openpi_full", 3.3e9, "bf16", "full", 70.0, "openpi README: Fine-Tuning (Full) > 70 GB"),
    ("groot_n17_infer", 3e9, "bf16", "infer", 16.0, "Isaac-GR00T: Inference 16 GB+"),
    ("groot_n17_ft", 3e9, "bf16", "full", 40.0, "Isaac-GR00T: Fine-tuning 40 GB+ recommended"),
    ("pi05_lerobot", 3.3e9, "bf16", "full", 80.0, "LeRobot pi05: sized for a single 80 GB GPU"),
]


def main() -> None:
    print("compute_budget: toy VRAM estimates vs official floors (no weights loaded)")
    print(f"{'name':<18} {'params':>8} {'prec':>4} {'mode':<5} {'est_GB':>7} {'floor_GB':>8} {'vs_floor':>10}")
    under = 0
    over = 0
    for name, n, prec, mode, floor, _note in OFFICIAL:
        est = estimate(n, prec, mode)
        tag = "UNDER" if est < floor else "OVER_OR_EQ"
        if est < floor:
            under += 1
        else:
            over += 1
        print(
            f"{name:<18} {n/1e9:7.2f}B {prec:>4} {mode:<5} {est:7.2f} {floor:8.1f} {tag:>10}"
        )
    print(f"rows={len(OFFICIAL)} under_floor={under} over_or_eq={over}")
    print("ok: floors from openpi README / Isaac-GR00T README / LeRobot smolvla+pi05 (page not run)")


if __name__ == "__main__":
    main()
```

## RPD 风格教师引导 RL 替身
已打开的 [RPD 论文](https://arxiv.org/abs/2503.05833)给 on-policy RL 加教师动作 / 均值 MSE：在 PPO 回报目标上补充模仿。论文用固定方差高斯说明动作均值 MSE 类似 BC。下面是**离散 Bernoulli** 替身，不声称这个高斯等价：使用 REINFORCE，没有 PPO clipping / critic；只在访问状态上做均值 MSE 监督梯度，不对状态占据分布求完整导数。ManiSkill 结果及论文百分比**以论文为准**；GPU / VLA / ManiSkill / 真机均为**本页未运行**。它不是 10.7 Cal-ConRFT 的离线保守 Q+BC，也不是 10.6 HIL-SERL，更不是 10.1 的扩散采样蒸馏。

论文中的 Hz 与成功率数字仅供阅读对比，不是本页实测或复现结果。

### 刻意次优、可超越的教师

状态 0…4 需按二元动作序列 `(1,0,1,1,0)` 前进；错动作回到 0，对动作以 0.05 概率打滑留在原处；5 是吸收成功态。起始以相同概率为 0 或 1，时域 12，终点到达 5 才奖励 1，gamma=1。随机教师一般以 0.9 概率选对，但在状态 2 仅 0.6。这是**刻意次优**的教师，不是真值 oracle。独立有限时域 DP 算得教师成功率 0.780842；总选对的策略接近 1（舍入为 1.000000）。DP 只作诊断，不给 RL 训练目标。

设策略动作 1 的均值为 `p=sigmoid(logit)`，查询教师标签 `aT∈{0,1}`；教师损失 `(p−aT)²` 的导数为 `2*(p−aT)*p*(1−p)`，期望梯度指向教师均值。每个非吸收决策的回报梯度 `G*(a−p)`，`G` 为该局终端回报。对访问状态求和，八局取平均，以 lr=0.25 上升。固定的**回报 / MSE 权重**如下：**BC-only=(0,1)**、**RL-from-scratch=(1,0)**、**RL+teacher-MSE=(1,0.5)**。这里 BC-only 指在线 MSE 模仿，交互预算相等但丢弃回报，**不是离线演示，也不是分类最大似然 BC**。

### 等预算、独立评估、不预定赢家

**11/23/37/53/71** 每种子三组初始 logits 相同。预生成不可变共享随机带，为各组提供相同起始、动作均匀数、打滑均匀数和教师标签均匀数，但访问的状态仍可能不同。每组 80 次更新 × 8 局 × 12 tick = **7680 环境步 / 80 次更新**；预先固定早期检查点均为 **768 步 / 8 次更新**。吸收后的 tick 仍计费；每 tick 都调用教师（含吸收态占位查询），scratch 丢弃标签。另报 active decisions：计费预算相同**不代表有效访问、计算量或墙钟时间相同**。

全部训练结束后，才在每种子相同、独立的 256 局留出评估上测两个预定快照；评估不选更新数，也不回馈训练。成功指 12 tick 内到达目标；`teacher_mean_MSE` 是对五个状态均匀平均的策略均值 / 教师均值差，只作诊断，不是成功率或反馈给拟合器的损失。stdout 列每种子、两检查点及均值 ± 总体标准差 / 范围。

```python
#!/usr/bin/env python3
"""RPD-style tabular Bernoulli/REINFORCE + action-mean teacher-MSE surrogate.
NOT PPO, VLA/HTTP/ManiSkill, Cal-ConRFT offline Q+BC, or HIL-SERL.
Teacher is deliberately SUBOPTIMAL: correct probability .9 except bottleneck .6.
Each arm starts from identical logits and consumes the same keyed random tape,
8 episodes x12 ticks per update, 80 updates. Goal stays absorbing to horizon;
those ticks are charged and active-decision counts are reported separately.
BC-only is online teacher-MSE imitation, NOT categorical maximum-likelihood BC.
Gamma=1; full-episode terminal return REINFORCE, no critic/clipping/replay.
Evaluation is separate and cannot select checkpoints or affect updates.
"""
from math import exp
from random import Random
from statistics import mean, pstdev

SEEDS = (11,23,37,53,71)
CORRECT = (1,0,1,1,0)
GOAL, HORIZON, BATCH, UPDATES = 5,12,8,80
CHECKPOINTS = (8,80)
SLIP, LR = .05,.25
# (reward-gradient weight, teacher-MSE weight), fixed before evaluation.
WEIGHTS = {"BC_only":(0.,1.), "RL_scratch":(1.,0.), "RL_teacher_MSE":(1.,.5)}


def sigmoid(x):
    return 1/(1+exp(-x))


def teacher_mean(state):
    if state == GOAL:
        return .5  # Charged dummy query while absorbing, ignored by all gradients.
    good = .6 if state == 2 else .9
    return good if CORRECT[state] else 1-good


def transition(state, action, slip_uniform):
    if state == GOAL:
        return GOAL
    if action != CORRECT[state]:
        return 0
    return state if slip_uniform < SLIP else state+1


def tape(seed, episodes):
    """One immutable common-random-numbers tape; independent of arm/trajectory."""
    rng = Random(seed)
    return tuple((int(rng.random()>=.5),
                  tuple((rng.random(),rng.random(),rng.random()) for _ in range(HORIZON)))
                 for _ in range(episodes))


def episode(probabilities, data):
    state, steps = data[0], []
    for action_u,slip_u,teacher_u in data[1]:
        p = .5 if state==GOAL else probabilities[state]
        label = int(teacher_u < teacher_mean(state))
        action = int(action_u < p)
        if state != GOAL:
            steps.append((state,action,p,label))
        state = transition(state,action,slip_u)
    return float(state==GOAL), tuple(steps)


def mse_logit_gradient(p, teacher_action):
    return 2*(p-teacher_action)*p*(1-p)


def update(logits, episodes, arm):
    reward_weight,mse_weight = WEIGHTS[arm]
    probabilities = tuple(sigmoid(x) for x in logits)
    gradient,active = [0.]*GOAL,0
    for data in episodes:
        reward,records = episode(probabilities,data)
        active += len(records)
        for state,action,p,label in records:
            gradient[state] += reward_weight*reward*(action-p)-mse_weight*mse_logit_gradient(p,label)
    # Objective is per-episode mean SUM of visited-state terms, not per-state normalization.
    return tuple(x+LR*g/len(episodes) for x,g in zip(logits,gradient)),active


def train(seed, arm, training_tape):
    rng=Random(seed)
    logits=tuple(rng.gauss(0,.05) for _ in range(GOAL))
    checkpoints,active = {},0
    for step in range(1,UPDATES+1):
        batch=training_tape[(step-1)*BATCH:step*BATCH]
        logits,visits=update(logits,batch,arm)
        active += visits
        if step in CHECKPOINTS:
            checkpoints[step]=(tuple(sigmoid(x) for x in logits), step*BATCH*HORIZON, step, active)
    return checkpoints


def exact_success(probabilities):
    """Independent finite-horizon DP baseline, not used by train/update."""
    value=[0.]*GOAL+[1.]
    for _ in range(HORIZON):
        previous=value
        value=[0.]*(GOAL+1);value[GOAL]=1.
        for state in range(GOAL):
            good=probabilities[state] if CORRECT[state] else 1-probabilities[state]
            value[state]=good*((1-SLIP)*previous[state+1]+SLIP*previous[state])+(1-good)*previous[0]
    return (value[0]+value[1])/2


def evaluate(probabilities, evaluation_tape):
    successes=sum(episode(probabilities,data)[0] for data in evaluation_tape)
    mismatch=mean((p-teacher_mean(s))**2 for s,p in enumerate(probabilities))
    return successes/len(evaluation_tape), mismatch


def main():
    teacher=tuple(teacher_mean(s) for s in range(GOAL))
    optimal=tuple(float(a) for a in CORRECT)
    print("RPD-style surrogate: discrete chain, deliberately suboptimal stochastic teacher")
    print(f"teacher_exact_success={exact_success(teacher):.6f} optimal_exact_success={exact_success(optimal):.6f}")
    print("weights BC_only=(RL0,MSE1) RL_scratch=(RL1,MSE0) RL_teacher_MSE=(RL1,MSE0.5)")
    print("paired tape:8 episodes x12 ticks/update; checkpoints8/80; gamma1 lr0.25")
    print("equal charged env_steps/updates; goal absorbing ticks included; active decisions reported")
    print("seed updates arm env_steps teacher_queries active_decisions success teacher_mean_MSE")
    aggregate={(step,arm):[] for step in CHECKPOINTS for arm in WEIGHTS}
    for seed in SEEDS:
        training=tape(1000+seed,UPDATES*BATCH)
        evaluation=tape(100000+seed,256)
        trained={arm:train(seed,arm,training) for arm in WEIGHTS}
        for step in CHECKPOINTS:
            for arm in WEIGHTS:
                probabilities,env_steps,updates,active=trained[arm][step]
                metrics=evaluate(probabilities,evaluation)
                aggregate[(step,arm)].append(metrics)
                success,mse=metrics
                print(f"{seed} {updates} {arm} {env_steps} {env_steps} {active} {success:.6f} {mse:.6f}")
    print("five-seed mean+-population_sd and range: updates arm metric mean sd min max")
    for (step,arm),rows in aggregate.items():
        for j,metric in enumerate(("success","teacher_mean_MSE")):
            values=[row[j] for row in rows]
            print(f"{step} {arm} {metric} {mean(values):.6f} {pstdev(values):.6f} {min(values):.6f} {max(values):.6f}")
    print("BC consumes equal online ticks/updates but ignores rewards; MSE imitation is not Bernoulli MLE")
    print("No presumed winner; teacher bias can hinder improvement; no PPO or VLA trained")
    print("Not Cal-ConRFT or HIL-SERL; real robot/GPU/ManiSkill: 本页未运行; paper 以论文为准")


if __name__ == "__main__":
    main()
```

### 模型假设

- **类预训练线性探针** 代替冻结骨干 + 轻量头。本页没有十亿参数 VLA。
- 臂 **A**（`A_sequential`）：只在新任务上微调——回放缓冲容量 **0**。
- 臂 **B**（`B_er_small`）：ER，缓冲容量 **8**，每步回放 **2**。
- 臂 **C**（`C_er_large`）：ER，缓冲容量 **48**，每步回放 **4**；切换阶段保留一半缓冲为旧任务样本（**保护前缀**）。
- **算力** = 各步 (|batch| + |replay|) 触及样本之和。该指标对墙钟 GPU 时间 **视而不见**。
- **共同随机数：** 一种子构建一条共享任务序 / 噪声带；A/B/C 看到相同任务与评估抽样。

### 公平设置

共享任务序、共享梯度步数预算，五多种子。不预设赢家：**no_shift** 下 ER 应与顺序微调 **接近打平**（max |Δavg| 很小；无任务切换时 ER 买不到收益）。在 **task_shift** 上，B 的平均成功率 0.7617±0.0336 略低于 A 的 0.7633±0.0529（差距在种子噪声之内）；NBT 遗忘为 A 0.1200±0.0617、B 0.0867±0.0398、C 0.0967±0.0545——ER 相对顺序微调可以降低遗忘，但更大的缓冲/算力并 **没有** 单调地降低 NBT（C 比 B 略差，也在噪声之内），并要付出明确的缓冲与算力代价。仅当间隙实质为 0 时称 **打平**；否则称 **接近打平 / 小间隙** 并写出间隙。

```python
#!/usr/bin/env python3
"""Synthetic continual VLA finetune + experience-replay literacy toy. Stdlib only.

Shared task sequence, shared total gradient steps, and shared sample noise for
every arm (common random numbers per seed). Every arm faces the **same** noise.

Arms:
  A_sequential   finetune task-by-task with **no** replay (buffer capacity 0)
  B_er_small     ER with small replay buffer (capacity B_SMALL)
  C_er_large     ER with larger replay buffer (capacity B_LARGE)

Regimes (≥5 seeds):
  no_shift     — single task; A/B/C **near-tie / small gap** (ER gives no gain)
  task_shift   — two tasks; sequential forgets; ER can cut NBT but not monotonically

Metrics: avg success, backward transfer (NBT-style drop on task0 after task1),
replay buffer capacity, and compute cost = grad_steps × (batch + replay_per_step).

Assumptions: linear probe stand-in of pretrained vs sequential BC — not real VLA
weights. Liu et al. arXiv:2603.03818 (abs+HTML+PDF): pretrained VLAs + small ER.
Paper LIBERO/NBT % 以论文为准; GPU/weights 本页未运行.
≠ compute_budget floors alone; ≠ SARM/DPO/RPD; ≠ MemoryVLA.
"""
from __future__ import annotations

import math
import random

SEEDS = (0, 1, 2, 3, 4)
ARMS = ("A_sequential", "B_er_small", "C_er_large")
DIM = 4
B_SMALL = 8
B_LARGE = 48
GRAD_STEPS_PER_TASK = 40
BATCH = 4
# Replay draws per step (compute cost explicit)
REPLAY_PER_STEP_SMALL = 2
REPLAY_PER_STEP_LARGE = 4
REGIMES = ("no_shift", "task_shift")


def shared_tasks(seed: int):
    """CRN: task datasets + eval noise identical for all arms."""
    rng = random.Random(60_000 + seed)
    # Two tasks: linear classifiers with different targets in feature space
    def make_task(offset: float, n: int = 60):
        X, y = [], []
        for _ in range(n):
            x = [rng.gauss(offset, 0.6) for _ in range(DIM)]
            # Label: sign of weighted sum (task-specific weights)
            w = [1.0, -0.5, 0.3, 0.8]
            s = sum(w[k] * (x[k] - offset) for k in range(DIM))
            y.append(1.0 if s >= 0.0 else 0.0)
            X.append(x)
        return X, y

    t0 = make_task(0.0)
    t1 = make_task(2.0)
    # Shared eval noise on logits
    eval_noise = [rng.gauss(0.0, 0.05) for _ in range(200)]
    return t0, t1, eval_noise


def init_pretrained(seed: int):
    """Shared pretrained-like prior (same for all arms)."""
    rng = random.Random(70_000 + seed)
    w = [0.2 * rng.gauss(0.0, 1.0) for _ in range(DIM)]
    b = 0.0
    return w, b


def predict_prob(x, w, b, noise=0.0):
    z = b + sum(w[k] * x[k] for k in range(DIM)) + noise
    # logistic
    if z >= 0:
        ez = math.exp(-z)
        return 1.0 / (1.0 + ez)
    ez = math.exp(z)
    return ez / (1.0 + ez)


def accuracy(X, y, w, b, eval_noise):
    ok = 0
    for i, (x, yi) in enumerate(zip(X, y)):
        p = predict_prob(x, w, b, eval_noise[i % len(eval_noise)])
        ok += 1 if (p >= 0.5) == (yi >= 0.5) else 0
    return ok / len(y)


def sgd_step(w, b, batch_X, batch_y, lr=0.2):
    gw = [0.0] * DIM
    gb = 0.0
    for x, yi in zip(batch_X, batch_y):
        p = predict_prob(x, w, b)
        err = p - yi
        gb += err
        for k in range(DIM):
            gw[k] += err * x[k]
    n = len(batch_X)
    b -= lr * gb / n
    w = [w[k] - lr * gw[k] / n for k in range(DIM)]
    return w, b


def train_on_task(w, b, X, y, buffer, buf_cap, replay_per_step, steps, rng, protect_old=0):
    """One task phase. Mutates buffer. Returns (w,b,compute_units).

    protect_old: number of leading buffer slots reserved for prior-task exemplars
    (not overwritten during the new task). Explicit ER capacity accounting.
    """
    compute = 0
    n = len(X)
    for step in range(steps):
        # Fresh batch from current task
        batch_X, batch_y = [], []
        for _ in range(BATCH):
            i = rng.randrange(n)
            batch_X.append(X[i])
            batch_y.append(y[i])
            # Add to replay buffer (reservoir-ish); never overwrite protected prefix
            if buf_cap > 0:
                if len(buffer) < buf_cap:
                    buffer.append((X[i], y[i]))
                else:
                    lo = max(0, min(protect_old, buf_cap - 1))
                    j = rng.randrange(lo, buf_cap)
                    buffer[j] = (X[i], y[i])
        # Replay mix
        if buffer and replay_per_step > 0:
            for _ in range(replay_per_step):
                bx, by = buffer[rng.randrange(len(buffer))]
                batch_X.append(bx)
                batch_y.append(by)
        compute += len(batch_X)  # samples touched this step
        w, b = sgd_step(w, b, batch_X, batch_y)
    return w, b, compute


def run_arm(arm: str, seed: int, regime: str):
    t0, t1, eval_noise = shared_tasks(seed)
    w, b = init_pretrained(seed)
    # Private RNG only for batch index order; data itself is shared CRN
    # Batch-index RNG: arm-salted under shift (private order); seed-only on no_shift so ER overhead
    # cannot look like a different fit when there is nothing to replay across tasks.
    if regime == "no_shift":
        rng = random.Random(80_000 + seed)
    else:
        rng = random.Random(80_000 + seed * 17 + {"A_sequential": 1, "B_er_small": 2, "C_er_large": 3}[arm])
    if arm == "A_sequential":
        cap, replay = 0, 0
    elif arm == "B_er_small":
        cap, replay = B_SMALL, REPLAY_PER_STEP_SMALL
    else:
        cap, replay = B_LARGE, REPLAY_PER_STEP_LARGE
    buffer: list[tuple[list[float], float]] = []
    total_compute = 0

    # Task 0
    w, b, c0 = train_on_task(w, b, t0[0], t0[1], buffer, cap, replay, GRAD_STEPS_PER_TASK, rng)
    total_compute += c0
    acc0_after0 = accuracy(t0[0], t0[1], w, b, eval_noise)

    if regime == "no_shift":
        # Only one task — ER cannot help relative to sequential
        avg = acc0_after0
        nbt = 0.0  # no second task
        return avg, nbt, cap, total_compute, replay

    # Seed / refresh buffer with task-0 exemplars before shift (ER's point)
    if cap > 0:
        buffer.clear()
        for i in range(min(cap, len(t0[0]))):
            buffer.append((t0[0][i], t0[1][i]))

    # Task 1 — protect half the buffer as prior-task exemplars
    protect = cap // 2
    w, b, c1 = train_on_task(
        w, b, t1[0], t1[1], buffer, cap, replay, GRAD_STEPS_PER_TASK, rng, protect_old=protect
    )
    total_compute += c1
    acc0_after1 = accuracy(t0[0], t0[1], w, b, eval_noise)
    acc1_after1 = accuracy(t1[0], t1[1], w, b, eval_noise)
    avg = 0.5 * (acc0_after1 + acc1_after1)
    # NBT-style: drop on task0 (positive = forgetting)
    nbt = acc0_after0 - acc0_after1
    return avg, nbt, cap, total_compute, replay


def mean_sd(xs):
    n = len(xs)
    mu = sum(xs) / n
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (n - 1))


def main():
    print(
        "Synthetic continual VLA + ER toy; shared tasks/steps/noise; "
        "not MemoryVLA / SARM / compute_budget redo"
    )
    print(
        f"DIM={DIM} GRAD_STEPS_PER_TASK={GRAD_STEPS_PER_TASK} BATCH={BATCH} "
        f"B_SMALL={B_SMALL} B_LARGE={B_LARGE} "
        f"REPLAY_PER_STEP={REPLAY_PER_STEP_SMALL}/{REPLAY_PER_STEP_LARGE} "
        f"seeds={len(SEEDS)}"
    )
    print(
        "model: pretrained-like linear probe; A=sequential (cap=0); "
        f"B=ER cap={B_SMALL}; C=ER cap={B_LARGE}; "
        "compute = sum over steps of |batch|+|replay|; "
        "Liu et al. arXiv:2603.03818; LIBERO/NBT % 以论文为准; GPU/weights 本页未运行"
    )
    print(
        "regime arm avg_success nbt_forget buffer_cap compute_samples "
        "replay_per_step (mean+-sd over seeds)"
    )
    summary = {}
    for regime in REGIMES:
        for arm in ARMS:
            avgs, nbts, caps, comps, rps = [], [], [], [], []
            for seed in SEEDS:
                avg, nbt, cap, comp, rp = run_arm(arm, seed, regime)
                avgs.append(avg)
                nbts.append(nbt)
                caps.append(cap)
                comps.append(comp)
                rps.append(rp)
            ma, sa = mean_sd(avgs)
            mn, sn = mean_sd(nbts)
            mc, sc = mean_sd(comps)
            summary[(regime, arm)] = (ma, mn, caps[0], mc, rps[0], sa, sn)
            print(
                f"{regime} {arm} {ma:.4f}+-{sa:.4f} {mn:.4f}+-{sn:.4f} "
                f"{caps[0]} {mc:.1f}+-{sc:.1f} {rps[0]}"
            )

    # no_shift: A vs B avg gap
    aa = summary[("no_shift", "A_sequential")][0]
    ab = summary[("no_shift", "B_er_small")][0]
    ac = summary[("no_shift", "C_er_large")][0]
    gap = max(abs(aa - ab), abs(aa - ac), abs(ab - ac))
    if gap < 1e-6:
        lab = "numerical equality (gap≈0)"
    elif gap < 0.02:
        lab = f"near-tie / small gap; max |Δavg|={gap:.4f}; no task shift ⇒ ER buys nothing"
    else:
        lab = f"gap; max |Δavg|={gap:.4f}"
    print(f"no_shift: A/B/C avg_success ({lab})")

    aa, fa, _, _, _, saa, sna = summary[("task_shift", "A_sequential")]
    ab, fb, _, _, _, sab, snb = summary[("task_shift", "B_er_small")]
    ac, fc, _, _, _, sac, snc = summary[("task_shift", "C_er_large")]
    print(
        f"task_shift: avg A={aa:.4f}+-{saa:.4f} B={ab:.4f}+-{sab:.4f} C={ac:.4f}+-{sac:.4f}; "
        f"NBT forget A={fa:.4f}+-{sna:.4f} B={fb:.4f}+-{snb:.4f} C={fc:.4f}+-{snc:.4f} "
        f"(B avg slightly below A within seed noise; ER can reduce forgetting vs sequential, "
        f"but larger buffer/compute does not monotonically cut NBT — C slightly worse than B "
        f"within seed noise; B compute={summary[('task_shift','B_er_small')][3]:.0f}, "
        f"C compute={summary[('task_shift','C_er_large')][3]:.0f} vs "
        f"A={summary[('task_shift','A_sequential')][3]:.0f})"
    )
    print(

        "buffer capacity and replay samples are real costs; "
```

## 分阶段对齐与混合专家（ChatVLA）

上面几节只为控制做微调。ChatVLA 报告说，用视觉-语言骨干这样做会抹掉它回答问题的能力，并提出两个补救办法：先训练控制，再加入多模态数据；给两种技能各自独立的前馈专家。本节在一个**很小的线性替身**上比较纯机器人数据微调、这种分阶段做法和朴素联合训练。它**不是** Qwen2-VL 或 ChatVLA 的代码，**不是** DPO 或 OpenHelix，**不是** CoT-VLA，**不是** Magma，也**不是**上面的经验回放 toy（没有回放缓冲，没有任务序列）。

**实际打开的来源（已读 arXiv 摘要页、HTML 和 PDF 正文）。**

- **Zhou 等，ChatVLA: Unified Multimodal Understanding and Robot Control with Vision-Language-Action Model（arXiv 2502.14420 v1，11 位作者，2025 年 2 月 20 日提交）** 的内容如下。
  - 只用机器人数据训练时，模型“在所有基准上的表现都是 0”（"yields a performance of 0 across all benchmarks"），被提问时“只会喃喃自语”（"only murmuring when asked a question"）。论文称之为**虚假遗忘**（spurious forgetting）：机器人训练覆盖的是视觉-文本对齐，而不是删除了知识；用固定推理模板训练似乎能把它“重新激活”（"reactivate"）。
  - **任务干扰**（task interference）。控制和理解争夺共享参数，所以两者联合训练（按 RT-2 的做法，机器人数据与视觉-文本数据为 3:1）会损失控制性能。
  - **分阶段对齐训练**（Phased Alignment Training）。模型“先掌握具身控制，再逐步加入多模态数据”（"first masters embodied control before incrementally integrating multimodal data"）。
  - **混合专家**（MoE）。自注意力层共享，MLP 拆成控制专家和理解专家；“第 1 阶段训练中只激活控制专家”（"in stage 1 training, only the control expert is activated"）。用 "Answer based on question" 和 "Predict robot action" 这样的系统提示选择路径。
  - 论文在 25 个真实任务上与 OpenVLA 等方法比较；其数字以论文为准。GPU 或真机运行**本页未运行**。

### 替身模型与度量

记输入为 `x`，共享层为 `A`，投影到 `M=2` 个特征的专家投影为 `P`，新的动作头为 `c`，预训练的回答头为 `V`：

<Formula label="先过共享层，再过按任务选择的专家">z = A x,  f = P_task z</Formula>

<Formula label="控制输出与回答 logits">â = c · f,  logits = V f</Formula>

<Formula label="回答标签就是预训练模型自己的回答，所以打印出它的基线">y = argmax V0 P0 A0 x</Formula>

<Formula label="遗忘差距，单位为百分点">gap = baseline VQA − arm VQA</Formula>

**算例。** toy 打印 `NQ=3`，所以随机猜的准确率是 `33.3333 %`；冻结的基线回答的是它自己的标签，每个情形都是 `100.0000 %`。回答头读取预训练投影给出的两个特征。在 `conflict` 中机器人目标是 `W = (0, 0, 1, 0, 0, 0)`，方向在这两个特征之外。只有一个共享 `P` 时，拟合动作必须把仅有的两行特征之一转向这个方向，而三个回答类别两行都需要：两个任务争夺同样的 `M=2` 个特征。用混合专家时，转动的是 `P_ctrl`，`P_vqa` 不动。在 `aligned` 中目标是 `W = (1, 0, 0, 0, 0, 0)`，本来就在回答特征之内，应该没什么可争的。

**正确性判据。** 冻结模型在自己的标签上必须得到 `100.0000 %`，toy 打印这个数，测试文件也检查它。测试文件还检查 B 的第 1 阶段中，机器人更新后回答专家和回答头都没变，所以 B 第 1 阶段 VQA 的任何下降都只来自共享层 `A`。**没有积分步。** 训练是离散的 SGD 更新，没有被控对象；预算就是批次数，每组都写明。

```python
#!/usr/bin/env python3
"""ChatVLA-style phased alignment + MoE vs robot-only SFT vs naive joint training, on one tiny shared model.

Synthetic, stdlib only.  Not Qwen2-VL, not DiffusionVLA, not ChatVLA's code: a linear stand-in
that keeps only the structure the paper describes (a shared layer like the shared self-attention,
task-specific FFN experts, an action head and a language head).  Not DPO / OpenHelix, not
CoT-VLA, not Magma, and not this lesson's experience-replay toy (no replay buffer, no task
sequence).  ChatVLA paper numbers (VQA, 25 real tasks) are 以论文为准; GPU or real-robot runs:
本页未运行.

Model (dimensions D input, M bottleneck, NQ answer classes):
  z = A x                 shared layer (D x D), pretrained A0 = I plus a small per-seed perturbation
  f = P z                 FFN projection to M features (M x D); P0 rows = e1, e2
  control  a_hat = c . f  action head, new (c = 0 at the start)
  VQA      logits = V f   language head, pretrained V0 (NQ prototypes at 0, 120, 240 degrees)
The "pretrained VLM" answers VQA by its own map: label y = argmax V0 P0 A0 x, so its own VQA
accuracy on fresh inputs is printed as the frozen baseline.  The robot target is a = W . x with
W set by the regime.
Arms (equal robot budget R batches; B and C also get the same Q VQA batches):
  A_robot   robot-only SFT through the single shared P; V is never updated (no VQA data)
  B_phased  MoE: P_ctrl and P_vqa both start at P0.  Stage 1: first R - R2 robot batches, only
            the control expert and A train.  Stage 2: remaining R2 robot batches and Q VQA
            batches, interleaved, robot through P_ctrl and VQA through P_vqa (A shared)
  C_joint   naive joint training through the single shared P from step 0, robot : VQA = 3 : 1
            interleaving (the ratio the paper used, following RT-2), same R and Q
Training: plain SGD, batch BATCH, learning rate LR, squared error for control, softmax cross
entropy for VQA.  No integration step (discrete SGD updates, no plant); the step count is the
budget, stated per arm.
Metrics per seed: control success % (|a_hat - a| <= TOL on TEST inputs), VQA accuracy % on TEST
inputs, forgetting gap = baseline VQA - arm VQA (points).  B's VQA after stage 1 is printed too.
Common random numbers: per (regime, seed) the init perturbation, every robot batch, every VQA
batch and the test set are drawn once; batch i is the same batch for every arm that uses it.
Regimes and expectations, written before the first run:
  conflict   W = e3 (outside the VQA features), Q 100   A consistent VQA collapse vs B;
                                                        A and B near-tie on control;
                                                        C vs B: no prediction, printed either way
  aligned    W = e1 (inside the VQA features), Q 100    all three near-tie on control and VQA
  few_vqa    W = e3, Q 10                               B still consistent VQA win over A;
                                                        C vs B: no prediction
"""
import math
import random

SEEDS = (0, 1, 2, 3, 4)
D, M, NQ = 6, 2, 3
R, R2 = 300, 100          # robot batches per arm; B's stage-2 robot batches
BATCH = 8
LR = 0.05
TOL = 0.25
TEST = 400
INIT_SD = 0.05
NUM_ZERO = 1e-9
ARMS = ("A_robot", "B_phased", "C_joint")
# name: (robot target direction W, VQA batches Q); written before the first run
REGIMES = {
    "conflict": ((0, 0, 1, 0, 0, 0), 100),
    "aligned": ((1, 0, 0, 0, 0, 0), 100),
    "few_vqa": ((0, 0, 1, 0, 0, 0), 10),
}
STREAM = {"conflict": 0, "aligned": 1, "few_vqa": 2}
V0 = [[math.cos(2 * math.pi * k / NQ), math.sin(2 * math.pi * k / NQ)] for k in range(NQ)]
P0 = [[1.0 if j == i else 0.0 for j in range(D)] for i in range(M)]


def matvec(mat, v):
    return [sum(a * b for a, b in zip(row, v)) for row in mat]


def argmax(v):
    return max(range(len(v)), key=lambda k: v[k])


def copy(mat):
    return [row[:] for row in mat]


def shared_draws(regime, seed):
    rng = random.Random(1000 * STREAM[regime] + seed)
    w, q = REGIMES[regime]
    a0 = [[(1.0 if i == j else 0.0) + rng.gauss(0.0, INIT_SD) for j in range(D)] for i in range(D)]
    teacher = lambda x: argmax(matvec(V0, matvec(P0, matvec(a0, x))))
    gx = lambda: [rng.gauss(0.0, 1.0) for _ in range(D)]
    robot = []
    for _ in range(R):
        xs = [gx() for _ in range(BATCH)]
        robot.append([(x, sum(a * b for a, b in zip(w, x))) for x in xs])
    vqa = []
    for _ in range(q):
        xs = [gx() for _ in range(BATCH)]
        vqa.append([(x, teacher(x)) for x in xs])
    test = []
    for _ in range(TEST):
        x = gx()
        test.append((x, sum(a * b for a, b in zip(w, x)), teacher(x)))
    return a0, robot, vqa, test


class Net:
    def __init__(self, a0, moe):
        self.A = copy(a0)
        self.P = {"ctrl": copy(P0), "vqa": copy(P0)} if moe else {"ctrl": copy(P0)}
        self.moe = moe
        self.c = [0.0] * M
        self.V = copy(V0)
        self.train_a = True
        self.frozen = set()

    def proj(self, task):
        return self.P["vqa"] if (self.moe and task == "vqa") else self.P["ctrl"]

    def step(self, batch, task):
        P = self.proj(task)
        gA = [[0.0] * D for _ in range(D)]
        gP = [[0.0] * D for _ in range(M)]
        gc = [0.0] * M
        gV = [[0.0] * M for _ in range(NQ)]
        for x, y in batch:
            z = matvec(self.A, x)
            f = matvec(P, z)
            if task == "ctrl":
                err = sum(a * b for a, b in zip(self.c, f)) - y
                df = [err * ci for ci in self.c]
                for i in range(M):
                    gc[i] += err * f[i]
            else:
                logits = matvec(self.V, f)
                mx = max(logits)
                ex = [math.exp(l - mx) for l in logits]
                s = sum(ex)
                g = [e / s - (1.0 if k == y else 0.0) for k, e in enumerate(ex)]
                df = [sum(g[k] * self.V[k][i] for k in range(NQ)) for i in range(M)]
                for k in range(NQ):
                    for i in range(M):
                        gV[k][i] += g[k] * f[i]
            dz = [sum(df[i] * P[i][j] for i in range(M)) for j in range(D)]
            for i in range(M):
                for j in range(D):
                    gP[i][j] += df[i] * z[j]
            for i in range(D):
                for j in range(D):
                    gA[i][j] += dz[i] * x[j]
        n = float(len(batch))
        for i in range(M):
            for j in range(D):
                P[i][j] -= LR * gP[i][j] / n
        for i in range(D):
            for j in range(D):
                self.A[i][j] -= LR * gA[i][j] / n
        if task == "ctrl":
            for i in range(M):
                self.c[i] -= LR * gc[i] / n
        else:
            for k in range(NQ):
                for i in range(M):
                    self.V[k][i] -= LR * gV[k][i] / n

    def evaluate(self, test):
        ok_c = ok_q = 0
        for x, a, y in test:
            z = matvec(self.A, x)
            fc = matvec(self.proj("ctrl"), z)
            ok_c += abs(sum(p * q for p, q in zip(self.c, fc)) - a) <= TOL
            ok_q += argmax(matvec(self.V, matvec(self.proj("vqa"), z))) == y
        return 100.0 * ok_c / len(test), 100.0 * ok_q / len(test)


def run_arm(arm, draws):
    a0, robot, vqa, test = draws
    net = Net(a0, moe=(arm == "B_phased"))
    out = {}
    if arm == "A_robot":
        for b in robot:
            net.step(b, "ctrl")
    elif arm == "B_phased":
        for b in robot[:R - R2]:
            net.step(b, "ctrl")
        out["stage1_vqa"] = net.evaluate(test)[1]
        rest, qs = robot[R - R2:], list(vqa)
        n = max(len(rest), len(qs))
        for i in range(n):
            if i < len(rest):
                net.step(rest[i], "ctrl")
            if i < len(qs):
                net.step(qs[i], "vqa")
    else:
        qi = 0
        for i, b in enumerate(robot):
            net.step(b, "ctrl")
            if i % 3 == 2 and qi < len(vqa):
                net.step(vqa[qi], "vqa")
                qi += 1
        while qi < len(vqa):
            net.step(vqa[qi], "vqa")
            qi += 1
    ctrl, q = net.evaluate(test)
    base = Net(a0, moe=False).evaluate(test)[1]
    out.update({"ctrl": ctrl, "vqa": q, "gap": base - q, "base": base})
    return out


def mean_sd(xs):
    mu = sum(xs) / len(xs)
    return mu, math.sqrt(sum((x - mu) ** 2 for x in xs) / (len(xs) - 1))


def num(x):
    if x != 0.0 and abs(x) < 5e-5:
        return f"{x:.3e}"
    return f"{x:.4f}"


def sign_note(diffs, m):
    n, opp, zero = len(diffs), sum(1 for d in diffs if d * m < 0), sum(1 for d in diffs if d == 0)
    if opp == 0 and zero == 0:
        return f"same sign on all {n} seeds"
    if opp == 0:
        return f"no seed has the opposite sign, {zero} of {n} equal"
    note = f"{opp} of {n} seeds {'has' if opp == 1 else 'have'} the opposite sign"
    return note + (f", {zero} of {n} equal" if zero else "")


def verdict(xa, xb):
    diffs = [a - b for a, b in zip(xa, xb)]
    if all(d == 0.0 for d in diffs):
        return "0.0000+-0.0000: tie: identical on every seed"
    m, s = mean_sd(diffs)
    if abs(m) < NUM_ZERO and s < NUM_ZERO:
        return f"{num(m)}+-{num(s)}: numerically zero (floating-point residue), effectively a tie"
    if abs(m) <= s:
        label = "near-tie / small gap, within seed noise (|m| <= sd)"
    elif abs(m) < 0.2 * min(abs(mean_sd(xa)[0]), abs(mean_sd(xb)[0])):
        label = "small but consistent gap, not a tie (|m| > sd)"
    else:
        label = "clear gap, not a tie (|m| > sd)"
    return f"{num(m)}+-{num(s)}: {label}; {sign_note(diffs, m)}"


def main():
    print("Synthetic ChatVLA-style probe: robot-only SFT vs phased alignment + MoE vs naive joint (3:1); linear stand-in, not Qwen2-VL")
    print(f"D={D} M={M} NQ={NQ} R={R} robot batches (B stage 2: {R2}) BATCH={BATCH} LR={LR} TOL={TOL} TEST={TEST} seeds={len(SEEDS)}")
    print(f"chance VQA accuracy = {100.0 / NQ:.4f} %; no integration step (discrete SGD updates); C interleaves robot:VQA = 3:1")
    print("regime arm control_% vqa_% forgetting_gap_pts  (mean+-sd over seeds)")
    for regime, (w, q) in REGIMES.items():
        draws = [shared_draws(regime, s) for s in SEEDS]
        res = {arm: [run_arm(arm, d) for d in draws] for arm in ARMS}
        base = "{}+-{}".format(*map(num, mean_sd([r["base"] for r in res["A_robot"]])))
        print(f"{regime} (W = {w}; Q = {q} VQA batches; frozen baseline VQA {base} %)")
        for arm in ARMS:
            cols = ["{}+-{}".format(*map(num, mean_sd([r[k] for r in res[arm]]))) for k in ("ctrl", "vqa", "gap")]
            print(f"  {arm} " + " ".join(cols))
        s1 = "{}+-{}".format(*map(num, mean_sd([r["stage1_vqa"] for r in res["B_phased"]])))
        print(f"  B_phased VQA after stage 1 (robot only, before any VQA batch): {s1} %")
        for x, y in (("B_phased", "A_robot"), ("C_joint", "B_phased")):
            for key in ("ctrl", "vqa"):
                print(f"  paired {x[0]}-{y[0]} {key}: {verdict([r[key] for r in res[x]], [r[key] for r in res[y]])}")
    print("No winner declared: phased alignment + MoE restores VQA lost to robot-only SFT; whether naive joint training costs control depends on whether the tasks compete for features")
    print("ChatVLA numbers 以论文为准; GPU / 25-task real robot: 本页未运行")


if __name__ == "__main__":
    main()
```
