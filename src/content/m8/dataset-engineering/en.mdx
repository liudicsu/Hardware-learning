Month 6 taught you to evaluate a policy on a fixed list of starts. Month 8 starts one step earlier: the dataset that makes that evaluation honest. A pretrained VLA is only as good as the episodes you fine-tune on. This lesson writes a **variation × repetition** recording protocol, reads what LeRobotDataset v3 changed about on-disk layout, audits a tiny synthetic fixture on the CPU, and links evaluation starts back to Month 6. Open X-Embodiment (arXiv 2310.08864) is the large-scale reason people care about shared formats at all.

## Variation × repetition

A single tidy demo that always starts from the same pose teaches a brittle policy. The SmolVLA docs suggest starting around **~50 episodes** for one task; their pick-and-place example used **5 cube positions × 10 episodes**. Under **25** episodes was not enough in that setup. Treat that as a starting rule of thumb, then write your own grid:

| variation factor | levels | episodes per level |
|------------------|--------|--------------------|
| object pose      | 5      | 10                 |
| lighting         | …      | …                  |

Fill `examples/month-8/dataset-engineering/protocol.md` before the first teleop session. Exclusion rules belong on the same page: lost tracking, e-stop, collision, teleop freeze, wrong instruction spoken.

## LeRobotDataset v3

Hugging Face's LeRobot **dataset v3** docs describe a layout where **metadata locates episode boundaries**, and **many episodes can share one Parquet and one MP4**. That is the opposite of "one file per episode". For large Hub datasets you stream with `StreamingLeRobotDataset` instead of downloading everything first. The toy in this lesson mirrors the idea with JSON: `meta/episodes.jsonl` names each episode and variation; `data/chunk-000.json` holds frames from several episodes with an `episode_index` field.

```python
#!/usr/bin/env python3
"""dataset_audit.py: audit a tiny synthetic LeRobot-v3-style dataset (stdlib only).

Layout (toy mirror of LeRobotDataset v3 ideas — many episodes per file, metadata
locates episode boundaries; not a Hub download):
  synthetic_lerobot/meta/info.json
  synthetic_lerobot/meta/episodes.jsonl
  synthetic_lerobot/data/chunk-000.json   # list of frame dicts; episode_index field

Reports: episodes per variation, mean duration, action min/max, missing-frame
gaps, and which variations are under-covered (< target_min episodes).
usage: python3 dataset_audit.py
"""
from __future__ import annotations

import json
import math
import os
from collections import defaultdict

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.join(HERE, "synthetic_lerobot")
TARGET_MIN = 8  # SmolVLA docs suggest ~50 for real tasks; toy threshold is smaller


def main() -> None:
    with open(os.path.join(ROOT, "meta", "info.json"), encoding="utf-8") as f:
        info = json.load(f)
    fps = float(info["fps"])
    episodes = []
    with open(os.path.join(ROOT, "meta", "episodes.jsonl"), encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                episodes.append(json.loads(line))
    with open(os.path.join(ROOT, "data", "chunk-000.json"), encoding="utf-8") as f:
        frames = json.load(f)

    by_ep: dict[int, list[dict]] = defaultdict(list)
    for fr in frames:
        by_ep[int(fr["episode_index"])].append(fr)

    print(f"dataset {info['repo_id']}  fps={fps:g}  episodes={len(episodes)}  frames={len(frames)}")
    print(f"{'variation':<16} {'n_ep':>5} {'mean_s':>8} {'action_min':>11} {'action_max':>11} {'miss':>5} {'flag':>10}")

    under = []
    for ep_meta in episodes:
        eid = int(ep_meta["episode_index"])
        var = ep_meta["variation"]
        seq = sorted(by_ep[eid], key=lambda r: int(r["frame_index"]))
        # duration from frame count / fps
        dur = len(seq) / fps if fps else float("nan")
        actions = [float(r["action"][0]) for r in seq]  # first joint for range
        amin = min(actions) if actions else float("nan")
        amax = max(actions) if actions else float("nan")
        # missing frames: gaps in frame_index
        idxs = [int(r["frame_index"]) for r in seq]
        miss = 0
        if idxs:
            expect = list(range(idxs[0], idxs[-1] + 1))
            miss = len(expect) - len(idxs)
        ep_meta["_dur"] = dur
        ep_meta["_amin"] = amin
        ep_meta["_amax"] = amax
        ep_meta["_miss"] = miss
        ep_meta["_n"] = len(seq)

    # aggregate by variation
    groups: dict[str, list[dict]] = defaultdict(list)
    for ep_meta in episodes:
        groups[ep_meta["variation"]].append(ep_meta)

    for var in sorted(groups):
        eps = groups[var]
        n = len(eps)
        mean_s = sum(e["_dur"] for e in eps) / n
        amin = min(e["_amin"] for e in eps)
        amax = max(e["_amax"] for e in eps)
        miss = sum(e["_miss"] for e in eps)
        flag = "UNDER" if n < TARGET_MIN else "ok"
        if flag == "UNDER":
            under.append(var)
        print(f"{var:<16} {n:5d} {mean_s:8.2f} {amin:11.3f} {amax:11.3f} {miss:5d} {flag:>10}")

    print(f"target_min_episodes_per_variation={TARGET_MIN}")
    if under:
        print("under_covered: " + ", ".join(under))
    else:
        print("under_covered: (none)")
    # pinned totals for check_diagrams
    print(f"total_variations={len(groups)} total_episodes={len(episodes)}")


if __name__ == "__main__":
    main()
```

## Common mistakes

- **Recording 20 identical episodes.** Count variations, not only total episodes.
- **Skipping the hold-out starts.** If every eval pose was in training, the success rate is not an evaluation.
- **Assuming one file per episode.** v3 packs many episodes per Parquet/MP4; boundaries live in metadata.
- **Ignoring missing frames.** One gap in `cube_C` is already a data bug; find it with an audit script, not by eye.

## CRC for logged chunks: corruption detection is not authentication (+150 min)

A byte-perfect log can still have poor labels, missing frames or a weak recording protocol; conversely, useful episodes can be silently corrupted. Add integrity checks at ingestion as a separate concern. This toy uses **synthetic 16-byte chunks**, not actual Parquet/MP4/bag files, and **does not assert that LeRobotDataset v3 stores this CRC**. Keep the existing dataset audit and held-out-start discipline; this is not CAN/EtherCAT error confinement, QoS acceptance or a Dynamixel packet tutorial.

<Resource id="crc-polynomial-reference" />

The opened CRC reference is **secondary**, not a protocol standard. A CRC is polynomial division over GF(2): addition is XOR, not integer carries. We choose **CRC-32/IEEE (CRC-32/ISO-HDLC)**, normal polynomial **0x04C11DB7** with its implicit degree32 term; the reflected bit loop uses **0xEDB88320**, init and xorout **0xFFFFFFFF**, bytes in order and bits **LSB-first**. This is not CRC-32C. Its ASCII check value is **CRC32("123456789") = 0xCBF43926**; our explicit bitwise implementation is cross-checked against Python stdlib `zlib.crc32`, not replaced by a hard-coded answer.

### Same payloads, same corruption, three checkers

`dataset_crc_integrity_toy.py` uses four shared 16-byte payloads: zeros, FFs, ascending bytes0…15, and `(37*j+11) mod256`. For every corrupted copy, **no check**, **sum8=sum(bytes) mod256**, and **CRC32** receive the same bytes and original stored check values; no-check ignores the tags. Tags, length and framing remain intact in this experiment; neither insertion/deletion nor a specific storage/channel failure probability is simulated. Corruption is fully deterministic, so no stochastic seeds or five-seed uncertainty claim is needed.

| corrupted corpus | cases | no-check undetected | sum8 undetected | CRC32 undetected |
|---|---|---|---|---|
| every single-bit position | 512 | 512/512 | 0/512 | 0/512 |
| all-ones bursts, spans2…32, every fitting position | 13888 | 13888/13888 | 116/13888 (0.008353) | 0/13888 |
| adjacent balanced bytes (+1,−1 mod256) | 32 | 32/32 | 32/32 | 0/32 |
| deliberate degree32 generator error, span33 | 4 | 4/4 | 0/4 | 4/4 |

**Undetected error rate = accepted corrupted cases / corrupted cases**, not fraction of clean records and not an estimated real-channel probability. All four clean chunks pass each checker. The balanced-byte case exposes addition's cancellation. The 33-bit case exposes a real CRC collision and even a case sum8 catches: do not claim CRC detects every error or always dominates every checksum on every pattern.

### Why any single burst of span ≤32 bits is detected

For fixed-length equal-framed messages, CRC init/xorout offsets cancel when comparing errors. An error burst of spanL has a polynomial `E(x)=x^k B(x)` with nonzero first/last bits and `degree(B)=L−1`. The generator has degree32 and constant term1, so it has no factor x: if L≤32, a nonzero B of degree less than 32 cannot be divisible by it. Thus **any single nonzero burst of span≤32 bits** is detected, including sparse interiors—not just runs of ones. This algebraic guarantee is not inferred from our finite all-ones enumeration. It does not cover multiple separated bursts with larger combined span or every longer error. The reflected 33-bit full-generator mask `(0xEDB88320<<1)|1` is an explicit undetected example.

**CRC ≠ authentication**: an adversary who can alter payload and tag can recompute them; the toy flips a payload bit, replaces both tags, and both checkers accept. Even an ordinary unkeyed hash alone cannot authenticate a replaceable log; use an appropriate authenticated mechanism/trust model when required, not this CRC toy.
```python
#!/usr/bin/env python3
"""Deterministic synthetic logged chunks, NOT a change to LeRobot/Parquet format.
Same original payload/tags and exact corrupted cases for no-check, sum8, CRC32.
CRC-32/IEEE (ISO-HDLC): reflected 0xEDB88320, init/xorout 0xFFFFFFFF.
Payload-only errors; stored tags/length/framing intact in this experiment.
CRC is error detection, NOT authentication: collisions exist and tags can be forged.
"""
import zlib

POLY=0xEDB88320
PAYLOADS=(bytes(16),bytes([255])*16,bytes(range(16)),bytes((37*j+11)%256 for j in range(16)))


def crc32(data):
    state=0xFFFFFFFF
    for byte in data:
        state^=byte
        for _ in range(8):
            state=(state>>1)^(POLY if state&1 else 0)
    return state^0xFFFFFFFF


def sum8(data):
    return sum(data)&0xFF


def flip(data, start, mask):
    result=bytearray(data)
    for j in range(mask.bit_length()):
        if mask>>j&1:
            bit=start+j
            result[bit//8]^=1<<(bit%8)  # Contiguous on the reflected, LSB-first bit stream.
    return bytes(result)


def corpus():
    """All locations; 1-bit errors and all-ones bursts are samples, not all shapes."""
    rows=[]
    for original in PAYLOADS:
        for start in range(128):
            rows.append(("single_bit",original,flip(original,start,1)))
        for length in range(2,33):
            for start in range(129-length):
                rows.append(("burst_2to32",original,flip(original,start,(1<<length)-1)))
        for j in range(0,16,2):
            changed=bytearray(original)
            changed[j]=(changed[j]+1)&255
            changed[j+1]=(changed[j+1]-1)&255
            rows.append(("balanced_bytes",original,bytes(changed)))
        # Full reversed generator spans33 bits: an explicit long-burst collision.
        rows.append(("generator_33",original,flip(original,0,(POLY<<1)|1)))
    return tuple(rows)


def accepted(arm, received, tags):
    if arm=="no_check":
        return True
    if arm=="sum8":
        return sum8(received)==tags[0]
    if arm=="CRC32":
        return crc32(received)==tags[1]
    raise ValueError("unknown integrity arm")


def main():
    assert crc32(b"123456789")==0xCBF43926
    print(f'CRC32_IEEE("123456789")=0x{crc32(b"123456789"):08X}; reflected_poly=0x{POLY:08X}')
    print("init=xorout=0xFFFFFFFF; same4 synthetic16-byte chunks; tags/framing uncorrupted")
    print("group arm corrupted undetected undetected_rate")
    rows=corpus()
    for group in ("single_bit","burst_2to32","balanced_bytes","generator_33"):
        cases=tuple((original,received) for kind,original,received in rows if kind==group)
        for arm in ("no_check","sum8","CRC32"):
            misses=0
            for original,received in cases:
                assert received!=original
                assert crc32(received)==zlib.crc32(received)&0xFFFFFFFF
                misses+=accepted(arm,received,(sum8(original),crc32(original)))
            print(f"{group} {arm} {len(cases)} {misses} {misses/len(cases):.6f}")
    clean=[sum(accepted(arm,p,(sum8(p),crc32(p))) for p in PAYLOADS) for arm in ("no_check","sum8","CRC32")]
    print(f"clean acceptance: no_check={clean[0]}/4 sum8={clean[1]}/4 CRC32={clean[2]}/4")
    original=PAYLOADS[2];tampered=flip(original,0,1)
    forged=(sum8(tampered),crc32(tampered))
    print(f"attacker_recomputed_tags: sum8_accept={int(accepted('sum8',tampered,forged))} CRC32_accept={int(accepted('CRC32',tampered,forged))}")
    print("CRC detects ANY single nonzero burst span<=32 bits by polynomial degree, not by this finite enumeration")
    print("33-bit collision is deliberate; CRC != authentication; no random/channel-probability claim")
    print("Real recording/storage faults and format integration: 本页未运行")


if __name__=="__main__":
    main()
```
